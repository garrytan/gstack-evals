// GEVA-540 kit (not committed): keyless tests for stage-gap.ts, the stage 2 controller's scoring-failure rule and its
// pressure gate. Run from the gstack-evals checkout at the run's head: bun test <kit>/stage-gap.test.ts
import { describe, expect, test } from "bun:test";
import { createHash } from "node:crypto";
import { existsSync, mkdirSync, mkdtempSync, readdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { dirname, join, resolve } from "node:path";
import { clearResult, clearedResults, pressure60, pressureHold, scoreFailure, writeGap, type Files } from "./stage-gap.ts";

const REPO = resolve(process.env.H2H_REPO ?? "/workspace/gstack-evals");
const { scoreSession } = await import(REPO + "/lanes/outcome-skills/h2h/score/score.ts");
const { readRecords } = await import(REPO + "/lanes/outcome-skills/h2h/analysis/commands.ts");

const temp = (p: string) => mkdtempSync(join(tmpdir(), p));
const sha = (s: string) => createHash("sha256").update(s).digest("hex");
const put = (root: string, files: Record<string, string>) => {
  for (const [p, b] of Object.entries(files)) {
    mkdirSync(dirname(join(root, p)), { recursive: true });
    writeFileSync(join(root, p), b);
  }
};
const json = (f: string, x: unknown) => writeFileSync(f, JSON.stringify(x, null, 1) + "\n");
const read = (f: string) => readFileSync(f, "utf8");

/** An sm-unit.v1 dir that readUnit accepts (as tests/outcome-skills/h2h-score.test.ts builds it). */
function unitDir(id: string) {
  const dir = temp("stage-gap-unit-");
  const patch = "diff --git a/tests/new_test.go b/tests/new_test.go\nnew file mode 100644\n--- /dev/null\n+++ b/tests/new_test.go\n@@ -0,0 +1 @@\n+package tests\n";
  const body: Record<string, string> = {
    "fix.patch": "diff --git a/src/a.go b/src/a.go\n", "test.patch": patch, "issue.md": "x\n", "parser.py": "def parser(log):\n  return {}\n",
    "harbor/task.toml": "x\n", "harbor/environment/Dockerfile": "FROM x\n", "harbor/instruction.md": "x\n",
  };
  put(dir, body);
  json(join(dir, "unit.json"), {
    schema: "sm-unit.v1", id, lang: "go", base_commit: "b".repeat(40), repo_dir: "/repo", work_dir: "/repo",
    image: { ref: "example/fixture@sha256:" + "c".repeat(64), image_id: null, runtime_id: null }, test_patch: { files: ["tests/new_test.go"], sha256: sha(patch) },
    tests: { f2p: ["t1"], p2p: ["t2"] }, files: Object.fromEntries(Object.entries(body).map(([p, b]) => [p, sha(b)])),
  });
  return dir;
}

/** A stage 2 out dir holding one session (index 68, arm none) that finished its agent run. */
function outDir() {
  const out = temp("stage-gap-out-");
  for (const d of ["workers", "scores", "records", "sessions/0068-none"]) mkdirSync(join(out, d), { recursive: true });
  const f: Files = { result: join(out, "workers/0068.result.json"), score: join(out, "scores/0068.json"), record: join(out, "records/0068.json"), session: join(out, "sessions/0068-none/session.json") };
  json(f.session, { schema: "h2h-session.v1", unit: "o__unit-1", arm: "none", model: "claude-opus-5-5", item: "h2h-opus", stage: "stage2", index: 68, end: "completed", gap: null, cost_usd: 0.195793 });
  return { out, f, unit: unitDir("o__unit-1") };
}
const fail = (f: Files, kind = "score_error", code = "E_SM_PARSER") =>
  json(f.result, { index: 68, unit: "o__unit-1", arm: "none", kind, code, detail: code + " o__unit-1-score3: the dataset parser failed (exit 1, timed out)" });
const base = { schema: "geva540-stage-record.v1", stage: "stage2", item: "h2h-opus", run: "test", index: 68, unit: "o__unit-1", arm: "none" };
const gap = (x: ReturnType<typeof outDir>) => writeGap(x.f, x.unit, base, scoreSession, "2026-11-17T22:23:55.658Z");

describe("a session whose scoring failed (section 7 step 6)", () => {
  test("a first score_error halts live, and a relaunch gives it exactly one rescore", () => {
    const x = outDir();
    fail(x.f);
    expect(scoreFailure(x.f, "live")).toBe("halt");
    expect(scoreFailure(x.f, "relaunch")).toBe("rescore");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
    expect(clearResult(x.f).code).toBe("E_SM_PARSER");
    expect(existsSync(x.f.result)).toBe(false);
    expect(clearedResults(x.f)).toHaveLength(1);
    fail(x.f);
    expect(scoreFailure(x.f, "live")).toBe("gap");
    expect(scoreFailure(x.f, "relaunch")).toBe("gap");
  });

  test("a second score_error is written as a gap, in the scorer's own shape and with its code", () => {
    const x = outDir();
    fail(x.f);
    clearResult(x.f);
    fail(x.f);
    const rec = gap(x);
    const score = JSON.parse(read(x.f.score));
    expect(score).toEqual({
      schema: "h2h-score.v1", unit: "o__unit-1", arm: "none", stage: "stage2", outcome: null, verified: false, verified_without_new_test_rule: false,
      adapted_verified: null, gap: "E_SM_PARSER", dropped_arm_tests: [], worktree_pass: null, lines_changed: 0, planning_lines: 0,
    });
    expect(score).toEqual(scoreSession({ ...JSON.parse(read(x.f.session)), gap: "E_SM_PARSER" }, dirname(x.f.session), x.unit).score);
    expect(readRecords(join(x.out, "scores"), "h2h-score.v1")).toEqual([score]);
    expect(JSON.parse(read(x.f.record))).toEqual(rec);
    expect(rec).toMatchObject({ ...base, kind: "session", cost_usd: 0.195793, gap: "E_SM_PARSER", score_file: x.f.score, session_file: x.f.session });
    expect(String(rec.detail)).toContain("timed out");
    expect(existsSync(x.f.result)).toBe(false);
    expect(clearedResults(x.f).map((r) => r.code)).toEqual(["E_SM_PARSER", "E_SM_PARSER"]);
    expect(scoreFailure(x.f, "live")).toBe("keep");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
  });

  test("a failure followed by a clean rescore writes no gap", () => {
    const x = outDir();
    fail(x.f);
    clearResult(x.f);
    const clean = { schema: "h2h-score.v1", unit: "o__unit-1", arm: "none", stage: "stage2", outcome: "clean_fix", verified: true, gap: null };
    json(x.f.score, clean);
    json(x.f.record, { ...base, kind: "session", cost_usd: 0.195793, gap: null });
    const before = [read(x.f.score), read(x.f.record)];
    expect(scoreFailure(x.f, "live")).toBe("keep");
    expect(scoreFailure(x.f, "relaunch")).toBe("keep");
    fail(x.f);
    expect(scoreFailure(x.f, "live")).toBe("keep");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
    expect([read(x.f.score), read(x.f.record)]).toEqual(before);
  });

  test("a session the scorer already recorded as a gap is never written over", () => {
    const x = outDir();
    const own = { schema: "h2h-score.v1", unit: "o__unit-1", arm: "none", stage: "stage2", outcome: null, verified: false, gap: "E_SM_INFRA_GAP", lines_changed: 5 };
    json(x.f.score, own);
    json(x.f.record, { ...base, kind: "session", cost_usd: 0.2, gap: "E_SM_INFRA_GAP" });
    const before = [read(x.f.score), read(x.f.record)];
    fail(x.f);
    clearResult(x.f);
    fail(x.f, "score_error", "E_SM_PARSER");
    expect(scoreFailure(x.f, "live")).toBe("keep");
    expect(scoreFailure(x.f, "relaunch")).toBe("keep");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
    expect([read(x.f.score), read(x.f.record)]).toEqual(before);
    rmSync(x.f.score);
    expect(scoreFailure(x.f, "live")).toBe("keep");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
  });

  test("an earlier failure of another kind does not count toward the gap", () => {
    const x = outDir();
    fail(x.f, "error", "KIT_STAGE_IMAGE");
    clearResult(x.f);
    fail(x.f);
    expect(scoreFailure(x.f, "live")).toBe("halt");
    expect(scoreFailure(x.f, "relaunch")).toBe("rescore");
    expect(() => gap(x)).toThrow("KIT_STAGE_GAP");
    expect(readdirSync(join(x.out, "scores"))).toEqual([]);
  });
});

describe("the start gate's pressure check", () => {
  const cpu = (a60: number) => "some avg10=1.00 avg60=" + a60.toFixed(2) + " avg300=3.00 total=1\nfull avg10=0.00 avg60=0.00 avg300=0.00 total=0\n";
  test("starts wait only above the cpu or io threshold", () => {
    expect(pressure60(cpu(24.9))).toBe(24.9);
    expect(pressureHold(cpu(20), cpu(40), 20, 40)).toBe(false);
    expect(pressureHold(cpu(20.13), cpu(1.08), 20, 40)).toBe(true);
    expect(pressureHold(cpu(5.9), cpu(40.5), 20, 40)).toBe(true);
    expect(pressureHold(cpu(0.9), cpu(9.28), 20, 40)).toBe(false);
  });
  test("a missing or unreadable pressure file reads as no pressure", () => {
    expect(pressure60(null)).toBe(0);
    expect(pressure60("")).toBe(0);
    expect(pressureHold(null, null, 20, 40)).toBe(false);
  });
});
