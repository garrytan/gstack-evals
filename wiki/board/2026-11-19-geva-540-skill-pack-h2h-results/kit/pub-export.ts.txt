// pub-export.ts (GEVA-540 kit): prereg section 12's published per-session material for stage 2 of the skill-pack
// head-to-head. For every session: its diff (the scorer's own main-checkout change, sessionTree + openTree at the
// session's base_head, as scoreSession calls it), score record, cost, exposure markers and a tool-event summary
// (tools called; skills launched, from Skill tool calls and the frozen detector's skill-start markers; continuations), with each repository's root licence and notice files beside the
// diffs. Tokens are left out: the session runner recorded them as 0 (published as missing). Usage:
//   bun pub-export.ts <shard>/<of>     writes pub/rows/<session>.json, pub/diffs/<session>.patch, pub/licences/<repo>/
//   bun pub-export.ts assemble         writes pub/sessions.jsonl and pub/MANIFEST.sha256
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { existsSync, mkdirSync, mkdtempSync, readdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join, relative } from "node:path";
const R = "/home/ubi/h2h/run/geva540-s2-7700a8f";
const P = R + "/pub";
const sha = (b: string | Buffer) => createHash("sha256").update(b).digest("hex");
if (process.argv[2] === "assemble") {
  const rows = readdirSync(P + "/rows").sort().map((f) => readFileSync(join(P, "rows", f), "utf8").trim());
  writeFileSync(P + "/sessions.jsonl", rows.join("\n") + "\n");
  const files: string[] = [];
  const walk = (d: string) => { for (const f of readdirSync(d, { withFileTypes: true })) f.isDirectory() ? walk(join(d, f.name)) : files.push(join(d, f.name)); };
  for (const d of ["diffs", "licences"]) walk(join(P, d));
  files.push(P + "/sessions.jsonl");
  writeFileSync(P + "/MANIFEST.sha256", files.sort().map((f) => sha(readFileSync(f)) + "  " + relative(P, f)).join("\n") + "\n");
  console.log(JSON.stringify({ rows: rows.length, files: files.length }));
  process.exit(0);
}
const { sessionTree } = await import("/home/ubi/h2h/repo/lanes/outcome-skills/h2h/score/score.ts");
const { openTree } = await import("/home/ubi/h2h/repo/lanes/outcome-skills/h2h/score/tree.ts");
const [shard, of] = (process.argv[2] ?? "0/1").split("/").map(Number) as [number, number];
for (const d of ["diffs", "rows", "licences"]) mkdirSync(join(P, d), { recursive: true });
const SPDX = new Map<string, string>();
const findSpdx = (o: any): void => { if (o && typeof o === "object") for (const [k, v] of Object.entries(o)) { if (v && typeof v === "object" && typeof (v as any).spdx === "string" && k.includes("/")) SPDX.set(k, (v as any).spdx); else findSpdx(v); } };
for (const f of ["o", "b1", "b2"]) findSpdx(JSON.parse(readFileSync("/home/ubi/h2h/repo/wiki/plans/2026-11-15-geva-538-skill-pack-h2h-prereg/" + f + "-licences.json", "utf8")));
const units = new Map<string, any>(JSON.parse(readFileSync(R + "/stage/units.json", "utf8")).map((u: any) => [u.id, u]));
const dirs = readdirSync(R + "/out/sessions").sort().filter((_, i) => i % of === shard);
for (const d of dirs) {
  const sd = join(R, "out/sessions", d);
  const s = JSON.parse(readFileSync(join(sd, "session.json"), "utf8"));
  const scoreFile = join(R, "out/scores", String(s.index).padStart(4, "0") + ".json");
  const score = existsSync(scoreFile) ? JSON.parse(readFileSync(scoreFile, "utf8")) : null;
  const u = units.get(s.unit);
  const ud = JSON.parse(readFileSync(join(R, "stage/units", s.unit, "unit.json"), "utf8"));
  const tools: Record<string, number> = {};
  const skills: string[] = [];
  const log = join(sd, "sdk-messages.jsonl");
  if (existsSync(log)) for (const line of readFileSync(log, "utf8").split("\n")) {
    let m: any;
    try { m = JSON.parse(line); } catch { continue; }
    const content = m?.message?.content ?? m?.content;
    if ((m?.type !== "assistant" && m?.type !== "AssistantMessage") || !Array.isArray(content)) continue;
    for (const b of content) {
      if (b?.type !== "tool_use") continue;
      tools[b.name] = (tools[b.name] ?? 0) + 1;
      if (b.name === "Skill") skills.push(String(b.input?.skill ?? b.input?.command ?? ""));
    }
  }
  for (const k of s.exposure?.markers ?? []) if (String(k).startsWith("skill-start:")) skills.push(String(k).slice("skill-start:".length));
  let diff: string | null = null, diffSha: string | null = null, diffBytes = 0, captureError: string | null = null;
  if (!s.gap && existsSync(join(sd, s.tree_bundle ?? "tree.tar"))) {
    const work = mkdtempSync(join(tmpdir(), "pub-"));
    try {
      const repo = sessionTree(sd, s.tree_bundle, work);
      const t = openTree(repo, s.base_head, { containerRepoDir: ud.repo_dir });
      try {
        diff = "diffs/" + d + ".patch";
        writeFileSync(join(P, diff), t.main.patch);
        diffSha = sha(t.main.patch);
        diffBytes = Buffer.byteLength(t.main.patch);
        const lic = join(P, "licences", String(ud.repo).replace(/\//g, "__"));
        let mine = false;
        try { mkdirSync(lic); mine = true; } catch {}
        if (mine) {
          const names = execFileSync("git", ["-C", repo, "ls-tree", "--name-only", s.base_head], { encoding: "utf8" }).split("\n").filter((n) => /(^|[-_.])(licen[cs]e|copying|notice)/i.test(n));
          for (const n of names) { const body = t.readBase(n); if (body !== null) writeFileSync(join(lic, n), body); }
          writeFileSync(join(lic, "SOURCE.txt"), "https://github.com/" + ud.repo + " at " + s.base_head + "\nLicence at the freeze-date read (prereg o-, b1- and b2-licences.json): " + (SPDX.get(String(ud.repo)) ?? "not listed") + "\nRoot licence and notice files at that commit: " + (names.join(", ") || "none") + "\n");
        }
      } finally { t.close(); }
    } catch (e) {
      captureError = String((e as Error).message).slice(0, 200);
      diff = null; diffSha = null; diffBytes = 0;
    } finally { rmSync(work, { recursive: true, force: true }); }
  }
  const row = {
    session: d, index: s.index, unit: s.unit, arm: s.arm, model: s.model, item: s.item, class: u?.kind ?? null, entry: u?.entry ?? null, repo: ud.repo, lang: ud.lang,
    end: s.end, gap: s.gap ?? score?.gap ?? null, cost_usd: s.cost_usd, wall_ms: s.wall_ms, attempts: s.attempts, turns: s.turns, scripted_turns: s.scripted_turns,
    continuations: s.continuations, spurious_continuations: s.spurious_continuations, questions: s.questions, reservation_refusals: s.reservation_refusals,
    claude_version: s.claude_version, exposure: s.exposure, tools, skills_launched: skills,
    score: score && { outcome: score.outcome, verified: score.verified, verified_without_new_test_rule: score.verified_without_new_test_rule, adapted_verified: score.adapted_verified, gap: score.gap, lines_changed: score.lines_changed, planning_lines: score.planning_lines, dropped_arm_tests: score.dropped_arm_tests, worktree_pass: score.worktree_pass },
    diff, diff_sha256: diffSha, diff_bytes: diffBytes, capture_error: captureError,
  };
  writeFileSync(join(P, "rows", d + ".json"), JSON.stringify(row) + "\n");
}
console.log("shard " + shard + "/" + of + " done " + dirs.length);
