#!/usr/bin/env bash
# GEVA-589 counted judging (change post section 4, under the rerun prereg section 7): passes A and B on the 104 counted sessions on the parent's fallback
# route (resumable: a session already in judged.jsonl is skipped; scoring waits for all 104), from the judge clone at its admitted head, then scoring. Plain: the rerun seals nothing but the pooled check.
set -uo pipefail
export PATH=$HOME/.local/bin:$HOME/.bun/bin:$PATH
L=$HOME/cso3-ledger; export GSTACK_EVALS_LEDGER=$L/ledger.sqlite GSTACK_EVALS_LEDGER_RECORD=$L/record
for k in ANTHROPIC_API_KEY OPENROUTER_API_KEY OPENAI_API_KEY; do export $k="$(cat $HOME/.keys/$k)"; done
unset CSO_READOUT_KEY
G=$HOME/work/judge/gstack-evals; cd $G; GE=$G/bin/gstack-evals
export CSO_PLAN=longlist
J=$HOME/judge-main3; mkdir -p $J
step() { echo "== $1 $(date -u +%FT%TZ)"; }
step head; git log --oneline -1; echo "dirty $(git status --porcelain --untracked-files=all | wc -l)"
N=$(wc -l < $HOME/main3/sessions.jsonl); echo "sessions $N"
step admit
GSTACK_EVALS_GH_BIN=$HOME/bin/ghshim sg docker -c "$GE rig admit --item cso3-judge --proof direct --prereg wiki/plans/2026-11-24-geva-589-cso-longlist-change/README.md --run cso3-judge --check-run $(cat $HOME/ci-run-judge) --repo garrytan/gstack-evals" || { echo "admit failed"; exit 1; }
step judge
$GE cso judge --work $HOME/units --sessions $HOME/main3/sessions.jsonl --out $J --run cso3-judge --route fallback-opus-particle; echo "judge exit $?"
step score
[ $N = 104 ] && [ ! -e $J/scores.jsonl ] && $GE cso score --work $HOME/units --sessions $HOME/main3/sessions.jsonl --judged $J/judged.jsonl --out $J/scores.jsonl; echo "score exit $?"
jq -r '"\(.model) gap=\(.gap)"' $J/scores.jsonl | sort | uniq -c
$GE ledger status --item cso3-judge 2>&1 | grep ITEM
echo JUDGE_MAIN_DONE
