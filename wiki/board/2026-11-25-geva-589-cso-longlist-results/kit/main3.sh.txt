#!/usr/bin/env bash
# GEVA-589 counted run (change post section 4, change 3): admit cso3-main, serve it with Opus's $16 allowance, run the
# parent's frozen stage 2 draw restricted to its cso cells (104 sessions) four at a time at gstack a1af71d5, holding
# cso3-main's $125 cap through each rerun's spend cap and the Captain's $170 for cso3 and cso3b together through
# combined-cap.ts before each session, then stop the driver and export its folder. The Captain's guard: once 20 Opus
# sessions have finished, an Opus average over $2.20 pauses the queue (no new session starts) and goes to the Captain.
# A session whose allowance does not fit the room while other sessions still run waits for them (their unspent
# allowances come back); the run cap-stops only when nothing is running and the room is under the next allowance.
set -uo pipefail
export PATH=$HOME/.local/bin:$HOME/.bun/bin:$PATH
L=$HOME/cso3-ledger; export GSTACK_EVALS_LEDGER=$L/ledger.sqlite GSTACK_EVALS_LEDGER_RECORD=$L/record
for k in ANTHROPIC_API_KEY OPENROUTER_API_KEY; do export $k="$(cat $HOME/.keys/$k)"; done
unset OPENAI_API_KEY CSO_READOUT_KEY
G=$HOME/work/gstack-evals; cd $G; GE=$G/bin/gstack-evals
export CSO_PLAN=longlist
S=$HOME/main3; mkdir -p $S/units $S/running
CAP=125; TOTAL=170; LANES=${LANES:-4}; GUARD_N=20; GUARD_USD=2.20
D=wiki/board/2026-11-23-geva-561-cso-advisories-results/stage2-draw.json
step() { echo "== $1 $(date -u +%FT%TZ)"; }
step head; git log --oneline -1; echo "dirty $(git status --porcelain --untracked-files=all | wc -l)"
jq -r '.order[] | select(.arm == "cso") | "\(.unit) \(.model) \(.arm)"' $D > $S/order.txt; echo "order $(wc -l < $S/order.txt) sessions, draw $(sha256sum $D | cut -c1-12)"
[ $(wc -l < $S/order.txt) = 104 ] || { echo "need 104 cso cells"; exit 1; }
if cut -d' ' -f1 $S/order.txt | grep -qxFf $HOME/pilot-units.txt; then echo "a pilot unit is in the counted order"; exit 1; fi
[ -e $S/next ] || echo 0 > $S/next
JUDGE_LEFT=$($GE ledger status --item cso3-judge 2>&1 | sed -n "s/^ITEM: .* left=\$\([0-9.]*\).*/\1/p"); echo "cso3-judge room held for counted judging \$$JUDGE_LEFT"
[ -n "$JUDGE_LEFT" ] || { echo "no cso3-judge room read"; exit 1; }
step admit
[ -e $S/admitted ] || { GSTACK_EVALS_GH_BIN=$HOME/bin/ghshim sg docker -c "$GE rig admit --item cso3-main --proof direct --prereg wiki/plans/2026-11-24-geva-589-cso-longlist-change/README.md --run cso3-main --check-run $(cat $HOME/ci-run) --repo garrytan/gstack-evals" && touch $S/admitted || { echo "admit failed"; exit 1; }; }
step serve
GW=$(sg docker -c "docker network inspect bridge -f '{{(index .IPAM.Config 0).Gateway}}'")
MID=$(cat $HOME/machine-id)
setsid -f $GE outcome serve --run cso3-main --envelope-host $MID --stage cso3-main=cso:cso --host $GW --dir $S/driver --planned 104 --reruns 8 --cso-allowance claude-opus-5-5=16 > $S/serve.log 2>&1 < /dev/null
for i in $(seq 60); do grep -q '^stage=' $S/serve.log && break; grep -q '^E_' $S/serve.log && break; sleep 2; done
cat $S/serve.log
export CSO_CONTROL=$(sed -n 's/^control=//p' $S/serve.log) CSO_PROXY=$(sed -n 's/^stage=.* proxy=//p' $S/serve.log)
[ -n "$CSO_CONTROL" ] && [ -n "$CSO_PROXY" ] || { echo "driver did not start"; exit 1; }
export CSO_WORK=$HOME/units CSO_HOME=$HOME/homes/cso CSO_STAGE=rerun CSO_OUT=$S/sessions
alive() { pgrep -f "[o]utcome serve --run cso3-main" > /dev/null; }
guard() {
  [ -e $S/guard-ok ] && return 0
  local n avg
  n=$(find $S/sessions -path "*cso-opus-*" -name session.json 2>/dev/null | wc -l)
  [ $n -lt $GUARD_N ] && return 0
  avg=$(find $S/sessions -path "*cso-opus-*" -name session.json -print0 | xargs -0 jq -s '[.[].cost_usd // 0] | add / length')
  if awk -v a=$avg -v g=$GUARD_USD 'BEGIN { exit !(a > g) }'; then echo "GUARD opus average \$$avg over $n sessions is over \$$GUARD_USD: pause" > $S/pause; cat $S/pause; return 1; fi
  echo "GUARD opus average \$$avg over $n sessions" > $S/guard-ok; cat $S/guard-ok
}
claim() {
  local lane=$1 al fin run n line cc
  exec 9> $S/queue.lock; flock 9
  n=$(cat $S/next); line=$(sed -n "$((n + 1))p" $S/order.txt)
  if [ -z "$line" ] || [ -e $S/capstop ] || [ -e $S/halt ] || [ -e $S/pause ] || ! guard; then flock -u 9; exec 9>&-; return 1; fi
  set -- $line
  al=3; [ $2 = opus ] && al=16
  fin=$(find $S/sessions -name session.json -print0 2>/dev/null | xargs -0 -r jq -s '[.[].cost_usd // 0] | add // 0'); [ -n "$fin" ] || fin=0
  run=$(cat $S/running/* 2>/dev/null | awk '{ s += $1 } END { printf "%.6f", s + 0 }')
  ROOM=$(awk -v c=$CAP -v f=$fin -v r=$run 'BEGIN { printf "%.2f", c - f - r }')
  if awk -v r=$ROOM -v a=$al 'BEGIN { exit !(r < a) }'; then
    if awk -v r=$run 'BEGIN { exit !(r > 0) }'; then flock -u 9; exec 9>&-; return 2; fi
    echo "room \$$ROOM in \$$CAP is under the next allowance \$$al with nothing running: cap stop before session $((n + 1))"; touch $S/capstop; flock -u 9; exec 9>&-; return 1
  fi
  bun $G/lanes/outcome-skills/cso/combined-cap.ts --cap $TOTAL --programs cso3 --spent $(awk -v f=$fin -v r=$run 'BEGIN { printf "%.6f", f + r }') --reserve $JUDGE_LEFT --next $al >> $S/combined.log 2>&1; cc=$?
  if [ $cc = 10 ]; then
    if awk -v r=$run 'BEGIN { exit !(r > 0) }'; then flock -u 9; exec 9>&-; return 2; fi
    tail -1 $S/combined.log; echo "combined \$$TOTAL check stops before session $((n + 1)) with nothing running"; touch $S/capstop; flock -u 9; exec 9>&-; return 1
  fi
  [ $cc != 0 ] && { tail -2 $S/combined.log; echo "combined check exit $cc"; touch $S/halt; flock -u 9; exec 9>&-; return 1; }
  echo $((n + 1)) > $S/next
  echo $al > $S/running/$lane
  CLAIM="$((n + 1)) $line $al"
  flock -u 9; exec 9>&-
}
lane() {
  local i=$1
  local c w=0
  while :; do
    claim $i; c=$?
    if [ $c = 2 ]; then [ $w = 0 ] && echo "lane $i waits: room \$$ROOM $(date -u +%T)"; w=1; sleep 30; continue; fi
    [ $c != 0 ] && break; w=0
    set -- $CLAIM
    local n=$1 U=$2 m=$3 a=$4 al=$5 extra="" rc t0
    echo "[\"$U\"]" > $S/units/$U.json
    [ $m = opus ] && extra="--allowance 16"
    step "session $n $U $a $m lane $i room $ROOM"
    t0=$(date +%s)
    sg docker -c "$GE cso rerun --arm $a --model $m --units $S/units/$U.json --spend-cap $ROOM $extra"; rc=$?
    rm -f $S/running/$i
    echo "rerun exit $rc wall $(( $(date +%s) - t0 ))s session $n"
    [ $rc = 10 ] && touch $S/capstop
    [ $rc != 0 ] && ! alive && { echo "the driver is gone after session $n"; touch $S/halt; }
  done
}
step sessions
for i in $(seq $LANES); do lane $i > $S/lane-$i.log 2>&1 & sleep 30; done
wait
echo "exits: $(grep -h '^rerun exit' $S/lane-*.log | awk '{ print $3 }' | sort | uniq -c | tr -s ' ' | tr '\n' ';')"
[ -e $S/pause ] && { echo "MAIN PAUSED: $(cat $S/pause); the driver stays up"; exit 3; }
step stop
curl -s -X POST $CSO_CONTROL/stop; sleep 15; alive && pkill -TERM -f "[o]utcome serve --run cso3-main"; sleep 10; tail -5 $S/serve.log
$GE ledger status --item cso3-main 2>&1 | grep ITEM
[ -e $S/capstop ] && echo "MAIN CAP-STOPPED: it ends incomplete"
[ -e $S/halt ] && echo "MAIN HALTED"
step sessions-file
find $S/sessions/rerun -name session.json | sort | xargs -I{} jq -c . {} > $S/sessions.jsonl; echo "sessions $(wc -l < $S/sessions.jsonl)"
jq -r '"\(.model) \(.end) \(.exposure.loaded) \(.exposure.procedure_ran)"' $S/sessions.jsonl | sort | uniq -c
step export
tar czf $HOME/export-main3.tgz -C $HOME main3/driver main3/sessions.jsonl main3/serve.log main3/combined.log main3/order.txt cso3-ledger main3.log 2>/dev/null; sha256sum $HOME/export-main3.tgz
echo MAIN_DONE
