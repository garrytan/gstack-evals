#!/usr/bin/env bash
# GEVA-589 pooled judge check (change post section 4, the rerun prereg section 7 check), with the readout key in /dev/shm only.
set -uo pipefail
export PATH=$HOME/.local/bin:$HOME/.bun/bin:$PATH
L=$HOME/cso3-ledger; export GSTACK_EVALS_LEDGER=$L/ledger.sqlite GSTACK_EVALS_LEDGER_RECORD=$L/record
for k in ANTHROPIC_API_KEY OPENROUTER_API_KEY OPENAI_API_KEY; do export $k="$(cat $HOME/.keys/$k)"; done
K=/dev/shm/cso-readout.key
trap 'rm -f $K; test ! -e $K && echo "KEY_REMOVED $(date -u +%FT%TZ)"' EXIT
step() { echo "== $1 $(date -u +%FT%TZ)"; }
[ "$(sha256sum $K | cut -c1-8)" = 32a5870e ] || { echo "key digest mismatch"; exit 1; }
G=$HOME/work/judge/gstack-evals; cd $G; GE=$G/bin/gstack-evals
export CSO_PLAN=longlist
step head; git log --oneline -1; echo "dirty $(git status --porcelain --untracked-files=all | wc -l)"
P=$HOME/work/parent-check/judge-e/check-fallback-opus-particle.sealed
[ "$(sha256sum $P | cut -d' ' -f1)" = "$(cut -d' ' -f1 $P.sha256)" ] || { echo "parent check digest mismatch"; exit 1; }
step admit
GSTACK_EVALS_GH_BIN=$HOME/bin/ghshim sg docker -c "$GE rig admit --item cso3-judge --proof direct --prereg wiki/plans/2026-11-24-geva-589-cso-longlist-change/README.md --run cso3-judge --check-run $(cat $HOME/ci-run-judge) --repo garrytan/gstack-evals" || { echo "admit failed"; exit 1; }
step pooled-check
$GE cso judge --pooled-check --work $HOME/units --sessions $HOME/main3/sessions.jsonl --parent-check $P --pilot-gate-file $HOME/judge-repilot2-6/pilot-gate.json --judged $HOME/judge-main3/judged.jsonl --key $K --out $HOME/judge-pooled3 --run cso3-judge --route fallback-opus-particle; pooled_rc=$?; echo "pooled exit $pooled_rc"
step analyze
if [ $pooled_rc = 0 ]; then
  cut -d' ' -f1 $HOME/main3/order.txt | sort -u | sed 's/.*/"&"/' > $HOME/judge-main3/units52.jsonl; echo "units $(wc -l < $HOME/judge-main3/units52.jsonl)"
  $GE cso rerun-analyze --work $HOME/units --units $HOME/judge-main3/units52.jsonl --old-dir $HOME/work/export/stage2/sessions/stage2 --old-scores $HOME/work/export/stage2/scores.jsonl --sessions $HOME/main3/sessions.jsonl --scores $HOME/judge-main3/scores.jsonl --new-dir $HOME/main3/sessions/rerun --check $HOME/judge-pooled3/check-fallback-opus-particle.sealed --key $K --judged $HOME/judge-main3/judged.jsonl --old-judged $HOME/work/parent-s2/judge-s2/judged.jsonl > $HOME/judge-main3/rerun-analysis.json 2> $HOME/judge-main3/rerun-analysis.err; echo "analyze exit $?"
  tail -n 3 $HOME/judge-main3/rerun-analysis.err
else echo "pooled check did not pass: no analysis"; fi
$GE ledger status --program cso3 2>&1 | grep -E "ITEM|ENVELOPE" | cut -c1-200
echo POOLED_DONE
