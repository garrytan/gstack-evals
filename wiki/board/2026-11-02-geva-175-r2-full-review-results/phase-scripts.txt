#!/usr/bin/env bash
# R2 real run, phase F2b: F2 resumed after its matching batch finished (finished, 0 unjudged, third judge 0 packets) and
# judge-join refused --third with no third reads. Same steps from the join on; --third goes in only when the batch
# log shows the third judge answered. The paid matching batch is not rerun.
# Then the adjudication batch, joins, ledger, score, secondaries, r2-decide.py, complete-case sensitivity, #108 context.
set -u
PH=f2b
. $HOME/.capy/work/r2/common.sh
G="$R/bin/gstack-evals noloss"
P76=$R/wiki/plans/2026-10-31-geva-76-stacks-round-1-prereg
SID=stacks-r1-out19
SRC=$W/r2src
X=$W/r2/real
RAWR=$W/r2/raw/$SID
ARMS="gstack-full none-r2"
COMP=$W/bin/comparisons.r2.json
GSRC=$B/gstack; GBSRC=$B/gbrain
JUDGES="--primary anthropic/claude-opus-5-5 --second openai/gpt-6.1-sol --third anthropic/claude-sonnet-5-5"
arms() { local root=$1; shift; for a in "$@"; do printf -- ' --arm %s' "$root/$a"; done; }
n=0
step() { local pat=$1; shift; n=$((n + 1)); echo; echo "\$ $*" | sed "$HIDE"; "$@" > "$W/log/$PH$n.txt" 2>&1; local c=$?; grep -E -- "$pat" "$W/log/$PH$n.txt" | head -${SHOW:-30} | sed "$HIDE"; echo "exit $c (log/$PH$n.txt)"; LAST=$c; }
need() { [ "$LAST" = 0 ] || { echo "E_PHASE step $n exited $LAST; stopping before any later step"; echo stop > $W/$PH.exit; exit 1; }; }

J=$(cat $W/match-args.txt)
third() { grep -q "third 0;" "$1" && return 0; printf -- " --third %s/third.json" "$2"; }
[ "$(jq -r '.state + " " + (.unjudged|tostring)' $X/judge-match/run.json)" = "finished 0" ] || { echo "E_PHASE the matching batch did not finish with 0 unjudged; stopping before adjudication"; echo stop > $W/f2b.exit; exit 1; }
step 'judge-join|E_' $G judge-join --suite $S2 --key $X/match-key.json --reply $X/judge-match/primary.json --second $X/judge-match/second.json $(third $W/log/f21.txt $X/judge-match) $(arms $RAWR $ARMS) --out $X/judge-match/joined; need
for a in $ARMS; do step 'published|E_' $G ingest $X/judge-match/joined/$a --suite $S2 --root $X/judged; need; done
step 'packet|E_' $G judge-packet --suite $S2 $(arms $X/judged/$SID $ARMS) --seed 20261031 --out $X/adj --key $X/adj-key.json --adjudicate; need
A="--suite $S2 --packets $X/adj --key $X/adj-key.json $(arms $X/judged/$SID $ARMS) --brief $P76/judge-brief.md $JUDGES --ledger $LED --record $REC --source garrytan/gstack=$GSRC --source garrytan/gbrain=$GBSRC"
echo "$A" > $W/adj-args.txt
step '^(worst case|item )|E_' $G judge-run $A --item r2-judge-adjudicate --out $X/judge-adj --dry-run; need
date -u +"adjudicate start %FT%TZ"
step '^(judge-run|RIG_|E_)|unjudged|unverifiable' $G judge-run $A --item r2-judge-adjudicate --out $X/judge-adj --concurrency 4
ADJ=$n
step . sh -c "jq -c '{state, unjudged, lease: (.lease // .ledger // null)}' $X/judge-adj/run.json"
[ "$(jq -r '.state + " " + (.unjudged|tostring)' $X/judge-adj/run.json)" = "finished 0" ] || { echo "E_PHASE the adjudication batch did not finish with 0 unjudged; stopping before scoring"; echo stop > $W/f2b.exit; exit 1; }
unset ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY
step 'judge-join|E_' $G judge-join --suite $S2 --key $X/adj-key.json --reply $X/judge-adj/primary.json --second $X/judge-adj/second.json $(third $W/log/$PH$ADJ.txt $X/judge-adj) $(arms $X/judged/$SID $ARMS) --out $X/judge-adj/joined; need
for a in $ARMS; do step 'published|E_' $G ingest $X/judge-adj/joined/$a --suite $S2 --root $X/final; need; done
SHOW=40 step 'ITEM: r2|committed|E_' $L status --ledger $LED --ledger-record $REC
echo; echo "## Score, secondaries, the cap rule's exit, r2-decide.py"
SHOW=60 step . $G score --suite $S2 --root $X/final --prereg $COMP --variant gstack-full --reference none-r2 --out $X/score-gstack-full-vs-none.json
SHOW=60 step . $G secondary --suite $S2 --root $X/final --variant gstack-full --reference none-r2 --seed 20261031
echo "r2-caps.py exit $(cat $W/caps.exit)"
if [ "$(cat $W/caps.exit)" = 4 ]; then echo "r2-caps.py exited 4: R2 gives no outcome (FIX-METHOD to the Captain); r2-decide.py is not consulted"; else SHOW=20 step . python3 $W/bin/r2-decide.py $X/score-gstack-full-vs-none.json $SRC/gstack-full $S2; echo $LAST > $W/decide.exit; fi
echo; echo "## Complete-case sensitivity (reported, never deciding)"
step '^(derive|suite|E_)|units|blocking|dropped' $G derive --suite $S2 --comparisons $COMP --id stacks-r1-out19-cc --out $W/cc --root $X/final --drop-file $W/caps-drop.tsv
SHOW=60 step . $G score --suite $W/cc/stacks-r1-out19-cc.json --root $W/cc/arms --prereg $W/cc/comparisons.stacks-r1-out19-cc.json --variant gstack-full --reference none-r2 --out $W/score-cc.json
echo; echo "## #108's none-opus and gstack-opus on the same 19 units (SDK default prompt, 60 turns; context only)"
SHOW=40 step . $G secondary --suite $S2 --root $W/r2/suite/arms --variant gstack-opus --reference none-opus --seed 20261031
echo 0 > $W/f2b.exit
