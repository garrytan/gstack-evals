// GEVA-543: assemble ep analyze's input from ep records and the run's step records (outside the repo).
import { readFileSync, writeFileSync, existsSync } from "node:fs";
const run = process.env.EP_RUN_DIR!;
const A = run + "/analysis";
const rec = JSON.parse(readFileSync(A + "/records.json", "utf8"));
const step = (n: string) => JSON.parse(readFileSync(run + "/steps/" + n + ".json", "utf8"));
const adm = step("admission"), sr = step("start-rule"), gate = step("exposure-gate");
if (!adm.ok || adm.detail !== "primary composite") throw new Error("admission record is not 'primary composite'");
const m = /re-printed MDD ([0-9.]+) is within/.exec(sr.detail);
if (!sr.ok || !m) throw new Error("start-rule record has no re-printed MDD");
const seed = JSON.parse(readFileSync(run + "/readings-seed.json", "utf8")).seed;
const units = rec.units.filter((u: any) => u.set === "comparison");
const ids = new Set(units.map((u: any) => u.id));
const builds = rec.builds.filter((b: any) => ids.has(b.unit));
const reviews = rec.reviews.filter((r: any) => ids.has(r.unit));
const n = units.length, k = 3;
const arms: string[] = gate.arms;
const planned: Record<string, number> = { plain: n * k, stock: n, none: n };
for (const a of ["engphase", "standalone"]) planned[a] = arms.includes(a) ? n * k : 0;
const J = (f: string) => JSON.parse(readFileSync(run + "/judge/" + f, "utf8"));
const r0 = J("judge.json"), r1 = J("judge-r1.json");
const why = "judge check failed: under the one allowed rubric revision kappa 0.373 against 0.60 (sensitivity 0.444, specificity 0.900, stock control 0); under the original rubric kappa 0.743 passed but the stock control credited stock on udbd167e5cbd0 from plan text the plan reading missed, a rubric fault";
const base = { units, population: "all", primary: "composite", admitted: true, builds, reviews, smoke: rec.smoke, reviewer: "opus", planned, seed, designMdd: Number(m[1]) };
const out = (f: string, v: unknown) => { if (existsSync(A + "/" + f)) throw new Error(f + " exists"); writeFileSync(A + "/" + f, JSON.stringify(v, null, 1) + "\n"); };
out("analysis.json", { ...base, judged: { reported: false, why, bundles: r1, leaks: [] } });
out("analysis-descriptive-r0.json", { ...base, judged: { reported: true, why: null, bundles: r0, leaks: ["ud93e096f8b7f"] } });
out("analysis-descriptive-r1.json", { ...base, judged: { reported: true, why: null, bundles: r1, leaks: [] } });
const cnt = (xs: any[], key: string) => Object.entries(xs.reduce((o: any, x: any) => ((o[String(x[key])] = (o[String(x[key])] ?? 0) + 1), o), {})).map(([a, c]) => a + " " + c).join(", ");
console.log("records units by set: " + cnt(rec.units, "set") + "; comparison " + n + ", pilot flag " + units.filter((u: any) => u.pilot).length + ", repos " + new Set(units.map((u: any) => u.repo)).size);
console.log("builds " + rec.builds.length + " -> comparison units " + builds.length + " by arm: " + cnt(builds, "arm"));
console.log("reviews " + rec.reviews.length + " -> comparison units " + reviews.length + " by arm: " + cnt(reviews, "arm") + "; smoke " + rec.smoke.length);
console.log("planned " + JSON.stringify(planned) + ", seed " + seed + ", designMdd " + m[1] + ", judge bundles r0 " + r0.length + " r1 " + r1.length);
