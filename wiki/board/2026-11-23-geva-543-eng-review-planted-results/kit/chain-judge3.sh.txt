#!/usr/bin/env bash
# GEVA-543 judge chain 3 at 915685f: section 5's one rubric revision of call 2 (plan change 2026-11-23).
set -u
source ~/ep/env.sh >/dev/null 2>&1
HEAD=915685fb960eb0443c94dbf1a8a0b6e2782a1fe4
G=$EP_RUN_DIR/checks/gold-bundles.json
J=$EP_RUN_DIR/judge
log=~/ep/logs/chain-judge3.log
step() {
  n=$1; name=$2; shift 2
  cd ~/work/head || exit 3
  { [ "$(git rev-parse HEAD)" = $HEAD ] && [ "$(git status --porcelain | wc -l)" = 0 ]; } || { echo "$name: head not clean at $HEAD" >> $log; exit 3; }
  [ ! -e ~/HOST-RETIRED ] || exit 3
  [ "$(pgrep -f '[s]tage\.sh' | wc -l)" = 0 ] || { echo "$name: busy" >> $log; exit 3; }
  [ ! -e ~/ep/logs/admit-judge-c3-$n.log ] || { echo "$name: admission log $n exists" >> $log; exit 3; }
  GSTACK_EVALS_GH_BIN=$HOME/ep/gh-read.sh timeout 900 bin/gstack-evals rig admit --item ep-judge --proof direct --prereg wiki/plans/2026-11-15-geva-537-eng-review-planted-units-prereg/README.md --ledger-record $GSTACK_EVALS_LEDGER_RECORD > ~/ep/logs/admit-judge-c3-$n.log 2>&1
  echo "exit=$?" >> ~/ep/logs/admit-judge-c3-$n.log
  { grep -q 'ADMIT: green item=ep-judge' ~/ep/logs/admit-judge-c3-$n.log && grep -q '^exit=0' ~/ep/logs/admit-judge-c3-$n.log; } || { echo "$name: admission $n not green" >> $log; exit 3; }
  left=$(bin/gstack-evals ledger status --ledger-record $GSTACK_EVALS_LEDGER_RECORD --item ep-judge 2>&1 | sed -n 's/.*left=\$\([0-9.]*\).*/\1/p' | head -1)
  cap=$(python3 -c "import math,sys; print('%.2f' % (math.floor(float(sys.argv[1])*100)/100))" "$left")
  echo "$(date -u +%FT%TZ) $name: admission $n green, left=$left cap=$cap" >> $log
  PAID=1 bash ~/ep/stage.sh $name judge "$@" --cache "$J/cache" --item ep-judge --cap-usd "$cap" < /dev/null > /dev/null 2>&1
  tail -n 1 ~/ep/logs/$name.log | grep -q '^exit 0 ' || { echo "$(date -u +%FT%TZ) $name: stage did not exit 0" >> $log; exit 3; }
  echo "$(date -u +%FT%TZ) $name: exit 0" >> $log
}
echo "$(date -u +%FT%TZ) chain start" >> $log
for f in judge-r1 judge-gold-r1 gold-opus-r1 gold-gemini-r1; do [ ! -e $J/$f.json ] || { echo "$f.json exists" >> $log; exit 3; }; done
[ "$(sha256sum ~/work/head/lanes/eng-planted/readings/prompts/judge-named-r1.md | cut -d' ' -f1)" = 176c383093a49f073cf7a6b372e7b10a95a31968c6c3263ed45d3e7ab1bdb499 ] || { echo "revised prompt sha does not match" >> $log; exit 3; }
step 1 judge-gold-r1 --role judge --bundles "$G" --revision 1
cp -p $J/judge-r1.json $J/judge-gold-r1.json && rm $J/judge-r1.json
step 2 gold-opus-r1 --role gold-opus --bundles "$G" --revision 1
step 3 gold-gemini-r1 --role gold-gemini --bundles "$G" --revision 1
step 4 judge-main-r1 --role judge --revision 1
echo "$(date -u +%FT%TZ) chain done" >> $log
