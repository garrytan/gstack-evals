===== scripts/batch.sh
#!/usr/bin/env bash
# Operator controller for one Run T batch (GEVA-128), GEVA-93's batch.sh with its owner, runner path and TTL changed. From GEVA-89: It does what the tag's launch.sh does, with two changes
# the Captain approved: vm-run.sh runs detached on the VM (so this controller can die and be re-attached), and the
# ledger service is one shared process started separately on port 9911.
# Usage: batch.sh start <run dir> <prefix> <batch cap usd>   |   batch.sh attach <run dir>
set -uo pipefail
S=<ubicloud-skill>/scripts/ubi-runner.sh
RUNNER=$HOME/.capy/work/rt/runner-export/quick/laneb/runner
state_root=${XDG_STATE_HOME:-$HOME/.local/state}/ubi-runner
export UBI_OWNER=geva128 UBI_ALLOWED_STORAGE='320 600 640 1200'
mode=$1
run=$(cd "$2" && pwd)
meta=$run.vm
log() { echo "$(date -u +%FT%TZ) $*"; }
if [ "$mode" = start ]; then
  prefix=$3; cap=$4
  [ -e "$meta" ] && { log "refusing: $meta exists"; exit 2; }
  name=$(UBI_TTL_HOURS=${UBI_TTL_HOURS:-12} "$S" up -s standard-30 -S 600) || { log "up failed"; exit 3; }
  echo "$name" > "$meta"
  log "vm: $name"
  if ! { "$S" sync "$name" "$RUNNER" work/runner && "$S" sync "$name" "$run" work/run && "$S" ssh "$name" 'bash work/runner/vm-bootstrap.sh'; }; then
    log "setup failed before any trial; destroying $name"; "$S" down "$name"; exit 4
  fi
  { printf 'ANTHROPIC_API_KEY=%s\n' "${ANTHROPIC_API_KEY:-}"; printf 'OPENROUTER_API_KEY=%s\n' "${OPENROUTER_API_KEY:-}"; } | "$S" ssh "$name" 'umask 077; cat > ~/.laneb-keys'
  printf '#!/bin/bash\numask 022\ncd ~\nbash work/runner/vm-run.sh work/run "$1" "$2" > work/vm-run.log 2>&1\necho $? > work/vm-run.exit\n' | "$S" ssh "$name" 'cat > work/detach.sh'
else
  name=$(cat "$meta")
  prefix=""; cap=""
fi
. "$state_root/$name/env"
key="$state_root/$name/key"
SSHO=(-i "$key" -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o LogLevel=ERROR -o ConnectTimeout=30)
rsh() { timeout 120 ssh "${SSHO[@]}" "ubi@$IP" "$@"; }
tunnel() {
  while [ ! -e "$run.tunnel-stop" ]; do
    ssh "${SSHO[@]}" -N -o ServerAliveInterval=30 -o ServerAliveCountMax=4 -o ExitOnForwardFailure=yes -R 127.0.0.1:9911:127.0.0.1:9911 "ubi@$IP" &
    echo $! > "$run.tunnel-pid"; wait $!
    [ -e "$run.tunnel-stop" ] || { echo "$(date -u +%FT%TZ) tunnel dropped; reconnecting"; sleep 5; }
  done
}
rm -f "$run.tunnel-stop"
tunnel & tloop=$!
stop_tunnel() { touch "$run.tunnel-stop"; kill "$(cat "$run.tunnel-pid" 2>/dev/null)" 2>/dev/null; wait "$tloop" 2>/dev/null; }
trap stop_tunnel EXIT
for i in $(seq 1 30); do rsh 'curl -fsS 127.0.0.1:9911/status' >/dev/null 2>&1 && break; sleep 4; done
rsh 'curl -fsS 127.0.0.1:9911/status' >/dev/null 2>&1 || log "ledger not reachable through the tunnel"
if [ "$mode" = start ]; then
  rsh "setsid nohup bash work/detach.sh $prefix $cap < /dev/null > /dev/null 2>&1 & echo detached-started" || { log "detached start failed"; exit 5; }
fi
n=0
while true; do
  ex=$(rsh 'cat work/vm-run.exit 2>/dev/null' 2>/dev/null)
  [ -n "$ex" ] && break
  n=$((n + 1))
  if [ $((n % 10)) = 1 ]; then log "progress: $(rsh 'ls -d work/run/jobs/*/*/result.json 2>/dev/null | wc -l; tail -c 300 work/vm-run.log 2>/dev/null' 2>/dev/null | tr '\n' ' ' | cut -c1-400)"; fi
  sleep 60
done
log "vm-run exit $ex"
stop_tunnel; trap - EXIT
"$S" ssh "$name" 'rm -f ~/.laneb-keys; tail -5 work/vm-run.log' || true
mkdir -p "$run/pulled"
pulled=no
for attempt in 1 2 3; do
  if "$S" pull "$name" work/run/out "$run/pulled" && "$S" pull "$name" work/run/jobs "$run/pulled" \
    && [ "$("$S" ssh "$name" 'cd work/run && find out jobs -type f | wc -l')" = "$(cd "$run/pulled" && find out jobs -type f | wc -l)" ]; then
    pulled=yes; break
  fi
  log "pull attempt $attempt failed"; sleep 20
done
"$S" pull "$name" work/vm-run.log "$run/pulled" || true
if [ "$pulled" != yes ]; then log "WARNING: results not fully pulled; VM $name kept"; exit 7; fi
"$S" down "$name" && log "destroyed $name" && echo "$ex" > "$run.exit"
exit 0

===== scripts/refusal-stop.sh
#!/usr/bin/env bash
# GEVA-128: the Captain's ledger-refusal stop (GSTA-2 ruling on GST-823, 2026-11-01). Polls the batch VM's proxy totals
# every 60 s. When the proxy has latched a ledger refusal (cap_stop set; the tag's proxy then refuses every new session
# itself), this waits for the sessions already open to close, then sends SIGINT to Harbor so the batch starts nothing
# more. batch.sh then pulls and destroys the VM as usual. Usage: refusal-stop.sh <run dir>
set -uo pipefail
run=$(cd "$1" && pwd)
state_root=${XDG_STATE_HOME:-$HOME/.local/state}/ubi-runner
log() { echo "$(date -u +%FT%TZ) $*"; }
alive() { kill -0 "$(cat $run.ctl.pid 2>/dev/null)" 2>/dev/null; }
while [ ! -s $run.vm ]; do [ -e $run.exit ] || alive || exit 0; sleep 20; done
name=$(cat $run.vm)
while [ ! -f "$state_root/$name/env" ]; do sleep 10; done
. "$state_root/$name/env"; key="$state_root/$name/key"
rsh() { timeout 60 ssh -i "$key" -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o LogLevel=ERROR -o ConnectTimeout=20 "ubi@$IP" "$@"; }
log "watching $name for a ledger refusal"
while [ ! -e $run.exit ] && alive; do
  t=$(rsh 'curl -fsS 127.0.0.1:8788/totals' 2>/dev/null)
  cs=$(echo "$t" | jq -r '.cap_stop // empty' 2>/dev/null)
  if [ -n "$cs" ]; then
    log "ledger refusal latched: $cs; totals $t"
    echo "$cs" > $run.refusal
    while t=$(rsh 'curl -fsS 127.0.0.1:8788/totals' 2>/dev/null); do
      [ "$(echo "$t" | jq -r .open_sessions)" = 0 ] && break
      sleep 30
    done
    log "sessions open before the refusal have closed; totals $t; stopping Harbor"
    rsh "pkill -INT -f 'bin/[h]arbor run -y -c'"; log "SIGINT sent (pkill exit $?)"
    sleep 120
    if rsh "pgrep -f 'bin/[h]arbor run -y -c'" >/dev/null; then rsh "pkill -TERM -f 'bin/[h]arbor run -y -c'"; log "Harbor still up after 2 min; SIGTERM sent"; fi
    exit 0
  fi
  sleep 60
done
log "batch ended with no ledger refusal"

===== scripts/launch-ab2.sh
#!/usr/bin/env bash
# GEVA-128 Run T, phase 2 (second try, after launch-ab.sh stopped at its first check): start-rule.ts check --note wrote
# a passing note row and then crashed on fsync (ENOTSUP on the Capy Drive FUSE mount, start-rule.ts:94), so its exit
# status is not the verdict. rule() takes the verdict from the note row the check just wrote. particle-1 launches on
# its passing row of 21:42:04.945Z; Haiku gets a fresh check with particle-1 listed. refusal-stop.sh runs beside each.
set -uo pipefail
W=$HOME/.capy/work/rt
L=<captain-drive>/ledger/texttax-r1.jsonl
log() { echo "$(date -u +%FT%TZ) $*"; }
rule() {
  local before; before=$(wc -l < $L)
  bun quick/laneb/runner/start-rule.ts check --run texttax-r1 --ledger $L "$@" --note > $W/logs/rule-last.out 2>&1; local rc=$?
  head -1 $W/logs/rule-last.out
  local row; row=$(tail -n +$((before + 1)) $L | jq -c 'select(.kind == "note" and (.id | startswith("start-rule-")))' | tail -1)
  [ -n "$row" ] || { log "start rule wrote no note row (exit $rc)"; return 1; }
  [ "$(echo "$row" | jq -r .start_rule.pass)" = true ] || { log "start rule failed: $row"; return 1; }
  [ $rc = 0 ] || log "start-rule.ts exited $rc after writing a passing note row ($(grep -o 'ENOTSUP[^\n]*fsync' $W/logs/rule-last.out | head -1)); the rule passed"
}
start_batch() { # run dir name, ledger prefix, batch cap
  (setsid bash $W/scripts/batch.sh start $W/runs/$1 $2 $3 >> $W/runs/$1.ctl.log 2>&1 < /dev/null & echo $! > $W/runs/$1.ctl.pid)
  sleep 1
  (setsid bash $W/scripts/refusal-stop.sh $W/runs/$1 >> $W/runs/$1.refusal.log 2>&1 < /dev/null &)
  log "$1 controller started, pid $(cat $W/runs/$1.ctl.pid), batch cap \$$3"
}
curl -fsS 127.0.0.1:9911/status >/dev/null || { log "ledger service not up"; exit 2; }
cd /workspace/rt-runner
p=$(jq -c 'select(.id == "start-rule-2026-11-01T21:42:04.945Z")' $L)
[ "$(echo "$p" | jq -r '.start_rule.pass')" = true ] && [ "$(echo "$p" | jq -c '[.start_rule.batches[].model]')" = '["particle-1"]' ] || { log "particle-1 note row missing or not passing"; exit 1; }
log "particle-1 start rule (note row 21:42:04.945Z): bound \$$(echo "$p" | jq -r .start_rule.bound_usd) of \$75, pass"
start_batch t-particle tp- 61.64
rule --batch claude-haiku-5-5 --batch particle-1 || { log "Haiku not launched: Run T ends incomplete"; exit 1; }
start_batch t-haiku th- 8.81

#!/usr/bin/env bash
# GEVA-128: waits until one batch's controller has ended (exit file written, or controller gone), at most 5.75 h.
# Usage: watch.sh <batch>. It keeps this machine awake while it runs as a watched Capy operation.
W=$HOME/.capy/work/rt; b=$1; end=$((SECONDS + 20700))
while [ $SECONDS -lt $end ]; do
  [ -e $W/runs/$b.exit ] && { echo "$b ended: exit $(cat $W/runs/$b.exit)"; break; }
  kill -0 "$(cat $W/runs/$b.ctl.pid 2>/dev/null)" 2>/dev/null || { echo "$b controller is gone and $b.exit is missing"; break; }
  sleep 60
done
[ $SECONDS -ge $end ] && echo "$b still running at the watch limit"
grep -v progress: $W/runs/$b.ctl.log | tail -6; grep progress: $W/runs/$b.ctl.log | tail -1 | cut -c1-300

===== scripts/analysis.sh
#!/usr/bin/env bash
# GEVA-128 Run T, phase 4: the tag's analysis/discovery-tax.ts on the pulled job folders and proxy session files.
# Primary config: the proxy-sessions.jsonl files as written. Recovered config adds the 8 sessions whose settle failed
# (ECONNRESET through the tunnel), whose records the proxy never wrote, rebuilt from proxy-requests.jsonl by the jq
# below, so the retried-after-response sensitivity can see the first attempts (section 7). DeepSeek is not in the
# models list: dropped under R54 before its batch.
set -euo pipefail
W=$HOME/.capy/work/rt; A=$W/analysis; mkdir -p $A
: > $A/recovered-sessions.jsonl
for id in $(cat $W/ledger/open-ids.txt); do
  case $id in th-*) b=t-haiku;; tp-*) b=t-particle;; esac; s=${id#t?-}
  jq -s -c --arg s $s '[.[] | select(.session == $s and .event == "request")] | {id: $s, trial: .[0].trial, model: .[0].model, requests: length, ok_responses: (map(select(.status == 200)) | length), spent_micros: (map(.cost_micros // 0) | add), opened_at: .[0].ts, recovered_from: "proxy-requests.jsonl"}' $W/runs/$b/pulled/out/proxy-requests.jsonl >> $A/recovered-sessions.jsonl

done

grep -rl 'jq.*sk-ant' $B >/dev/null 2>&1 || true
cd $W/bundle && tar czf geva128-runt-data.tar.gz geva128-runt-data && sha256sum geva128-runt-data.tar.gz && ls -la geva128-runt-data.tar.gz
