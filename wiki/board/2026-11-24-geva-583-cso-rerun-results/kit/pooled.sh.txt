#!/usr/bin/env bash
# GEVA-583 pooled judge check (rerun prereg section 7), with the readout key in /dev/shm only.
set -uo pipefail
export PATH=$HOME/.local/bin:$HOME/.bun/bin:$PATH
L=$HOME/cso2-ledger; export GSTACK_EVALS_LEDGER=$L/ledger.sqlite GSTACK_EVALS_LEDGER_RECORD=$L/record
for k in ANTHROPIC_API_KEY OPENROUTER_API_KEY OPENAI_API_KEY; do export $k="$(cat $HOME/.keys/$k)"; done
K=/dev/shm/cso-readout.key
trap 'rm -f $K; test ! -e $K && echo "KEY_REMOVED $(date -u +%FT%TZ)"' EXIT
step() { echo "== $1 $(date -u +%FT%TZ)"; }
[ "$(sha256sum $K | cut -c1-8)" = 32a5870e ] || { echo "key digest mismatch"; exit 1; }
G=$HOME/work/judge/gstack-evals; cd $G; GE=$G/bin/gstack-evals
export CSO_PLAN=rerun
step head; git log --oneline -1; echo "dirty $(git status --porcelain --untracked-files=all | wc -l)"
P=$HOME/work/parent-check/judge-e/check-fallback-opus-particle.sealed
[ "$(sha256sum $P | cut -d' ' -f1)" = "$(cut -d' ' -f1 $P.sha256)" ] || { echo "parent check digest mismatch"; exit 1; }
step admit
GSTACK_EVALS_GH_BIN=$HOME/bin/ghshim sg docker -c "$GE rig admit --item cso2-judge --proof direct --prereg wiki/plans/2026-11-23-geva-583-cso-rerun-prereg/README.md --run cso2-judge --check-run $(cat $HOME/ci-run-judge) --repo garrytan/gstack-evals" || { echo "admit failed"; exit 1; }
step pooled-check
$GE cso judge --pooled-check --work $HOME/units --sessions $HOME/main/sessions.jsonl --parent-check $P --pilot-gate-file $HOME/judge-pilot/pilot-gate.json --judged $HOME/judge-main/judged.jsonl --key $K --out $HOME/judge-pooled --run cso2-judge --route fallback-opus-particle; echo "pooled exit $?"
$GE ledger status --program cso2 2>&1 | grep -E "ITEM|ENVELOPE" | cut -c1-200
echo POOLED_DONE
