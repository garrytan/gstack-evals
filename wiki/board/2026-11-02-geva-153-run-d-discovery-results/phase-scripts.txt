===== scripts/batch.sh
#!/usr/bin/env bash
# Run D (GEVA-153) batch controller, from GEVA-93 batch.sh: only owner, runner path, TTL and the pid file differ.
# Was: operator controller for one TB2 grid batch (GEVA-89). It does what the tag's launch.sh does, with two changes
# the Captain approved: vm-run.sh runs detached on the VM (so this controller can die and be re-attached), and the
# ledger service is one shared process started separately on port 9911.
# Usage: batch.sh start <run dir> <prefix> <batch cap usd>   |   batch.sh attach <run dir>
set -uo pipefail
S=<ubicloud-skill>/scripts/ubi-runner.sh
RUNNER=<work>/runner-export/quick/laneb/runner
state_root=${XDG_STATE_HOME:-$HOME/.local/state}/ubi-runner
export UBI_OWNER=geva153 UBI_ALLOWED_STORAGE='320 600 640 1200'
mode=$1
run=$(cd "$2" && pwd)
meta=$run.vm
echo $$ > "$run.ctl.pid"
log() { echo "$(date -u +%FT%TZ) $*"; }
if [ "$mode" = start ]; then
  prefix=$3; cap=$4
  [ -e "$meta" ] && { log "refusing: $meta exists"; exit 2; }
  name=$(UBI_TTL_HOURS=${UBI_TTL_HOURS:-12} "$S" up -s standard-30 -S 600) || { log "up failed"; exit 3; }
  echo "$name" > "$meta"
  log "vm: $name"
  if ! { "$S" sync "$name" "$RUNNER" work/runner && "$S" sync "$name" "$run" work/run && "$S" ssh "$name" 'bash work/runner/vm-bootstrap.sh'; }; then
    log "setup failed before any trial; destroying $name"; "$S" down "$name"; exit 4
  fi
  { printf 'ANTHROPIC_API_KEY=%s\n' "${ANTHROPIC_API_KEY:-}"; printf 'OPENROUTER_API_KEY=%s\n' "${OPENROUTER_API_KEY:-}"; } | "$S" ssh "$name" 'umask 077; cat > ~/.laneb-keys'
  printf '#!/bin/bash\numask 022\ncd ~\nbash work/runner/vm-run.sh work/run "$1" "$2" > work/vm-run.log 2>&1\necho $? > work/vm-run.exit\n' | "$S" ssh "$name" 'cat > work/detach.sh'
else
  name=$(cat "$meta")
  prefix=""; cap=""
fi
. "$state_root/$name/env"
key="$state_root/$name/key"
SSHO=(-i "$key" -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o LogLevel=ERROR -o ConnectTimeout=30)
rsh() { timeout 120 ssh "${SSHO[@]}" "ubi@$IP" "$@"; }
tunnel() {
  while [ ! -e "$run.tunnel-stop" ]; do
    ssh "${SSHO[@]}" -N -o ServerAliveInterval=30 -o ServerAliveCountMax=4 -o ExitOnForwardFailure=yes -R 127.0.0.1:9911:127.0.0.1:9911 "ubi@$IP" &
    echo $! > "$run.tunnel-pid"; wait $!
    [ -e "$run.tunnel-stop" ] || { echo "$(date -u +%FT%TZ) tunnel dropped; reconnecting"; sleep 5; }
  done
}
rm -f "$run.tunnel-stop"
tunnel & tloop=$!
stop_tunnel() { touch "$run.tunnel-stop"; kill "$(cat "$run.tunnel-pid" 2>/dev/null)" 2>/dev/null; wait "$tloop" 2>/dev/null; }
trap stop_tunnel EXIT
for i in $(seq 1 30); do rsh 'curl -fsS 127.0.0.1:9911/status' >/dev/null 2>&1 && break; sleep 4; done
rsh 'curl -fsS 127.0.0.1:9911/status' >/dev/null 2>&1 || log "ledger not reachable through the tunnel"
if [ "$mode" = start ]; then
  rsh "setsid nohup bash work/detach.sh $prefix $cap < /dev/null > /dev/null 2>&1 & echo detached-started" || { log "detached start failed"; exit 5; }
fi
n=0
while true; do
  ex=$(rsh 'cat work/vm-run.exit 2>/dev/null' 2>/dev/null)
  [ -n "$ex" ] && break
  n=$((n + 1))
  if [ $((n % 10)) = 1 ]; then log "progress: $(rsh 'ls -d work/run/jobs/*/*/result.json 2>/dev/null | wc -l; tail -c 300 work/vm-run.log 2>/dev/null' 2>/dev/null | tr '\n' ' ' | cut -c1-400)"; fi
  sleep 60
done
log "vm-run exit $ex"
stop_tunnel; trap - EXIT
"$S" ssh "$name" 'rm -f ~/.laneb-keys; tail -5 work/vm-run.log' || true
mkdir -p "$run/pulled"
pulled=no
for attempt in 1 2 3; do
  if "$S" pull "$name" work/run/out "$run/pulled" && "$S" pull "$name" work/run/jobs "$run/pulled" \
    && [ "$("$S" ssh "$name" 'cd work/run && find out jobs -type f | wc -l')" = "$(cd "$run/pulled" && find out jobs -type f | wc -l)" ]; then
    pulled=yes; break
  fi
  log "pull attempt $attempt failed"; sleep 20
done
"$S" pull "$name" work/vm-run.log "$run/pulled" || true
if [ "$pulled" != yes ]; then log "WARNING: results not fully pulled; VM $name kept"; exit 7; fi
"$S" down "$name" && log "destroyed $name" && echo "$ex" > "$run.exit"
exit 0

===== scripts/watch.sh
#!/usr/bin/env bash
# Exits when every named batch has ended (its .exit file exists) or its controller is gone. Keeps the machine awake while it runs.
W=<work>
while true; do
  pending=0
  for b in "$@"; do
    r=$W/runs/$b
    [ -e $r.exit ] && continue
    pid=$(cat $r.ctl.pid 2>/dev/null)
    if [ -n "$pid" ] && kill -0 $pid 2>/dev/null; then pending=1; continue; fi
    echo "$(date -u +%FT%TZ) controller of $b is gone and $b.exit is missing"; exit 3
  done
  [ $pending = 0 ] && { echo "$(date -u +%FT%TZ) every batch ended: $*"; for b in "$@"; do echo "$b exit $(cat $W/runs/$b.exit)"; done; exit 0; }
  sleep 60
done

===== scripts/ledger-sup.sh
#!/usr/bin/env bash
# The one discovery-r1 ledger service on 127.0.0.1:9911 (GEVA-153, Run D), from GEVA-93's ledger-sup.sh. It starts the
# service only when nothing listens on 9911, so it never runs a second service on the file, and ends when ledger.stop exists.
W=<work>
L=<captain-drive>/ledger/discovery-r1.jsonl
cd $W/runner-export/quick/laneb/runner || exit 1
while [ ! -e $W/ledger.stop ]; do
  if ! ss -ltnH 'sport = :9911' | grep -q '127.0.0.1:9911'; then
    echo "$(date -u +%FT%TZ) ledger-sup: starting the ledger service"
    bun ledger.ts serve --path $L --item discovery-r1 --port 9911 --prereg 9820b149c3c5b9f12b82edc796871464b3c157c2 &
    wait $!
    echo "$(date -u +%FT%TZ) ledger-sup: service exited $?"
  fi
  sleep 3
done

# GEVA-153 Run D chart from the analysis output: left, trials that opened a listed skill per preset cell (counts, no
# interval: the analysis prints per-cell opens without one); right, pass rate per preset cell with its 95% interval and cost per trial.
import re, sys
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
src, out = sys.argv[1], sys.argv[2]
rows, opens, model = [], {}, None
for l in open(src):
    h = re.match(r'## (\S+)$', l.strip())
    if h: model = h.group(1)
    m = re.match(r'(\S+) (none|gstack|gstack-rewritten|placebo)@claude-code-preset: .*?pass ([\d.]+)% \[([\d.]+)%, ([\d.]+)%\].*?cost per trial \$([\d.]+)', l)

#!/usr/bin/env bash
# GEVA-153 Run D data bundle for the Captain drive: pulled job folders and proxy files per batch, the route check,
# the ledger, analysis inputs and the run record, with TB2 fixture secrets redacted as GEVA-93 and GEVA-128 did.
set -euo pipefail
W=<work>
C=<captain-drive>
B=$W/bundle/geva153-rund-data
rm -rf $W/bundle && mkdir -p $B/runs $B/analysis
for b in particle-1 claude-opus-5-5-b claude-opus-5-5-r2-b gpt-6.1-sol; do mkdir -p $B/runs/$b && cp -r $W/runs/$b/pulled $B/runs/$b/ && cp $W/runs/$b.ctl.log $B/runs/; done
mkdir -p $B/runs/attempt1 && cp $W/runs/attempt1/*.ctl.log $B/runs/attempt1/
cp -r $W/route-d $B/route-d && cp $W/route-d.log $W/prep.log $W/RECORD.md $W/sr-*-gate.json $B/
cp $C/ledger/discovery-r1.jsonl $B/ledger-discovery-r1.jsonl
cp $W/analysis/lost-session.jsonl $W/analysis/config-primary.json $W/analysis/config-recovered.json $W/analysis/output-primary.txt $W/analysis/output-recovered.txt $B/analysis/
grep -rlZ -e '<home>/' $B | xargs -0 -r sed -i -e "s#<captain-drive>#<captain-drive>#g" -e "s#<work>#<work>#g" -e "s#<ubicloud-skill>#<ubicloud-skill>#g"
PAT='sk-ant-[A-Za-z0-9_-]{20,}|sk-or-v1-[a-f0-9]{20,}|sk-proj-[A-Za-z0-9_-]{20,}|BEGIN (RSA |EC |OPENSSH |ENCRYPTED )?PRIVATE KEY|ghp_[A-Za-z0-9]{30,}|AKIA[0-9A-Z]{16}'
n=0
while IFS= read -r -d '' f; do
  perl -0pi -e 's/-----BEGIN ([A-Z ]*)PRIVATE KEY-----.*?-----END \1PRIVATE KEY-----/<redacted-fixture-private-key>/gs; s/-----BEGIN ([A-Z ]*)PRIVATE KEY-----(?:[A-Za-z0-9+\/=\s]|\\n)*/<redacted-fixture-private-key>/g; s/(sk-ant-[A-Za-z0-9_-]{20,}|sk-or-v1-[a-f0-9]{20,}|sk-proj-[A-Za-z0-9_-]{20,}|ghp_[A-Za-z0-9]{30,}|AKIA[0-9A-Z]{16})/<redacted-fixture-secret>/g' "$f"; n=$((n+1))
done < <(grep -rlEZ "$PAT" $B || true)
echo "redacted files: $n; remaining pattern matches: $(grep -rlE "$PAT" $B | wc -l)"
python3 - "$B" <<'PYEOF'
import os, sys
root = sys.argv[1]
for k in ['ANTHROPIC_API_KEY', 'OPENAI_API_KEY', 'OPENROUTER_API_KEY', 'UBICLOUD_API_KEY']:
    v = os.environ.get(k, '').strip().encode()
    if not v: print(k, 'not set, not scanned'); continue
    n = 0
    for d, _, fs in os.walk(root):
        for f in fs:
            try:
                if v in open(os.path.join(d, f), 'rb').read(): n += 1
            except OSError: pass
    print(k, 'live-value matches:', n)
PYEOF
cd $W/bundle && tar -czf geva153-rund-data.tar.gz geva153-rund-data && sha256sum geva153-rund-data.tar.gz && ls -l geva153-rund-data.tar.gz
