#!/usr/bin/env python3
"""Read each guard's real hook command from its source of truth and write eps.json."""
import json, os, sys
home = os.path.expanduser("~")
g = os.path.join(home, ".claude/skills/gstack")

def frontmatter(path):
    txt = open(path).read()
    assert txt.startswith("---\n")
    fm = txt[4:txt.index("\n---\n", 4)]
    import subprocess
    p = subprocess.run(["bun", "-e", "console.log(JSON.stringify(Bun.YAML.parse(await Bun.stdin.text())))"],
                       input=fm, capture_output=True, text=True, check=True)
    return json.loads(p.stdout)

spec = {}
for skill in ("careful", "guard"):
    fm = frontmatter(os.path.join(g, skill, "SKILL.md"))
    m = {}
    for ent in fm["hooks"]["PreToolUse"]:
        if ent["matcher"] in ("Bash", "PowerShell"):
            cmds = [h["command"] for h in ent["hooks"] if h.get("type") == "command"]
            assert len(cmds) == 1, (skill, ent)
            m[ent["matcher"]] = cmds[0]
    spec[skill] = m
settings = sys.argv[1] if len(sys.argv) > 1 else os.path.join(home, ".claude/settings.json")
st = json.load(open(settings))
dg = []
for ent in st["hooks"]["PreToolUse"]:
    for h in ent["hooks"]:
        if "destructive-guard-hook" in h.get("command", ""):
            dg.append((ent["matcher"], h["command"], h.get("timeout")))
assert len(dg) == 1, dg
matcher, cmd, timeout = dg[0]
spec["default"] = {"Bash": cmd, "PowerShell": cmd, "_matcher": matcher, "_timeout": timeout}
spec["_e1_e2_identical"] = spec["careful"] == spec["guard"]
json.dump(spec, sys.stdout, indent=1)
print()
