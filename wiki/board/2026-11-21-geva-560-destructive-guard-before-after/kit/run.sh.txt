#!/usr/bin/env bash
# GEVA-560 re-measure of gstack ab6a38b7 (old = 17303862) on a Ubicloud VM. Feeds commands to hooks as stdin; never runs them.
set -euo pipefail
B="$HOME/b"
cd "$B"
exec >>"$B/run.log" 2>&1
echo "start $(date -u +%FT%TZ) nproc=$(nproc)"
sudo DEBIAN_FRONTEND=noninteractive apt-get update -qq >/dev/null
sudo DEBIAN_FRONTEND=noninteractive apt-get install -y -qq unzip zstd >/dev/null
if ! command -v bun >/dev/null; then
  curl -fsSL https://bun.sh/install | bash -s "bun-v1.4.4" >/dev/null
fi
export PATH="$HOME/.bun/bin:$PATH"
bun --version
mkdir -p "$HOME/gstack" && tar -xf gstack.tar -C "$HOME/gstack"
mkdir -p "$HOME/.claude/skills" && ln -sfn "$HOME/gstack" "$HOME/.claude/skills/gstack"
(cd "$HOME/gstack" && sha256sum -c "$B/matchers.sha256")
mkdir -p "$B/old/gstack" "$B/homeold/.claude/skills" && tar -xf gstack-old.tar -C "$B/old/gstack"
ln -sfn "$B/old/gstack" "$B/homeold/.claude/skills/gstack"
(cd "$B/old/gstack" && sha256sum -c "$B/matchers-old.sha256")
mkdir -p fx && tar -xf fixtures.tar -C fx
for r in testbed app work; do
  sudo mkdir -p "/$r" && sudo chown "$(id -u):$(id -g)" "/$r" && cp -a fx/repo/. "/$r/"
  echo "$r $(git -C /$r branch --show-current) $(git -C /$r symbolic-ref refs/remotes/origin/HEAD) dirty=$(git -C /$r status --porcelain | wc -l)"
done
printf '{"clean":"%s","main":"%s","testbed":"/testbed","app":"/app","work":"/work"}\n' "$B/fx/repo" "$B/fx/main" > fixtures.json
sed "s#<home>/#$HOME/#g" settings.local.json > settings.vm.json
python3 specgen.py settings.vm.json > eps.json
sed "s#<home>/#$B/homeold/#g" settings.local.json > settings.old.json
HOME="$B/homeold" python3 specgen.py settings.old.json > eps-old.json
cat eps.json eps-old.json
python3 -c "import json;assert json.load(open('eps.json'))['_e1_e2_identical']"
zstd -dqf items.jsonl.zst -o items.jsonl
wc -l items.jsonl
mkdir -p state
W=$(nproc)
echo "phase1 $(date -u +%FT%TZ)"
python3 runner.py --items items.jsonl --spec eps.json --fixtures fixtures.json --eps E1,E3 --out r1.jsonl --state "$B/state" --workers "$W"
echo "phase2 $(date -u +%FT%TZ)"
python3 - <<'PY'
import json, random, re
hits = set()
for l in open('r1.jsonl'):
    r = json.loads(l)
    if r['dec'] != 'none':
        hits.add(r['id'])
items = [json.loads(l) for l in open('items.jsonl')]
ids = [it['id'] for it in items]
random.seed(560)
samp = set(random.sample(ids, 5000))
open('ids-e2.txt', 'w').write('\n'.join(sorted(samp | hits)) + '\n')
open('ids-e45.txt', 'w').write('\n'.join(sorted(hits | set(random.sample(sorted(samp), 2000)))) + '\n')
random.seed(5601)
open('ids-lat.txt', 'w').write('\n'.join(random.sample(ids, 10000)) + '\n')
gl = re.compile(r'\b(reset|checkout|switch|restore|clean)\b')
nm = nd = 0
with open('items-main.jsonl', 'w') as fm, open('items-dirty.jsonl', 'w') as fd:
    for it in items:
        c = it['tool_input'].get('command') or ''
        if 'push' in c and 'git' in c:
            fm.write(json.dumps(it) + '\n'); nm += 1
        if 'git' in c and gl.search(c):
            fd.write(json.dumps(it) + '\n'); nd += 1
print('hits', len(hits), 'e2', len(samp | hits), 'main', nm, 'dirty', nd)
PY
python3 runner.py --items items.jsonl --spec eps.json --fixtures fixtures.json --eps E2 --only ids-e2.txt --out r2.jsonl --state "$B/state" --workers "$W"
python3 runner.py --items items.jsonl --spec eps.json --fixtures fixtures.json --eps E4,E5,E5u --only ids-e45.txt --out r45.jsonl --state "$B/state" --workers "$W"
echo "phase-lat $(date -u +%FT%TZ)"
for round in 1 2; do
  HOME="$B/homeold" python3 runner.py --items items.jsonl --spec eps-old.json --fixtures fixtures.json --eps E1 --only ids-lat.txt --out rlat-old-$round.jsonl --state "$B/state" --workers "$W"
  python3 runner.py --items items.jsonl --spec eps.json --fixtures fixtures.json --eps E1 --only ids-lat.txt --out rlat-new-$round.jsonl --state "$B/state" --workers "$W"
done
echo "phase-main $(date -u +%FT%TZ)"
for r in testbed app work; do git -C "/$r" checkout -q main; echo "$r $(git -C /$r branch --show-current)"; done
python3 runner.py --items items-main.jsonl --spec eps.json --fixtures fixtures.json --eps E1 --out rmain.jsonl --state "$B/state" --workers "$W"
for r in testbed app work; do git -C "/$r" checkout -q agent-work; done
echo "phase-dirty $(date -u +%FT%TZ)"
for r in testbed app work; do echo dirty >> "/$r/a.txt"; echo u > "/$r/u.txt"; echo "$r $(git -C /$r status --porcelain | tr '\n' ' ')"; done
python3 runner.py --items items-dirty.jsonl --spec eps.json --fixtures fixtures.json --eps E1 --out rdirty.jsonl --state "$B/state" --workers "$W"
rm -f items.jsonl items-main.jsonl items-dirty.jsonl
echo "done $(date -u +%FT%TZ)"
touch DONE
