#!/usr/bin/env python3
"""Feed shell commands to gstack's command guards as PreToolUse hook payloads.

Never executes a command: each one goes to the hook as JSON on stdin, the way
Claude Code sends it, and the hook's stdout decision is recorded.
"""
import argparse, json, os, re, subprocess, sys, threading, time, uuid
from concurrent.futures import ThreadPoolExecutor, FIRST_COMPLETED, wait

EP_ENV = {
    "E1": {"CLAUDE_CODE_ENTRYPOINT": "cli"},
    "E2": {"CLAUDE_CODE_ENTRYPOINT": "cli"},
    "E3": {"CLAUDE_CODE_ENTRYPOINT": "cli"},
    "E4": {"CLAUDE_CODE_ENTRYPOINT": "sdk-cli"},
    "E5": {"CLAUDE_CODE_ENTRYPOINT": "sdk-py"},
    "E5u": {"CLAUDE_CODE_ENTRYPOINT": "sdk-py", "GSTACK_SESSION_KIND": "unattended"},
}
EP_SPEC = {"E1": "careful", "E2": "guard", "E3": "default", "E4": "default", "E5": "default", "E5u": "default"}


def pattern_map(careful_sh):
    src = open(careful_sh).read()
    m = {}
    for warn, pat in re.findall(r'WARN="([^"]+)"\s*\n\s*PATTERN="(\w+)"', src):
        m[warn] = pat
    for pat, warn in re.findall(r'^\s*(\w+)\) WARN="([^"]+)" ;;', src, re.M):
        m[warn] = pat
    return m


def classify(stdout, pmap):
    out = stdout.strip()
    if not out:
        return "none", None, None
    try:
        d = json.loads(out.splitlines()[-1])
    except Exception:
        return "unparsed", None, out[:200]
    o = d.get("hookSpecificOutput") or {}
    if o.get("permissionDecision"):
        dec, reason = o["permissionDecision"], o.get("permissionDecisionReason", "")
    elif d.get("systemMessage"):
        dec, reason = "warn", d["systemMessage"]
    elif d == {}:
        return "none", None, None
    else:
        return "other", None, out[:200]
    pat = None
    mm = re.search(r"\(pattern: (\w+)\)", reason)
    if mm:
        pat = mm.group(1)
    elif "[HIGH] Recursive delete of /" in reason:
        pat = "high_rm_root"
    elif "[HIGH] Force-push to the default branch" in reason:
        pat = "high_force_push_default"
    elif "[HIGH] Writing to, formatting or wiping a whole disk" in reason:
        pat = "high_disk"
    elif "The destructive-command check failed" in reason:
        pat = "check_failed"
    elif "Shell obfuscation detected" in reason:
        pat = "obfuscation"
    elif "Could not parse the tool payload" in reason:
        pat = "parse_fail"
    elif "Project rule matched" in reason:
        pat = "project_rule"
    else:
        for warn, p in pmap.items():
            if warn in reason:
                pat = p
                break
    return dec, pat, reason


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--items", required=True)
    ap.add_argument("--spec", required=True)
    ap.add_argument("--fixtures", required=True, help="json map fixture name -> dir")
    ap.add_argument("--eps", required=True)
    ap.add_argument("--out", required=True)
    ap.add_argument("--state", required=True)
    ap.add_argument("--workers", type=int, default=4)
    ap.add_argument("--default-fixture", default="clean")
    ap.add_argument("--only", help="file of ids to run (one per line)")
    ap.add_argument("--keep-reason", action="store_true")
    a = ap.parse_args()

    spec = json.load(open(a.spec))
    fixtures = json.load(open(a.fixtures))
    home = os.path.expanduser("~")
    pmap = pattern_map(os.path.join(home, ".claude/skills/gstack/careful/bin/check-careful.sh"))
    eps = a.eps.split(",")
    only = set(l.strip() for l in open(a.only)) if a.only else None

    base_env = {
        "HOME": home,
        "PATH": os.environ["PATH"],
        "LANG": "C.UTF-8",
        "USER": os.environ.get("USER", "user"),
        "LOGNAME": os.environ.get("USER", "user"),
        "SHELL": "/bin/bash",
        "GSTACK_STATE_ROOT": a.state,
        "GSTACK_HOME": a.state,
    }
    done = set()
    if os.path.exists(a.out):
        for l in open(a.out):
            try:
                r = json.loads(l)
                done.add((r["id"], r["ep"]))
            except Exception:
                pass
    jobs = []
    for l in open(a.items):
        it = json.loads(l)
        if only is not None and it["id"] not in only:
            continue
        for ep in eps:
            if (it["id"], ep) not in done:
                jobs.append((it, ep))
    print(f"jobs {len(jobs)} (skipped {len(done)} done)", file=sys.stderr, flush=True)

    lock = threading.Lock()
    fout = open(a.out, "a")
    tpath = os.path.join(a.state, "transcript.jsonl")

    def run(it, ep):
        tool = it.get("tool", "Bash")
        matcher = "PowerShell" if tool == "PowerShell" else "Bash"
        cmd = spec[EP_SPEC[ep]][matcher]
        fx = it.get("fixture") or a.default_fixture
        payload = {
            "session_id": str(uuid.uuid4()),
            "transcript_path": tpath,
            "cwd": fixtures[fx],
            "permission_mode": "default",
            "hook_event_name": "PreToolUse",
            "tool_name": tool,
            "tool_input": it["tool_input"],
            "tool_use_id": "toolu_" + uuid.uuid4().hex[:24],
        }
        env = dict(base_env)
        env.update(EP_ENV[ep])
        t0 = time.monotonic()
        to = False
        try:
            p = subprocess.run(["/bin/sh", "-c", cmd], input=json.dumps(payload), cwd=fixtures[fx], env=env,
                               capture_output=True, text=True, timeout=10)
            rc, so, se = p.returncode, p.stdout, p.stderr
        except subprocess.TimeoutExpired:
            rc, so, se, to = None, "", "", True
        ms = int((time.monotonic() - t0) * 1000)
        dec, pat, reason = ("timeout", None, None) if to else classify(so, pmap)
        rec = {"id": it["id"], "ep": ep, "fx": fx, "rc": rc, "dec": dec, "pat": pat, "ms": ms}
        if to:
            rec["to"] = True
        if se.strip():
            rec["err"] = se.strip()[:160]
        if reason and (a.keep_reason or pat is None):
            rec["reason"] = reason
        with lock:
            fout.write(json.dumps(rec) + "\n")
            fout.flush()

    n = 0
    t0 = time.time()
    with ThreadPoolExecutor(a.workers) as ex:
        pending = set()
        for job in jobs:
            pending.add(ex.submit(run, *job))
            if len(pending) >= a.workers * 4:
                fin, pending = wait(pending, return_when=FIRST_COMPLETED)
                for f in fin:
                    f.result()
                    n += 1
                    if n % 5000 == 0:
                        print(f"{n} done {time.time()-t0:.0f}s", file=sys.stderr, flush=True)
        for f in pending:
            f.result()
    fout.close()
    print(f"finished {len(jobs)} in {time.time()-t0:.0f}s", file=sys.stderr, flush=True)


if __name__ == "__main__":
    main()
