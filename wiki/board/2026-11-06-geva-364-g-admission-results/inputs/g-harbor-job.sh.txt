#!/usr/bin/env bash
# g-harbor-job.sh: G admission's Harbor job (GEVA-336), in the shape of quick/laneb/runner/smoke.sh. One Harbor trial per
# (arm, task, trial n), real_agent:RealAgent as the agent in U4 mode, on the run commit and the envelope host.
#
#   g-harbor-job.sh prepare                        sets each bound run copy's network_mode lines to "public" (below)
#   g-harbor-job.sh order                          the frozen session order (G prereg section 5), one "arm label n" a line
#   g-harbor-job.sh one <arm> <label> <n> [repair] one trial; a repair names the gap rerun it is (real_agent's rerun)
#   g-harbor-job.sh batch <workers>                every trial in order, <workers> at a time, skipping trials already run
#
# Every path comes from the run env file named by G_RUN_ENV (no keys there): GE (the checkout), VENV (Harbor f3a5a92 and
# claude-agent-sdk 0.2.174), ARMS (the arm tars), TASKS, COPIES, BINDS and LABELS (in /dev/shm, from outcome images
# bind), JOBS (Harbor's output, in /dev/shm), MANIFEST, DRIVER_OUT (outcome serve's stdout: control= and stage= lines),
# SIDECAR (the sandbox sidecar's control URL) and HOST (the trial-network address the proxies and sidecar listen on).
# No provider key reaches Harbor: the session's bearer comes from its U4 slot.
#
# Network: the run copy's task.toml asks Harbor for its egress-control sidecar (environment "no-network", agent
# "allowlist"), which puts the trial container in the sidecar's network namespace with no network of its own, and
# real_agent refuses that (E_REAL_SANDBOX_SETUP: the container must sit on exactly one network). real_agent writes its
# own IP-and-port egress table (lanes/real-outcome/agent/egress.py) and probes it before the sidecar prep and before the
# session opens, and egress.py states that Harbor runs the trial with network_mode "public". So prepare rewrites those two
# lines to "public", as smoke.sh's NETWORK=public does, keeping the bound file beside the copy, and one refuses a run copy
# whose task.toml differs from the bound one in anything else. Harbor's own verifier is off: the lane's verifier is real
# score trial, in the pinned image.
set -euo pipefail
. "$G_RUN_ENV"
TOLD_SHA=e2d70ed830d97e0881eb6a071efbe6d3606cd0c48636f275154e3b0b186a0853
NONE_SHA=84ff92691f909a05b224e1c56abb4864f01b4f8e3c854e4bb4c7baf1d3f6d652
MANIFEST_SHA=7b1ab47b539ee573ea353be777a7b6787e7168d94742b5e43477305e32f4ebe9
REGISTRY=$GE/lanes/outcome-skills/arms/registry.json
PUBLIC_LINE='network_mode = "public"'

order() {
  python3 -I -c '
import random, sys
labels = sorted(l.strip() for l in open(sys.argv[1]) if l.strip())
assert len(labels) == 12, len(labels)
for trial in (1, 2):
    pairs = sorted((arm, label) for arm in ("none", "plan-eng-review:told", "autoplan:told") for label in labels)
    random.Random(20261104 + trial).shuffle(pairs)
    for arm, label in pairs:
        print(arm, label, trial)
' "$LABELS"
}

public() { sed -E 's/^network_mode = "(allowlist|no-network)"$/network_mode = "public"/'; }

name() {
  local short
  case "$1" in none) short=none ;; plan-eng-review:told) short=pe ;; autoplan:told) short=ap ;; *) echo "E_USAGE unknown arm $1" >&2; exit 2 ;; esac
  echo "g-$short-$2-t$3$([ -n "$4" ] && echo -rerun)"
}

one() {
  local arm=$1 label=$2 n=$3 repair=$4 item tar sha
  case "$arm" in
    none) item=g-none tar=$ARMS/none.tar sha=$NONE_SHA ;;
    plan-eng-review:told) item=g-pe tar=$ARMS/told.tar sha=$TOLD_SHA ;;
    autoplan:told) item=g-ap tar=$ARMS/told.tar sha=$TOLD_SHA ;;
    *) echo "E_USAGE unknown arm $arm" >&2; exit 2 ;;
  esac
  grep -qx "$label" "$LABELS" || { echo "E_USAGE $label is not a final T-G label" >&2; exit 2; }
  case "$n" in 1|2) ;; *) echo "E_USAGE trial must be 1 or 2" >&2; exit 2 ;; esac
  local toml=$COPIES/$label/task.toml
  [ -f "$COPIES/$label.task.toml.bound" ] && cmp -s "$toml" <(public < "$COPIES/$label.task.toml.bound") && [ "$(grep -cxF "$PUBLIC_LINE" "$toml")" = 2 ] || { echo "E_USAGE $toml is not the bound task.toml with public network modes; run prepare" >&2; exit 2; }
  local control proxy image wall agent_sec mult job
  control=$(sed -n 's/^control=\(http:\/\/127\.0\.0\.1:[0-9]*\)$/\1/p' "$DRIVER_OUT" | tail -1)
  proxy=$(sed -n "s/^stage=$item arm=[^ ]* lease=[^ ]* reserved=[0-9]* proxy=\([0-9.]*:[0-9]*\)$/\1/p" "$DRIVER_OUT" | tail -1)
  image=$(awk -F'\t' -v l="$label" '$1 == l { print $2 }' "$BINDS")
  [ -n "$control" ] && [ -n "$proxy" ] && [ "${proxy%%:*}" = "$HOST" ] && [ -n "$image" ] || { echo "E_USAGE no control URL, no $item proxy on $HOST or no bound image for $label" >&2; exit 2; }
  wall=$(jq -er --arg a "$arm" '.arms[$a].limits.wall_limit_sec' "$REGISTRY")
  agent_sec=$(jq -er '.limits.agent_sec' "$TASKS/$label/task.json")
  mult=$(python3 -I -c 'import math, sys; print(max(1.0, math.ceil((float(sys.argv[1]) + 900) / float(sys.argv[2]) * 100) / 100))' "$wall" "$agent_sec")
  job=$(name "$arm" "$label" "$n" "$repair")
  [ ! -e "$JOBS/$job" ] || { echo "E_USAGE $JOBS/$job exists; a trial runs once, and a gap reruns once under its own name with a repair" >&2; exit 2; }
  mkdir -p "$JOBS"
  echo "$(date -u +%FT%TZ) start $job proxy=$proxy image=$image agent_timeout_multiplier=$mult" >> "$JOBS/launch.log"
  set +e
  env -u ANTHROPIC_BASE_URL PYTHONPATH="$GE/lanes/real-outcome/agent:$GE/quick/laneb/runner/agent" PYTHONDONTWRITEBYTECODE=1 ANTHROPIC_API_KEY=placeholder-not-a-key \
    "$VENV/bin/harbor" run -y -p "$COPIES/$label" -a real_agent:RealAgent -m anthropic/claude-opus-5-5 --base-url "http://$proxy" \
    --ak skip_install=true --ak ledger=u4 --ak control_url="$control" --ak sidecar_url="$SIDECAR" \
    --ak arm="$arm" --ak arms_path="$REGISTRY" --ak arm_tar="$tar" --ak arm_sha256="$sha" --ak upstream_model=claude-opus-5-5 \
    --ak task_json="$TASKS/$label/task.json" --ak task_label="$label" --ak trial_n="$n" $([ -n "$repair" ] && printf -- '--ak rerun=%s' "$repair") \
    --ak image_id="$image" --ak manifest="$MANIFEST" --ak manifest_sha256="$MANIFEST_SHA" \
    --disable-verification --agent-timeout-multiplier "$mult" --max-retries 0 \
    -e docker --extra-docker-compose "$GE/quick/laneb/runner/agent-caps.compose.yaml" -o "$JOBS" --job-name "$job" > "$JOBS/$job.log" 2>&1
  local c=$?
  set -e
  echo "$(date -u +%FT%TZ) end $job exit $c" >> "$JOBS/launch.log"
  return 0
}

case "${1:-}" in
  prepare)
    for l in $(cat "$LABELS"); do
      b=$COPIES/$l.task.toml.bound
      [ -f "$b" ] || cp "$COPIES/$l/task.toml" "$b"
      public < "$b" > "$COPIES/$l/task.toml"
      echo "$l $(sha256sum < "$b" | cut -c1-12) -> $(sha256sum < "$COPIES/$l/task.toml" | cut -c1-12)"
    done
    ;;
  order) order ;;
  one) [ $# -ge 4 ] || { echo "E_USAGE g-harbor-job.sh one <arm> <label> <n> [repair]" >&2; exit 2; }; one "$2" "$3" "$4" "${5:-}" ;;
  batch)
    w=${2:?E_USAGE g-harbor-job.sh batch <workers>}
    order | while read -r arm label n; do [ -e "$JOBS/$(name "$arm" "$label" "$n" "")" ] || echo "$arm $label $n"; done | xargs -r -P "$w" -L 1 bash "$0" one
    ;;
  *) echo "E_USAGE g-harbor-job.sh prepare | order | one <arm> <label> <n> [repair] | batch <workers>" >&2; exit 2 ;;
esac
