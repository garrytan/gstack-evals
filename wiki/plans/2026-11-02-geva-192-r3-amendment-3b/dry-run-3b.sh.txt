#!/usr/bin/env bash
# R3 amendment 3b dry run (GEVA-192), at the 3b pin (GST-981's merge commit, for Sol and Opus attempt 4). It reruns
# amendment 3's dry-run-a3.sh.txt with the pin's export, both models on temporary ledgers, and prints its whole output.
# Then it drives each rule 3b adds. It shows the diff from 73042c5 to the pin and which pinned files changed, and that
# UBI_LOCATION reaches the runner with no code change. It drives capacity-probe.sh.txt against a stub runner (no VM, no
# network, no key) through a failing window, a pass after failures, a VM it can't confirm gone and a probe ledger that
# refuses, and the run step's 30-minute and location guard. It drives a no-start through the export's own screen.ts run and
# launch.sh with the stub runner standing in for a VM that never reaches running: Opus on a temporary a4 ledger, then a5
# on a fresh one; Sol on a temporary ledger, then its second attempt on a fresh ledger with the route record carried
# forward. no-start.sh.txt classifies each and four records that are not no-starts. It drives section 10 step 5's
# rule for a route-check exit 1 with three fixtures on a copy of the export (GST-1005). Last, it runs one real probe with
# the pinned runner copy at eu-central-h1, standard-16, 640 GB, on the temporary item r3-capacity-probe-dry, which
# destroys the VM at once, and checks list --mine. Model keys are unset first; only the real probe uses UBICLOUD_API_KEY.
# Usage: PIN=<the 3b pin> G=<gstack-evals git checkout holding PIN and 73042c5> SWEBV=<harbor-datasets checkout>
#   ARMS=<the three arm tars> R=<read-only copy of the pinned ubi-runner.sh> HARBOR_DIR=<harbor at f3a5a92>
#   DROPS=<r3-drop-list.json> A2=<attempt 2's plan.json> A3=<amendment 3's post folder> B=<this post's folder>
#   PROBE_LEDGER=<the dry-run probe ledger> UBI_OWNER=<thread code> bash dry-run-3b.sh.txt <empty work dir>
# Paths print as $W, $G, $SWEBV, $ARMS, $R, $HARBOR_DIR, $DROPS, $A2, $A3, $B and $PROBE_LEDGER.
set -u
W=$(mkdir -p "$1" && cd "$1" && pwd)
unset ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY
OLD=73042c56a0107eab47ff66dd8804983931d22b7c
E=$W/a3/export
L=$E/quick/laneb
mkdir -p "$W/log" "$W/bin"
HIDE="s#$B#\$B#g; s#$A3#\$A3#g; s#$W#\$W#g; s#$G#\$G#g; s#$SWEBV#\$SWEBV#g; s#$ARMS#\$ARMS#g; s#$R#\$R#g; s#$HARBOR_DIR#\$HARBOR_DIR#g; s#$DROPS#\$DROPS#g; s#$A2#\$A2#g; s#$PROBE_LEDGER#\$PROBE_LEDGER#g"
n=0
step() {
  local pat=$1; shift
  n=$((n + 1))
  echo
  echo "\$ $*" | sed "$HIDE"
  "$@" > "$W/log/$n.txt" 2>&1
  local c=$?
  grep -E -- "$pat" "$W/log/$n.txt" | head -"${SHOW:-30}" | sed "$HIDE"
  echo "exit $c (log/$n.txt)"
}
cat > "$W/bin/stub-runner.sh" <<'EOF'
#!/usr/bin/env bash
# Stub of the pinned runner for the keyless fixtures: no network and no VM. Each call is logged with the location and
# size settings it saw. up fails the way attempt 3's did for its first STUB_FAIL_UPS calls (all of them by default).
st=${STUB_STATE:?}; mkdir -p "$st"
cmd=$1; shift
echo "$cmd $* UBI_LOCATION=${UBI_LOCATION:-unset} UBI_SIZE=${UBI_SIZE:-unset} UBI_STORAGE=${UBI_STORAGE:-unset}" >> "$st/calls"
case $cmd in
  up)
    k=$(( $(cat "$st/ups" 2>/dev/null || echo 0) + 1 )); echo "$k" > "$st/ups"
    loc=${UBI_LOCATION:-eu-central-h1}; name=ubirun-fixture-$(date +%s)-$k-t1
    echo "ubi-runner: reap: no expired VMs" >&2
    echo "ubi-runner: creating $loc/$name (standard-16)" >&2
    if [ "$k" -le "${STUB_FAIL_UPS:-999}" ]; then
      echo "ubi-runner: VM did not reach running in 10 minutes (last state: waiting for capacity)" >&2
      echo "ubi-runner: destroyed $loc/$name" >&2
      exit 1
    fi
    echo "$loc  $name" > "$st/alive"; echo "ubi-runner: ready: $name (fixture)" >&2; echo "$name" ;;
  down)
    if [ "${STUB_DOWN_FAILS:-0}" = 1 ]; then echo "ubi-runner: FAILED to destroy $1 (HTTP 500, fixture)" >&2; exit 1; fi
    rm -f "$st/alive"; echo "ubi-runner: gone: ?/$1 (fixture)" >&2 ;;
  list) [ ! -s "$st/alive" ] || cat "$st/alive" ;;
  *) echo "stub: $cmd is not stubbed" >&2; exit 1 ;;
esac
EOF
chmod +x "$W/bin/stub-runner.sh"
cat > "$W/bin/approve.ts" <<'EOF'
const { Ledger } = await import(process.argv[2] + "/quick/laneb/runner/ledger.ts");
new Ledger(process.env.LEDGER!, process.argv[3]).approve(Number(process.argv[4]), { words: process.argv[5], by: process.argv[6], source: "dry run" });
EOF
S=$W/bin/stub-runner.sh
export S UBI_OWNER
DRY="Dry run: a temporary ledger, no spend"
# the run step's guard, as section 10 step 9 writes it: a probe pass in this location, written in the last 30 minutes.
guard() { test -f "$1/probe-pass" && grep -qx "location=$2" "$1/probe-pass" && test $(( $(date +%s) - $(stat -c %Y "$1/probe-pass") )) -le 1800; }
# The run command's settings from section 10 step 9, with the stub as the runner. UBICLOUD_API_KEY is removed, so no
# fixture can reach Ubicloud even if the stub were missing.
runenv() { [ -x "$S" ] || { echo "runenv: no stub runner" >&2; return 2; }; env -u UBICLOUD_API_KEY LEDGER="$1" UBI_OWNER="$UBI_OWNER" UBI_RUNNER="$S" STUB_STATE="$2" UBI_TTL_HOURS=24 VM_USD_PER_HOUR=0.60 VM_RESERVE_HOURS=4 UBI_SIZE=standard-16 UBI_STORAGE=640 LEDGER_PORT="$3" VM_EXTEND_LEAD_SEC=1800 UBI_LOCATION="$4" "${@:5}"; }

echo "## Amendment 3's dry run at the 3b pin"
step . sh -c "sha256sum $A3/dry-run-a3.sh.txt $B/capacity-probe.sh.txt $B/no-start.sh.txt"
SHOW=1000 step . env PIN="$PIN" G="$G" SWEBV="$SWEBV" ARMS="$ARMS" R="$R" HARBOR_DIR="$HARBOR_DIR" DROPS="$DROPS" A2="$A2" bash "$A3/dry-run-a3.sh.txt" "$W/a3"

echo; echo "## What changed from 73042c5 to the pin"
step . git -C "$G" diff --stat "$OLD" "$PIN"
step . git -C "$G" diff --name-only "$OLD" "$PIN" -- quick
step . sh -c "cd $L && for f in r3/screen.ts r3/exposure.ts r3/route-check.ts runner/launch.sh runner/vm-run.sh runner/ledger.ts runner/materialize.ts runner/prepare-run.ts analysis/analyze.ts config/cells.r3.json config/proxy.json tasks/swebv-r3.txt tasks/swebv-r3.json tasks/swebv-r3-drop.json; do o=\$(git -C $G show $OLD:quick/laneb/\$f | sha256sum | cut -c1-64); p=\$(sha256sum \$f | cut -c1-64); [ \$o = \$p ] && echo \"same     \$f\" || echo \"changed  \$f\"; done"

echo; echo "## UBI_LOCATION reaches the runner with no code change"
step . sh -c "sha256sum $R; grep -n 'DEFAULT_LOCATION=' $R; grep -n '\"\$S\" up' $L/runner/launch.sh; grep -c UBI_LOCATION $L/runner/launch.sh; grep -n 'runner/launch.sh\"), dir' $L/r3/screen.ts | cut -c1-140"

echo; echo "## Probe fixtures (stub runner, no VM)"
F=$W/probe-fixtures; mkdir -p "$F"
step . sh -c "LEDGER=$F/fail.jsonl bun $W/bin/approve.ts $E r3-capacity-probe 5 '$DRY' GEVA-192 && STUB_STATE=$F/s-fail PROBE_WINDOW_SEC=5 PROBE_GAP_SEC=2 UBI_OWNER=$UBI_OWNER bash $B/capacity-probe.sh.txt $S $L $F/fail.jsonl eu-central-h1 $F/fail; echo probe exit \$?; test ! -e $F/fail/probe-pass && echo 'no probe-pass'; bun $L/runner/ledger.ts status --path $F/fail.jsonl --item r3-capacity-probe"
step . sh -c "LEDGER=$F/late.jsonl bun $W/bin/approve.ts $E r3-capacity-probe 5 '$DRY' GEVA-192 && STUB_STATE=$F/s-late STUB_FAIL_UPS=2 PROBE_WINDOW_SEC=60 PROBE_GAP_SEC=2 UBI_OWNER=$UBI_OWNER bash $B/capacity-probe.sh.txt $S $L $F/late.jsonl eu-north-h1 $F/late; echo probe exit \$?; cat $F/late/probe-pass; cat $F/s-late/calls"
step . sh -c "LEDGER=$F/left.jsonl bun $W/bin/approve.ts $E r3-capacity-probe 5 '$DRY' GEVA-192 && STUB_STATE=$F/s-left STUB_FAIL_UPS=0 STUB_DOWN_FAILS=1 UBI_OWNER=$UBI_OWNER bash $B/capacity-probe.sh.txt $S $L $F/left.jsonl eu-central-h1 $F/left; echo probe exit \$?; test ! -e $F/left/probe-pass && echo 'no probe-pass'"
step . sh -c "LEDGER=$F/cap.jsonl bun $W/bin/approve.ts $E r3-capacity-probe 0.2 '$DRY' GEVA-192 && STUB_STATE=$F/s-cap UBI_OWNER=$UBI_OWNER bash $B/capacity-probe.sh.txt $S $L $F/cap.jsonl eu-central-h1 $F/cap; echo probe exit \$?; test ! -e $F/s-cap/calls && echo 'the runner was never called'"
step . bash -c "$(declare -f guard); guard $F/late eu-north-h1 && echo 'fresh pass in eu-north-h1: the run step may start'; guard $F/late eu-central-h1 || echo 'pass in another location: the run step refuses'; touch -d '31 minutes ago' $F/late/probe-pass; guard $F/late eu-north-h1 || echo 'pass 31 minutes old: the run step refuses'; guard $F/fail eu-central-h1 || echo 'no pass: the run step refuses'"

echo; echo "## No-start fixtures: the export's screen.ts run with the stub runner"
N=$W/no-start; mkdir -p "$N"
step . bash -c "cd $E && $(declare -f runenv); runenv $W/a3/r3a-screen.jsonl $N/s-a4 9911 eu-central-h1 bun quick/laneb/r3/screen.ts run $W/a3/work-r3a-screen --model claude-opus-5-5 > $N/run-a4.log 2>&1; echo run exit \$?; cat $W/a3/work-r3a-screen/screen.launch-exit; cat $N/run-a4.log; cat $N/s-a4/calls"
step . bash "$B/no-start.sh.txt" "$W/a3/work-r3a-screen" "$N/run-a4.log" "$W/a3/r3a-screen.jsonl" claude-opus-5-5
step . sh -c "cd $E && LEDGER=$W/a3/r3a-screen.jsonl bun quick/laneb/r3/screen.ts analyze $W/a3/work-r3a-screen --model claude-opus-5-5; echo analyze exit \$?"
step . sh -c "LEDGER=$N/r3a-screen-a5.jsonl bun $W/bin/approve.ts $E r3a-screen 150 '$DRY' GEVA-192 && cd $E && LEDGER=$N/r3a-screen-a5.jsonl SWEBV=$SWEBV ARMS=$ARMS ORACLE=1 CONCURRENT=8 bun quick/laneb/r3/screen.ts prepare $N/work-a5 --model claude-opus-5-5 && python3 $W/a3/bin/jobtasks.py $DROPS $N/work-a5"
step . bash -c "cd $E && $(declare -f runenv); runenv $N/r3a-screen-a5.jsonl $N/s-a5 9911 eu-north-h1 bun quick/laneb/r3/screen.ts run $N/work-a5 --model claude-opus-5-5 > $N/run-a5.log 2>&1; echo run exit \$?; cat $N/s-a5/calls; bash $B/no-start.sh.txt $N/work-a5 $N/run-a5.log $N/r3a-screen-a5.jsonl claude-opus-5-5"
step . bash -c "cd $E && $(declare -f runenv); runenv $W/a3/r3a-sol.jsonl $N/s-sol 9912 eu-central-h1 bun quick/laneb/r3/screen.ts run $W/a3/work-r3a-sol --model gpt-6.1-sol > $N/run-sol.log 2>&1; echo run exit \$?; cat $N/run-sol.log; bash $B/no-start.sh.txt $W/a3/work-r3a-sol $N/run-sol.log $W/a3/r3a-sol.jsonl gpt-6.1-sol"
step . sh -c "LEDGER=$N/r3a-sol-a2.jsonl bun $W/bin/approve.ts $E r3a-sol 140 '$DRY' GEVA-192 && cd $E && (LEDGER=$N/r3a-sol-a2.jsonl SWEBV=$SWEBV ARMS=$ARMS ORACLE=1 CONCURRENT=8 bun quick/laneb/r3/screen.ts prepare $N/work-sol-a2 --model gpt-6.1-sol; echo 'prepare without the route record: exit' \$?) 2>&1 | tail -2"
step . sh -c "cp $W/a3/r3a-sol.jsonl.route.json $N/r3a-sol-a2.jsonl.route.json && sha256sum $W/a3/r3a-sol.jsonl.route.json $N/r3a-sol-a2.jsonl.route.json && cd $E && LEDGER=$N/r3a-sol-a2.jsonl SWEBV=$SWEBV ARMS=$ARMS ORACLE=1 CONCURRENT=8 bun quick/laneb/r3/screen.ts prepare $N/work-sol-a2b --model gpt-6.1-sol && python3 $W/a3/bin/jobtasks.py $DROPS $N/work-sol-a2b"
step . bash -c "cd $E && $(declare -f runenv); runenv $N/r3a-sol-a2.jsonl $N/s-sol-a2 9912 eu-central-h1 bun quick/laneb/r3/screen.ts run $N/work-sol-a2b --model gpt-6.1-sol > $N/run-sol-a2.log 2>&1; echo run exit \$?; bash $B/no-start.sh.txt $N/work-sol-a2b $N/run-sol-a2.log $N/r3a-sol-a2.jsonl gpt-6.1-sol"

echo; echo "## Records that are not no-starts"
step . bash -c "cp -r $W/a3/work-r3a-screen $N/x1 && echo 6 > $N/x1/screen.launch-exit && { cat $N/run-a4.log; echo 'vm: ubirun-fixture-x1-t24'; } > $N/x1.log && bash $B/no-start.sh.txt $N/x1 $N/x1.log $W/a3/r3a-screen.jsonl claude-opus-5-5"
step . bash -c "cp -r $W/a3/work-r3a-screen $N/x2 && echo 9 > $N/x2/screen.launch-exit && bash $B/no-start.sh.txt $N/x2 $N/run-a4.log $W/a3/r3a-screen.jsonl claude-opus-5-5"
step . bash -c "cp -r $W/a3/work-r3a-screen $N/x3 && echo 0 > $N/x3/pilot.launch-exit && bash $B/no-start.sh.txt $N/x3 $N/run-a4.log $W/a3/r3a-screen.jsonl claude-opus-5-5"
step . bash -c "cp $W/a3/r3a-screen.jsonl $N/x4.jsonl && echo '{\"ts\":\"1970-01-01T00:00:00.000Z\",\"item\":\"r3a-screen\",\"kind\":\"reserve\",\"id\":\"r3a-screen/django__django-10973__a1\",\"usd_micros\":4000000}' >> $N/x4.jsonl && bash $B/no-start.sh.txt $W/a3/work-r3a-screen $N/run-a4.log $N/x4.jsonl claude-opus-5-5"

echo; echo "## Route check exit 1 (GST-1005): a drop only when the record says drop"
# A copy of the export whose r3-sol.test.ts gains two stub modes and three fixture tests; the export is unchanged. Each
# fixture runs the pin's route-check.ts through that test file's own rig (stub harbor, the real proxy and ledger
# service, fake keys). rc-rule.sh then applies section 10 step 5's rule to each exit and saved record.
C=$W/rc-fixture; X3=$C/out; mkdir -p "$X3"
cat > "$W/bin/rc-fixture.py" <<'EOF'
# rc-fixture.py <r3-sol.test.ts copy>: add stub modes dirsessions (the proxy's sessions file becomes a directory once the
# session closes: EISDIR on read, after the stat) and badsessions (a line that is not JSON), and three fixture tests.
import sys
p = sys.argv[1]; s = open(p).read()
anchor = 'if (mode === "loopsessions")'
i = s.index(anchor)
s = s[:i] + 'if (mode === "dirsessions") { const f = join(process.cwd(), "proxy-sessions.jsonl"); rmSync(f, { force: true }); mkdirSync(f); }\nif (mode === "badsessions") appendFileSync(join(process.cwd(), "proxy-sessions.jsonl"), "not json\\\\n");\n' + s[i:]
s += '''
describe("R3 amendment 3b fixture", () => {
  for (const [name, plan] of [["drop", "smoke1=harbor,smoke2=harbor"], ["dirsessions", "told1=dirsessions"], ["badsessions", "told1=badsessions"]] as const) {
    test("3b fixture " + name, async () => {
      const r = rig();
      const res = await routeCheck(r, { STUB_PLAN: plan });
      const out = process.env.FIX_OUT!;
      writeFileSync(join(out, name + ".exit"), String(res.code) + "\\n");
      writeFileSync(join(out, name + ".out"), res.out);
      if (existsSync(routeRecordPath(r.ledger))) copyFileSync(routeRecordPath(r.ledger), join(out, name + ".route.json"));
    }, 60_000);
  }
});
'''
open(p, "w").write(s)
EOF
cat > "$W/bin/rc-rule.sh" <<'EOF'
# rc-rule.sh <dir> <case>...: section 10 step 5's rule on each case's route-check exit and saved record.
d=$1; shift
for c in "$@"; do
  rc=$(cat "$d/$c.exit")
  if [ -f "$d/$c.route.json" ]; then rec=$(jq -c '{verdict, reason, incomplete}' "$d/$c.route.json" | cut -c1-160); else rec="no record"; fi
  echo "$c: exit $rc, record $rec"
  grep -m1 -o 'E_INTERNAL.*' "$d/$c.out" | cut -c1-160 | sed 's/^/  /'
  [ "$rc" = 1 ] || continue
  if [ -f "$d/$c.route.json" ] && jq -e '.verdict == "drop"' "$d/$c.route.json" > /dev/null; then
    echo "  rule: exit 1 and the record says drop, so Sol's R3 ends (#133 section 6.2)"
  else
    echo "  rule: exit 1 with no drop recorded, so it is not a drop; it goes to the Captain (GST-1005)"
  fi
done
EOF
step . bash -c "cp -a $E $C/export && python3 $W/bin/rc-fixture.py $C/export/quick/laneb/tests/r3-sol.test.ts && diff <(cd $E && find . -type f | sort) <(cd $C/export && find . -type f | sort) > /dev/null && echo 'same file list as the export'; diff -r $E/quick/laneb/r3 $C/export/quick/laneb/r3 && diff -r $E/quick/laneb/runner $C/export/quick/laneb/runner && echo 'r3/ and runner/ unchanged in the copy'; diff $E/quick/laneb/tests/r3-sol.test.ts $C/export/quick/laneb/tests/r3-sol.test.ts | grep -c '^>'"
step '^ *[0-9]+ (pass|fail)|^Ran |3b fixture' sh -c "cd $C/export && FIX_OUT=$X3 bun test quick/laneb/tests/r3-sol.test.ts -t '3b fixture'"
step . bash "$W/bin/rc-rule.sh" "$X3" drop dirsessions badsessions

echo; echo "## One real probe: eu-central-h1, standard-16, 640 GB, temporary item r3-capacity-probe-dry"
step . sh -c "test -s $PROBE_LEDGER || LEDGER=$PROBE_LEDGER bun $W/bin/approve.ts $E r3-capacity-probe-dry 5 'GSTA-2, approvals/2026-11-02-r3-amendment-3b-caps.md: \$5 of VM time in all for the author probes, no model calls' GSTA-2; bun $L/runner/ledger.ts status --path $PROBE_LEDGER --item r3-capacity-probe-dry"
step . sh -c "echo '$(sha256sum "$R" | cut -c1-64)' | grep -qx e53f11384dd6faa14bab96aff14ea42577787df285470e8a192ef24de62753a8 && echo 'runner copy OK' && PROBE_ITEM=r3-capacity-probe-dry UBI_OWNER=$UBI_OWNER bash $B/capacity-probe.sh.txt $R $L $PROBE_LEDGER eu-central-h1 $W/probe-real; echo probe exit \$?; cat $W/probe-real/probe-pass; grep -h '^ubi-runner: \(creating\|ready\|destroy\|destroyed\|gone\|VM did\)' $W/probe-real/probe-*.log | sed 's/ready: \([a-z0-9-]*\) (.*)/ready: \1/'"
step . sh -c "UBI_OWNER=$UBI_OWNER $R list --mine; echo list --mine exit \$?; bun $L/runner/ledger.ts status --path $PROBE_LEDGER --item r3-capacity-probe-dry; grep -c '\"kind\":\"settle\"' $PROBE_LEDGER"
echo; echo "## Done"
