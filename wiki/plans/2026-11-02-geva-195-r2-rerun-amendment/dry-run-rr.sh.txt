n=0
step() {
  local pat=$1; shift
  n=$((n + 1))
  echo
  echo "\$ $*" | sed "$HIDE" | cut -c1-400
  "$@" > "$W/log/$n.txt" 2>&1
  local c=$?
  grep -E -- "$pat" "$W/log/$n.txt" | head -"${SHOW:-24}" | sed "$HIDE" | cut -c1-400
  echo "exit $c (log/$n.txt)"
}

init_items() { local rec=$1; shift; for p in "$@"; do ITEM="${p%%:*}" CAP="${p#*:}" REC="$rec" bun -e 'const { recordApproval } = await import(process.env.R + "/src/runner/ledger/ledger.ts"); recordApproval(process.env.REC, { action: "init", item: process.env.ITEM, cap_micros: Math.round(Number(process.env.CAP) * 1e6), quotes: [{ words: "Dry run: keyless item, no spend", by: "GEVA-195", source: "dry-run" }], admission: "warn" });' > /dev/null || echo "E_INIT ${p%%:*}"; echo "item ${p%%:*} at \$${p#*:}"; done; }
front() { local dir=$1; shift; for f in "$@"; do h=$(sha256sum "$dir/$f" | cut -c1-64); grep -q "$h" "$dir/README.md" && echo "front matter ok $f $h" || echo "E_PIN $f $h is not in its post's front matter"; done; }
mapcheck() {
  M="$MAP" bun -e 'const { loadReaderMap } = await import(process.env.B + "/src/trial/review/readers.ts"); const r = loadReaderMap(process.env.M); console.log("problem" in r ?   local c; c=$(jq -r .gstack.sha "$MAP")
  for p in $(jq -r '.checklists | keys[]' "$MAP"); do git -C "$GSRC" ls-tree --name-only "$c" -- "$p" | grep -qx "$p" && echo "present at the pin: $p" || echo "E_MAP_PATH $p"; done
  for f in review/sections/adversarial.md ship/sections/adversarial.md; do echo "marker in $f: $(git -C "$GSRC" show "$c:$f" | grep -c 'Think like an attacker and a chaos engineer')"; done
  echo "gstack VERSION at the pin: $(git -C "$GSRC" show "$c:VERSION")"
}

scrub_arm() {
  local A=$1 k=0 s
  bun "$SEVMAP" "$A" || return $?
  hits() { $G ingest "$A" --suite "$S" --check 2>&1 | sed -n 's/^E_NOLOSS_ARM \(findings\/[^ ]*\.jsonl\) line \([0-9]*\) (finding .*breaks the publication scan.*/\1:\2/p' | sort -t: -k1,1 -k2,2n -u; }
  hits > "$A.hits0"; echo "$(basename "$A") scan hits before the scrub: $(wc -l < "$A.hits0")"
  for s in "$SCRUB/scrub.ts" "$SCRUB/scrub2.ts" "$SCRUB/r1-scrub3.ts"; do
    k=$((k + 1)); if [ -s "$A.hits$((k - 1))" ]; then bun "$s" "$A" "$A.hits$((k - 1))"; else echo "$(basename "$s"): no hits left, not run"; fi
    hits > "$A.hits$k"; echo "$(basename "$A") scan hits after step $k: $(wc -l < "$A.hits$k")"
  done
  rm -f "$A".hits*
}

  local X=$W/pilot/$1; mkdir -p "$X"; cp -r "$W/record" "$X/record"

  step 'packet|E_' $G judge-packet --suite "$S" $AR --seed 20261031 --out "$X/match" --key "$X/match-key.json"

decide() { step '^(R2 rerun|exposure|gate|E_)' python3 "$W/bin/rr-decide.py" "$1" "$2" "$S" "${3:-$W/src/gstack-full}" "${4:-$W/src/none-rr}" "$W/run/rr-full/parses.tsv" gstack-full-opus; }
score_on() { step '^([A-Za-z0-9-]+ vs [a-z0-9-]+ at a |verdict|E_)' $G score --suite "$1/$2.json" --root "$1/arms" --prereg "$1/comparisons.$2.json" --variant gstack-full --reference none-rr --out "$1/score.json"; }
derive_on() { local X=$1 id=$2; shift 2; step '^(derive|E_)|dropped' $G derive --suite "$S" --comparisons "$W/bin/comparisons.rr.json" --id "$id" --out "$X/$id" --root "$X/final" --drop-gaps "$@"; score_on "$X/$id" "$id"; }
judge_round() {
  local X=$W/grid/$1; mkdir -p "$X"; cp -r "$W/record" "$X/record"
  step 'packet|E_' $G judge-packet --suite "$S" $(arms "$W/raw/stacks-r1" $ARMS) --seed 20261031 --out "$X/match" --key "$X/match-key.json"
  local J="--suite $S --packets $X/match --key $X/match-key.json $(arms $W/raw/stacks-r1 $ARMS) --brief $P/judge-brief.md $JUDGES --ledger $X/ledger.sqlite --record $X/record"
  [ "$1" = keep ] && step '^(worst case|item )|E_' sh -c "$G judge-run $J --item rr-judge-match --out $X/judge-match --dry-run | tee $W/judge/grid-match.txt"
  FAKE_RATES=$2 step '^(judge-run|E_)|unjudged' bun "$W/bin/judge-fake.ts" $J --item rr-judge-match --out "$X/judge-match" --concurrency 4
  step 'judge-join|E_' $G judge-join --suite "$S" --key "$X/match-key.json" --reply "$X/judge-match/primary.json" --second "$X/judge-match/second.json" $(third "$X/judge-match") $(arms "$W/raw/stacks-r1" $ARMS) --out "$X/judge-match/joined"
  for a in $ARMS; do step 'published|E_' $G ingest "$X/judge-match/joined/$a" --suite "$S" --root "$X/judged"; done
  step 'packet|E_' $G judge-packet --suite "$S" $(arms "$X/judged/stacks-r1" $ARMS) --seed 20261031 --out "$X/adj" --key "$X/adj-key.json" --adjudicate
  local A="--suite $S --packets $X/adj --key $X/adj-key.json $(arms $X/judged/stacks-r1 $ARMS) --brief $P/judge-brief.md $JUDGES --ledger $X/ledger.sqlite --record $X/record --source garrytan/gstack=$GSRC --source garrytan/gbrain=$GBSRC"
  [ "$1" = keep ] && step '^(worst case|item )|E_' sh -c "$G judge-run $A --item rr-judge-adjudicate --out $X/judge-adj --dry-run | tee $W/judge/grid-adj.txt"
  FAKE_RATES=$2 step '^(judge-run|unverifiable|E_)' bun "$W/bin/judge-fake.ts" $A --item rr-judge-adjudicate --out "$X/judge-adj" --concurrency 4
  step 'judge-join|E_' $G judge-join --suite "$S" --key "$X/adj-key.json" --reply "$X/judge-adj/primary.json" --second "$X/judge-adj/second.json" $(third "$X/judge-adj") $(arms "$X/judged/stacks-r1" $ARMS) --out "$X/judge-adj/joined"
  for a in $ARMS; do step 'published|E_' $G ingest "$X/judge-adj/joined/$a" --suite "$S" --root "$X/final"; done
  derive_on "$X" stacks-r1-rr
  decide "$X/stacks-r1-rr/score.json" "$X/stacks-r1-rr/derive.json"
}

echo "# Stratum: the cohort's 14 units."; derive_on "$X" stacks-r1-rr-cohort14 --drop-file "$W/out19.tsv"

echo; echo "## Judge items' worst-case caps (section 8)"
step '^shape' python3 - "$W/raw/stacks-r1" "$R2F" "$W/judge/shape.json" <<'EOF'
import glob, json, os, sys
stub, r2, out = sys.argv[1:4]
def stats(root):
    n, size = [], []
    for f in glob.glob(os.path.join(root, "*", "findings", "*.jsonl")):
        lines = [l for l in open(f) if l.strip()]

import json, math, re, sys
item, log, key, sessions, shape, out, packets = sys.argv[1:8]
real = float(sessions) * json.load(open(shape))["per_session"]
per = {}
for e in json.load(open(key))["entries"]:
    per[e["packet"]] = per.get(e["packet"], 0) + 1
rows, sends = {}, 0
for line in open(log):
    m = re.match(r"^(primary|second|third)\s+\S+\s+(\S+)\s+worst case \$([0-9.,]+) \((\d+) sends", line)
    if m:
        rows.setdefault(m.group(1), []).append((per.get(m.group(2), 0), float(m.group(3).replace(",", ""))))
        sends = int(m.group(4))
stub = total = 0.0
reqs = 0
for role, xs in rows.items():
    k = len(xs); mx = sum(x for x, _ in xs) / k; my = sum(y for _, y in xs) / k
    vx = sum((x - mx) ** 2 for x, _ in xs)
    b = sum((x - mx) * (y - my) for x, y in xs) / vx if vx else 0.0
    a = my - b * mx
    stub += sum(y for _, y in xs); reqs += k * sends
    total += int(packets) * (a + b * real)
    print("judge cap %s %s: %d packets, $%.4f a packet plus $%.5f a finding (least squares on the stub batch's lines)" % (item, role, k, a, b))
cap = math.ceil(total)

Sized at %.0f findings a packet (%s sessions a packet) on %s packets, the worst case is $%.2f: cap $%d" % (item, sum(per.values()), max(len(v) for v in rows.values()), reqs, sends, stub, real, sessions, packets, total, cap))
open(out, "a").write("%s\t%d\n" % (item, cap))
