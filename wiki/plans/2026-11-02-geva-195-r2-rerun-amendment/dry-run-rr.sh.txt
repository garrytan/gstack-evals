n=0
step() {
  local pat=$1; shift
  n=$((n + 1))
  echo
  echo "\$ $*" | sed "$HIDE" | cut -c1-400
  "$@" > "$W/log/$n.txt" 2>&1
  local c=$?
  grep -E -- "$pat" "$W/log/$n.txt" | head -"${SHOW:-24}" | sed "$HIDE" | cut -c1-400
  echo "exit $c (log/$n.txt)"
}

ST() { env B="$B" MAP="$MAP" TOLD="$PC/told-gstack-full.txt" bun "$W/bin/rr-stub-arms.ts" "$@"; }
# STE appends a stub run's stderr (2>>) as the README's run step does, closed by its exit line. review-units' CLI exits 2 on any
# stop (src/trial/review/cli.ts at 176fdc7), so a stub run that logged a stop line and exited 0 closes with exit 2.
STE() { local f=$1 rc n; shift; mkdir -p "$(dirname "$f")"; ST "$@" 2>> "$f"; rc=$?; n=$rc; [ $rc -eq 0 ] && grep -qE '^[A-Z][A-Z0-9_]*[A-Z0-9]( |:|$)' "$f" && n=2; echo "review-units exit $n" >> "$f"; cat "$f" >&2; return $rc; }
GS() { printf 'E_COVERAGE_GAP arm none-opus stopped with %s, so it gets no verdict (synthetic)\nreview-units exit 2\n' "$1"; }
third() { [ -s "$1/third.json" ] && echo "--third $1/third.json"; }
init_items() { local rec=$1; shift; for p in "$@"; do ITEM="${p%%:*}" CAP="${p#*:}" REC="$rec" bun -e 'const { recordApproval } = await import(process.env.R + "/src/runner/ledger/ledger.ts"); recordApproval(process.env.REC, { action: "init", item: process.env.ITEM, cap_micros: Math.round(Number(process.env.CAP) * 1e6), quotes: [{ words: "Dry run: keyless item, no spend", by: "GEVA-195", source: "dry-run" }], admission: "warn" });' > /dev/null || echo "E_INIT ${p%%:*}"; echo "item ${p%%:*} at \$${p#*:}"; done; }
front() { local dir=$1; shift; for f in "$@"; do h=$(sha256sum "$dir/$f" | cut -c1-64); grep -q "$h" "$dir/README.md" && echo "front matter ok $f $h" || echo "E_PIN $f $h is not in its post's front matter"; done; }
mapcheck() {

  local c; c=$(jq -r .gstack.sha "$MAP")
  for p in $(jq -r '.checklists | keys[]' "$MAP"); do git -C "$GSRC" ls-tree --name-only "$c" -- "$p" | grep -qx "$p" && echo "present at the pin: $p" || echo "E_MAP_PATH $p"; done
  for f in review/sections/adversarial.md ship/sections/adversarial.md; do echo "marker in $f: $(git -C "$GSRC" show "$c:$f" | grep -c 'Think like an attacker and a chaos engineer')"; done
  echo "gstack VERSION at the pin: $(git -C "$GSRC" show "$c:VERSION")"
}

scrub_arm() {
  local A=$1 k=0 s
  bun "$SEVMAP" "$A" || return $?
  hits() { $G ingest "$A" --suite "$S" --check 2>&1 | sed -n 's/^E_NOLOSS_ARM \(findings\/[^ ]*\.jsonl\) line \([0-9]*\) (finding .*breaks the publication scan.*/\1:\2/p' | sort -t: -k1,1 -k2,2n -u; }
  hits > "$A.hits0"; echo "$(basename "$A") scan hits before the scrub: $(wc -l < "$A.hits0")"
  for s in "$SCRUB/scrub.ts" "$SCRUB/scrub2.ts" "$SCRUB/r1-scrub3.ts"; do
    k=$((k + 1)); if [ -s "$A.hits$((k - 1))" ]; then bun "$s" "$A" "$A.hits$((k - 1))"; else echo "$(basename "$s"): no hits left, not run"; fi
    hits > "$A.hits$k"; echo "$(basename "$A") scan hits after step $k: $(wc -l < "$A.hits$k")"
  done
  rm -f "$A".hits*
}

  step 'judge-join|E_' $G judge-join --suite "$S" --key "$X/match-key.json" --reply "$X/judge-match/primary.json" --second "$X/judge-match/second.json" $(third "$X/judge-match") $(arms "$W/raw/stacks-r1" $ARMS) --out "$X/judge-match/joined"

  step 'packet|E_' $G judge-packet --suite "$S" $(arms "$X/judged/stacks-r1" $ARMS) --seed 20261031 --out "$X/adj" --key "$X/adj-key.json" --adjudicate

import glob, json, os, sys

size_judge() {
python3 - "$@" <<'EOF'
import json, math, re, sys
item, log, key, sessions, shape, out, packets = sys.argv[1:8]
real = float(sessions) * json.load(open(shape))["per_session"]
per = {}
for e in json.load(open(key))["entries"]:
    per[e["packet"]] = per.get(e["packet"], 0) + 1
rows, sends = {}, 0
for line in open(log):
    m = re.match(r"^(primary|second|third)\s+\S+\s+(\S+)\s+worst case \$([0-9.,]+) \((\d+) sends", line)
    if m:
        rows.setdefault(m.group(1), []).append((per.get(m.group(2), 0), float(m.group(3).replace(",", ""))))
        sends = int(m.group(4))
stub = total = 0.0
reqs = 0
for role, xs in rows.items():
    k = len(xs); mx = sum(x for x, _ in xs) / k; my = sum(y for _, y in xs) / k
    vx = sum((x - mx) ** 2 for x, _ in xs)
    b = sum((x - mx) * (y - my) for x, y in xs) / vx if vx else 0.0
    a = my - b * mx
    stub += sum(y for _, y in xs); reqs += k * sends
    total += int(packets) * (a + b * real)
    print("judge cap %s %s: %d packets, $%.4f a packet plus $%.5f a finding (least squares on the stub batch's lines)" % (item, role, k, a, b))
cap = math.ceil(2 * total)
print("judge cap %s: the stub-shaped batch holds %d findings in %d packets and counts at most %d requests (packets x 3 judges x %d sends); its worst case $%.2f. Sized at %.0f findings a packet (%s sessions a packet) on %s packets, the worst case is $%.2f, and $%.2f if every answer is asked again (judge-run asks an answer again once): cap $%d" % (item, sum(per.values()), max(len(v) for v in rows.values()), reqs, sends, stub, real, sessions, packets, total, 2 * total, cap))
open(out, "a").write("%s\t%d\n" % (item, cap))
EOF
}

import { appendFileSync, readdirSync, readFileSync } from "fs";

const wide = big.text.repeat(Math.ceil(CHARS / big.text.length)).slice(0, CHARS);
let total = 0;
for (const role of ["primary", "second", "third"]) {
  const m = lines.match(new RegExp("^" + role + "\\s+(\\w+)/(\\S+)\\s+" + big.file.replace(/\./g, "\\.") + "\\s+worst case \\$[0-9.]+ \\((\\d+) sends at \\$([0-9.]+)\\)", "m"));
