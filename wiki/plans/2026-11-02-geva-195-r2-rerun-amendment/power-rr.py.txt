#!/usr/bin/env python3
# Power for the R2 rerun (GEVA-195 amendment, section 1), keyless and seeded. #112's R1 method (power.py in
# 2026-11-01-geva-101-r48-follow-ups-prereg): each blocker gets its own detection chance from a beta with the control rate
# as its mean and concentration 2 (the scorer's "wide" spread), the treatment adds a 30-point lead, each pass is a
# Bernoulli draw, and a blocker's value is its share of found passes. The standard error of the arms' mean difference
# over 4000 simulated runs, times 2.8, is the MDD (80% power, two-sided 0.05). Blockers are drawn one by one, so the
# result depends on the blocker count and not on how blockers group into PRs. rr-admit.py calls mdd() and clusters_of().
# Usage: python3 power-rr.py <stacks-r1.json>
import json, math, random, sys
from collections import Counter

SEED, SIMS, LEAD = 20261031, 4000, 0.30


def mdd(clusters, p0, passes, sims=SIMS, seed=SEED):
    """The MDD in points for blocking blockers grouped in clusters (sizes), control rate p0, passes per arm."""
    rng = random.Random(seed)
    n = sum(clusters)
    est = []
    for _ in range(sims):
        tot = 0.0
        for size in clusters:
            for _ in range(size):
                p = rng.betavariate(max(p0 * 2, 1e-6), max((1 - p0) * 2, 1e-6))
                q = min(1.0, max(0.0, p + LEAD))
                tot += sum(rng.random() < q for _ in range(passes)) / passes - sum(rng.random() < p for _ in range(passes)) / passes
        est.append(tot / n)
    m = sum(est) / sims
    return 280 * math.sqrt(sum((x - m) ** 2 for x in est) / (sims - 1))


def clusters_of(suite):
    """Blocking blockers per PR cluster, for every cluster that holds one, largest first."""
    c = Counter()
    for u in suite["units"]:
        c[u["cluster"]] += sum(1 for b in u.get("blockers", []) if b.get("counts") == "blocking")
    return sorted((v for v in c.values() if v), reverse=True)


if __name__ == "__main__":
    all33 = clusters_of(json.load(open(sys.argv[1])))
    SETS = [("out19, blocking", [1] * 13, 0.05), ("out19, any severity", [1] * 13, 0.25), ("cohort14, blocking", [2, 2, 2, 1, 4, 4, 4, 1, 1, 1, 3, 1, 1, 2], 0.103), ("all 33 units, blocking", all33, 0.071)]
    print("power: all 33 units' blocking blockers per PR cluster: " + ",".join(map(str, all33)))
    for name, cl, p0 in SETS:
        print("power: %s: %d blockers on %d PR clusters, control %.1f%%: MDD %.0f points at 1 pass, %.0f at 2 passes" % (name, sum(cl), len(cl), 100 * p0, mdd(cl, p0, 1), mdd(cl, p0, 2)))
