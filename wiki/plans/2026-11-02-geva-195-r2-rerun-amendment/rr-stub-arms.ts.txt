function stubSandbox(env: (task: string, e: Record<string, string>) => Record<string, string>) {
  const run = (command: string, cwd: string, e: Record<string, string>, timeout: number) => {
    const r = spawnSync("sh", ["-c", command], { cwd, env: e, timeout, encoding: "utf8" });
    return { exitCode: r.status ?? 1, timedOut: false, stdout: r.stdout ?? "", stderr: r.stderr ?? "" };
  };
  return {
    isolation: "process",
    workPath: (work: string) => work,
    fixturesPath: (fixtures: string) => fixtures,
    homePath: (dir: string) => join(dir, "home"),
    step: (_role: string, command: string, _task: string, o: { work: string; env: Record<string, string>; timeoutMs: number }) => run(command, o.work, o.env, o.timeoutMs),
    agent(task: string, o: { work: string; env: Record<string, string> }) {
      mkdirSync(o.env.HOME!, { recursive: true });
      return {
        exec: (command: string, timeoutMs: number) => run(command, o.work, o.env, timeoutMs),
        spawn: (argv: string[], extra: Record<string, string> = {}) => (argv[0] === "claude" ? spawn(process.execPath, [STUB, ...argv.slice(1)], { cwd: o.work, env: { ...o.env, PATH: process.env.PATH ?? "", ...extra, ...env(task, o.env) }, stdio: ["pipe", "pipe", "pipe"] }) : spawn(argv[0]!, argv.slice(1), { cwd: o.work, env: { ...o.env, PATH: process.env.PATH ?? "", ...extra }, stdio: ["pipe", "pipe", "pipe"] })),
        stop() {},
      };
    },
  };
}
