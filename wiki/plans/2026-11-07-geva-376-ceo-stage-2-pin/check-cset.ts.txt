// GEVA-376: checks cset-1 against the custody addendum before the suite build. Prints counts and digests only.
// usage: bun check-cset.ts <cset dir holding cset.json and units/> <ADDENDUM.md>
import { createHash } from "node:crypto";
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { ceoUnitsDigest } from "./build/lanes/review-noloss/suite.ts";
const [dir, addendum] = process.argv.slice(2) as [string, string];
const sha = (b: string | Buffer) => createHash("sha256").update(b).digest("hex");
const c = JSON.parse(readFileSync(join(dir, "cset.json"), "utf8"));
const table = new Map<string, string[]>();
for (const l of readFileSync(addendum, "utf8").split("\n")) {
  const m = l.match(/^\| (C\d{3}) \| (changed|held) \| ([0-9a-f]{64}) \| ([0-9a-f]{64}) \| ([0-9a-f]{64}) \|$/);
  if (m) table.set(m[1]!, [m[2]!, m[3]!, m[4]!, m[5]!]);
}
const set = c.set as { slot: string; cid: string; kind: string }[];
const per = new Map((c.per_unit as any[]).map((p) => [p.slot, p]));
let order = 0, kindKey = 0, tab = 0, perOk = 0;
const counts = { changed: 0, held: 0 };
set.forEach((u, i) => {
  if (u.slot === "C" + String(i + 1).padStart(3, "0")) order++;
  const d = join(dir, "units", u.cid);
  const f = (n: string) => sha(readFileSync(join(d, n)));
  const plan = f("plan.md"), key = f("key.json"), proj = f("context/project.md");
  const unit = sha(["plan.md " + plan, "context/project.md " + proj, "key.json " + key, "namemap.json " + f("namemap.json")].join("\n"));
  const kind = JSON.parse(readFileSync(join(d, "key.json"), "utf8")).kind;
  if (kind === u.kind) kindKey++;
  (counts as any)[kind]++;
  const t = table.get(u.slot);
  if (t && t[0] === kind && t[1] === unit && t[2] === plan && t[3] === key) tab++;
  const p = per.get(u.slot);
  if (p && p.kind === kind && p.unit_sha256 === unit && p.plan_sha256 === plan && p.key_sha256 === key && p.project_sha256 === proj) perOk++;
});
const dg = ceoUnitsDigest(join(dir, "units"), "units", set.map((u) => u.cid));
console.log(JSON.stringify({ slots: set.length, addendum_rows: table.size, slot_order_ok: order, kind_matches_key_json: kindKey, addendum_table_matches: tab, cset_json_per_unit_matches: perOk, counts, units_digest: "sha256" in dg ? dg.sha256 : dg, cset_json_digest: c.units_digest.sha256, reserve: c.reserve.length, capped: c.capped.length }));
