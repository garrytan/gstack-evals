// GEVA-376: part 2's mkunits.ts (GEVA-338) for the stage 2 comparison set. It builds ceo-outcome unit sources from a units dir: one git repo whose main is an empty root commit and whose
// branch pr-<n> adds plan.md and context/project.md for the n-th unit, a units dir of plan.md and key.json under the
// suite's folder names, the suite, and order.json (each unit's id and its key.json kind, in slot order) beside the sealed
// root. Prints counts and digests only; it never prints unit text. It differs from part 2's copy only in the suite's title
// and purpose, the slot label (C and three digits) and order.json.
// usage: bun mkunits.ts <in units dir> <ids file, one id per line in slot order> <out dir> <repo> <suite id> <sealed: 0|1> <date>
import { spawnSync } from "node:child_process";
import { createHash } from "node:crypto";
import { copyFileSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
const [inDir, idsFile, out, repo, suiteId, sealedFlag, date] = process.argv.slice(2) as [string, string, string, string, string, string, string];
const sha = (b: string | Buffer) => createHash("sha256").update(b).digest("hex");
const git = (dir: string, ...a: string[]) => { const r = spawnSync("git", ["-C", dir, ...a], { encoding: "utf8" }); if (r.status !== 0) throw new Error("git " + a[0] + " failed: " + r.stderr.slice(0, 200)); return r.stdout.trim(); };
const ids = readFileSync(idsFile, "utf8").split("\n").map((l) => l.trim()).filter(Boolean);
const src = join(out, "source");
mkdirSync(src, { recursive: true });
git(src, "init", "-q", "-b", "main");
git(src, "config", "user.name", "eval");
git(src, "config", "user.email", "eval@example.com");
git(src, "config", "commit.gpgsign", "false");
git(src, "commit", "-q", "--allow-empty", "--date", date + "T00:00:00Z", "-m", "init");
const base = git(src, "rev-parse", "HEAD");
const units: any[] = [];
const order: { id: string; kind: string }[] = [];
const rows: string[] = [];
ids.forEach((cid, i) => {
  const pr = i + 1;
  git(src, "checkout", "-q", "-b", "pr-" + pr, base);
  mkdirSync(join(src, "context"), { recursive: true });
  copyFileSync(join(inDir, cid, "plan.md"), join(src, "plan.md"));
  copyFileSync(join(inDir, cid, "context", "project.md"), join(src, "context", "project.md"));
  git(src, "add", "plan.md", "context/project.md");
  git(src, "commit", "-q", "--date", date + "T00:00:00Z", "-m", "plan");
  const head = git(src, "rev-parse", "HEAD");
  const id = repo + "#" + pr + "@" + head.slice(0, 12);
  const folder = id.replaceAll("/", "__");
  const ud = join(out, "sealed", "units", folder);
  mkdirSync(ud, { recursive: true });
  copyFileSync(join(inDir, cid, "plan.md"), join(ud, "plan.md"));
  copyFileSync(join(inDir, cid, "key.json"), join(ud, "key.json"));
  units.push({ id, repo, pr, head, base, cluster: repo + "#" + pr, group: null, blockers: [] });
  order.push({ id, kind: JSON.parse(readFileSync(join(ud, "key.json"), "utf8")).kind });
  rows.push(["C" + String(pr).padStart(3, "0"), sha(readFileSync(join(ud, "plan.md"))), sha(readFileSync(join(ud, "key.json"))), sha(readFileSync(join(src, "context", "project.md")))].join(" "));
});
git(src, "checkout", "-q", "main");
const sealed = sealedFlag === "1";
const suite = { schema_version: 1, id: suiteId, title: sealed ? "CEO outcome comparison set" : "CEO outcome public smoke unit", kind: "ceo-outcome", development_only: !sealed, purpose: sealed ? "The CEO outcome eval's stage 2 comparison units: plan.md and key.json per unit, scored by the CEO outcome readings." : "Invented public units for the CEO outcome eval's keyless dry run.", sealed, catalogs: ["units"], sources: [], prereg_freezes: [], judging_rule: "judge-brief.md of the CEO outcome ladder amendment (sha256 d7b3dce1498e01e874f1dae024ac90e79e5ff5e3af12cc5ed37e2f8a48a14fb5)", shelf_life: { rule: "held out until the Captain retires the set", released: null }, groups: {}, units, ineligible: [], excluded: [], frozen_bars: [] };
const text = JSON.stringify(suite, null, 2) + "\n";
mkdirSync(join(out, "sealed", "suites"), { recursive: true });
writeFileSync(join(out, "sealed", "suites", suiteId + ".json"), text);
const orderText = JSON.stringify(order) + "\n";
writeFileSync(join(out, "order.json"), orderText);
console.log(JSON.stringify({ units: units.length, changed: order.filter((o) => o.kind === "changed").length, held: order.filter((o) => o.kind === "held").length, base_sha256: sha(base), suite_sha256: sha(text), order_sha256: sha(orderText) }));
for (const r of rows) console.log("slot " + r);
