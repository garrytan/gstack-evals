// LOSS power under the stacks round 1 Holm family (wiki/plans/2026-10-31-geva-76-stacks-round-1-prereg, section 7).
// Same simulation as lanes/review-noloss/power.ts (wide spread, beta concentration 2, reference recall 0.65, seed 1,
// 2,000 suites), with the LOSS test's alpha as a parameter: 0.05 reproduces the lane's table, and 0.0125 is Holm's
// strictest step across the four primary comparisons. Keyless.
import { readFileSync } from "node:fs";
import { seededRandom } from "/workspace/gstack-evals/src/runner/stats/paired.ts";
import { pairedTests } from "/workspace/gstack-evals/lanes/review-noloss/stats.ts";
import { suiteClusters } from "/workspace/gstack-evals/lanes/review-noloss/power.ts";
function gamma(shape: number, rng: () => number): number {
  if (shape < 1) return gamma(shape + 1, rng) * Math.pow(rng(), 1 / shape);
  const d = shape - 1 / 3, c = 1 / Math.sqrt(9 * d);
  for (;;) {
    let x: number, v: number;
    do { const u1 = rng(), u2 = rng(); x = Math.sqrt(-2 * Math.log(u1 || 1e-12)) * Math.cos(2 * Math.PI * u2); v = 1 + c * x; } while (v <= 0);
    v = v * v * v; const u = rng();
    if (u < 1 - 0.0331 * x ** 4 || Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
  }
}
const suite = JSON.parse(readFileSync(process.argv[2]!, "utf8"));
const clusters = suiteClusters(suite);
const rows: string[] = ["| Loss (points) | LOSS at alpha 0.05 | LOSS at alpha 0.0125 (Holm, strictest step) |", "|---|---|---|"];
for (const delta of [-10, -15, -20, -23, -25, -30]) {
  const rng = seededRandom(1);
  const cluster = clusters.flatMap((n, c) => Array.from({ length: n }, () => "c" + c));
  const share = (r: number, p: number) => { let h = 0; for (let i = 0; i < p; i++) if (rng() < r) h++; return h / p; };
  let a = 0, b = 0;
  const sims = 2000;
  for (let s = 0; s < sims; s++) {
    const rates = cluster.map(() => { const x = gamma(0.65 * 2, rng); return x / (x + gamma(0.35 * 2, rng)); });
    const mean = rates.reduce((x, r) => x + r, 0) / rates.length;
    const scale = Math.max(0, mean + delta / 100) / mean;
    const pairs = rates.map((r, i) => ({ id: String(i), cluster: cluster[i]!, a: share(r, 2), b: share(Math.min(1, r * scale), 2) }));
    const t = pairedTests(pairs, 10, [2, 2]);
    if (t.loss.p !== null && t.loss.p < 0.05) a++;
    if (t.loss.p !== null && t.loss.p < 0.0125) b++;
  }
  rows.push("| " + -delta + " | " + (100 * a / sims).toFixed(1) + "% | " + (100 * b / sims).toFixed(1) + "% |");
}
console.log("PR clusters " + clusters.length + ", blocking blockers " + clusters.reduce((x, n) => x + n, 0) + ", 2 passes per arm, 2,000 suites, seed 1");
console.log(rows.join("\n"));
