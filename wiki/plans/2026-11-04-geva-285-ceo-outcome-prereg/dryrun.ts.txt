#!/usr/bin/env bun
// dryrun.ts: keyless dry run of the CEO outcome eval's preregistration (GEVA-285). It drives every frozen rule in
// ceo-rule.ts on synthetic readings and opens and settles one ledger lease per frozen item on a temporary ledger.
// Every unit, review, label and cost here is synthetic. No model is called and no key is in the environment.
// Run from a directory holding the bundle's files, with ceo-rule.ts.txt and dryrun.ts.txt renamed to drop .txt:
//   env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY bun dryrun.ts <scorer checkout> <scratch dir>
// It writes dryrun.txt beside itself.
import { createHash, randomBytes } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { join, resolve } from "node:path";
import { spawnSync } from "node:child_process";

const [scorer, OUT] = process.argv.slice(2).map((p) => resolve(p)) as [string, string];
const HERE = import.meta.dir;
const HOME = process.env.HOME ?? "";
const KEYLESS = Object.fromEntries(Object.entries(process.env).filter(([k]) => !/^(ANTHROPIC|OPENAI|OPENROUTER)_API_KEY$/.test(k))) as Record<string, string>;
rmSync(OUT, { recursive: true, force: true });
mkdirSync(OUT, { recursive: true });
const lines: string[] = [];
let pass = 0, fail = 0;
const scrub = (t: string) => t.replaceAll(OUT, "$OUT").replaceAll(scorer, "$SCORER").replaceAll(HERE, "$BUNDLE").replaceAll(HOME, "~");
const say = (t: string) => lines.push(scrub(t));
const check = (label: string, ok: boolean) => { ok ? pass++ : fail++; say((ok ? "PASS " : "FAIL ") + label); };
const section = async (title: string, fn: () => Promise<void> | void) => {
  say("\n## " + title);
  try { await fn(); } catch (e) { check(title + ": ran to the end (threw " + (e instanceof Error ? e.message.split("\n")[0] : String(e)) + ")", false); }
};
const runRule = (cmd: string, data: unknown, tag: string) => {
  const f = join(OUT, tag + ".json");
  writeFileSync(f, JSON.stringify(data));
  return spawnSync("bun", [join(HERE, "ceo-rule.ts"), cmd, f], { encoding: "utf8", env: { ...KEYLESS, GSTACK_EVALS_SCORER: scorer }, maxBuffer: 1 << 26 });
};
const refusal = (cmd: string, data: unknown, tag: string) => {
  const r = runRule(cmd, data, tag);
  return { status: r.status, msg: (r.stderr.split("\n").find((l) => l.startsWith("error: ")) ?? "").slice(7) };
};
const rule = (cmd: string, data: unknown, tag: string) => {
  const r = runRule(cmd, data, tag);
  if (r.status !== 0) throw new Error("ceo-rule.ts " + cmd + " exited " + r.status + ": " + r.stderr.trim().split("\n").slice(-2).join(" "));
  return r.stdout;
};
const PLAN = "E_USAGE readings must carry one row per planned session";
const verdictOf = (o: string) => (o.split("\n").find((l) => l.startsWith("Verdict: ")) ?? "").slice(9);
const show = (o: string, keep: RegExp) => { for (const l of o.split("\n").filter((l) => keep.test(l))) say("    " + l); };
const tagOf = (s: string) => s.slice(0, 48).replace(/[^a-z0-9]+/gi, "-");

type Label = "D" | "R1" | "R2" | "R3" | "none";
const agree = (l: Label) => ({ sol: l, opus: l, sonnet: l });
const missLabels = () => [agree("none"), agree("R1"), agree("none")];
const hitLabels = (rank = 1) => [0, 1, 2].map((i) => agree(i === rank - 1 ? "D" : "none"));
const PLANNED: Record<string, Record<string, number>> = { opus: { none: 0.25, "autoplan-ceo": 3, standalone: 0.7, lens: 0.3 }, sol: { none: 0.15, "autoplan-ceo": 1.8, standalone: 0.42 }, particle: { none: 0.06, "autoplan-ceo": 0.33, standalone: 0.15 } };
const POSTED = ["2026-07-20", "2026-09-12", "2026-10-03", "2026-08-02"];
const CATS = ["scope", "premise or demand", "user impact or migration", "direction or strategy fit"];
let rid = 0;
const review = (unit: string, arm: string, model: string, passNo: number, isHit: boolean, extra: Record<string, unknown> = {}) => ({ id: "r" + ++rid, unit, arm, model, pass: passNo, status: "scored", exposed: arm === "none" || arm === "lens" ? null : true, labels: isHit ? hitLabels(1 + (rid % 3)) : missLabels(), words: [28, 35, 41], leak: { sol: false, opus: false }, forecast: isHit ? 30 : 70, usd: PLANNED[model]![arm]!, ...extra });
const unitsOf = (nc: number, nh: number) => [...Array(nc)].map((_, i) => ({ id: "c" + i, kind: "changed", strict: POSTED[i % 4], category: CATS[i % 4], role: i % 3 === 0 ? "maintainer" : "outside" })).concat([...Array(nh)].map((_, i) => ({ id: "h" + i, kind: "held", strict: "2026-07-10", category: "held", role: "outside" })));
const stockOf = (units: any[], hitA: (i: number) => boolean, hitB: (i: number) => boolean) => units.filter((u) => u.kind === "changed").flatMap((u, i) => [{ unit: u.id, row: "A", labels: hitA(i) ? hitLabels(2) : missLabels() }, { unit: u.id, row: "B", labels: hitB(i) ? hitLabels(3) : missLabels() }]);

// Stage 1: 16 changed + 4 held calibration units; Opus none 3 passes, Sol and particle-1 none 2 passes; smoke: 2 sessions
// per gstack arm per model on c0 and c1.
type S1 = { m: number; incomplete?: number; stockA?: number; stockB?: number; stockSame?: boolean; leakUnits?: number; gaps?: number; smokeFail?: string[]; smokeSlow?: string[]; smokeNoBlock?: string[]; splitDecides?: boolean; judgeFail?: boolean; lowAgree?: boolean; smokeUsd?: Record<string, [number, number]>; noneUsd?: Record<string, number> };
function stage1(o: S1) {
  rid = 0;
  const units = unitsOf(16, 4);
  const reviews: any[] = [];
  const usdNone = (m: string) => ({ usd: o.noneUsd?.[m] ?? PLANNED[m]!.none! });
  let k = 0;
  for (let p = 1; p <= 3; p++) for (let i = 0; i < 16; i++) { const r = review("c" + i, "none", "opus", p, k < o.m, usdNone("opus")); if (k >= 48 - (o.gaps ?? 0)) { r.status = "gap"; r.labels = []; } if (k >= 48 - (o.incomplete ?? 0)) { r.status = "incomplete"; r.labels = []; r.forecast = null; r.usd = 0; } reviews.push(r); k++; }
  const misses = () => reviews.filter((r) => r.status === "scored" && r.labels[0]?.sol === "none" && r.labels[1]?.sol === "R1");
  if (o.splitDecides) {
    const [r0, r1, r2] = misses();
    r0.labels = [{ sol: "D", opus: "none", sonnet: "D" }, agree("none"), agree("none")];
    r1.labels = [{ sol: "D", opus: "none", sonnet: "none" }, agree("none"), agree("none")];
    r2.labels = [{ sol: "R1", opus: "none", sonnet: "D" }, agree("none"), agree("none")];
  }
  if (o.judgeFail) {
    const [r0, r1] = misses();
    r0.labels = [{ sol: null, opus: "D", sonnet: "D" }, agree("none"), agree("none")];
    r1.labels = [{ sol: null, opus: "D", sonnet: "none" }, agree("none"), agree("none")];
  }
  for (let i = 0; i < 4; i++) for (let p = 1; p <= 3; p++) reviews.push(review("h" + i, "none", "opus", p, false, usdNone("opus")));
  for (const model of ["sol", "particle"]) for (let p = 1; p <= 2; p++) for (const u of units) reviews.push(review(u.id, "none", model, p, u.kind === "changed" && Number(u.id.slice(1)) % 2 === 0, usdNone(model)));
  if (o.lowAgree) for (const r of reviews) if (r.status === "scored" && r.labels[0]?.sol === "none" && r.labels[1]?.sol === "R1") r.labels[0] = { sol: "D", opus: "none", sonnet: "none" };
  for (let l = 0; l < (o.leakUnits ?? 0); l++) for (const r of reviews.filter((r) => r.unit === "c" + (15 - l)).slice(0, 2)) r.leak = { sol: true, opus: true };
  for (const model of ["opus", "sol", "particle"]) for (const arm of ["autoplan-ceo", "standalone"]) ["c0", "c1"].forEach((u, i) => {
    const key = arm + ":" + model;
    const usd = (o.smokeUsd?.[key] ?? [0.85 * PLANNED[model]![arm]!, PLANNED[model]![arm]!])[i];
    const nb = (o.smokeNoBlock ?? []).includes(key) && i === 1;
    reviews.push(review(u, arm, model, 1, false, { smoke: true, exposed: !(o.smokeFail ?? []).includes(key), status: nb ? "no-block" : "scored", labels: nb ? [] : missLabels(), wall_s: (o.smokeSlow ?? []).includes(key) && i === 0 ? 2400 : 900, usd }));
  });
  const stock = stockOf(units, (i) => i < (o.stockA ?? 0) || (!!o.stockSame && i === 0), (i) => (i >= 8 && i < 8 + (o.stockB ?? 0)) || (!!o.stockSame && i === 0));
  return { stage: 1, units, reviews, stock };
}

// Stage 2: changed + 16 held units, 2 passes. hits[model][arm] is the number of changed units hit in both passes.
type S2 = { hits: Record<string, Record<string, number>>; passes?: Record<string, number>; exposure?: Record<string, number>; gapUnits?: Record<string, number>; incomplete?: Record<string, number>; noBlock?: Record<string, number>; unjudged?: Record<string, number>; nullForecast?: Record<string, number>; changed?: number; skip?: string[]; family?: Record<string, string[]>; stockB?: number };
function stage2(o: S2) {
  rid = 0;
  const nc = o.changed ?? 48;
  const units = unitsOf(nc, 16);
  const reviews: any[] = [];
  for (const [model, arms] of Object.entries(o.hits)) for (const [arm, h] of Object.entries(arms)) {
    const key = arm + ":" + model;
    if ((o.skip ?? []).includes(key)) continue;
    const exp = o.exposure?.[key] ?? 1;
    let n = 0;
    for (const u of units) for (let p = 1; p <= (o.passes?.[model] ?? 2); p++) {
      const idx = Number(u.id.slice(1)), ch = u.kind === "changed";
      const r: any = review(u.id, arm, model, p, ch && idx < h, { forecast: ch ? 25 : 80 });
      if (arm !== "none" && arm !== "lens") r.exposed = ch ? n++ < Math.round(exp * nc * 2) : true;
      if (ch && idx >= nc - (o.gapUnits?.[key] ?? 0)) { r.status = "gap"; r.labels = []; r.forecast = null; }
      if (ch && idx >= nc - (o.incomplete?.[key] ?? 0)) { r.status = "incomplete"; r.labels = []; r.forecast = null; r.usd = 0; r.exposed = null; }
      if (ch && idx < (o.noBlock?.[key] ?? 0)) { r.status = "no-block"; r.labels = []; r.forecast = null; }
      if (ch && p === 1 && idx >= h && idx < h + (o.unjudged?.[key] ?? 0)) r.labels[0] = { sol: null, opus: "D", sonnet: "none" };
      if (ch && idx < (o.nullForecast?.[key] ?? 0)) r.forecast = null;
      reviews.push(r);
    }
  }
  return { stage: 2, units, reviews, stock: stockOf(units, () => false, (i) => i < (o.stockB ?? 0)), ...(o.family ? { family: o.family } : {}), passes: Object.fromEntries(["opus", "sol", "particle"].map((m) => [m, o.passes?.[m] ?? 2])) };
}

await section("a. Frozen files", () => {
  for (const f of ["template.txt", "judge-brief.md", "stock-list.txt", "unit-brief.md", "power.py", "power.txt", "ceo-rule.ts"]) {
    const p = join(HERE, f);
    check(f + (existsSync(p) ? " sha256 " + createHash("sha256").update(readFileSync(p)).digest("hex") : " is missing"), existsSync(p));
  }
  const t = readFileSync(join(HERE, "template.txt"), "utf8");
  check("the template names the ceo-outcome fence, at most 3 ranked issues of at most 40 words, and a 0 to 100 forecast", t.includes("info string is ceo-outcome") && t.includes("at most 3 objects") && t.includes("at most 40 words") && t.includes("from 0 to 100"));
  const s = readFileSync(join(HERE, "stock-list.txt"), "utf8").trim().split("\n");
  check("the stock list holds two rows of three generic concerns (A: demand, migration, scope; B: direction fit, upkeep, an existing alternative)", s.length === 6 && s.filter((l) => l.startsWith("A: ")).length === 3 && s.filter((l) => l.startsWith("B: ")).length === 3);
  const b = readFileSync(join(HERE, "judge-brief.md"), "utf8");
  check("the judge brief reads each issue's first 40 words and judges it on its first problem", b.includes("first 40 words") && b.includes("the first problem it states"));
  const head = spawnSync("git", ["-C", scorer, "rev-parse", "HEAD"], { encoding: "utf8" }).stdout.trim();
  say("    scorer HEAD " + head + " (the pin post freezes the scorer pin; this dry run uses only src/runner/stats and src/runner/ledger)");
});

await section("b. Stage 1 headroom rule: every outcome", () => {
  const cases: [string, S1, RegExp][] = [
    ["m 20 of 48", { m: 20 }, /^HEADROOM \(20 of 48 = 0\.417; band edges at this n: 8 to 28\)$/],
    ["top edge, m 28 of 48", { m: 28 }, /^HEADROOM \(28 of 48/],
    ["over the top edge, m 29 of 48", { m: 29 }, /^CEILING \(29 of 48 = 0\.604 > 0\.60\): stop$/],
    ["bottom edge, m 8 of 48", { m: 8 }, /^HEADROOM \(8 of 48/],
    ["under the bottom edge, m 7 of 48", { m: 7 }, /^FLOOR \(7 of 48 = 0\.146 < 0\.15\): stop, FIX-METHOD$/],
    ["stock row A hits 1 unit", { m: 20, stockA: 1 }, /^HEADROOM/],
    ["stock row A hits 2 units", { m: 20, stockA: 2 }, /^STOCK FAIL \(a stock row hit D on 2 of 16 changed units/],
    ["stock row B (direction fit) hits 2 units", { m: 20, stockB: 2 }, /^STOCK FAIL \(a stock row hit D on 2 of 16 changed units/],
    ["rows A and B both hit the same single unit", { m: 20, stockSame: true }, /^HEADROOM/],
    ["one unit leak-dropped (it held 1 hit)", { m: 20, leakUnits: 1 }, /^HEADROOM \(19 of 45 = 0\.422; band edges at this n: 7 to 27\)$/],
    ["five units leak-dropped", { m: 20, leakUnits: 5 }, /^NO READ \(fewer than 12 changed units \(11\)\)/],
    ["5 gaps of 48 (10.4%)", { m: 20, gaps: 5 }, /^NO READ \(gaps, unjudged and incomplete 5 of 48 planned Opus none readings, over 10%\)/],
    ["4 gaps of 48 (8.3%)", { m: 20, gaps: 4 }, /^HEADROOM \(20 of 44/],
    ["Opus none's last 6 readings incomplete (its item's cap stopped it; 6 of 48 planned)", { m: 20, incomplete: 6 }, /^NO READ \(gaps, unjudged and incomplete 6 of 48 planned Opus none readings, over 10%\)/],
    ["splits: Sol D/Opus none/Sonnet D is D; Sol D/Opus none/Sonnet none is not; Sol R1/Opus none/Sonnet D is not (20 + 1)", { m: 20, splitDecides: true }, /^HEADROOM \(21 of 48/],
    ["judge failure: Sol missing, Opus and Sonnet D is D; Sol missing, Opus D, Sonnet none is unjudged and dropped (21 of 47)", { m: 20, judgeFail: true }, /^HEADROOM \(21 of 47 = 0\.447; band edges at this n: 8 to 28\)$/],
  ];
  for (const [label, opts, want] of cases) {
    const v = verdictOf(rule("headroom", stage1(opts), "hr-" + tagOf(label)));
    check(label + " -> " + v, want.test(v));
  }
  const s1Missing = stage1({ m: 20 });
  s1Missing.reviews = s1Missing.reviews.filter((r: any) => !(r.arm === "none" && r.model === "opus" && r.unit === "c3" && r.pass === 2));
  const r1 = refusal("headroom", s1Missing, "hr-missing-row");
  check("an Opus none row missing from the readings (c3, pass 2) -> exit " + r1.status + ": " + r1.msg, r1.status === 1 && r1.msg === PLAN + "; first missing: none on opus, unit c3, pass 2");
  const s1Dup = stage1({ m: 20 });
  s1Dup.reviews.push({ ...s1Dup.reviews.find((r: any) => r.arm === "none" && r.model === "opus" && r.unit === "c4" && r.pass === 1), id: "dup" });
  const r1d = refusal("headroom", s1Dup, "hr-duplicate-row");
  check("a duplicated Opus none row in stage 1 (c4, pass 1) -> exit " + r1d.status + ": " + r1d.msg, r1d.status === 1 && r1d.msg === PLAN + "; first duplicate: none on opus, unit c4, pass 1");
  const s1Smoke = stage1({ m: 20 });
  s1Smoke.reviews.find((r: any) => r.smoke && r.arm === "autoplan-ceo" && r.model === "opus" && r.unit === "c0").pass = 2;
  const r1s = refusal("headroom", s1Smoke, "hr-smoke-pass2");
  check("a smoke row past its one planned pass -> exit " + r1s.status + ": " + r1s.msg, r1s.status === 1 && r1s.msg === PLAN + "; first row outside the plan: smoke autoplan-ceo on opus, unit c0, pass 2");
  const ok = rule("headroom", stage1({ m: 20 }), "hr-agree-ok");
  check("judges in full agreement -> positive agreement on D 1.000 and no brief revision", /positive agreement on D 1\.000/.test(ok) && !/under 0\.70/.test(ok));
  const low = rule("headroom", stage1({ m: 20, lowAgree: true }), "hr-agree-low");
  check("Sol alone says D on every missed review's first issue -> hits unchanged (20 of 48), positive agreement under 0.70 sets one brief revision", /^HEADROOM \(20 of 48/.test(verdictOf(low)) && /positive agreement on D 0\.[0-6][0-9]{2}.*one brief revision before stage 2's judging/.test(low));
  show(low, /^(Opus none|Stock|Measured|Descriptive|Agreement)/);
});

await section("c. Stage 1 smoke admissions", () => {
  const ok = rule("headroom", stage1({ m: 20 }), "smoke-ok");
  check("all six smoke cells admitted -> stage 2 may start", /may start after the projection and all-in checks; untested arms: none/.test(ok));
  const cases: [string, S1, RegExp][] = [
    ["standalone not exposed on particle-1 -> that cell UNTESTED, stage 2 may start", { m: 20, smokeFail: ["standalone:particle"] }, /untested arms: standalone:particle$/m],
    ["autoplan-ceo smoke session at 2400 s on Sol (over 1800 s) -> UNTESTED", { m: 20, smokeSlow: ["autoplan-ceo:sol"] }, /untested arms: autoplan-ceo:sol$/m],
    ["standalone smoke session with no block on Opus -> UNTESTED", { m: 20, smokeNoBlock: ["standalone:opus"] }, /untested arms: standalone:opus$/m],
    ["autoplan-ceo not exposed on Opus -> stage 2 does not start", { m: 20, smokeFail: ["autoplan-ceo:opus"] }, /Stage 2: does not start; autoplan-ceo failed its smoke on Opus/],
    ["CEILING with clean smoke -> stage 2 does not start", { m: 30 }, /^Stage 2: does not start$/m],
  ];
  for (const [label, opts, want] of cases) check(label, want.test(rule("headroom", stage1(opts), "smoke-" + tagOf(label))));
  show(rule("headroom", stage1({ m: 20, smokeSlow: ["autoplan-ceo:sol"] }), "smoke-show"), /^Smoke/);
});

await section("d. Projection check: per-item caps and the cut order", () => {
  const base = { envelope: 1410, changed: 48, held: 16, stage1_judges: 4.7, untested: [] as string[] };
  const fits = rule("projection", { ...base, stage1: stage1({ m: 20 }) }, "proj-plan");
  check("planned per-session costs -> PASS as planned, each cap 1.25 times its item's projection plus one full round at allowance, total $1322.00 of $1410.00, every round admits", /Projection: PASS at 'as planned'/.test(fits) && /as planned: caps ceo-o-cmp-opus-p \$222\.00, ceo-o-cmp-opus-a \$516\.00, ceo-o-cmp-sol-p \$99\.00, ceo-o-cmp-sol-a \$306\.00, ceo-o-cmp-particle-p \$34\.00, ceo-o-cmp-particle-a \$54\.00, judges \$91\.00; total \$1322\.00 -> fits/.test(fits) && /round admission: every item's rounds admit at 1\.25 times its projected spend/.test(fits));
  show(fits, /^(Stage 2 projection|  per session|  as planned|  round admission)/);
  const steps: [string, S1, RegExp][] = [
    ["autoplan smoke sessions $2.00 and $3.60 on Opus -> projected from $3.60, the larger -> held units at 1 pass", { m: 20, smokeUsd: { "autoplan-ceo:opus": [2.0, 3.6] } }, /PASS at 'held units at 1 pass'/],
    ["autoplan $4.70 on Opus, standalone $0.19 on particle-1 -> standalone on particle-1 dropped", { m: 20, smokeUsd: { "autoplan-ceo:opus": [3.0, 4.7], "standalone:particle": [0.15, 0.19] } }, /PASS at 'standalone on particle-1 dropped'/],
    ["autoplan $4.60 on Opus, standalone $1.20 on Sol -> standalone on Sol dropped", { m: 20, smokeUsd: { "autoplan-ceo:opus": [3.0, 4.6], "standalone:sol": [1.0, 1.2] } }, /PASS at 'standalone on Sol dropped'/],
    ["autoplan $4.90 and none $0.60 on Opus, standalone $1.20 on Sol -> lens dropped", { m: 20, noneUsd: { opus: 0.6 }, smokeUsd: { "autoplan-ceo:opus": [3.0, 4.9], "standalone:sol": [1.0, 1.2] } }, /PASS at 'lens dropped'/],
    ["autoplan $6.00 on Opus -> Sol and particle-1 cells at 1 pass", { m: 20, smokeUsd: { "autoplan-ceo:opus": [3.0, 6.0] } }, /PASS at 'Sol and particle-1 cells at 1 pass'.*1 pass on sol, particle/],
    ["autoplan $8.00 on Opus -> stop", { m: 20, smokeUsd: { "autoplan-ceo:opus": [3.0, 8.0] } }, /Projection: STOP, over the envelope after every cut/],
  ];
  for (const [label, s1, want] of steps) { const o = rule("projection", { ...base, stage1: stage1(s1) }, "proj-" + tagOf(label)); check(label + (want.test(o) ? "" : " [got: " + (o.split("\n").find((l) => l.startsWith("Projection")) ?? "") + "]"), want.test(o)); if (label.startsWith("autoplan smoke")) check("the per-session line projects autoplan-ceo on Opus at $3.60", /autoplan-ceo:opus \$3\.60/.test(o)); }
  const un = rule("projection", { ...base, untested: ["standalone:particle"], stage1: stage1({ m: 20 }) }, "proj-untested");
  check("an UNTESTED cell is left out of the projection and the cells to run", !/standalone:particle/.test(un.split("\n").find((l) => l.startsWith("Projection: PASS")) ?? "standalone:particle"));
});

await section("d2. Round admission at stage 1's fixed caps", () => {
  const items = (caps: Record<string, number>, k = 1) => [
    { item: "ceo-o-hr-opus-none", model: "opus", cap: caps.on, up: 60, arms: ["none"], usd: { none: 0.25 * k } },
    { item: "ceo-o-hr-sol-none", model: "sol", cap: caps.sn, up: 40, arms: ["none"], usd: { none: 0.15 * k } },
    { item: "ceo-o-hr-particle-none", model: "particle", cap: caps.pn, up: 40, arms: ["none"], usd: { none: 0.06 * k } },
    { item: "ceo-o-hr-smoke-opus-a", model: "opus", cap: caps.oa, up: 2, arms: ["autoplan-ceo"], usd: { "autoplan-ceo": 3 * k } },
    { item: "ceo-o-hr-smoke-opus-p", model: "opus", cap: caps.op, up: 2, arms: ["standalone"], usd: { standalone: 0.7 * k } },
    { item: "ceo-o-hr-smoke-sol-a", model: "sol", cap: caps.sa, up: 2, arms: ["autoplan-ceo"], usd: { "autoplan-ceo": 1.8 * k } },
    { item: "ceo-o-hr-smoke-sol-p", model: "sol", cap: caps.sp, up: 2, arms: ["standalone"], usd: { standalone: 0.42 * k } },
    { item: "ceo-o-hr-smoke-particle-a", model: "particle", cap: caps.pa, up: 2, arms: ["autoplan-ceo"], usd: { "autoplan-ceo": 0.33 * k } },
    { item: "ceo-o-hr-smoke-particle-p", model: "particle", cap: caps.pp, up: 2, arms: ["standalone"], usd: { standalone: 0.15 * k } },
  ];
  const frozen = { on: 52, sn: 23, pn: 6, oa: 25, op: 15, sa: 13, sp: 8, pa: 1, pp: 1 };
  const at1 = rule("admission", { items: items(frozen) }, "adm-frozen");
  check("the frozen stage 1 caps admit every round at the projected spend", /^Admission: PASS$/m.test(at1));
  show(at1, /^  /);
