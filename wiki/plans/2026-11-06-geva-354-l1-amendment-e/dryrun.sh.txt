#   Part E  ceiling.ts at L1's $3,300 line on the pinned ceiling-plan.json against L1's final Opus ledgers, with route and
#           item leases opened and settled on temporary ledgers in the pinned order.
#   Part F  the keyless checks in a work tree at the commit: bun test, typecheck, errors:check, wiki:check.
# Every rule line reads "ok" or "FAIL" with its expected and observed outcome; the last line is DRYRUN: ok or DRYRUN: fail.
# Usage: bash dryrun.sh <commit> [<work dir>]   env: G (the gstack-evals checkout, default /workspace/gstack-evals), ARMS
#   (quick/laneb/arms/build.sh l1's output at 802c2e4), SWEREBENCH (the SWE-rebench leaderboard export), TB2
#   (terminal-bench-2 at 69671fba), OPUS (a folder with L1's final ledgers pilot-opus-d.jsonl, f1-opus.jsonl, tb2-opus.jsonl
#   and the voided pilot-opus.jsonl). The plan and the crew file are the post's ceiling-plan.json and whatif-crew.json.

fails=0 rules=0
# rule <label> <expected> <observed> <test command...>
rule() { local label=$1 want=$2 got=$3; shift 3; rules=$((rules + 1)); if "$@" > /dev/null 2>&1; then echo "ok   $label | expected: $want | observed: $got" | sed "$HIDE"; else fails=$((fails + 1)); echo "FAIL $label | expected: $want | observed: $got" | sed "$HIDE"; fi; }
sha() { sha256sum "$1" | cut -c1-64; }
at() { git -C "$G" show "$1:$2"; }

# approve <ledger> <item> <cap usd>: a temporary ledger's approval row, in the shape approve.ts writes
approve() { mkdir -p "$(dirname "$1")"; printf '{"words":"dry run only, no cap recorded","by":"dry-run","ts":"2026-11-06T00:00:00.000Z","item":"%s","kind":"approval","id":"approval","usd_micros":%s}\n' "$2" "$(($3 * 1000000))" > "$1"; }
LG() { bun "$L/runner/ledger.ts" "$@"; }
# keep <prepared dir> <dest>: the sha256 of every file but plan.json, then plan.json, each part's manifest.json and the job
# files; the prepared dir itself goes, since one f1 item holds several GB of task copies
keep() { mkdir -p "$2/kept" && (cd "$1" && find . -type f ! -path ./plan.json -print0 | sort -z | xargs -0 sha256sum) > "$2/files.sha256" && cp "$1/plan.json" "$2/" && (cd "$1" && find . \( -name manifest.json -o -name 'job*.json' \) -exec cp --parents {} "$2/kept/" \;); rm -rf "$1"; }

// Prints the repo-relative path of every .ts file reachable from the entry points through relative imports that load at
// run time: "import type" lines are left out, since Bun erases them.
import { existsSync, readFileSync } from "node:fs";
import { dirname, relative, resolve } from "node:path";
const root = resolve(process.argv[2]!);
const seen = new Set<string>();
const walk = (f: string) => {
  if (seen.has(f)) return;
  seen.add(f);
  for (const m of readFileSync(f, "utf8").matchAll(/(?:^|\n)\s*(import|export)\s+(type\s+)?[^;]*?from\s*"(\.{1,2}\/[^"]+)"|import\(\s*"(\.{1,2}\/[^"]+)"\s*\)/g)) {

    const c = [p, p + ".ts", p + "/index.ts"].find((x) => existsSync(x) && x.endsWith(".ts"));
    if (c) walk(c);
  }
};
for (const e of process.argv.slice(3)) walk(resolve(root, e));
console.log([...seen].map((f) => relative(root, f)).sort().join("\n"));
TS
