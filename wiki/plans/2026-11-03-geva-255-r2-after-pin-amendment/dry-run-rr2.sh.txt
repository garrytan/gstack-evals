        for i, v in enumerate(x): yield from leaves(v, p + "/" + str(i))
    else: yield p, x

d = sorted(k for k in set(a) | set(b) if a.get(k) != b.get(k))
print("leaves that differ: %d of %d: %s" % (len(d), len(set(a) | set(b)), d))' "$1" "$2"; }

sev() { # <case> <arm dir, or - for a fixture> <map file> <arm name> [fixture labels]
  local c=$1 src=$2 map=$3 name=$4; shift 4; local X=$W/sev/$c; rm -rf "$X"; mkdir -p "$X/findings"
  if [ "$src" = - ]; then
    echo '{"id": "fixture"}' > "$X/arm.json"; local i=0
    for l in "$@"; do i=$((i + 1)); jq -cn --arg s "$l" --arg i "p1-$i" '{severity: $s, path: "src/a.ts", line: 1, summary: "dry-run fixture finding", pass: 1, id: $i, evidence: "fixture"}'; done > "$X/findings/fixture.jsonl"
  else cp "$src/arm.json" "$X/arm.json"; cp "$src"/findings/*.jsonl "$X/findings/"; fi
  jq --slurpfile m "$map" --arg a "$name" 'if $m[0][$a] then .severity_map = $m[0][$a] else del(.severity_map) end' "$X/arm.json" > "$X/a.json" && mv "$X/a.json" "$X/arm.json"
  echo "$c: $(cat "$X"/findings/*.jsonl | grep -c .) findings, severity_map from the $name entry of $(basename "$map")"
  bun "$SEVMAP" "$X"; local rc=$?
  python3 -c '
import json, sys, glob
d = sys.argv[1]; m = (json.load(open(d + "/arm.json")).get("severity_map") or {}).get("default", {})
raw = lambda f: str(f.get("label") or f.get("severity")).split(",")[0].strip()
counts = {}
for p in glob.glob(d + "/findings/*.jsonl"):
    for line in open(p):
        if line.strip():
            l = raw(json.loads(line)); counts[l] = counts.get(l, 0) + 1
to = lambda l: l if l in ("CRITICAL", "INFORMATIONAL") else m.get(l, "UNMAPPED")
out = {}
for l, k in counts.items(): out.setdefault(to(l), []).append("%s x%d" % (l, k))
for v in sorted(out): print("maps to %s: %s" % (v, ", ".join(sorted(out[v]))))' "$X"
  return $rc
}
