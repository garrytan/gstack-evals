#!/usr/bin/env bun
// hr-rules-dry.ts: the hr-rule, hr-kappa and hr-exposure part of the headroom prereg's dry run (GEVA-238). Keyless; no
// request leaves the machine. Run from a dir holding hr-rule.ts, hr-exposure.ts, hr-kappa.ts and e0-readings.ts (each its
// .txt with the suffix removed):
//   bun hr-rules-dry.ts <pinned scorer b7b6aec> <scratch dir>
// hr-rule fixtures: readings.json files built by e0-readings.ts's own exported functions (armReadings, decide, checkReaudit,
// e0NoRead, kappaRead) from synthetic sessions and judge replies, and assembled in the shape its run() writes. hr-kappa runs
// as run, on a synthetic E0-shaped run root whose 2x2 tables reproduce E0's printed kappa and n (its self-check), through a
// stub scorer that re-exports the pin's journal, unit and plan code and its packetAnswers, with loadKey reading the key file
// and readArmDir a no-op. hr-exposure runs on synthetic transcripts in the review-units layout. Prints PASS or FAIL lines
// with counts only, then "PASS n, FAIL m". Exit 0 when nothing fails, else 1.

// ---------- hr-rule fixtures ----------
interface St { m: number; items?: number; anchors?: number; stopped?: Array<[number, number]>; gap?: Array<[number, number]>; cap?: Array<[number, number]>; unjudged?: boolean; failing?: number }
interface Spec { strata: Record<string, St>; reaudit?: boolean; shapes?: boolean; passes?: number }
function suiteOf() {
  return { id: "hr-fixture", units: STRATA.flatMap((st, si) => [1, 2].map((i) => { const u = uid(st, i); return { id: u, repo: "example/base" + (si * 2 + i), pr: si * 2 + i, head: sha(u).slice(0, 40), blockers: ["B1", "B2", "B3", "B4"].map((b) => ({ id: b, case: cid(u, b) })) }; })) };
}
function build(spec: Spec) {
  const P = spec.passes ?? 4, blockers: any[] = [], sessions: any[] = [];
  for (const st of STRATA) {
    const s = spec.strata[st]!, keep = s.items ?? 6;
    const drop = new Set(["2 B3", "2 B2", "2 B1", "1 B3", "1 B2", "1 B1"].slice(0, 6 - keep));
    for (const i of [1, 2]) {
      for (const [k, b] of ["B1", "B2", "B3"].entries()) if (!drop.has(i + " " + b)) blockers.push({ unit: uid(st, i), blocker: b, stratum: st, kind: spec.shapes ? SHAPE[st]![k] : KIND[st], role: "candidate", flagged: false });
      blockers.push({ unit: uid(st, i), blocker: "B4", stratum: st, kind: i === 1 ? "A-X" : "A-F", role: "anchor", flagged: false });
      for (let p = 1; p <= P; p++) {
        const has = (l?: Array<[number, number]>) => (l ?? []).some(([a, b]) => a === i && b === p);
        const cls = has(s.stopped) ? "TURN_CAP" : has(s.gap) ? "GAP" : has(s.cap) ? "CAP" : "SCORED";
        sessions.push({ unit: uid(st, i), pass: p, cls, block: cls === "SCORED" });
      }
    }
  }
  const kinds = { suite_sha256: "fixture", blockers };
  const matched = new Set<string>(), unjudgedOut = new Set<string>();
  for (const st of STRATA) {
    const s = spec.strata[st]!;
    const scored = (role: string) => sessions.filter((x) => x.unit.startsWith("zu-" + st + "-") && x.cls === "SCORED").flatMap((x) => blockers.filter((b) => b.unit === x.unit && b.role === role).map((b) => b.unit + " " + b.blocker + " " + x.pass));
    const cand = scored("candidate"), anc = scored("anchor");
    if (s.m > cand.length) throw new Error("fixture asks for more matches than readings");
    cand.slice(0, s.m).forEach((k) => matched.add(k));
    anc.slice(0, s.anchors ?? anc.length).forEach((k) => matched.add(k));
    if (s.unjudged) { const k = cand.find((c) => !matched.has(c))!; unjudgedOut.add(k.split(" ").slice(0, 2).join(" ")); }
  }
  const replies: Record<string, Map<string, Set<string>>> = {}, entries = new Map<string, any>();
  for (const j of ["sol", "opus", "sonnet"]) {
    const m = new Map<string, Set<string>>();
    for (const b of blockers) {
      const k = b.unit + " " + b.blocker;
      if (j === "sol" && unjudgedOut.has(k)) continue;
      m.set(k, new Set([...matched].filter((x) => x.startsWith(k + " ")).map((x) => "o-" + b.blocker + "-" + x.split(" ")[2])));
    }
    replies[j] = m;
  }
  for (const x of matched) { const [u, b, p] = x.split(" "); entries.set("p-" + u + " o-" + b + "-" + p, { arm: ARM, pass: Number(p), finding: "f1" }); }
  const key = { packets: [...new Set(blockers.map((b) => b.unit))].map((u) => ({ unit: u, file: "p-" + u })) };
  const cells = R.armReadings({ sessions }, kinds, key, replies, entries, () => "critical", ARM);
  const dead = Object.keys(R.decide(cells, kinds, null).kinds).filter((k) => R.decide(cells, kinds, null).kinds[k].verdict === "DEAD");
  let reaudit = null;
  if (spec.reaudit) {
    const r: any = {};
    for (const k of dead) { const st = STRATA.find((x) => KIND[x] === k)!; r[k] = blockers.filter((b) => b.kind === k).map((b, i) => ({ unit: b.unit, blocker: b.blocker, result: i < (spec.strata[st]!.failing ?? 0) ? "fail" : "pass" })); }
    reaudit = R.checkReaudit(r, kinds, dead);
  }
  const d = R.decide(cells, kinds, reaudit);
  const cnt = (c: string) => sessions.filter((x) => x.cls === c).length;
  const stops = sessions.filter((x) => R.STOP_CLASSES.has(x.cls));
  const incomplete = cells.filter((c: any) => c.cls === "SCORED" && c.block && c.final === null).length;
  const reasons = R.e0NoRead({ gaps: cnt("GAP"), caps: cnt("CAP"), judgingIncomplete: incomplete, judgeStopped: [], anchor: d.anchor_pooled, initBad: 0, armBad: [] });
  return { e0: reasons.length ? "NO READ" : "READ", reasons, kinds: d.kinds, strata: d.strata, anchor_pooled: d.anchor_pooled, sensitivity: { e0: reasons.length ? "NO READ" : "READ", reasons, kinds: d.kinds, strata: d.strata, anchor_pooled: d.anchor_pooled, diffs: [] }, kappa: R.kappaRead(cells), judge_stopped: [], stops: stops.length, scheduled: sessions.length, classes: Object.fromEntries(["SCORED", "TURN_CAP", "ALLOWANCE", "TIMEOUT_OUT", "GAP", "CAP"].map((c) => [c, [cnt(c), 0]])), particle: { kinds: {}, anchor_pooled: null }, costs: {}, audit: {}, cells };
}
let fx = 0;
function rule(readings: any, extra: string[] = []) {
  const f = join(S, "readings-" + ++fx + ".json");
  writeFileSync(f, JSON.stringify(readings, null, 1) + "\n");
  const r = run(["hr-rule.ts", "--readings", f, ...extra]);
  const got: Record<string, { reading: string; line: string }> = {};
  for (const l of r.out.split("\n")) { const m = /^Stratum (\w+) \([^)]*\): (.*?)(?: \(|;)/.exec(l); if (m) got[m[1]!] = { reading: m[2]!, line: l }; }
  return { ...r, got };
}

  return r;
};
