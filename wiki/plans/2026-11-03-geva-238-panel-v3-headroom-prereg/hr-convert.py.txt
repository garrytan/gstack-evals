#!/usr/bin/env python3
# hr-convert.py <slice dir> <run dir> [<dry-run seed>]: converts the extracted panel v3 headroom slice (plan section 3)
# into the headroom check's run material: plan cases in the one catalog schema under <run>/sealed-root/catalog/hr/ (lens
# ceo), one combined source clone (panel/plans, a branch per unit number: base commit context/ and src/, head commit adds
# plan.md), map.json and, in <run>/sealed-root/, hr-kinds.json (each candidate mapped to its stratum's headroom kind: ceo
# C-RC, eng E-RC, dx X-RC, design G-RC; anchors keep their shape), hr-shapes.json (each candidate with its E0 shape), both
# in e0-kinds.json's schema, and hr-traces.json, all three filled with suite ids by hr-build.sh. Kept units only; excluded
# items are no blockers. Each candidate's blocker cause is E0's (lever content, then the item's cause) followed by its
# trace: each hop's file and symbol, ending in the planted behavior.
# Unit numbers are an OS-random permutation; the optional seed draws them from random.Random(seed) instead, for the keyless
# dry run only: it is honored only for a slice whose manifest seal_kind is dry-run, and says so. Prints counts and digests
# only. Refuses (exit 2, E_USAGE) a pending seal (or any seal_kind but final or dry-run), a seed with a final slice, a row
# not final or not kept, a kit digest other than kit v3.1's, a missing audit_date, a kept unit whose key.json items or
# traces differ from the manifest, and a slice that breaks the recipe: other than 8 units, 2 a stratum; a unit that does
# not key one item of each of its stratum's three E0 shapes plus one anchor in A-X, A-F or A-T (exclusions allowed);
# a stratum whose anchors are not one context anchor (its evidence names a context/ file) and one src anchor (its
# evidence names a src/ file plan.md names); bases other than 4 distinct MIT, BSD-2-Clause or BSD-3-Clause repos at a
# 40-hex commit, two hosting eng and dx and two hosting ceo and design, one unit of each; a kept candidate's trace with
# fewer than 2 hops past its origin, hop files that are not 2 or more distinct src/ files, a hop file missing from the
# unit tree or named in plan.md with or without its src/ prefix, a terminal that is not the last hop's file, an added
# terminal file (upstream_path null), or a terminal whose git hash-object equals the recorded upstream blob; and a unit
# tree (head commit, git ls-tree -r -t -l HEAD) outside 307200 to 491520 bytes, 150 to 600 blobs or under 12 trees; a kept unit that excludes its anchor; an anchor whose key.json anchor_place is not its
# manifest anchor_type; or a symlink, hidden path or special file under a kept unit's context/ or src/.
# e0-kinds.json's flagged rule holds: a blocker is flagged when its unit's failing_unit_codes is non-empty, never the
# codes. It creates the run dir mode 0700, refuses to go on when the dir is not 0700, and removes the run dir it made when
# it refuses after making it.
#   hr-convert.py --smoke <smoke unit dir> <run dir> <scorer>: the public smoke unit (plan section 4): plan.md, context/,
# src/ and a key.json holding the base record and one dummy blocker. Applies only the layout refusals (top level holds
# plan.md, key.json, context/ and src/ only; no symlink, hidden path or special file; an MIT or BSD base at a 40-hex
# commit; a dummy blocker on a plan.md heading) and the size refusals above, then writes the case (visibility public),
# the source clone, map.json and the suite planrev-v3-hr-smoke with the pinned scorer, keyless. No kinds, traces or seal
# check. GEVA-238, from convert-e0.py (GEVA-219).

def git(d, *a):
    r = subprocess.run(["git", "-C", d, "-c", "gc.auto=0", "-c", "maintenance.auto=false", *a], capture_output=True)
    if r.returncode: refuse("git " + a[0] + " failed while building a unit branch")
    return r.stdout
def make_run():
    if os.path.exists(out): refuse("the run dir exists")
    os.makedirs(out, mode=0o700); MADE.append(out)
    if os.stat(out).st_mode & 0o777 != 0o700: refuse("the run dir is mode %o, not 0700" % (os.stat(out).st_mode & 0o777))
    src = os.path.join(out, "work-source"); subprocess.run(["git", "init", "-q", "-b", "main", src], check=True, capture_output=True)
    return src, tempfile.mkdtemp(dir=out)
def build_unit(ud, tmp, src, pr):
    c = os.path.join(tmp, "u"); shutil.rmtree(c, ignore_errors=True); os.makedirs(c)
    git(c, "init", "-q", "-b", "main")
    for sub in ("context", "src"):
        if os.path.isdir(os.path.join(ud, sub)): shutil.copytree(os.path.join(ud, sub), os.path.join(c, sub))
    git(c, "add", "-A"); git(c, "commit", "-q", "-m", "Working documents and code")
    base = git(c, "rev-parse", "HEAD").decode().strip()
    shutil.copy(os.path.join(ud, "plan.md"), os.path.join(c, "plan.md"))
    git(c, "add", "plan.md"); git(c, "commit", "-q", "-m", "Plan")
    head = git(c, "rev-parse", "HEAD").decode().strip()
    size = blobs = trees = other = 0
    for e in git(c, "ls-tree", "-r", "-t", "-l", "-z", "HEAD").decode().split("\0"):
        if not e: continue
        meta = e.split("\t", 1)[0].split()
        if meta[1] == "tree": trees += 1
        elif meta[1] == "blob": blobs += 1; size += int(meta[3])
        else: other += 1
    git(src, "fetch", "-q", "--no-tags", c, "main:refs/heads/unit-" + str(pr))
    return base, head, (size, blobs, trees, other)
def outside(n):
    size, blobs, trees, other = n
    return [not 307200 <= size <= 491520, not 150 <= blobs <= 600, trees < 12, other > 0]
def size_refusal(stats):
    bad = [outside(n) for n in stats]
    if any(any(b) for b in bad):
        refuse("%d unit trees fall outside the size range (git ls-tree -r -t -l HEAD): bytes outside 307200 to 491520 %d, blobs outside 150 to 600 %d, under 12 trees %d, other entries %d"
               % (sum(any(b) for b in bad), sum(b[0] for b in bad), sum(b[1] for b in bad), sum(b[2] for b in bad), sum(b[3] for b in bad)))
def heading_line(planb, section):
    hits = [n + 1 for n, l in enumerate(planb.decode().split("\n")) if re.match(r"^#{1,6}\s", l) and strip(l) == strip(section)]
    if not hits: refuse("no plan.md heading for a keyed section in one unit")
    return hits[0]
def finish(src, tmp, mapping):
    shutil.rmtree(tmp, ignore_errors=True)
    os.makedirs(os.path.join(out, "source"))
    subprocess.run(["git", "-C", src, "-c", "gc.auto=0", "bundle", "create", "-q", os.path.join(out, "source", "plans.bundle"), "--all"], check=True, capture_output=True)
    shutil.rmtree(src, ignore_errors=True)
    mapping.sort(key=lambda m: m["pr"])
    with open(os.path.join(out, "map.json"), "w") as fh: fh.write(json.dumps({"repo": REPO, "units": mapping}, indent=2) + "\n")
    os.chmod(os.path.join(out, "map.json"), 0o600)

def trace_fault(r, iid):
    ud = os.path.join(sl, "units", r["unit"]); t = (r.get("traces") or {}).get(iid)
    if not isinstance(t, dict) or not isinstance(t.get("hops"), list): return 4
    hops = t["hops"]
    if len(hops) < 2: return 0
    files = [h.get("file") if isinstance(h, dict) else None for h in hops]
    if not all(isinstance(f, str) and f.startswith("src/") and ".." not in f.split("/") for f in files) or len(set(files)) < 2: return 1
    if not all(os.path.isfile(os.path.join(ud, f)) and not os.path.islink(os.path.join(ud, f)) for f in files): return 2
    if any(f in plans[r["unit"]] or f[4:] in plans[r["unit"]] for f in files): return 3
    term = t.get("terminal")
    if not (isinstance(t.get("origin"), str) and t["origin"].strip() and all(isinstance(h.get("symbol"), str) and h["symbol"].strip() for h in hops) and isinstance(term, dict)
            and term.get("file") == files[-1] and isinstance(term.get("line"), int) and isinstance(term.get("behavior"), str) and term["behavior"].strip()): return 4
    if not 1 <= term["line"] <= len(open(os.path.join(ud, files[-1]), "rb").read().split(b"\n")): return 4
    if term.get("upstream_path") is None: return 5
    if not (isinstance(term["upstream_path"], str) and isinstance(term.get("upstream_blob"), str) and HEX40.match(term["upstream_blob"])): return 4
    got = subprocess.run(["git", "hash-object", "--no-filters", "--", os.path.join(ud, files[-1])], capture_output=True, text=True).stdout.strip()
    return 6 if got == term["upstream_blob"] else None
