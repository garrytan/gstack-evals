#!/usr/bin/env bash
# Rebuild routes.tsv (Amendment C routing table, L1 prereg section 2 and B1) keylessly, and print the counts and the
# routed-stratum digest. Steps already done in WORK are kept.
# Usage: CHECKOUT=<gstack-evals checkout> ARM_TAR=<after arm tar> WORK=<dir> bash routes.sh
set -euo pipefail
CHECKOUT=${CHECKOUT:-/workspace/gstack-evals}; ARM_TAR=${ARM_TAR:?set ARM_TAR to the after-pin arm tar}; WORK=${WORK:?set WORK}
HERE="$(cd "$(dirname "$0")" && pwd)"
K="env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY -u UBICLOUD_API_KEY"
GSTACK=699a41cb6808e01240d02e82247693f9170d401a; TB2=69671fbaac6d67a7ef0dfec016cc38a64ef7a77c; HARBOR=f3a5a923f6d722472c9e797af836ae4425fb215f
REB='swe-rebench/swe-rebench-leaderboard@sha256:ebe7444e313a0d8db94fa541139826eaebe2b0abcd4900c6f73e750494910dca'
S=$WORK/src; mkdir -p $S
pin() { [ -d "$S/$1" ] || git clone -q --no-checkout "$2" "$S/$1"; git -C "$S/$1" fetch -q origin "$3"; git -C "$S/$1" -c advice.detachedHead=false checkout -q "$3"; test "$(git -C "$S/$1" rev-parse HEAD)" = "$3"; }
pin gstack https://github.com/garrytan/gstack.git $GSTACK
pin terminal-bench-2 https://github.com/laude-institute/terminal-bench-2.git $TB2
[ -d $S/gstack-evals ] || git clone -q --filter=blob:none --no-checkout https://github.com/garrytan/gstack-evals.git $S/gstack-evals
git -C $S/gstack-evals fetch -q origin tag laneb-discovery-3-runner
git -C $S/gstack-evals show laneb-discovery-3-runner:quick/laneb/tasks/final.json > $S/final-discovery3.json
git -C $S/gstack-evals show laneb-discovery-3-runner:quick/laneb/tasks/tasks.json > $S/tasks-discovery3.json
[ -x $S/harbor-venv/bin/harbor ] || $K bash -c "uv venv -q $S/harbor-venv --python 3.12 && VIRTUAL_ENV=$S/harbor-venv uv pip install -q 'harbor @ git+https://github.com/harbor-framework/harbor@$HARBOR'"
[ -d $S/rebench-export/swe-rebench-leaderboard ] || (mkdir -p $S/rebench-export && cd $S/rebench-export && $K $S/harbor-venv/bin/harbor datasets download "$REB" -o . >/dev/null)
ARM=$WORK/arm-home
[ -f $ARM/.claude/settings.json ] || { mkdir -p $ARM && tar xf "$ARM_TAR" -C $ARM .claude .gstack; }
echo "after arm tar sha256 $(sha256sum "$ARM_TAR" | cut -c1-64)"
echo "arm gstack vs clean clone at $GSTACK (lib, hosts/claude/hooks, bin/gstack-skill-route): $(diff -rq $ARM/.claude/skills/gstack/lib $S/gstack/lib 2>&1 | wc -l) + $(diff -rq $ARM/.claude/skills/gstack/hosts/claude/hooks $S/gstack/hosts/claude/hooks 2>&1 | wc -l) + $(cmp -s $ARM/.claude/skills/gstack/bin/gstack-skill-route $S/gstack/bin/gstack-skill-route && echo 0 || echo 1) differing"
python3 - "$S/final-discovery3.json" > $WORK/tb2-tasks.txt <<'PY'
import json, sys
d = json.load(open(sys.argv[1]))
drop = {"crack-7z-hash", "adaptive-rejection-sampler", "largest-eigenval"}
assert len(d["tasks"]) == 40 and drop <= set(d["tasks"])
print("\n".join(sorted(t for t in d["tasks"] if t not in drop)))
PY
echo "final.json at laneb-discovery-3-runner sha256 $(sha256sum $S/final-discovery3.json | cut -c1-64); tb2-tasks.txt $(wc -l < $WORK/tb2-tasks.txt) tasks, sha256 $(sha256sum $WORK/tb2-tasks.txt | cut -c1-64)"
F=$CHECKOUT/wiki/plans/2026-11-03-geva-254-l1-overhaul-prereg-2000/f-tasks.txt
echo "f-tasks.txt sha256 $(sha256sum $F | cut -c1-64)"
rm -rf $WORK/prompts
$K $S/harbor-venv/bin/python $HERE/build_prompts.py $CHECKOUT $S/rebench-export/swe-rebench-leaderboard $S/terminal-bench-2 $S/tasks-discovery3.json $F $WORK/tb2-tasks.txt $WORK/prompts
$K bun $HERE/route.ts $WORK/prompts $ARM $S/gstack $WORK
echo "routes.tsv sha256 $(sha256sum $WORK/routes.tsv | cut -c1-64)"
