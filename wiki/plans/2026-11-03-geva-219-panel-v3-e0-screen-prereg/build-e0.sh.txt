#!/usr/bin/env bash
# build-e0.sh <bundle tar> <slice dir> <run dir> <scorer> [<bundle sha256>]: builds E0's run material from the extracted
# calibration slice with the pinned scorer: refuses unless the bundle's sha256 is the frozen one (the fifth argument binds
# another digest, for the keyless dry run only, and says so), then plan cases and the source clone (convert-e0.py), the
# suite planrev-v3-e0 (noloss suite --from-catalog), the mirror clone and e0-kinds.json with the suite's blockers and
# sha256. Keyless. Runs under umask 077, so the run root and every file in it are private. Prints counts and digests only.
# GEVA-219.
set -euo pipefail
umask 077
FROZEN=ea69c111bc5d781d507ee296d84202702eb628375c7310bfb26eb93523644934
BUNDLE=$(realpath "$1"); SLICE=$(realpath "$2"); RUN=$(realpath -m "$3"); SCORER=$(realpath "$4"); WANT=${5:-$FROZEN}
GOT=$(sha256sum "$BUNDLE" | cut -c1-64)
if [ "$GOT" != "$WANT" ]; then echo "E_USAGE build-e0: the bundle's sha256 is $GOT, not $WANT" >&2; exit 2; fi
[ "$WANT" = "$FROZEN" ] || echo "build-e0: bundle bound to $WANT, not the frozen digest (dry run only)"
G="env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY $SCORER/bin/gstack-evals"
HERE=$(cd "$(dirname "$0")" && pwd)
python3 "$HERE/convert-e0.py" "$SLICE" "$RUN"
SR=$RUN/sealed-root; mkdir -p "$SR/suites"
if ! (cd "$SCORER" && $G noloss suite --from-catalog catalog/e0 --id planrev-v3-e0 --sealed-root "$SR" --out "$SR/suites/planrev-v3-e0.json") > "$RUN/suite.log" 2>&1; then
  echo "E_USAGE build-e0: noloss suite refused; its output is in \$RUN/suite.log" >&2; exit 2
fi
git clone -q --mirror "$RUN/source/plans.bundle" "$RUN/source/plans.git" 2>/dev/null
python3 - "$SR" <<'PY'
import hashlib, json, os, sys
from collections import Counter
sr = sys.argv[1]
sp = os.path.join(sr, "suites", "planrev-v3-e0.json"); kp = os.path.join(sr, "e0-kinds.json")
suite = json.load(open(sp)); kinds = json.load(open(kp))
rows = [dict(unit=u["id"], blocker=b["id"], **kinds["cases"][b["case"]]) for u in suite["units"] for b in u["blockers"]]
assert len(rows) == len(kinds["cases"]), "every case is one suite blocker"
s = hashlib.sha256(open(sp, "rb").read()).hexdigest()
with open(kp, "w") as fh: fh.write(json.dumps({"suite": suite["id"], "suite_sha256": s, "blockers": rows}, indent=2) + "\n")
c = Counter((r["stratum"], r["kind"], r["role"]) for r in rows)
print("suite", suite["id"], "units", len(suite["units"]), "blockers", len(rows), "flagged blockers", sum(r["flagged"] for r in rows))
for st in ("ceo", "eng", "dx", "design"): print("  " + st + ": " + ", ".join("%s %d" % (k, n) for (x, k, ro), n in sorted(c.items()) if x == st))
print("suite sha256", s)
print("e0-kinds.json sha256", hashlib.sha256(open(kp, "rb").read()).hexdigest())
PY
