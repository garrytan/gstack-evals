#!/usr/bin/env python3
# convert-e0.py <slice dir> <run dir>: converts the extracted panel v3 calibration slice (GEVA-217's layout) into E0's run
# material: plan cases in the one catalog schema under <run>/sealed-root/catalog/e0/ (lens ceo), one combined source clone
# (panel/plans, a branch per unit number: base commit context/ and src/, head commit adds plan.md), map.json and
# <run>/sealed-root/e0-kinds.json (filled with suite ids by build-e0.sh). Kept units only; excluded items are no blockers.
# Unit numbers are an OS-random permutation. Prints counts and digests only. Refuses (exit 2, E_USAGE) a pending seal, a
# row not final or not kept, a kit digest other than the kit's, a kept unit whose key.json items differ from the manifest, or
# counts other than the frozen ones. e0-kinds.json marks each blocker flagged when its unit's failing_unit_codes is
# non-empty, never the codes. Fields it does not use (drop_rule, counts) are ignored. It creates the run dir mode 0700 and
# refuses to go on when the dir is not 0700. GEVA-219.
import hashlib, json, os, random, re, shutil, subprocess, sys, tempfile
from collections import Counter
KIT = "6f21a709317f889b2472cc1e3f0637b36e4057499c12da7a5e31e055b83e0722"
REPO = "panel/plans"
def refuse(msg):
    print("E_USAGE convert-e0: " + msg, file=sys.stderr); sys.exit(2)
if len(sys.argv) != 3: refuse("usage: convert-e0.py <slice dir> <run dir>")
sl, out = os.path.abspath(sys.argv[1]), os.path.abspath(sys.argv[2])
sha = lambda b: hashlib.sha256(b).hexdigest()
man = json.load(open(os.path.join(sl, "manifest.json")))
if man.get("seal_kind") in (None, "pending"): refuse("the manifest's seal_kind is %s, not a final seal" % man.get("seal_kind"))
if any(r.get("state") != "final" for r in man["units"]): refuse("%d manifest rows are not in state final" % sum(r.get("state") != "final" for r in man["units"]))
if man.get("kit_digest") != KIT: refuse("the manifest's kit_digest is not the kit's digest " + KIT)
kept = [r for r in man["units"] if r.get("kept") is True]
if len(kept) != len(man["units"]): refuse("%d manifest rows are not kept" % (len(man["units"]) - len(kept)))
bad = 0
for r in kept:
    key = json.load(open(os.path.join(sl, "units", r["unit"], "key.json")))
    if {i["id"]: [i["item_kind"], i["role"]] for i in key["items"]} != r["items"] or key.get("unit") != r["unit"] or key.get("stratum") != r["stratum"]: bad += 1
    if not set(r.get("excluded_items", [])) <= set(r["items"]): bad += 1
if bad: refuse("%d kept units have key.json items that do not match the manifest" % bad)
FROZEN = {k: 6 for k in ("C-Q", "C-U2", "C-OM", "E-CD", "E-OM", "E-Q", "X-OM", "X-CD", "X-Q", "G-OM", "G-A11Y", "G-X")}
FROZEN.update({"C-U2": 5, "G-A11Y": 5, "E-OM": 5})
got, anchors = Counter(), 0
for r in kept:
    for iid, (k, ro) in r["items"].items():
        if iid in set(r.get("excluded_items", [])): continue
        if ro == "anchor": anchors += 1
        else: got[k] += 1
if len(kept) != 24 or Counter(r["stratum"] for r in kept) != Counter({"ceo": 6, "eng": 6, "dx": 6, "design": 6}) or anchors != 24 or dict(got) != FROZEN or Counter(r["stratum"] for r in kept if r.get("failing_unit_codes")) != Counter({"ceo": 2, "eng": 1}):
    refuse("the slice's counts differ from the frozen ones (24 kept units, 6 per stratum, 24 anchors, 69 candidates with C-U2, G-A11Y and E-OM at 5, 3 flagged units: 2 ceo, 1 eng)")
if os.path.exists(out): refuse("the run dir exists")
os.makedirs(out, mode=0o700)
if os.stat(out).st_mode & 0o777 != 0o700: refuse("the run dir is mode %o, not 0700" % (os.stat(out).st_mode & 0o777))
SR = os.path.join(out, "sealed-root"); CAT = os.path.join(SR, "catalog", "e0"); os.makedirs(CAT)
src = os.path.join(out, "work-source"); subprocess.run(["git", "init", "-q", "-b", "main", src], check=True, capture_output=True)
def git(d, *a):
    r = subprocess.run(["git", "-C", d, "-c", "gc.auto=0", "-c", "maintenance.auto=false", *a], capture_output=True)
    if r.returncode: refuse("git " + a[0] + " failed while building a unit branch")
    return r.stdout
FLAW = {"C-Q": "wrong-thing", "C-U2": "wrong-thing", "A-F": "wrong-thing"}
MODE = {"greenfield": "greenfield", "added": "added-capability", "fix": "fix"}
LEAD = {"H2": ("Deciding fact (written nowhere): ", "intermediate"), "QT": ("Computation: ", "computation"), "OM": ("Missing step: ", "absent_step"), "CD": ("Code behavior: ", "code_fact")}
strip = lambda s: s.strip().lstrip("#").strip()
def cause(i):
    k, c = i["item_kind"], i["cause"].strip()
    if k == "A-X": return "A conflict between two places (" + strip(i["plan_section"]) + " and " + " and ".join(strip(p) for p in i.get("other_places", [])) + "). " + c
    if k == "A-F": return "A false fact. " + c
    if k == "A-T": return "A hidden commitment. " + c
    parts = [LEAD[lv][0] + i[LEAD[lv][1]].strip() for lv in i["levers"]]
    if k in ("C-Q", "C-U2"): parts.append("The plan decides: " + i["decision"].strip() + " It should become: " + i["should_become"].strip())
    return " ".join(parts + [c])
nums = list(range(1, len(kept) + 1)); random.SystemRandom().shuffle(nums)
tmp = tempfile.mkdtemp(dir=out)
cases, mapping = {}, []
counts = Counter()
for r, pr in zip(sorted(kept, key=lambda r: r["unit"]), nums):
    uid = r["unit"]; ud = os.path.join(sl, "units", uid)
    keyb = open(os.path.join(ud, "key.json"), "rb").read(); key = json.loads(keyb)
    planb = open(os.path.join(ud, "plan.md"), "rb").read()
    c = os.path.join(tmp, "u"); shutil.rmtree(c, ignore_errors=True); os.makedirs(c)
    git(c, "init", "-q", "-b", "main")
    for sub in ("context", "src"):
        if os.path.isdir(os.path.join(ud, sub)): shutil.copytree(os.path.join(ud, sub), os.path.join(c, sub))
    git(c, "add", "-A"); git(c, "commit", "-q", "-m", "Working documents and code")
    base = git(c, "rev-parse", "HEAD").decode().strip()
    shutil.copy(os.path.join(ud, "plan.md"), os.path.join(c, "plan.md"))
    git(c, "add", "plan.md"); git(c, "commit", "-q", "-m", "Plan")
    head = git(c, "rev-parse", "HEAD").decode().strip()
    git(src, "fetch", "-q", "--no-tags", c, "main:refs/heads/unit-" + str(pr))
    lines = planb.decode().split("\n")
    def heading_line(section):
        hits = [n + 1 for n, l in enumerate(lines) if re.match(r"^#{1,6}\s", l) and strip(l) == strip(section)]
        if not hits: refuse("no plan.md heading for a keyed section in one unit")
        return hits[0]
    excl = set(r.get("excluded_items", []))
    ids = []
    for k2, i in enumerate([i for i in key["items"] if i["id"] not in excl], 1):
        cid = "e0-" + str(pr) + "-" + str(k2); ids.append(cid)
        case = {"schema_version": 1, "kind": "plan", "id": cid, "visibility": "sealed", "repo": REPO, "pr": pr, "reviewed_head": head, "merge_base": base,
                "plan": {"digest": sha(planb), "lens": "ceo", "stratum": "seeded", "source": "panel v3 calibration kind " + i["item_kind"], "plan_type": MODE[key["mode"]], "flaw_class": FLAW.get(i["item_kind"], "right-thing-badly")},
                "defect": {"file": "plan.md", "line": heading_line(i["plan_section"]), "section": strip(i["plan_section"]), "cause": cause(i), "summary": i["item_kind"]},
                "severity": {"deserved": "CRITICAL", "basis": "Kit key deserved severity " + i["deserved_severity"] + "; Critical and High are must-fix under the plan told prompt."},
                "receipt": {"source": "panel v3 calibration units/" + uid + "/key.json (sha256 " + sha(keyb) + "), " + i["id"], "evidence": [e["path"] + ": " + e["what"] for e in i["evidence"]]},
                "answer_key_audit": {"by": "GEVA-217 custody audit (" + ", ".join(r.get("audit_files", [])) + ")", "at": "2026-11-02", "verdict": "confirmed", "notes": "Manifest verdict " + str(r.get("verdict")) + ", state final."},
                "credit": "Panel v3 calibration slice (GEVA-202 kit, GEVA-217 custody), written blind to gstack"}
        with open(os.path.join(CAT, cid + ".json"), "w") as fh: fh.write(json.dumps(case, indent=2) + "\n")
        cases[cid] = {"stratum": r["stratum"], "kind": i["item_kind"], "role": i["role"], "flagged": bool(r.get("failing_unit_codes"))}
        counts[(r["stratum"], i["item_kind"], i["role"])] += 1
    mapping.append({"unit": uid, "pr": pr, "stratum": r["stratum"], "head": head, "base": base, "cases": ids})
shutil.rmtree(tmp, ignore_errors=True)
os.makedirs(os.path.join(out, "source"))
subprocess.run(["git", "-C", src, "-c", "gc.auto=0", "bundle", "create", "-q", os.path.join(out, "source", "plans.bundle"), "--all"], check=True, capture_output=True)
shutil.rmtree(src, ignore_errors=True)
mapping.sort(key=lambda m: m["pr"])
with open(os.path.join(out, "map.json"), "w") as fh: fh.write(json.dumps({"repo": REPO, "units": mapping}, indent=2) + "\n")
os.chmod(os.path.join(out, "map.json"), 0o600)
with open(os.path.join(SR, "e0-kinds.json"), "w") as fh: fh.write(json.dumps({"suite": "planrev-v3-e0", "suite_sha256": None, "cases": cases}, indent=2) + "\n")
print("kept units", len(kept), "of", len(man["units"]), "; blockers", len(cases), "; excluded items", sum(len(r.get("excluded_items", [])) for r in kept))
for s in ("ceo", "eng", "dx", "design"):
    print("stratum", s, "units", sum(m["stratum"] == s for m in mapping), "; " + ", ".join("%s %s %d" % (k, ro, n) for (st, k, ro), n in sorted(counts.items()) if st == s))
print("plans bundle sha256", sha(open(os.path.join(out, "source", "plans.bundle"), "rb").read()))
