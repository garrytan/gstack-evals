#!/usr/bin/env bun
// e0-dryrun.ts: keyless dry run of the panel v3 E0 preregistration (GEVA-219). Drives the frozen rules through the pinned
// scorer on a synthetic slice: (a) the pin; (b) gen-synth-e0.py, the kit's check_units.py, build-e0.sh and its refusals;
// (c) review-units --dry-run with the frozen commands; (d) real stops through the real SDK, CLI and metering proxy
// (e0-stops.ts) and judge-packet on those arms; (g) the ledger, one lease per item; (e) judge-packet over synthetic arms,
// judge-run --dry-run per judge item, the combine rule and kappa; (f) rule edges; (h) the start rule; (i) the read audit;
// (j) a leak check; (k) the readings end to end in e0-e2e.ts, its own process. Every unit, finding and judgment is synthetic. Writes e0-dryrun.txt beside itself.
//   env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY bun e0-dryrun.ts <scorer> <scratch>
import { createHash, randomBytes } from "node:crypto";
import { cpSync, existsSync, mkdirSync, readFileSync, readdirSync, rmSync, statSync, writeFileSync } from "node:fs";
import { join, resolve } from "node:path";
import { spawnSync } from "node:child_process";
const [scorer, OUT] = process.argv.slice(2).map((p) => resolve(p)) as [string, string];
const HERE = import.meta.dir;
const BRIEF = join(HERE, "judge-brief-v3.md");
const PIN = "b7b6aeccec868e5127e951b187ca4b637da13114";
const HOME = process.env.HOME ?? "";
const KEYLESS = Object.fromEntries(Object.entries(process.env).filter(([k]) => !/^(ANTHROPIC|OPENAI|OPENROUTER)_API_KEY$/.test(k))) as Record<string, string>;
rmSync(OUT, { recursive: true, force: true });
mkdirSync(OUT, { recursive: true });
const lines: string[] = [];
let pass = 0, fail = 0, IDS: string[] = [];
const scrub = (t: string) => {
  let s = t.replaceAll(OUT, "$OUT").replaceAll(scorer, "$SCORER").replaceAll(HERE, "$BUNDLE").replaceAll(HOME, "~");
  for (const id of IDS) s = s.replaceAll(id, "<unit>");
  return s.replace(/packets\/[^\s"']+\.json/g, "packets/<n>.json").replace(/panel\/plans#\d+(@[0-9a-f]+)?/g, "<unit>").replace(/\be0-\d+-\d+\b/g, "<case>").replace(/plans-\d+-[0-9a-f]+/g, "<unit>").replace(/unit-\d+\b/g, "<unit>");
};
const say = (t: string) => lines.push(scrub(t));
const check = (label: string, ok: boolean) => { ok ? pass++ : fail++; say((ok ? "PASS " : "FAIL ") + label); };
const run = (cmd: string[], cwd = scorer, env = KEYLESS) => spawnSync(cmd[0]!, cmd.slice(1), { encoding: "utf8", cwd, env, maxBuffer: 1 << 28 });
const section = async (title: string, fn: () => Promise<void> | void) => {
  say("\n## " + title);
  try { await fn(); } catch (e) { check(title + ": ran to the end (threw " + (e instanceof Error ? e.message.split("\n")[0] : String(e)) + ")", false); }
};
const sha = (b: string | Buffer) => createHash("sha256").update(b).digest("hex");
const R = await import(join(HERE, "e0-readings.ts"));
const { loadSuite } = await import(join(scorer, "lanes/review-noloss/suite.ts"));
const { nolossCli } = await import(join(scorer, "lanes/review-noloss/cli.ts"));
const noloss = async (args: string[]) => { const o: string[] = [], e: string[] = []; const code = await nolossCli(args, { out: (t: string) => o.push(t), err: (t: string) => e.push(t) }); return { code, out: o.join("\n"), err: e.join("\n") }; };
const SL = join(OUT, "slice"), RUN = join(OUT, "run"), SR = join(RUN, "sealed-root"), SUITE = join(SR, "suites/planrev-v3-e0.json"), SRC = join(RUN, "source/plans.git");
let STOPS: any[] = [], REC = "", LENV: Record<string, string> = {};

await section("a. Pin", () => {
  const head = run(["git", "-C", scorer, "rev-parse", "HEAD"]).stdout.trim();
  const dirty = run(["git", "-C", scorer, "status", "--porcelain", "--untracked-files=no"]).stdout.trim();
  check("the scorer is at " + PIN + " with no tracked change", head === PIN && dirty === "");
});

await section("b. Synthetic slice, the kit's check, build-e0.sh and its refusals", async () => {
  check("gen-synth-e0.py writes a slice in the sealed layout", run(["python3", join(HERE, "gen-synth-e0.py"), SL, "20261103"]).status === 0);
  IDS = JSON.parse(readFileSync(join(SL, "manifest.json"), "utf8")).units.map((u: any) => u.unit);
  const KIT = join(OUT, "kit");
  mkdirSync(KIT);
  for (const [f, t] of [["check_units.py.txt", "check_units.py"], ["echo-terms.txt", "echo-terms.txt"]]) cpSync(join(scorer, "wiki/plans/2026-11-02-geva-202-panel-v3-author-kit", f!), join(KIT, t!));
  const c = run(["python3", join(KIT, "check_units.py"), join(SL, "units"), "--wave", "calibration"]);
  const zero = (c.stdout.match(/^unit \S+ problems 0$/gm) ?? []).length;
  check("the kit's check_units.py --wave calibration: " + zero + " of 24 units at problems 0 (exit " + c.status + ")", c.status === 0 && zero === 24);
  const tar = join(OUT, "slice.tar");
  run(["tar", "-cf", tar, "-C", SL, "."], OUT);
  const wrong = run(["bash", join(HERE, "build-e0.sh"), tar, SL, join(OUT, "run-wrongsha"), scorer]);
  check("build-e0.sh refuses a bundle whose sha256 is not the frozen ea69c111... (exit " + wrong.status + ")", wrong.status === 2 && /^E_USAGE build-e0: the bundle's sha256/m.test(wrong.stderr));
  const b = run(["bash", join(HERE, "build-e0.sh"), tar, SL, RUN, scorer, sha(readFileSync(tar))]);
  for (const l of b.stdout.split("\n").filter((l) => l.trim() && !/sha256/.test(l))) say("    " + l);
  if (b.status !== 0) for (const l of b.stderr.split("\n").filter((l) => l.trim()).slice(-4)) say("    stderr: " + l.slice(0, 300));
  check("build-e0.sh builds the synthetic slice, bound to its own sha256 by the dry-run-only fifth argument (exit " + b.status + ")", b.status === 0);
  if (run(["bash", "-c", "command -v age age-keygen"]).status !== 0) say("    custody-e0.sh: age is not installed here, so its end-to-end check is skipped");
  else {
    const ident = join(OUT, "age-identity.txt"), enc = join(OUT, "slice.tar.age"), clog = join(OUT, "CUSTODY.md"), RC = join(OUT, "run-custody");
    run(["age-keygen", "-o", ident]);
    run(["age", "-r", run(["age-keygen", "-y", ident]).stdout.trim(), "-o", enc, tar]);
    const shm = () => readdirSync("/dev/shm").filter((f) => f.startsWith("e0-slice.")).length, before = shm(), es = sha(readFileSync(enc));
    const bad = run(["bash", join(HERE, "custody-e0.sh"), enc, ident, join(OUT, "run-custody-wrongsha"), scorer, clog]);
    check("custody-e0.sh refuses an age bundle whose sha256 is not the frozen ea69c111... before it decrypts (exit " + bad.status + ")", bad.status === 2 && /^E_USAGE custody-e0: the bundle's sha256/m.test(bad.stderr) && !existsSync(clog) && shm() === before);
    const c = run(["bash", join(HERE, "custody-e0.sh"), enc, ident, RC, scorer, clog, es]);
    if (c.status !== 0) for (const l of c.stderr.split("\n").filter((l) => l.trim()).slice(-4)) say("    stderr: " + l.slice(0, 300));
    const sp = join(RC, "sealed-root", "suites", "planrev-v3-e0.json"), ss = existsSync(sp) ? sha(readFileSync(sp)) : "", cl = existsSync(clog) ? readFileSync(clog, "utf8").trim().split("\n") : [];
    check("custody-e0.sh decrypts the age bundle into /dev/shm and runs build-e0.sh on it, bound by the dry-run-only sixth argument (exit " + c.status + ")", c.status === 0 && /\(dry run only\)/.test(c.stdout));
    check("custody-e0.sh builds the suite (" + (ss ? JSON.parse(readFileSync(sp, "utf8")).units.length : 0) + " units) and appends one CUSTODY.md line with a UTC time, the bundle's sha256 and the suite's sha256 (" + cl.length + " line)", ss !== "" && JSON.parse(readFileSync(sp, "utf8")).units.length === 24 && cl.length === 1 && new RegExp("^\\d{4}-\\d\\d-\\d\\dT\\d\\d:\\d\\d:\\d\\dZ custody-e0 bundle " + es + " suite " + ss + "$").test(cl[0] ?? ""));
    check("custody-e0.sh leaves no extraction under /dev/shm (" + (shm() - before) + " left)", shm() === before);
  }
  const open = readdirSync(RUN, { recursive: true }).filter((f) => statSync(join(RUN, String(f))).mode & 0o044).length;
  check("build-e0.sh's run root is mode " + (statSync(RUN).mode & 0o777).toString(8) + " (0700 wanted), and " + open + " files or dirs under it are group- or world-readable", (statSync(RUN).mode & 0o777) === 0o700 && open === 0);
  const um = run(["bash", "-c", "umask 0277 && python3 \"$0\" \"$1\" \"$2\"", join(HERE, "convert-e0.py"), SL, join(OUT, "run-umask")]);
  check("convert-e0.py refuses a run dir it could not make mode 0700 (umask 0277; exit " + um.status + ": " + (um.stderr.split("\n")[0] ?? "").slice(0, 150) + ")", um.status === 2 && /not 0700/.test(um.stderr));
  const s = loadSuite(SUITE, { sealedRoot: SR }).suite;
  const kinds = JSON.parse(readFileSync(join(SR, "e0-kinds.json"), "utf8"));
  const cand = kinds.blockers.filter((x: any) => x.role === "candidate");
  check("the suite loads with the sealed root: 24 units, 93 blockers (24 anchors, 69 candidates), 12 blockers in 3 flagged units, and e0-kinds.json names the suite's sha256", s.units.length === 24 && s.units.flatMap((u: any) => u.blockers).length === 93 && kinds.blockers.length === 93 && cand.length === 69 && kinds.blockers.filter((x: any) => x.flagged).length === 12 && kinds.suite_sha256 === sha(readFileSync(SUITE)) && kinds.blockers.every((x: any) => !("codes" in x)));
  const refuse = (label: string, mutate: (m: any, d: string) => void) => {
    const d = join(OUT, "ref-" + label.replace(/\W+/g, "-"));
    cpSync(SL, d, { recursive: true });
    const m = JSON.parse(readFileSync(join(d, "manifest.json"), "utf8"));
    mutate(m, d);
    writeFileSync(join(d, "manifest.json"), JSON.stringify(m));
    const r = run(["python3", join(HERE, "convert-e0.py"), d, d + "-run"]);
    check("convert-e0.py refuses " + label + " (exit " + r.status + ": " + (r.stderr.split("\n")[0] ?? "").slice(0, 150) + ")", r.status === 2 && /^E_USAGE/.test(r.stderr));
  };
  refuse("a pending manifest", (m) => { m.seal_kind = "pending"; });
  refuse("a bad kit digest", (m) => { m.kit_digest = "0".repeat(64); });
  refuse("a key/manifest mismatch", (m, d) => { const k = join(d, "units", m.units[3].unit, "key.json"); const key = JSON.parse(readFileSync(k, "utf8")); key.items[0].item_kind = "C-OM"; writeFileSync(k, JSON.stringify(key)); });
  refuse("a row not in state final", (m) => { m.units[5].state = "draft"; });
  refuse("a unit with kept false", (m) => { m.units[5].kept = false; });
  refuse("wrong counts (a fourth excluded candidate)", (m) => { m.units[4].excluded_items = [Object.keys(m.units[4].items)[0]]; });
  refuse("wrong counts (a fourth flagged unit)", (m) => { m.units[12].failing_unit_codes = ["x"]; });
});

await section("c. The frozen commands (review-units --dry-run, paths swapped)", async () => {
  const { MAX_TURNS } = await import(join(scorer, "src/adapters/claude-code/driver.ts"));
  for (const [model, item, cap] of [["claude-opus-5-5", "e0-opus-none", "43.30"], ["particle-ai/particle-1", "e0-particle-none", "6.12"]] as const) {
    const r = run([join(scorer, "bin/gstack-evals"), "review-units", "--suite", SUITE, "--arms", "none", "--models", model, "--passes", "2", "--sealed-root", SR, "--plan", "ceo", "--units", "all", "--source", "panel/plans=" + SRC, "--base-branch", "panel/plans=main", "--system-prompt", "none=claude-code-preset", "--parallel", "3", "--timeout", "3600", "--item", item, "--spend-cap", cap, "--out", join(OUT, "rundir", item), "--dry-run"]);
    const o = r.stdout + r.stderr;
    for (const l of o.split("\n").filter((l) => l.trim() && !/^\s+\S*(plans|<unit>|#)/.test(scrub(l))).slice(0, 9)) say("    " + l.slice(0, 260));
    check(item + ": review-units --dry-run runs keyless (exit " + r.status + ")", r.status === 0);
    check(item + ": 48 sessions", /sessions 48\b/.test(o));
    check(item + ": the none arm's line on " + model, /arms: none \(/.test(o) && o.includes(model));
    check(item + ": claude-code-preset appears in the plan", /claude-code-preset/.test(o));
    check(item + ": 60 turns (no --max-turns; the driver's MAX_TURNS is " + MAX_TURNS + " and the dry run prints a max-turns line only when it differs)", MAX_TURNS === 60 && !/max turns/.test(o));
    check(item + ": the spend cap prints as $" + cap, o.includes("$" + cap));
    say("    timeout 3600 " + (/3600/.test(o) ? "printed" : "accepted, not printed by the pin's dry run") + "; parallel 3 " + (/parallel 3/.test(o) ? "printed" : "accepted, not printed") + "; item " + (o.includes(item) ? "printed" : "accepted, not printed (the dry run returns before it reads --item)"));
  }
});

await section("d. Real stops: the real SDK and CLI against the real metering proxy with a scripted upstream", async () => {
  const r = run(["bun", join(HERE, "e0-stops.ts"), scorer, SUITE, SRC, join(OUT, "stops"), "all"], HERE);
  STOPS = r.stdout.split("\n").filter((l) => l.startsWith("{")).map((l) => JSON.parse(l));
  check("e0-stops.ts ran as its own process (exit " + r.status + ")", r.status === 0);
  const want: Record<string, string> = { block: "SCORED", noblock: "SCORED", turncap: "TURN_CAP", allowance: "ALLOWANCE", timeout_out: "TIMEOUT_OUT", timeout_pre: "GAP", upstream: "GAP" };
  for (const [name, cls] of Object.entries(want)) {
    const s = STOPS.find((x) => x.kind === "stop" && x.name === name);
    if (!s || s.error) { check(name + ": the scenario ran" + (s?.error ? " (" + s.error + ")" : ""), false); continue; }
    say("    " + name + ": main records " + s.termination + "; planScore " + s.planScore + "; arm.json " + s.status + (s.gaps ? ", " + s.gaps + " coverage gap" : "") + "; review-units stops [" + s.stops.join(", ") + "]; init " + s.init + "; e0-readings reads " + s.cls + (s.block ? " with a parsed block" : ""));
    check(name + ": e0-readings's termination read gives " + cls + (name === "noblock" ? " with no parsed block (a miss)" : ""), s.cls === cls && (name !== "block" || s.block) && (name !== "noblock" || !s.block));
  }
  for (const name of ["block", "allowance", "timeout_pre"]) {
    const s = STOPS.find((x) => x.kind === "stop" && x.name === name);
    if (!s?.out) continue;
    const p = await noloss(["judge-packet", "--suite", SUITE, "--arm", join(s.out, "arms", "none-opus"), "--seed", "20261103", "--out", join(OUT, "jp-" + name), "--key", join(OUT, "jp-" + name + ".key.json"), "--source", "panel/plans=" + SRC, "--sealed-root", SR]);
    say("    judge-packet on the " + name + " arm (one unit of the suite" + (s.status === "stopped" ? ", arm.json stopped with a coverage gap" : "") + "): " + (p.code === 0 ? "accepted: " + p.out.split("\n")[0] : "refused (exit " + p.code + "): " + p.err.split("\n")[0]!.slice(0, 220)));
  }
});

await section("g. Ledger: every frozen item at its cap, one lease opened and settled per item", async () => {
  const { ledgerCli } = await import(join(scorer, "src/runner/ledger/cli.ts"));
  const { openLease, settleLease, ledgerStatus } = await import(join(scorer, "src/runner/ledger/ledger.ts"));
  const { openLocalLedger } = await import(join(scorer, "src/runner/ledger/local.ts"));
  const { ledgerPath } = await import(join(scorer, "src/runner/ledger/paths.ts"));
  const { rigAdmission } = await import(join(scorer, "src/rig/admission.ts"));
  const { newRunId } = await import(join(scorer, "src/skill-eval/cli/call.ts"));
  const LD = join(OUT, "ledger");
  mkdirSync(LD);
  REC = join(LD, "record");
  writeFileSync(join(LD, "approvals-stub"), "#!/bin/sh\nexit 0\n", { mode: 0o755 });
  LENV = { ...KEYLESS, GSTACK_EVALS_LEDGER_RECORD: REC, GSTACK_EVALS_LEDGER: join(LD, "local.sqlite"), GSTACK_EVALS_APPROVALS_BIN: join(LD, "approvals-stub") };
  say("A temporary approval record and local ledger under $OUT/ledger; the gstack-approvals mirror is a stub. Leases use the pinned entry points' own openLease calls (review-units: amount = --spend-cap, rig admission proof trial; judge-run: proof direct), with --admission warn.");
  const local = openLocalLedger(ledgerPath(undefined, LENV));
  let calls = 0, settled = 0;
  const warned: string[] = [];
  const leaseFor = (item: string, run_id: string, proof: string, amount?: number) => openLease(REC, local, { item, run_id, ...(amount === undefined ? {} : { amount_micros: amount }), requireAdmission: rigAdmission({ proof }), warn: (code: string) => warned.push(code) });
  const codeOf = (f: () => unknown) => { try { f(); return "opened"; } catch (e: any) { return e?.code ?? String(e).split(" ")[0]; } };
  const ITEMS: Array<[string, number | null, string, number, string]> = [["e0-opus-none", 43.3, "claude-opus-5-5", 21_600_000, "trial"], ["e0-particle-none", 6.12, "particle-ai/particle-1", 2_400_000, "trial"], ["e0-judge-sol", null, "gpt-6.1-sol", 1_100_000, "direct"], ["e0-judge-opus", null, "claude-opus-5-5", 1_100_000, "direct"], ["e0-judge-sonnet", null, "claude-sonnet-5-5", 1_100_000, "direct"]];
  for (const [item, cap, model, micros, proof] of ITEMS) {
    const capMicros = cap === null ? Math.floor((62_860_000 - settled) / 10_000) * 10_000 : Math.round(cap * 1e6);
    const o: string[] = [], e: string[] = [];
    const code = await ledgerCli(["init", "--item", item, "--cap", (capMicros / 1e6).toFixed(2), "--admission", "warn", "--words", "dry run only", "--by", "e0-dryrun.ts", "--source", "E0 keyless dry run"], { out: (t: string) => o.push(t), err: (t: string) => e.push(t) }, LENV);
    check(item + " recorded at $" + (capMicros / 1e6).toFixed(2) + (cap === null ? " ($62.86 less $" + (settled / 1e6).toFixed(2) + " settled by every earlier E0 item, rounded down to the cent)" : ""), code === 0);
    const runId = proof === "trial" ? newRunId("review-units") : "judge-" + randomBytes(6).toString("hex");
    const lease = leaseFor(item, runId, proof, proof === "trial" ? capMicros : undefined);
    calls++;
    const rv = await local.reserve({ idempotency_key: "dry-" + calls, lease_id: lease.lease_id, cell: { run_id: runId, arm: "none", task: "synthetic", model, trial: 1, attempt: 1, meter_id: "dry-" + calls }, max_micros: micros });
    await local.settle(rv.reservation.reservation_id, { kind: "spent", spent_micros: micros, usage: { input: 1000, output: 100, reasoning: 0, cache_read: 0, cache_write: 0 } });
    settleLease(REC, local, lease.lease_id);
    const st = ledgerStatus(REC, item)[0];
    settled += st.committed_micros;
    check(item + ": one lease opened (" + warned.at(-1) + ") and settled at $" + (st.committed_micros / 1e6).toFixed(2) + ", the item bound to its run", st.run_id === runId && st.committed_micros === micros && !st.open_lease);
    const again = codeOf(() => leaseFor(item, proof === "trial" ? newRunId("review-units") : "judge-" + randomBytes(6).toString("hex"), proof));
    check(item + ": a second run on the settled item refuses with " + again, again === "E_ITEM_BOUND");
  }
});

await section("e. Judging: one blind batch, judge-run --dry-run per judge item, the combine rule and kappa", async () => {
  const { writePlanArm } = await import(join(scorer, "tests/noloss/plan-panel.ts"));
  const suite = loadSuite(SUITE, { sealedRoot: SR }).suite;
  const raw = join(OUT, "arms-raw");
  for (const [id, rate] of [["none-opus", 0.5], ["none-particle-1", 0.3]] as const) writePlanArm(join(raw, id), suite, { id, passes: 2, raw: true, noise: 0, outcome: (u: any, b: any, p: number) => (((u.pr * 7 + p * 3 + b.id.length) % 10) / 10 < rate ? "F" : "M") });
  const packets = join(OUT, "packets"), keyPath = join(OUT, "key.json");
  const p = await noloss(["judge-packet", "--suite", SUITE, "--arm", join(raw, "none-opus"), "--arm", join(raw, "none-particle-1"), "--seed", "20261103", "--out", packets, "--key", keyPath, "--source", "panel/plans=" + SRC, "--sealed-root", SR]);
  say("    " + (p.code === 0 ? p.out.split("\n")[0] : p.err.split("\n")[0]));
  check("judge-packet builds one batch over both arms, seed 20261103", p.code === 0);
  {
    const st = join(OUT, "arms-stopped");
    rmSync(st, { recursive: true, force: true });
    cpSync(raw, st, { recursive: true });
    const aj = join(st, "none-opus", "arm.json"), arm = JSON.parse(readFileSync(aj, "utf8"));
    arm.status = "stopped";
    arm.coverage_gaps = ["panel/plans#1 pass 2: incomplete, the session ended missing (E_ALLOWANCE)"];
    writeFileSync(aj, JSON.stringify(arm, null, 2));
    const q = await noloss(["judge-packet", "--suite", SUITE, "--arm", join(st, "none-opus"), "--arm", join(st, "none-particle-1"), "--seed", "20261103", "--out", join(OUT, "jp-stopped"), "--key", join(OUT, "jp-stopped.key.json"), "--source", "panel/plans=" + SRC, "--sealed-root", SR]);
    check("judge-packet accepts a full Opus arm that review-units marked stopped with a coverage gap, as an allowance stop leaves it (a findings file for every unit): " + (q.code === 0 ? q.out.split("\n")[0] : q.err.split("\n")[0]), q.code === 0);
  }
  say("    judge brief v3: judge-brief-v3.md, sha256 " + sha(readFileSync(BRIEF)));
  const J: Record<string, string[]> = { sol: ["openai/gpt-6.1-sol", "anthropic/claude-opus-5-5", "anthropic/claude-sonnet-5-5"], opus: ["anthropic/claude-opus-5-5", "openai/gpt-6.1-sol", "anthropic/claude-sonnet-5-5"], sonnet: ["anthropic/claude-sonnet-5-5", "openai/gpt-6.1-sol", "anthropic/claude-opus-5-5"] };
  for (const [j, [pr, se, th]] of Object.entries(J)) {
    const args = ["judge-run", "--suite", SUITE, "--packets", packets, "--key", keyPath, "--arm", join(raw, "none-opus"), "--arm", join(raw, "none-particle-1"), "--brief", BRIEF, "--primary", pr!, "--second", se!, "--third", th!, "--item", "e0-judge-" + j, "--out", join(OUT, "jr-" + j), "--ledger", LENV.GSTACK_EVALS_LEDGER!, "--record", REC, "--source", "panel/plans=" + SRC, "--dry-run"];
    let r = await noloss([...args, "--sealed-root", SR]);
    if (r.code && /sealed-root/.test(r.err)) r = await noloss(args);
    const worst = r.out.split("\n").filter((l) => /^worst case:|^item /.test(l));
    for (const l of worst) say("    " + j + ": " + l);
    if (r.code) say("    " + j + ": " + r.err.split("\n")[0]!.slice(0, 220));
