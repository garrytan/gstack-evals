#!/usr/bin/env bash
# G O5 pin post dry run (GEVA-282, GST-1361), keyless: no key, no VM, no model call. It drives every start-rule check of
# the G prereg (2026-11-04-geva-262-g-family-admission-prereg, section 10 item 6) and plan v3's O5 line (section 13 item 6)
# at the commit given, exported with git archive outside any git work tree, plus the checks the Part 2 brief adds.
#   Part A  the told arm: gstack 699a41cb fetched by sha, told-install.sh into a clean HOME, checkToldHome, the lite
#           render reproduced byte for byte by gstack-modules render (l1.ts liteMatch), and O1's pin tests.
#   Part C  the clean room: cleanRoomPreflight in a stand-in image (busybox, by image id) refusing a seeded drive mount
#           and each path O1's second scan reads, passing a clean room, refusing another digest.
#   Part L  O6's envelope and U4 leases on a temporary ledger: one lease opened and settled per frozen stage (prereg
#           section 9), a lease the envelope refuses, and each admission arm's session limit at a metering proxy slot
#           whose upstream is a stub, so the overshoot past the limit is measured.
#   Part F  the futility look both ways, on made-up trials.
#   Part R  the metered inputs as GSTA-2 decided (A): resize.py in both directions, and admission.sh.txt.
#   Part D  O2's Docker suite with the clean-room Docker tests (GSTACK_EVALS_DOCKER=1), as #210 counted it (57 + 26 tests),
#           compared with #210's counts, and the tool images it built.
# Each check prints one line; the first miss prints DRYRUN: fail and exits 1. The last line is DRYRUN: ok.
# Usage: bash dryrun.sh.txt <commit> [<work dir>]   env: G (the gstack-evals checkout, default /workspace/gstack-evals),
#   GSTACK_REPO (default https://git.capy.ai/garrytan/gstack.git), CLEAN_ROOM_IMAGE (default busybox:1.36).
set -u -o pipefail
COMMIT=${1:?usage: bash dryrun.sh.txt <commit> [<work dir>]}
G=${G:-/workspace/gstack-evals}
HERE=$(cd "$(dirname "$0")" && pwd)
FULL=$(git -C "$G" rev-parse "$COMMIT^{commit}") || exit 2
W=${2:-$HOME/.capy/work/g-o5-dryrun/run-${FULL:0:7}}
rm -rf "$W" && mkdir -p "$W" && W=$(cd "$W" && pwd)
GSTACK_REPO=${GSTACK_REPO:-https://git.capy.ai/garrytan/gstack.git}
IMAGE=${CLEAN_ROOM_IMAGE:-busybox:1.36}
PIN=699a41cb6808e01240d02e82247693f9170d401a
CR_HOME=/home/agent
unset ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY UBICLOUD_API_TOKEN AGENT_USER MULT M_PER M_AP AUTH_PAID AUTH_EXTRA YIELD_G TOLD_SCALE
E=$W/export
export E W CR_HOME
mkdir -p "$E" "$W/log" "$W/drv" "$W/s"
HIDE="s#$W#\$W#g; s#$HOME#~#g; s#$CR_HOME#\$CR_HOME#g"
rules=0
say() { printf '%s\n' "$1" | sed "$HIDE"; }
stop() { say "FAIL $1"; echo; echo "rules: $rules, failed: 1"; echo "DRYRUN: fail"; exit 1; }
# rule <label> <expected> <observed> <test command...>
rule() { local label=$1 want=$2 got=$3; shift 3; rules=$((rules + 1)); if "$@" > /dev/null 2>&1; then say "ok   $label | expected: $want | observed: $got"; else stop "$label | expected: $want | observed: $got"; fi; }
# drive <name> <command...>: a driver prints its own ok lines and stops at its first FAIL line.
drive() { local name=$1; shift; (cd "$E" && "$@") > "$W/log/$name.out" 2> "$W/log/$name.err"; local c=$?; sed "$HIDE" "$W/log/$name.out" | grep -v '^FAIL'; rules=$((rules + $(grep -c '^ok ' "$W/log/$name.out"))); if [ $c != 0 ] || grep -q '^FAIL' "$W/log/$name.out"; then stop "$(grep -m1 '^FAIL' "$W/log/$name.out" | cut -c6- || true)$( [ $c = 0 ] || printf ' (driver %s exited %s: %s)' "$name" "$c" "$(tail -c 300 "$W/log/$name.err" | tr '\n' ' ')")"; fi; }
first() { grep -E -m1 -- "$1" "$2" 2>/dev/null | cut -c1-240 || true; }
for k in ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY AGENT_USER; do echo "$k: $([ -n "${!k+x}" ] && echo set || echo unset)"; done
echo "commit $FULL; bun $(bun --version); docker $(docker version --format '{{.Server.Version}}' 2>/dev/null); python $(python3 -c 'import sys; print(sys.version.split()[0])')"
cat > "$W/drv/lib.ts" <<'EOF'
export const line = (ok: boolean, label: string, want: string, got: string) => {
  console.log((ok ? "ok   " : "FAIL ") + label + " | expected: " + want + " | observed: " + got);
  if (!ok) process.exit(1);
};
export const usd = (m: number) => "$" + (m / 1e6).toFixed(2);
EOF

echo; echo "## Part 0: export"
git -C "$G" archive "$FULL" | tar -x -C "$E"
(cd "$E" && bun install --frozen-lockfile) > "$W/log/bun-install.txt" 2>&1; c=$?
rule "P0 the export is outside any git work tree, and bun install --frozen-lockfile passes" "git rev-parse fails; install exit 0" "install exit $c; $(cd "$E" && git rev-parse --show-toplevel 2>&1 | head -1)" sh -c "test $c = 0 && cd '$E' && ! git rev-parse --show-toplevel"

echo; echo "## Part A: the told arm at gstack $PIN"
SRC=$W/gstack.git
{ git init -q --bare "$SRC" && git -C "$SRC" fetch -q --depth 1 "$GSTACK_REPO" "$PIN" && git -C "$SRC" update-ref refs/heads/pin "$PIN" && git -C "$SRC" symbolic-ref HEAD refs/heads/pin; } > "$W/log/fetch.txt" 2>&1; c=$?
ARCH=$(git -C "$SRC" archive --format=tar "$PIN" 2>/dev/null | sha256sum | cut -c1-64)
L1M=$E/quick/laneb/arms/manifests/l1/after.json
rule "A1 gstack fetched by sha (git fetch --depth 1 <repo> <sha>); its git archive matches #202's after-pin manifest" "$(jq -r .git_archive_sha256 "$L1M")" "fetch exit $c, $ARCH" test "$c" = 0 -a "$ARCH" = "$(jq -r .git_archive_sha256 "$L1M")"
H=$W/told-home
TP="$(dirname "$(command -v bun)"):/usr/local/bin:/usr/bin:/bin"
TOLD_PATH=$TP bash "$E/lanes/outcome-skills/arms/told-install.sh" "$SRC" "$H" > "$W/log/told-install.txt" 2>&1; c=$?
rule "A2 told-install.sh (./setup --host claude --loadout lite) into a clean HOME" "exit 0" "exit $c, $(tail -1 "$W/log/told-install.txt" | cut -c1-80)" test "$c" = 0
R=$H/.gstack/render/claude
mv "$R" "$R.installed" && (cd "$H/.claude/skills/gstack" && env -i HOME="$H" PATH="$TP" USER=user LANG=C.UTF-8 TERM=dumb ./bin/gstack-modules render --host claude --loadout lite --out-dir "$R") > "$W/log/render.txt" 2>&1; c=$?
mv "$R" "$W/fresh-render" 2>/dev/null; mv "$R.installed" "$R"
cat > "$W/drv/told.ts" <<'EOF'
import { readFileSync } from "node:fs";
import { line } from "../drv/lib.ts";
const E = process.env.E!, W = process.env.W!, H = W + "/told-home";
const { checkToldHome, TOLD_PIN } = await import(E + "/lanes/outcome-skills/arms/told.ts");
const { liteMatch, liteLock } = await import(E + "/quick/laneb/arms/l1.ts");
const after = JSON.parse(readFileSync(E + "/quick/laneb/arms/manifests/l1/after.json", "utf8"));
const row = readFileSync(E + "/wiki/plans/2026-11-03-geva-254-l1-overhaul-prereg-2000/install-manifest.md", "utf8").split("\n").find((l) => l.includes("lite tree digest, normalized (setup render)")) ?? "";
let r: any;
try { r = checkToldHome(H); } catch (e) { r = { ok: false, why: (e as Error).message.slice(0, 300) }; }
line(r.ok === true, "A3 checkToldHome: pin sha and version, the pin's source and hook sha256s, the hooks, a clean checkout", "pass at " + TOLD_PIN.sha.slice(0, 8) + " " + TOLD_PIN.version, r.ok ? "pass, hooks " + JSON.stringify(r.hooks) : r.why);
const ok4 = r.files === 204 && r.digest === TOLD_PIN.lite_tree_digest && r.digest === after.lite.tree_digest_normalized && row.includes(r.digest);
line(ok4, "A4 the installed lite render's tree digest equals told.ts TOLD_PIN, #202's after.json and #193's install-manifest.md (setup render row)", "204 files, " + TOLD_PIN.lite_tree_digest, r.files + " files, " + r.digest + (row.includes(r.digest) ? " (in #193's row)" : " (not in #193's row)"));
const fresh = W + "/fresh-render";
let m: any;
try { m = liteMatch(H + "/.gstack/render/claude", fresh); } catch (e) { m = { problems: [(e as Error).message.slice(0, 200)], matched: 0, files: 0 }; }
const lock = (() => { try { return liteLock(fresh + "/loadout.lock.json"); } catch { return null; } })();
line(m.problems.length === 0 && m.matched === 204 && lock?.spec_sha256 === TOLD_PIN.lite_spec_sha256, "A5 a fresh gstack-modules render of the pin's lite lock matches the installed render byte for byte, nothing extra", "204 of 204 lock files identical, spec " + TOLD_PIN.lite_spec_sha256.slice(0, 12), m.matched + " of " + (lock?.files ?? "?") + " identical, spec " + (lock?.spec_sha256 ?? "none").slice(0, 12) + ", lock sha256 " + (lock?.lock_sha256 ?? "none") + (lock?.lock_sha256 === after.lite.lock_sha256 ? " (= #202's)" : " (#202's " + after.lite.lock_sha256 + ")") + (m.problems.length ? "; " + m.problems.slice(0, 3).join("; ") : ""));
EOF
rule "A5a gstack-modules render at the pin" "exit 0" "exit $c" test "$c" = 0
drive told bun "$W/drv/told.ts"
docker image inspect "$IMAGE" > /dev/null 2>&1 || docker pull -q "$IMAGE" > "$W/log/pull.txt" 2>&1
(cd "$E" && GSTACK_EVALS_DOCKER=1 OUTCOME_TOLD_HOME="$H" OUTCOME_GSTACK_PIN="$H/.claude/skills/gstack" bun test tests/outcome-skills/told.test.ts tests/outcome-skills/autoplan.test.ts) > "$W/log/pin-tests.txt" 2>&1; c=$?
PT="$(grep -Eo '^ *[0-9]+ (pass|skip|fail)' "$W/log/pin-tests.txt" | tr -s ' ' | tr '\n' ',' | sed 's/^ //')"
rule "A6 O1's pin tests with OUTCOME_TOLD_HOME and OUTCOME_GSTACK_PIN set (told.test.ts, autoplan.test.ts)" "exit 0, 0 fail, 0 skip" "exit $c,$PT" sh -c "test $c = 0 && grep -Eq '^ *0 fail' '$W/log/pin-tests.txt' && ! grep -Eq '^ *[1-9][0-9]* skip' '$W/log/pin-tests.txt'"

echo; echo "## Part C: the clean room (cleanRoomPreflight)"
ID=$(docker image inspect --format '{{.Id}}' "$IMAGE" 2>/dev/null)
echo "stand-in image $IMAGE, id $ID"
export IMAGE ID
cat > "$W/drv/clean.ts" <<'EOF'

cat > "$W/drv/spend.ts" <<'EOF'

import { readFileSync } from "node:fs";
