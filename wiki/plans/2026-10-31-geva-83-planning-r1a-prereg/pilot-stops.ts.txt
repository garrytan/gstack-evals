// R1a pilot stops (b), (c) and (e) (GSTA-2, 2026-11-01): the three section 8 readings no pinned command prints, computed
// from the files the pinned runner and noloss write. Keyless; it prints counts and shares and never a unit id.
//   bun pilot-stops.ts <scorer checkout at e305ed0> <pilot review-units out dir> <adjudicated pilot arms dir, judge-join's
//   --out for the adjudication batch>
// (b) Every ask and prose event in the journaled sessions' runs/<model>/<arm>/<task>/p<n>/host/usersim.ndjson, the user
//     simulator's log, which tags each reply's source (src/adapters/claude-code/driver.ts:371, src/usersim/sim.ts:200 and
//     :224): the share whose source is fallback. A run directory no journal line names is not read. Fires above 0.10.
// (c) The particle-1 pilot sessions the pinned planScore (src/trial/review/run.ts:172-177) scores as a miss: a missing or
//     malformed plan-findings block, or a session that ended incomplete after model output. A gap (an infra failure, or a
//     session that ended before any model output) is never a failure, so an infra outage cannot drop the rung (GSTA-2,
//     2026-11-01, narrowing E2). Each of parses.tsv's 12 particle-1 rows is scored from its own cells.ndjson line. Fires
//     above 0.20, which is 3 or more of the 12.
// (e) The six Opus and particle-1 arms of the pilot, none, gstack and lens on Opus and none, gstack and placebo on
//     particle-1, by arm id (<arm>-<model tag>, src/trial/review/run.ts:157-162), each with its 4 findings files: the
//     must-fix findings blind adjudication ruled not-defect (the not-defect rows of the arm's adjudication.tsv), divided by
//     the arm's plans times its passes. adjudication.tsv rules only the blocking findings matching left unmatched
//     (lanes/review-noloss/contract.ts:5 and :364), so an arm with none has no adjudication.tsv and reads 0. Fires above
//     0.25 in any arm.
// It refuses, and prints no reading, when the scorer checkout is not at e305ed0 with no tracked change; when any session in
// the pilot's cells.ndjson scores gap (the stops are read only once the pilot has no gap left, after its one resume); when
// parses.tsv does not hold exactly 12 particle-1 rows; when parses.tsv and cells.ndjson do not hold exactly the pilot's 32
// sessions (8 cells x 4 plans x 1 pass), each parses.tsv row naming its own cells.ndjson line, so a cell that never ran
// (a cap stop or a killed resume) refuses too (GSTA-2, 2026-11-01); when no journaled session left a usersim.ndjson; or
// when one of the six arm ids is missing or holds other than 4 findings files.
import { execFileSync } from "node:child_process";
import { existsSync, readdirSync, readFileSync } from "node:fs";
import { join, resolve } from "node:path";

const PIN = "e305ed087b3f17bd211287b0e2d5e3f3960b5a4f";
const P1 = "particle-ai/particle-1", P1_SESSIONS = 12, PILOT_SESSIONS = 32, PLANS = 4;
const ARMS = ["none-opus", "gstack-opus", "lens-opus", "none-particle-1", "gstack-particle-1", "placebo-particle-1"];
const fail = (message: string): never => {
  console.error("E_USAGE " + message);
  process.exit(2);
};
const [scorer, out, arms] = process.argv.slice(2).map((p) => resolve(p));
if (!scorer || !out || !arms) fail("usage: bun pilot-stops.ts <scorer checkout at e305ed0> <pilot out dir> <adjudicated pilot arms dir>");
const git = (...args: string[]) => {
  try {
    return execFileSync("git", ["-C", scorer!, ...args], { encoding: "utf8", stdio: ["ignore", "pipe", "ignore"] }).trim();
  } catch {
    return null;
  }
};
const head = git("rev-parse", "HEAD");
if (head !== PIN) fail("the scorer checkout is at " + (head ?? "no git commit") + ", not " + PIN + "; stop (c) scores sessions with the pinned planScore only");
if (git("status", "--porcelain", "--untracked-files=no") !== "") fail("the scorer checkout at " + PIN + " has tracked changes; use a clean checkout");
const { planScore, readReviewJournal, reviewCellId } = await import(join(scorer!, "src/trial/review/run.ts"));

const fired: string[] = [];
const verdict = (fires: boolean, stop: string) => {
  if (fires) fired.push(stop);
  return fires ? "fires" : "does not fire";
};
const tsv = (path: string) => {
  const [head, ...rows] = readFileSync(path, "utf8").split("\n").filter((l) => l.trim());
  const cols = head!.split("\t");
  return rows.map((r) => Object.fromEntries(r.split("\t").map((v, i) => [cols[i], v])));
};
for (const f of ["cells.ndjson", "parses.tsv"]) if (!existsSync(join(out!, f))) fail(f + " is missing from " + out);
const journal = readReviewJournal(out!);
const score = (e: unknown): string => {
  try {
    return planScore(e);
  } catch {
    return fail("a cells.ndjson line has no record or parse the pinned planScore can read");
  }
};
const gaps = [...journal.values()].filter((e) => score(e) === "gap").length;
if (gaps) fail(gaps + " of the pilot's " + journal.size + " journaled sessions score gap by the pinned planScore; the stops are read only once the pilot has no gap left, after its one resume (section 10)");
const rows = tsv(join(out!, "parses.tsv"));
const cellOf = (r: Record<string, string>) => reviewCellId({ model: r.model, arm: r.arm, unit: r.unit, pass: Number(r.pass) }) as string;
const p1Cells = rows.filter((r) => r.model === P1).map(cellOf);
if (p1Cells.length !== P1_SESSIONS) fail("parses.tsv holds " + p1Cells.length + " particle-1 rows, not the pilot's " + P1_SESSIONS + " (3 arms x 4 plans x 1 pass)");
const named = new Set(rows.map(cellOf));
if (rows.length !== PILOT_SESSIONS || named.size !== PILOT_SESSIONS || journal.size !== PILOT_SESSIONS || [...named].some((c) => !journal.has(c))) fail("parses.tsv holds " + rows.length + " rows and cells.ndjson " + journal.size + " sessions; the readings need exactly the pilot's " + PILOT_SESSIONS + " sessions (8 cells x 4 plans x 1 pass), each parses.tsv row naming its own cells.ndjson line, so a cell that never ran refuses them");
const logs = [...journal.values()].map((e: any) => {
  if (typeof e.record?.task !== "string" || !e.record.task) fail("a cells.ndjson line has no record.task, so its run directory cannot be found");
  return join(out!, "runs", e.cell.model.replaceAll("/", "_"), e.cell.arm, e.record.task, "p" + e.cell.pass, "host", "usersim.ndjson");
}).filter((p) => existsSync(p));
if (!logs.length) fail("no journaled session in " + out + " left host/usersim.ndjson, so stop (b) cannot be read");

for (const a of ARMS) {
  if (!existsSync(join(arms!, a, "arm.json"))) fail("arm " + a + " is missing from " + arms + "; stop (e) reads all six Opus and particle-1 arms");
  const files = existsSync(join(arms!, a, "findings")) ? readdirSync(join(arms!, a, "findings")).filter((f) => f.endsWith(".jsonl")).length : 0;
  if (files !== PLANS) fail("arm " + a + " holds " + files + " findings files, not the pilot's " + PLANS);
}

let events = 0, fallback = 0;
for (const log of logs)
  for (const line of readFileSync(log, "utf8").split("\n").filter((l) => l.trim())) {
    const e = JSON.parse(line);
    if (e.kind !== "ask" && e.kind !== "prose") continue;
    events++;
    if (e.source === "fallback") fallback++;
  }
const bShare = events ? fallback / events : 0;
console.log("(b) user simulator: " + events + " ask and prose events in " + logs.length + " journaled session logs, " + fallback + " fallback, share " + bShare.toFixed(3) + " (stop above 0.10): " + verdict(bShare > 0.1, "b"));

const misses = p1Cells.filter((c) => score(journal.get(c)) === "miss").length;
const cShare = misses / P1_SESSIONS;
console.log("(c) the pilot's cells.ndjson holds " + journal.size + " sessions and no gap; particle-1: " + P1_SESSIONS + " pilot sessions, " + misses + " scored miss by the pinned planScore, share " + cShare.toFixed(3) + " (stop above 0.20): " + verdict(cShare > 0.2, "c"));

let worst = 0;
for (const a of ARMS) {
  const arm = JSON.parse(readFileSync(join(arms!, a, "arm.json"), "utf8"));
  const adjudicated = existsSync(join(arms!, a, "adjudication.tsv"));
  const notDefect = adjudicated ? tsv(join(arms!, a, "adjudication.tsv")).filter((r) => r.ruling === "not-defect").length : 0;
  const rate = notDefect / (PLANS * arm.passes);
  worst = Math.max(worst, rate);
  console.log("(e) " + a + ": " + notDefect + " must-fix findings ruled not-defect over " + PLANS + " plans x " + arm.passes + " pass" + (arm.passes === 1 ? "" : "es") + " = " + rate.toFixed(3) + " per plan per pass" + (adjudicated ? "" : " (no adjudication.tsv: no must-fix finding was left unmatched)"));
}
console.log("(e) highest rate " + worst.toFixed(3) + " (stop above 0.25 in any cell): " + verdict(worst > 0.25, "e"));
console.log("stops fired: " + (fired.length ? fired.join(", ") : "none"));
