// R1a keyless dry run (GEVA-83, planning lane B8, plan 2026-10-31-geva-79-planning-lane-plan). Runs every frozen rule of
// the R1a preregistration through the pinned scorer on the sealed panel's real suites with synthetic arms, keyless:
//   bun dryrun.ts <scorer checkout> <sealed root> <source clone> <post dir> <scratch dir>
// The stub must sit in the scorer checkout at lanes/review-noloss/suites/planrev-ceo-r1a.json. Every finding, judgment
// and ruling here is synthetic. The transcript keeps each step's command (paths shortened), exit code and the scorer's
// summary lines, drops every line that names a unit, and ends with a leak check over every unit's head and base. The
// ledger checks run on a temporary approval record and local ledger under the scratch dir, with the approval mirror stubbed.

const check = (label: string, ok: boolean) => {

const show = (text: string, max = 40) => {
  const kept = text.split("\n").filter((l) => l.trim() && !UNIT.test(l) && !/^\|\s*\S+\s*\|/.test(l.trim()) || /\| planrev/.test(l));
  for (const l of kept.slice(0, max)) say("    " + l);
};
async function cli(label: string, argv: string[], o: { show?: number; display?: string[] } = {}) {
  const out: string[] = [], err: string[] = [];
  const code = await nolossCli(argv, { out: (t: string) => out.push(t), err: (t: string) => err.push(t) });
  say("$ noloss " + (o.display ?? argv).map((a) => (/\s/.test(a) ? JSON.stringify(a) : a)).join(" "));
  say("  exit " + code + (label ? "  (" + label + ")" : ""));
  if (o.show !== 0) show(out.join("\n") + "\n" + err.join("\n").split("\n").filter((l) => /^E_[A-Z_]+ /.test(l) && !/context:|fix:|docs:|correlation:/.test(l)).join("\n"), o.show ?? 40);
  return { code, out: out.join("\n"), err: err.join("\n") };
}
function units(argv: string[], label: string, show_ = 30) {
  const r = spawnSync(join(scorer, "bin/gstack-evals"), ["review-units", ...argv], { encoding: "utf8", cwd: scorer, env: Object.fromEntries(Object.entries(process.env).filter(([k]) => !/^(ANTHROPIC|OPENAI|OPENROUTER)_API_KEY$/.test(k))) });
  say("$ review-units " + argv.join(" "));
  say("  exit " + r.status + "  (" + label + ")");
  show((r.stdout ?? "") + "\n" + (r.stderr ?? "").split("\n").filter((l) => /^E_[A-Z_]+ |^Usage/.test(l) && !/context:|fix:|docs:|correlation:/.test(l)).join("\n"), show_);
  return { code: r.status, out: r.stdout ?? "", err: r.stderr ?? "" };
}
const draw = (...parts: unknown[]) => parseInt(createHash("sha256").update(parts.join("|")).digest("hex").slice(0, 8), 16) / 0x100000000;
const RATE: Record<string, [number, number]> = { [id("none", OPUS)]: [0.3, 0.1], [id("gstack", OPUS)]: [0.65, 0.15], [id("lens", OPUS)]: [0.35, 0.1], [id("none", P1)]: [0.15, 0.05], [id("gstack", P1)]: [0.4, 0.1], [id("placebo", P1)]: [0.15, 0.05], [id("none", FABLE)]: [0.35, 0.1], [id("gstack", FABLE)]: [0.6, 0.15], [id("flagall", OPUS)]: [1, 1], [id("flagbait", OPUS)]: [1, 1] };
const isDecoy = (u: any) => u.blockers.every((b: any) => b.counts === "over-blocking");
const save = (name: string, value: unknown) => {
  const p = join(OUT, name);
  writeFileSync(p, typeof value === "string" ? value : JSON.stringify(value, null, 2) + "\n");
  return p;
};

async function judged(name: string, suite: any, suiteArg: string, cells: string[], passes: number, adjudicate: boolean, diag: Record<string, string[]> = {}, extra: Record<string, number> = {}) {
  const raw = join(OUT, name, "raw"), rawRoot = join(OUT, name, "raw-root"), judgedDir = join(OUT, name, "judged"), adjDir = join(OUT, name, "adjudicated"), root = join(OUT, name, "arms");
  for (const c of cells) {
    const [hit, bait] = RATE[c]!;
    writePlanArm(join(raw, c), suite, { id: c, passes, raw: true, noise: c.startsWith("flagall") ? 3 : 0, outcome: (u: any, b: any, pass: number) => (draw(c, u.pr, pass, b.id) < (isDecoy(u) ? bait : hit) ? "F" : draw(c, u.pr, pass, b.id, "d") < 0.25 ? "D" : "M") });
    for (const u of suite.units.slice(0, extra[c] ?? 0)) {
      const line = [1, 2, 4, 5].find((n) => !u.blockers.some((b: any) => b.line === n));
      appendFileSync(join(raw, c, "findings", u.pr + ".jsonl"), JSON.stringify({ id: "p1-x1", severity: "Critical", path: "plan.md", line, summary: "Synthetic finding that names no flaw.", section: "Users", claim: "Synthetic finding that names no flaw.", evidence: "plan.md and context/goals.md", fix: "Synthetic.", pass: 1 }) + "\n");
    }
    if (diag[c]) {
      const a = JSON.parse(readFileSync(join(raw, c, "arm.json"), "utf8"));
      writeFileSync(join(raw, c, "arm.json"), JSON.stringify({ ...a, diagnostics: diag[c] }, null, 2) + "\n");
    }
    const r = await nolossCli(["ingest", join(raw, c), "--suite", suiteArg, "--root", rawRoot, ...S], { out: () => {}, err: () => {} });
    check(name + ": raw arm " + c + " ingests", r === 0);
  }
  const armDir = (r: string, c: string) => join(r, suite.id, c);
  const join_ = async (keyPath: string, packets: string, arms: string[], outDir: string, item: "blockers" | "findings") => {
    const key = JSON.parse(readFileSync(keyPath, "utf8"));
    const all = key.packets.map((p: any) => {
      const pk = JSON.parse(readFileSync(join(packets, p.file), "utf8"));
      const items = item === "blockers" ? pk.blockers.map((b: any) => ({ id: b.id, matches: pk.findings.filter((f: any) => f.line === b.line).map((f: any) => f.id) })) : pk.findings.map((f: any) => ({ id: f.id, ruling: "not-defect" }));
      return { p, items };
    });
    const body = (judge: string, only?: Set<string>) => ({ batch: key.batch, judge, packets: all.map(({ p, items }: any) => ({ file: p.file, sha256: p.sha256, [item]: items.filter((x: any) => !only || only.has(p.file + " " + x.id)) })).filter((p: any) => p[item].length) });
    const named = (text: string) => new Set([...text.matchAll(/(packets\/\S+\.json) ([BF]\d+)/g)].map((m) => m[1] + " " + m[2]));
    const args = ["judge-join", "--suite", suiteArg, "--key", keyPath, "--out", outDir, ...arms.flatMap((a) => ["--arm", a]), ...S];
    const replies = ["--reply", save(name + "-" + item + "-primary.json", body("judge one"))];
    const attempt = async () => {
      const o: string[] = [], e: string[] = [];
      const code = await nolossCli([...args, ...replies], { out: (t: string) => o.push(t), err: (t: string) => e.push(t) });
      return { code, out: o.join("\n"), err: e.join("\n") };
    };
    let r = await attempt();
    for (const [flag, judge] of [["--second", "judge two"], ["--third", "judge three"]] as const) {
      if (r.code === 0) break;
      say("  judge-join waits: " + (r.err.split("\n")[0] ?? "").replace(/:.*$/, ""));
      replies.push(flag, save(name + "-" + item + flag + ".json", body(judge, named(r.err))));
      r = await attempt();
    }
    say("$ noloss " + [...args, ...replies].join(" "));
    say("  exit " + r.code + "  (" + name + ": judge-join " + item + " for every arm at once)");
    show(r.out + "\n" + r.err.split("\n").filter((l) => /^E_[A-Z_]+ /.test(l) && !/context:|fix:|docs:|correlation:/.test(l)).join("\n"), 3);
    return r;
  };
  const packets = join(OUT, name, "packets"), keyPath = join(OUT, name, "key.json");
  const p = await cli(name + ": blind packets over every arm, plan trees from the source clone", ["judge-packet", "--suite", suiteArg, ...cells.flatMap((c) => ["--arm", armDir(rawRoot, c)]), "--seed", "20261031", "--out", packets, "--key", keyPath, "--source", "panel/plans=" + SRC, ...S], { show: 3 });
  check(name + ": judge-packet exit 0", p.code === 0);
  const j = await join_(keyPath, packets, cells.map((c) => armDir(rawRoot, c)), judgedDir, "blockers");
  check(name + ": judge-join exit 0", j.code === 0);
  let from = judgedDir;
  if (adjudicate) {
    const adjPackets = join(OUT, name, "adj-packets"), adjKey = join(OUT, name, "adj-key.json");
    const a = await cli(name + ": blind adjudication packets", ["judge-packet", "--suite", suiteArg, ...cells.flatMap((c) => ["--arm", join(judgedDir, c)]), "--seed", "20261101", "--out", adjPackets, "--key", adjKey, "--adjudicate", "--source", "panel/plans=" + SRC, ...S], { show: 3 });
    check(name + ": adjudication packets build, or there is nothing to adjudicate", a.code === 0 || /nothing to adjudicate/.test(a.err));
    if (existsSync(adjKey)) {
      const k = await join_(adjKey, adjPackets, cells.map((c) => join(judgedDir, c)), adjDir, "findings");
      check(name + ": adjudication join exit 0", k.code === 0);
      from = adjDir;
    } else say("  no must-fix finding is left unmatched, so there is no adjudication batch");
  }
  for (const c of cells) check(name + ": judged arm " + c + " ingests", (await nolossCli(["ingest", join(from, c), "--suite", suiteArg, "--root", root, ...S], { out: () => {}, err: () => {} })) === 0);
  return { root, from, armDir: (c: string) => armDir(root, c) };
}

const outcome = (r: { out: string; err: string; code: number }) => (r.out.match(/verdict: (\S+)( NOLOSS_UNDERPOWERED)?/) ?? [])[0] ?? (r.err.match(/^E_[A-Z_]+/m) ?? ["?"])[0];
async function score(label: string, file: string, v: string, r: string, expect: string, o: { suite?: string; root: string; extra?: string[] }) {
  const argv = ["score", "--suite", o.suite ?? "planrev-ceo-r1a", "--root", o.root, "--prereg", file, "--variant", v, "--reference", r, ...S, ...(o.extra ?? [])];
  const s = await cli(label, argv, { show: 0 });
  show(s.out.split("\n").filter((l) => /verdict:|Holm|claim|false alarm|recall|on \d+ blockers|NOLOSS|delta|FALSE_ALARM/i.test(l)).join("\n") + "\n" + s.err.split("\n").filter((l) => /^E_[A-Z_]+ /.test(l) && !/context:|fix:|docs:|correlation:/.test(l)).join("\n"), 14);
  const e = await nolossCli([...argv, "--expect", expect], { out: () => {}, err: () => {} });
  check(label + ": --expect " + expect, e === 0);
  return s;
}
async function derive(label: string, name: string, file: string, root: string, extra: string[], shown?: string[]) {
  const out = join(OUT, "derived", name);
  const head = ["derive", "--suite", "planrev-ceo-r1a", "--comparisons", file, "--root", root, "--id", name, "--out", out];
  const d = await cli(label, [...head, ...extra, ...S], { show: 4, display: [...head, ...(shown ?? extra), ...S] });
  check(label + ": derive exit 0", d.code === 0);
  return { suite: join(out, name + ".json"), root: join(out, "arms"), file: join(out, "comparisons." + name + ".json") };
}
