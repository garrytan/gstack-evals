// R1a false-alarm gate size (GEVA-96, 2026-11-01). How often does an arm whose must-fix false alarms on decoys match
// none's pass the frozen gate? Keyless and synthetic: no panel material is read.
//   bun gate-size.ts <scorer checkout> [sims]
// Each simulated reading uses the pinned scorer's own clusteredPairedDelta with the comparisons files' seed and draws,
// paired by decoy with each decoy its own cluster, and the gate's own rule (lanes/review-noloss/score.ts falseAlarmGate):
// the reference's excess per decoy, its 95% interval rounded to 4 places, PASS when the upper end is at or under the margin.
// Both arms draw from the same per-decoy rates, so the true excess is zero. Per decoy and pass an arm's must-fix false
// alarms are the bait flagged must-fix (Bernoulli, rate p_d) plus other must-fix findings ruled not-defect (count, mean m_d).
// p_d ~ Beta with mean p and concentration 2, and m_d = m x Gamma(shape 2, scale 0.5), are shared by both arms. The other
// findings are Poisson, or overdispersed (negative binomial, shape 1) per pass.
import { join, resolve } from "node:path";
const scorer = resolve(process.argv[2] ?? ".");
const SIMS = Number(process.argv[3] ?? 2000);
const { clusteredPairedDelta } = await import(join(scorer, "src/runner/stats/paired.ts"));
const SEED = 20261031, DRAWS = 10000;
let s = 0x9e3779b9 ^ 20261101;
const rand = () => {
  s = (s + 0x6d2b79f5) | 0;
  let t = Math.imul(s ^ (s >>> 15), 1 | s);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
const normal = () => Math.sqrt(-2 * Math.log(1 - rand())) * Math.cos(2 * Math.PI * rand());
const gamma = (k: number): number => {
  if (k < 1) return gamma(k + 1) * Math.pow(rand(), 1 / k);
  const d = k - 1 / 3, c = 1 / Math.sqrt(9 * d);
  for (;;) {
    let x = 0, v = 0;
    do { x = normal(); v = 1 + c * x; } while (v <= 0);
    v = v * v * v;
    const u = rand();
    if (u < 1 - 0.0331 * x ** 4 || Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
  }
};
const beta = (a: number, b: number) => { const x = gamma(a); return x / (x + gamma(b)); };
const poisson = (m: number) => { if (m <= 0) return 0; const L = Math.exp(-m); let k = 0, p = 1; do { k++; p *= rand(); } while (p > L); return k - 1; };
const round = (x: number, d = 4) => Math.round(x * 10 ** d) / 10 ** d;

function passRate(D: number, P: number, p: number, m: number, over: boolean, margins: number[]): number[] {
  const passed = margins.map(() => 0);
  for (let i = 0; i < SIMS; i++) {
    const vals = [];
    for (let d = 0; d < D; d++) {
      const pd = p === 0 ? 0 : beta(2 * p, 2 * (1 - p));
      const md = m === 0 ? 0 : m * gamma(2) * 0.5;
      const arm = () => {
        let n = 0;
        for (let k = 0; k < P; k++) n += (rand() < pd ? 1 : 0) + poisson(over ? md * gamma(1) : md);
        return n / P;
      };
      vals.push({ id: "d" + d, cluster: "d" + d, a: arm(), b: arm() });
    }
    const r = clusteredPairedDelta(vals, { seed: SEED, draws: DRAWS });
    const hi = r.ci95 ? round(r.ci95[1]) : null;
    margins.forEach((g, j) => { if (hi !== null && hi <= g + 1e-9) passed[j]++; });
  }
  return passed.map((x) => x / SIMS);
}

const designs = [
  { name: "headline suite (primary, secondary, secondary-opus): 12 decoys, 3 passes", D: 12, P: 3 },
  { name: "Fable subset: 6 decoys, 1 pass", D: 6, P: 1 },
];
const margins = [0.5, 1];
console.log("# R1a false-alarm gate size: share of matched readings that PASS (" + SIMS + " simulated readings per row, seed " + SEED + ", " + DRAWS + " draws each)");
console.log("p: bait flagged must-fix per decoy per pass (mean over decoys). m: other must-fix findings ruled not-defect per decoy per pass.");
for (const z of designs) {
  console.log("\n## " + z.name + "\n");
  console.log("| p | m | other findings | pass at margin 0.5 | pass at margin 1 |");
  console.log("|---|---|---|---|---|");
  for (const over of [false, true]) for (const p of [0.1, 0.3, 0.5]) for (const m of [0, 0.1, 0.25, 0.5, 0.75, 1, 1.5, 2]) {
    if (m === 0 && over) continue;
    const [a, b] = passRate(z.D, z.P, p, m, over, margins);
    console.log("| " + p + " | " + m + " | " + (m === 0 ? "none" : over ? "overdispersed" : "Poisson") + " | " + (a * 100).toFixed(1) + "% | " + (b * 100).toFixed(1) + "% |");
  }
}
