#!/usr/bin/env bun
// dryrun.ts: the keyless dry run of the trust-but-verify eval's frozen rules (GEVA-313). It drives every command of
// tbv-rule.ts on synthetic readings and opens and settles one ledger lease per frozen item on a temporary ledger.
//   env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY bun dryrun.ts <scorer checkout> <scratch dir>
// Run it from a directory holding the bundle's files with tbv-rule.ts.txt and dryrun.ts.txt renamed to drop .txt.
import { randomBytes } from "node:crypto";
import { mkdirSync, rmSync, writeFileSync } from "node:fs";
import { join, resolve } from "node:path";
import { spawnSync } from "node:child_process";

const [scorer, OUT] = process.argv.slice(2).map((p) => resolve(p)) as [string, string];
const HERE = import.meta.dir;
const HOME = process.env.HOME ?? "";
const KEYLESS = Object.fromEntries(Object.entries(process.env).filter(([k]) => !/^(ANTHROPIC|OPENAI|OPENROUTER)_API_KEY$/.test(k))) as Record<string, string>;
rmSync(OUT, { recursive: true, force: true });
mkdirSync(OUT, { recursive: true });
const lines: string[] = [];
let pass = 0, fail = 0;
const scrub = (t: string) => t.replaceAll(OUT, "$OUT").replaceAll(scorer, "$SCORER").replaceAll(HERE, "$BUNDLE").replaceAll(HOME, "~");
const say = (t: string) => lines.push(scrub(t));
const check = (label: string, ok: boolean) => { ok ? pass++ : fail++; say((ok ? "PASS " : "FAIL ") + label); };
const section = async (title: string, fn: () => Promise<void> | void) => {
  say("\n## " + title);
  try { await fn(); } catch (e) { check(title + ": ran to the end (threw " + (e instanceof Error ? e.message.split("\n")[0] : String(e)) + ")", false); }
};
let n = 0;
const run = (cmd: string, data: unknown) => {
  const f = join(OUT, cmd + "-" + ++n + ".json");
  writeFileSync(f, JSON.stringify(data));
  return spawnSync("bun", [join(HERE, "tbv-rule.ts"), cmd, f], { encoding: "utf8", env: { ...KEYLESS, GSTACK_EVALS_SCORER: scorer }, maxBuffer: 1 << 26 });
};
const rule = (cmd: string, data: unknown) => { const r = run(cmd, data); if (r.status !== 0) throw new Error("tbv-rule.ts " + cmd + " exited " + r.status + ": " + r.stderr.trim().split("\n").slice(-1)[0]); return r.stdout; };
const verdict = (o: string) => (o.split("\n").find((l) => l.startsWith("Verdict: ")) ?? "").slice(9);
const show = (o: string, keep: RegExp) => { for (const l of o.split("\n").filter((l) => keep.test(l))) say("    " + l); };

// A deterministic uniform draw from string parts (FNV-1a).
const u = (...p: (string | number)[]) => { let h = 0x811c9dc5; for (const c of p.join("|")) { h ^= c.charCodeAt(0); h = Math.imul(h, 0x01000193) >>> 0; } return (h % 100000) / 100000; };

type Opt = { nA?: number; nB?: number; clean?: number; facts?: number };
function tasksOf(o: Opt) {
  const t: any[] = [];
  const nA = o.nA ?? 50, nB = o.nB ?? 50, clean = o.clean ?? 0;
  for (let i = 0; i < nA + nB; i++) t.push({ id: "t" + i, cluster: "c" + (i % 5 === 1 ? i - 1 : i), source: i < nA ? "A" : "B", stratum: i % 3 === 0 ? "long" : i % 3 === 1 ? "short" : "other", nfacts: o.facts ?? 3 + (i % 6) });
  for (let i = 0; i < clean; i++) t.push({ id: "k" + i, cluster: "k" + i, source: i % 2 ? "B" : "A", stratum: "clean", nfacts: 0 });
  return t;
}
let rid = 0;
function reading(task: any, family: string, arm: string, model: string, passNo: number, rate: number, x: Record<string, unknown> = {}) {
  const facts = [...Array(task.nfacts)].map((_, i) => { const v = u(task.id, family, arm, model, passNo, i) < rate; return { in: v, any: v || u("any", task.id, i) < 0.3 }; });
  const r: any = { id: "r" + ++rid, task: task.id, family, arm, model, pass: passNo, status: "scored", judge: model === "opus" ? "sol" : "opus", facts, unsupported: 0, exposed: arm === "none" || arm === "lens" ? null : true, words: 200, usd: model === "opus" ? 0.4 : model === "sol" ? 0.2 : 0.02, cacheRead: 0.8 };
  if (task.stratum === "long") r.list = { named: rate > 0.6, order: true, rest: rate > 0.6, offer: rate > 0.6 };
  if (task.stratum === "short") r.shortShare = rate;
  if (u("dbl", r.id) < 0.2 && task.nfacts) r.second = { judge: r.judge === "sol" ? "opus" : "sol", facts: facts.map((f) => ({ ...f })) };
  return Object.assign(r, x);
}
type Rates = Record<string, number>;
function held(rates: Rates, o: { stage?: string; models?: string[]; passes?: number; capy?: boolean; mutate?: (r: any) => void; clean?: number; devGap?: number } = {}) {
  rid = 0;
  const tasks = tasksOf({ clean: o.clean ?? 10 });
  const models = o.models ?? ["opus", "sol", "particle"];
  const rs: any[] = [];
  for (const m of models) for (const arm of ["none", "current", "principle"]) for (const t of tasks) for (let p = 1; p <= (o.passes ?? 2); p++) rs.push(reading(t, t.source === "A" ? "investigate" : "ship", arm, m, p, rates[arm + ":" + m] ?? rates[arm]!));
  if (o.capy) for (const arm of ["current", "principle"]) for (const t of tasks.filter((t) => t.source === "B")) for (let p = 1; p <= (o.passes ?? 2); p++) rs.push(reading(t, "capy", arm, "opus", p, rates["capy-" + arm] ?? rates[arm]!));
  if (o.mutate) rs.forEach(o.mutate);
  const controls = tasks.slice(0, 20).flatMap((t) => ["sol", "opus"].flatMap((j) => [{ task: t.id, kind: "stock", judge: j, facts: [...Array(t.nfacts)].map(() => ({ in: false, any: false })) }, { task: t.id, kind: "oracle", judge: j, facts: [...Array(t.nfacts)].map(() => ({ in: true, any: true })) }]));
  return { stage: o.stage ?? "held", tasks, readings: rs, controls, plan: { models, passes: Object.fromEntries(models.map((m) => [m, o.passes ?? 2])), capyPasses: o.passes ?? 2, skillArms: ["none", "current", "principle"], ...(o.devGap === undefined ? {} : { devGap: { opusPrincipleMinusNone: o.devGap, devMdd: 13.1 } }) } };
}
const line = (o: string, re: RegExp) => o.split("\n").find((l) => re.test(l)) ?? "";

say("# Trust-but-verify eval: keyless dry run");
const head = spawnSync("git", ["-C", scorer, "rev-parse", "HEAD"], { encoding: "utf8" }).stdout.trim();

await section("a. The word rule and the cuts", () => {
  const table = "| Test | State |\n|---|---|\n| parse | failed |";
  const code = "~~~ts\nconst x = 1;\n~~~";
  const list = "- one\n- two\n  * three";
  const long = [...Array(300)].map((_, i) => "w" + i).join(" ");
  const o = rule("words", { texts: [table, code, list, long, "-- | *** ---"] });
  show(o, /^words/);
  const ls = o.split("\n");
  check("a table counts only its cells (4 words), not its pipes and rules", ls[0]!.startsWith("words 4,"));
  check("a code block counts tokens with a letter or digit (~~~ts, const, x, 1;), not = or the bare fence", ls[1]!.startsWith("words 4,"));
  check("bullets don't count (3 words)", ls[2]!.startsWith("words 3,"));
  check("a 300-word reply is over 250 and its boundary falls right after word 250", ls[3]!.includes("over 250 true") && ls[3]!.endsWith("before boundary 250"));
  check("punctuation alone is no word (0)", ls[4]!.startsWith("words 0,"));
});

await section("b. End states, in the frozen precedence", () => {
  const T = (kind: string, turn = "report") => ({ kind, turn });
  const cases = [
    [T("text")], [T("5xx"), T("text")], [T("5xx"), T("timeout"), T("broken-stream")], [T("tool-only"), T("text", "followup")], [T("tool-only"), T("tool-only", "followup")], [T("refusal")], [T("4xx")], [T("validator")], [T("closer")], [T("budget")],
    [T("text"), T("budget", "d1")], [T("text"), T("allowance", "d1"), T("text", "d2")], [T("truncated"), T("text")], [T("text"), T("5xx", "d1"), T("5xx", "d1"), T("no-final", "d1"), T("text", "d2")], [T("allowance")], [T("no-final"), T("no-final"), T("text")],
  ];
  const o = rule("endstate", { cases });
  show(o, /^endstate/);
  const ls = o.split("\n");
  const want = ["endstate scored, sends 1", "endstate scored, sends 2", "endstate gap, sends 3", "endstate scored, sends 2", "endstate no-report, sends 2", "endstate no-report, sends 1", "endstate harness, sends 1", "endstate harness, sends 0", "endstate harness, sends 0", "endstate cap, sends 1", "endstate scored, sends 2", "endstate scored, sends 3", "endstate scored, sends 2", "endstate scored, sends 5", "endstate gap, sends 1", "endstate scored, sends 3"];
  const names = ["a text report scores", "a 5xx then text: one retry, scored", "three transport failures: two retries, then a gap", "a tool-only reply gets the one follow-up, which scores", "no text after the follow-up is a no-report", "a refusal is a no-report", "a non-refusal 4xx is a harness failure", "a validator failure stops before any send", "a closer byte-check failure stops before any send", "a cap refusal on the report turn is a cap stop", "a cap refusal on D1: the item stops, the report stands and D1 drops", "an allowance refusal on D1 drops only D1", "a truncated stream is retried as a transport failure", "D1 failing after two retries drops only D1, and D2 still runs", "an allowance refusal on the report turn is a gap", "two missing final responses are retried and the third send scores"];
  want.forEach((w, i) => check(names[i]! + " (" + w.slice(9) + ")", ls[i]!.startsWith(w)));
  check("the D1 cap case names the item stop", ls[10]!.includes("the item stops"));
});

await section("c. Round admission against a cap", () => {
  check("settled $50 + reserved $5 + unknown $2 + a round's worst case $7.70 fits a $65 cap", rule("admit", { settled: 50, reserved: 5, unknown: 2, roundWorst: 7.7, cap: 65 }).trim() === "admit true");
  check("the same round is refused under a $64 cap (unknown charges count at their reservation)", rule("admit", { settled: 50, reserved: 5, unknown: 2, roundWorst: 7.7, cap: 64 }).trim() === "admit false");
});

function stage1(o: { recallFacts?: number; facts?: number; rate?: number; drop?: number; gaps?: number; stock?: number; oracle?: number; badJudge?: string; disagree?: boolean; harness?: boolean }) {
  rid = 0;
  const tasks = tasksOf({ nA: 10, nB: 10, facts: o.facts });
  const rs: any[] = [];
  let k = 0;
  for (let p = 1; p <= 3; p++) for (const t of tasks) {
    const r = reading(t, t.source === "A" ? "investigate" : "ship", "none", "opus", p, o.rate ?? 0.5);
    if (o.recallFacts !== undefined) { r.facts = r.facts.map((f: any, i: number) => ({ in: i < o.recallFacts!, any: true })); if (r.second) r.second.facts = r.facts.map((f: any) => ({ ...f })); }
    if (o.drop !== undefined && Number(t.id.slice(1)) < o.drop) { r.status = "gap"; }
    if (o.gaps !== undefined && Number(t.id.slice(1)) >= 10 && k++ < o.gaps) r.status = "gap";
    if (o.disagree && r.second) r.second.facts = r.facts.map((f: any) => ({ in: !f.in, any: !f.any }));
    rs.push(r);
  }
  for (const t of tasks.slice(0, 2)) for (let p = 1; p <= 2; p++) rs.push(reading(t, "investigate", "current", "opus", p, 0.5, { usd: 0.45 }));
  for (const t of tasks.slice(0, 4)) { rs.push(reading(t, "investigate", "none", "sol", 1, 0.5, { usd: 0.2 })); rs.push(reading(t, "investigate", "none", "particle", 1, 0.5, { usd: 0.02 })); }
  if (o.harness) rs[5].status = "harness";
  const ctl = tasks.flatMap((t) => ["sol", "opus"].flatMap((j) => [{ task: t.id, kind: "stock", judge: j, facts: [...Array(t.nfacts)].map((_, i) => ({ in: j === o.badJudge && i < Math.round((o.stock ?? 0) * t.nfacts), any: false })) }, { task: t.id, kind: "oracle", judge: j, facts: [...Array(t.nfacts)].map((_, i) => ({ in: j === o.badJudge ? i < Math.round((o.oracle ?? 1) * t.nfacts) : true, any: true })) }]));
  return { stage: "hr", tasks, readings: rs, controls: ctl, plan: { models: ["opus"], passes: { opus: 3 }, skillArms: ["none"] } };
}

await section("d. Stage 1: the headroom rule", () => {
  const o = rule("headroom", stage1({}));
  show(o, /^(Stage 1|Agreement|controls|cost|current smoke|R0|Verdict)/);
  check("plain Opus at about 0.5 reads HEADROOM", verdict(o).startsWith("HEADROOM"));
  check("it prints measured cost and cache-read share for every model and arm path stage 1 ran (Opus none and current, Sol none, particle-1 none)", ["cost none:opus", "cost current:opus", "cost none:sol", "cost none:particle"].every((k) => o.includes(k)));
  const edge = (facts: number, inF: number) => verdict(rule("headroom", stage1({ facts, recallFacts: inF })));
  check("R0 exactly 0.75 (15 of 20 facts) is HEADROOM, the top edge included", edge(20, 15).startsWith("HEADROOM"));
  check("R0 0.80 is CEILING", edge(20, 16).startsWith("CEILING"));
  check("R0 exactly 0.15 (3 of 20) is HEADROOM, the bottom edge included", edge(20, 3).startsWith("HEADROOM"));
  check("R0 0.10 is FLOOR", edge(20, 2).startsWith("FLOOR"));
  check("only 15 readable tasks is NO READ", verdict(rule("headroom", stage1({ drop: 5 }))).startsWith("NO READ"));
  check("7 gaps in 60 planned Opus none readings (over 10%) is NO READ", verdict(rule("headroom", stage1({ gaps: 7 }))).startsWith("NO READ"));
  check("6 gaps (exactly 10%) still reads", !verdict(rule("headroom", stage1({ gaps: 6 }))).startsWith("NO READ"));
  for (const j of ["sol", "opus"]) {
    check("the stock report at 0.2 on the " + j + " judge is JUDGE FAIL", verdict(rule("headroom", stage1({ badJudge: j, stock: 0.2 }))).startsWith("JUDGE FAIL"));
    check("the oracle report at 0.6 on the " + j + " judge is JUDGE FAIL", verdict(rule("headroom", stage1({ badJudge: j, oracle: 0.6 }))).startsWith("JUDGE FAIL"));
  }
  const dis = rule("headroom", stage1({ disagree: true }));
  check("positive agreement under 0.70 on the double read is JUDGE FAIL, FIX-METHOD, with no brief revision bought", verdict(dis).startsWith("JUDGE FAIL") && verdict(dis).includes("no brief revision"));
  check("a harness failure stops stage 1 for a B1 fix", verdict(rule("headroom", stage1({ harness: true }))).startsWith("HARNESS FAILURE"));
});

await section("e. The projection check from measured costs", () => {
  const prices = { opus: { in: 4, out: 20 }, sol: { in: 2, out: 10 }, particle: { in: 0.1, out: 1 } };
  const it = (item: string, rs: any[], judgeN: number) => ({ item, readings: rs, judgeN, inputTokens: { opus: 100000, sol: 100000, particle: 100000 } });
  const measured = { "none:opus": 0.4, "current:opus": 0.45, "none:sol": 0.2, "none:particle": 0.02, "lens:opus": 0.4, "current:sol": 0.22, "current:particle": 0.02 };
  const o = rule("projection", { items: [it("tbv-dev-base", [{ model: "opus", arm: "none", n: 88 }, { model: "opus", arm: "current", n: 132 }, { model: "opus", arm: "lens", n: 88 }, { model: "sol", arm: "none", n: 88 }, { model: "sol", arm: "current", n: 88 }, { model: "particle", arm: "none", n: 88 }, { model: "particle", arm: "current", n: 88 }], 660), it("tbv-climb-1", [{ model: "opus", arm: "principle", n: 132 }], 132)], measured, judgePerReading: 0.07, prices });

  const r = run("projection", { items: [it("x", [{ model: "sol", arm: "principle", n: 10 }], 0)], measured: { "none:opus": 0.4 }, judgePerReading: 0.07, prices });
  check("a later item on a model and arm path stage 1 didn't measure is refused", r.status === 2 && r.stderr.includes("no measured cost for principle:sol"));
});

await section("i. The treatment-diff leak check", () => {
