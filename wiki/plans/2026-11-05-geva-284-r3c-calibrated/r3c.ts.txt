const sha = (b: Buffer | string) => createHash("sha256").update(b).digest("hex");
export const idsSha = (ids: string[]) => sha([...ids].sort().map((i) => i + "\n").join(""));
const pinned = (path: string, want: string) => {
  const raw = readFileSync(path);
  if (sha(raw) !== want) throw new Error("E_PIN_MISMATCH " + path + ": sha256 " + sha(raw) + " is not the pinned " + want + "; fix: copy the record again from the drive path named beside INHAND (or L1_FILES' post folder) and rerun");
  return raw.toString("utf8");
};

export type TrialRow = { part: string; task: string; arm: string; outcome: string; cost_usd?: number | null };
const graded = (t: TrialRow) => t.outcome === "pass" || t.outcome === "fail";
const tsv = (text: string): Record<string, string>[] => {
  const [head, ...lines] = text.trim().split("\n");
  const cols = head!.split("\t");
  return lines.map((l) => { const v = l.split("\t"); return Object.fromEntries(cols.map((c, i) => [c, v[i] ?? ""])); });
};
const tsvRows = (text: string): TrialRow[] => tsv(text).map((o) => ({ part: o.part!, task: o.task!, arm: o.arm!, outcome: o.outcome!, cost_usd: o.cost_usd === "" ? null : Number(o.cost_usd) }));
const jsonlRows = (text: string): TrialRow[] => text.split("\n").filter((l) => l.trim()).map((l) => JSON.parse(l));

// None trials only: R3a4's screen trials, R3b's grid none trials and R3c's calibration trials. Told trials, R3a4's pilot
// and R3b's re-check never choose a task.
export function noneCounts(rows: TrialRow[]): Map<string, Counts> {
  const m = new Map<string, Counts>();
  for (const t of rows) {
    if (t.arm !== "none" || !graded(t) || !/^(screen|grid|cal)-/.test(t.part)) continue;
    const c = m.get(t.task) ?? { s: 0, n: 0 };
    c.s += t.outcome === "pass" ? 1 : 0; c.n += 1; m.set(t.task, c);
  }
  return m;
}
export function screenPass(rows: TrialRow[]): Map<string, boolean> {
  const m = new Map<string, boolean>();
  for (const t of rows) if (t.arm === "none" && t.part.startsWith("screen-") && graded(t)) m.set(t.task, t.outcome === "pass");
  return m;
}

const other = (m: Model): Model => (m === "opus" ? "sol" : "opus");
// The same upstream change under a moved owner (L1's rule): the owner dropped, the repository name and number kept.
export const changeKey = (id: string) => { const i = id.indexOf("__"); return (i < 0 ? id : id.slice(i + 2)).replace(/_interface$/, ""); };

// Frozen tiers over the kept pool. floor: this model has 4 or more none trials and passed none, and the other model
// passed none of its none trials. t1: 4 or more none trials, not floor. t2: one screen trial, which failed, or which
// passed while the other model's screen trial failed. t4: both models passed their screen trial. t3: kept tasks no
// screen drew, less L1's 90 and any task that is the same change as a screen task (out).
export function tiers(h: InHand, m: Model) {
  const own = h.none[m], oth = h.none[other(m)], scr = h.screen[m], oscr = h.screen[other(m)];
  const keys = new Set([...scr.keys()].map(changeKey));
  const out: Record<Tier | "floor" | "out", string[]> = { t1: [], t2: [], t3: [], t4: [], floor: [], out: [] };
  for (const t of h.kept) {
    const c = own.get(t);
    if (!scr.has(t)) { (h.l1.has(t) || keys.has(changeKey(t)) ? out.out : out.t3).push(t); continue; }
    if (c!.n >= 4) { (c!.s === 0 && (oth.get(t)?.s ?? 0) === 0 ? out.floor : out.t1).push(t); continue; }
    if (scr.get(t) && oscr.get(t)) out.t4.push(t); else out.t2.push(t);
  }
  for (const k of Object.keys(out) as (keyof typeof out)[]) out[k].sort();
  return out;
}

// Calibration: fresh none trials per candidate, by tier (cal-1), then R3C.t4Second more for each t4 task whose cal-1
// trial failed (cal-2). They choose tasks and size the grid; no grid contrast reads them.
export function calibrationPlan(tr: ReturnType<typeof tiers>) {
  const cal1 = (["t1", "t2", "t3", "t4"] as Tier[]).flatMap((k) => tr[k].map((task) => ({ task, tier: k, trials: R3C.calTrials[k] })));
  return { cal1, cal1Trials: cal1.reduce((z, x) => z + x.trials, 0) };
}
export function cal2Tasks(tr: ReturnType<typeof tiers>, cal: TrialRow[]): string[] {
  const failed = new Set(cal.filter((t) => /^cal-1/.test(t.part) && t.arm === "none" && t.outcome === "fail").map((t) => t.task));
  return tr.t4.filter((t) => failed.has(t));
}

// mainText: closingBlock's input, the session's main-thread assistant text blocks in order. From Claude Code's message
// stream it takes the text content blocks of assistant messages that carry no parent_tool_use_id; a thinking block, a
// tool call and any subagent message (one with a parent_tool_use_id) are never read. Every arm is fed this way.
export function mainText(stream: { type?: string; parent_tool_use_id?: string | null; message?: { content?: unknown } }[]): string[] {
  return stream.filter((x) => x.type === "assistant" && !x.parent_tool_use_id).flatMap((x): any[] => { const c = x.message?.content; return Array.isArray(c) ? c : []; }).filter((b) => b?.type === "text" && typeof b.text === "string").map((b) => b.text as string);
}
// Localization (report only, ladder D1) reads only mainText's blocks, in order (a list of blocks is joined with a blank
// line between them): no thinking block and no subagent text. The last closing block in that text counts, so a block written
// before a Stop hook continuation still counts and a DEBUG REPORT never does: the last line that, trimmed of
// non-alphanumeric characters at both ends, reads "root cause files" in any case, then the non-empty lines after it up
// to the first blank line, at most three. Each entry loses a list marker, backticks and asterisks; a markdown link
// becomes its target; then it loses everything after its first token, trailing punctuation (a colon included),
// wrapping quotes and brackets, a line reference (:12 or :12:3) and a leading ./ or /testbed/. No block, or no entry,
// is a miss.
export function closingBlock(assistantText: string | string[]): string[] {
  const lines = (typeof assistantText === "string" ? assistantText : assistantText.join("\n\n")).split("\n");
  let at = -1;
  lines.forEach((l, i) => { if (l.replace(/^[^A-Za-z0-9]+|[^A-Za-z0-9]+$/g, "").toLowerCase() === "root cause files") at = i; });
  if (at < 0) return [];
  const out: string[] = [];
  for (const l of lines.slice(at + 1)) {
    if (!l.trim()) { if (out.length) break; continue; }
    const tok = l.replace(/^\s*(?:[-*+]|\d+[.)])\s+/, "").replace(/[`*]/g, "").trim().replace(/^\[[^\]]*\]\(([^)\s]+)\)/, "$1").split(/\s+/)[0] ?? "";
    const p = tok.replace(/[,;.:)]+$/, "").replace(/^["'(\[<]+|["')\]>]+$/g, "").replace(/[,;.:]+$/, "").replace(/(?::\d+)+$/, "").replace(/^\.\//, "").replace(/^\/testbed\//, "");
    if (p) out.push(p);
    if (out.length === 3) break;
  }
  return out;
}
// A task's root-cause files: the files its oracle patch (the oracle gate's model.patch) changes, less test files (a path
// segment test or tests, or a file named test_*.py or *_test.py). A task left with none is out of the measure, counted.
export function goldFiles(patch: string): string[] {
  const files = new Set<string>();
  for (const x of patch.matchAll(/^diff --git a\/(\S+) b\/(\S+)$/gm)) files.add(x[2]!);
  return [...files].filter((f) => !/(^|\/)tests?\//.test(f) && !/(^|\/)(test_[^/]*\.py|[^/]*_test\.py)$/.test(f)).sort();
}
export const locHit = (block: string[], gold: string[], k: number) => block.slice(0, k).some((f) => gold.includes(f));
export const locRank = (block: string[], gold: string[]) => { const i = block.slice(0, 3).findIndex((f) => gold.includes(f)); return i < 0 ? 0 : R3C.locWeights[i]!; };
// Graded secondary (report only): the share of the task's FAIL_TO_PASS tests that passed, under the PASS_TO_PASS gate:
// 0 when any PASS_TO_PASS test did not pass. status maps each test the verifier ran to "pass" when its status is one the
// task's own grading counts toward its reward, and to anything else otherwise; a test missing from it did not pass.
// null, never 1, when the task lists no FAIL_TO_PASS test (out of the measure, counted). A trial whose hidden tests all
// pass scores 1, as its reward does. On the oracle gate, oracleF2p counts how many oracle trials score 1 and how many
// score null, report only: it never stops a part.
export function f2pShare(status: Record<string, string>, f2p: string[], p2p: string[]): number | null {
  if (!f2p.length) return null;
  if (!p2p.every((t) => status[t] === "pass")) return 0;
  return f2p.filter((t) => status[t] === "pass").length / f2p.length;
}
export const oracleF2p = (shares: (number | null)[]) => ({ trials: shares.length, one: shares.filter((s) => s === 1).length, nulls: shares.filter((s) => s === null).length });

// Admission: every none trial of this model on the task (in hand and calibration), at least R3C.minTrials of them, and a
// measured none rate inside the band, both ends included.
export const admitted = (c: Counts | undefined) => !!c && c.n >= R3C.minTrials && c.s >= R3C.band[0] * c.n - 1e-9 && c.s <= R3C.band[1] * c.n + 1e-9;

// The task model: a nonparametric maximum-likelihood mixing law for per-task none chances on the grid i / 40, fitted by
// EM from a uniform start for a fixed number of iterations, so it is deterministic.
export const GRID = Array.from({ length: R3C.npmle.grid + 1 }, (_, i) => i / R3C.npmle.grid);
const lik = (p: number, c: Counts) => Math.pow(p, c.s) * Math.pow(1 - p, c.n - c.s);
export function npmle(data: Counts[], iters = R3C.npmle.iters): number[] {
  let g = GRID.map(() => 1 / GRID.length);
  const L = data.map((c) => GRID.map((p) => lik(p, c)));
  for (let it = 0; it < iters; it++) {
    const acc = GRID.map(() => 0);
    for (const row of L) { let z = 0; for (let j = 0; j < GRID.length; j++) z += g[j]! * row[j]!; for (let j = 0; j < GRID.length; j++) acc[j]! += (g[j]! * row[j]!) / z; }
    g = acc.map((a) => a / data.length);
  }
  return g;
}
export function posterior(g: number[], c: Counts | undefined): number[] {
  const w = GRID.map((p, j) => g[j]! * (c ? lik(p, c) : 1)), z = w.reduce((a, b) => a + b, 0);
  return w.map((x) => x / z);
}
// Report-only second law (the reviewer's finding d3): the beta-binomial maximum-likelihood law, on a log grid of a and b
// from R3C.beta.lo to R3C.beta.hi, put on GRID by the beta's mass in each grid point's cell.
export function lgamma(x: number): number {
  if (x < 0.5) return Math.log(Math.PI / Math.abs(Math.sin(Math.PI * x))) - lgamma(1 - x);
  const c = [0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313, -176.61502916214059, 12.507343278686905, -0.13857109526572012, 9.9843695780195716e-6, 1.5056327351493116e-7];
  x -= 1;
  let a = c[0]!;
  const t = x + 7.5;
  for (let i = 1; i < 9; i++) a += c[i]! / (x + i);
  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(a);
}
function betacf(a: number, b: number, x: number): number {
  const tiny = 1e-300;
  let c = 1, d = 1 - ((a + b) * x) / (a + 1);
  d = 1 / (Math.abs(d) < tiny ? tiny : d);
  let h = d;
  for (let m = 1; m <= 500; m++) {
    for (const aa of [(m * (b - m) * x) / ((a - 1 + 2 * m) * (a + 2 * m)), (-(a + m) * (a + b + m) * x) / ((a + 2 * m) * (a + 1 + 2 * m))]) {
      d = 1 + aa * d; d = 1 / (Math.abs(d) < tiny ? tiny : d);
      c = 1 + aa / c; if (Math.abs(c) < tiny) c = tiny;
      h *= d * c;
    }
    if (Math.abs(d * c - 1) < 1e-13) break;
  }
  return h;
}
export function betai(a: number, b: number, x: number): number {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const bt = Math.exp(lgamma(a + b) - lgamma(a) - lgamma(b) + a * Math.log(x) + b * Math.log(1 - x));
  return x < (a + 1) / (a + b + 2) ? (bt * betacf(a, b, x)) / a : 1 - (bt * betacf(b, a, 1 - x)) / b;
}
export const betaLaw = (a: number, b: number) => GRID.map((p) => betai(a, b, Math.min(1, p + 0.5 / R3C.npmle.grid)) - betai(a, b, Math.max(0, p - 0.5 / R3C.npmle.grid)));
export function betaFit(data: Counts[]): { a: number; b: number } {
  const steps = R3C.beta.grid, l0 = Math.log(R3C.beta.lo), l1 = Math.log(R3C.beta.hi);
  const at = (i: number) => Math.exp(l0 + ((l1 - l0) * i) / steps);
  let best = { a: 1, b: 1, ll: -Infinity };
  for (let i = 0; i <= steps; i++) for (let j = 0; j <= steps; j++) {
    const a = at(i), b = at(j), lb = lgamma(a) + lgamma(b) - lgamma(a + b);
    let ll = 0;
    for (const c of data) ll += lgamma(c.s + a) + lgamma(c.n - c.s + b) - lgamma(c.n + a + b) - lb;
    if (ll > best.ll) best = { a, b, ll };
  }
  return { a: best.a, b: best.b };
}
const inBand = (p: number) => p >= R3C.band[0] - 1e-9 && p <= R3C.band[1] + 1e-9;
export const bandMass = (w: number[]) => w.reduce((z, x, j) => z + (inBand(GRID[j]!) ? x : 0), 0);
const draw = (w: number[], u: () => number) => { let r = u(), j = 0; for (; j < w.length - 1; j++) { r -= w[j]!; if (r <= 0) break; } return GRID[j]!; };

// Forecast (the admission check from data in hand; it decides nothing): draw each candidate's none chance from its
// posterior under the law fitted to this model's none trials, simulate calibration and admission, count what is kept.
function candidatePosteriors(h: InHand, m: Model, law: "npmle" | "beta" = "npmle") {
  const tr = tiers(h, m), own = h.none[m];
  const g = law === "beta" ? (({ a, b }) => betaLaw(a, b))(betaFit([...own.values()])) : npmle([...own.values()]);
  return { tr, g, cands: (["t1", "t2", "t3", "t4"] as Tier[]).flatMap((k) => tr[k].map((t) => ({ t, k, w: posterior(g, own.get(t)) }))) };
}
// One simulated calibration: each candidate's true none chance drawn from its posterior, then cal-1 and cal-2 as frozen.
function simulateCalibration(h: InHand, m: Model, cands: { t: string; k: Tier; w: number[] }[], u: () => number) {
  const own = h.none[m], rows: TrialRow[] = [], truth = new Map<string, number>();
  let kept = 0, band = 0;
  for (const c of cands) {
    const p = draw(c.w, u), base = own.get(c.t) ?? { s: 0, n: 0 };
    truth.set(c.t, p);
    let k = 0, n = R3C.calTrials[c.k];
    for (let i = 0; i < n; i++) { const pass = u() < p; k += +pass; rows.push({ part: "cal-1", task: c.t, arm: "none", outcome: pass ? "pass" : "fail", cost_usd: R3C.cost[m].planNoneUsd }); }
    if (c.k === "t4" && k === 0) for (let i = 0; i < R3C.t4Second; i++) { const pass = u() < p; k += +pass; n++; rows.push({ part: "cal-2", task: c.t, arm: "none", outcome: pass ? "pass" : "fail", cost_usd: R3C.cost[m].planNoneUsd }); }
    if (admitted({ s: base.s + k, n: base.n + n })) { kept++; band += +inBand(p); }
  }
  return { rows, truth, kept, band };
}
export const calSpendUsd = (m: Model, rows: TrialRow[]) => rows.reduce((z, r) => z + (typeof r.cost_usd === "number" ? r.cost_usd : NaN), 0) + rows.length * R3C.vmUsdPerHour * R3C.cost[m].vmHoursNone;

export function forecast(h: InHand, m: Model, sims: number) {
  const { tr, g, cands } = candidatePosteriors(h, m);
  const u = rng(R3C.forecastSeed);
  const ns: number[] = [], trueBand: number[] = [], trials: number[] = [];
  let last: TrialRow[] = [], rateSum = 0, rateN = 0;
  for (let s = 0; s < sims; s++) {
    const c = simulateCalibration(h, m, cands, u);
    ns.push(c.kept); trueBand.push(c.band); trials.push(c.rows.length); last = c.rows;
    for (const [t, k] of noneCounts(c.rows)) { const b = h.none[m].get(t) ?? { s: 0, n: 0 }; if (admitted({ s: b.s + k.s, n: b.n + k.n })) { rateSum += c.truth.get(t)!; rateN++; } }
  }
  const q = (xs: number[], f: number) => [...xs].sort((a, b) => a - b)[Math.floor(f * (xs.length - 1))]!;
  const mean = (xs: number[]) => xs.reduce((a, b) => a + b, 0) / xs.length;
  const exp = (k: Tier) => cands.filter((c) => c.k === k).reduce((z, c) => z + bandMass(c.w), 0);
  return {
    tiers: tr, lawBandMass: bandMass(g), expectedInBand: { t1: exp("t1"), t2: exp("t2"), t3: exp("t3"), t4: exp("t4") },
    admittedMean: mean(ns), admittedP10: q(ns, 0.1), admittedP50: q(ns, 0.5), admittedP90: q(ns, 0.9), inBandShare: mean(trueBand) / Math.max(1e-9, mean(ns)),
    pAtLeastNMin: ns.filter((x) => x >= R3C.nMin).length / sims, calTrialsMean: mean(trials), lastSim: last, admittedRate: rateSum / Math.max(1, rateN),
  };
}

// Ladder (GEVA-292, report only): R3c is /investigate's 2x tier. k is a strict none rate over the forecast's mean true
// none rate on admitted tasks (pooled over simulations), against R3b's grid (in hand) and against R3a's SWE-bench
// Verified tasks. The session-cost ratio is the median cost of the in-hand graded none trials (R3a4's screen, R3b's grid)
// on the candidates over the median of R3b's graded grid none trials.
export function ladderLines(dir: string, m: Model, f: ReturnType<typeof forecast>): string[] {
  const r3b = tsvRows(pinned(join(dir, m + "-r3b.tsv"), INHAND[m + "-r3b.tsv"]!)).filter((t) => /^grid-/.test(t.part) && t.arm === "none" && graded(t));
  const a4 = jsonlRows(pinned(join(dir, m + "-r3a4.jsonl"), INHAND[m + "-r3a4.jsonl"]!));
  const cand = new Set((["t1", "t2", "t3", "t4"] as Tier[]).flatMap((k) => f.tiers[k]));
  const onCand = [...a4, ...r3b].filter((t) => t.arm === "none" && graded(t) && /^(screen|grid)-/.test(t.part) && cand.has(t.task));
  const med = (xs: number[]) => { if (!xs.length || xs.some((x) => !Number.isFinite(x))) return NaN; const v = [...xs].sort((a, b) => a - b), i = v.length >> 1; return v.length % 2 ? v[i]! : (v[i - 1]! + v[i]!) / 2; };
  const s0 = r3b.filter((t) => t.outcome === "pass").length, v = R3C.ladder.verified[m];
  const cb = med(onCand.map((t) => t.cost_usd ?? NaN)), cr = med(r3b.map((t) => t.cost_usd ?? NaN));
  return [
    "ladder: R3c is /investigate's 2x tier (GEVA-292); forecast mean true none rate of admitted tasks " + f.admittedRate.toFixed(3),
    "k against R3b's grid none rate (" + s0 + " of " + r3b.length + ", " + (100 * s0 / r3b.length).toFixed(1) + "%): " + (s0 / r3b.length / f.admittedRate).toFixed(2) + "; k against R3a's SWE-bench Verified none rate (" + v.s + " of " + v.n + ", " + (100 * v.s / v.n).toFixed(1) + "%): " + (v.s / v.n / f.admittedRate).toFixed(2),
    "session cost: median in-hand none trial on the candidates $" + cb.toFixed(3) + " (" + onCand.length + " trials) over R3b's grid none median $" + cr.toFixed(3) + " (" + r3b.length + " trials): " + (cb / cr).toFixed(2),
  ];
}

// Power of the frozen KEEP rule on a grid of the admitted tasks: each task's none chance drawn from its posterior,
// told = min(1, p + d), R replicates per arm, KEEP when the paired task-clustered 95% interval (10,000 draws, seed
// 20261101, the bootstrap analyze()'s contrast runs) is above zero. split (report only): each task's lift is 2d or 0
// with even odds, so the mean lift is d but it sits on half the tasks. The generator is re-seeded per call.
export function keepRate(posts: number[][], R: number, d: number, sims: number, split = false) {
  const u = rng(R3C.power.seed);
  let keep = 0;
  for (let s = 0; s < sims; s++) {
    const none = new Map<string, number>(), told = new Map<string, number>(), tasks: string[] = [];
    posts.forEach((w, i) => {
      const p = draw(w, u), lift = split ? (u() < 0.5 ? 2 * d : 0) : d, q = Math.min(1, p + lift);
      let kn = 0, kt = 0;
      for (let r = 0; r < R; r++) { kn += +(u() < p); kt += +(u() < q); }
      const k = "t" + String(i).padStart(3, "0");
      none.set(k, kn / R); told.set(k, kt / R); tasks.push(k);
    });
    const avg = (map: Map<string, number>, smp: string[]) => smp.reduce((z, k) => z + map.get(k)!, 0) / smp.length;
    if (bootstrap(tasks, (smp: string[]) => avg(told, smp) - avg(none, smp), { seed: R3C.power.bootSeed, b: R3C.power.draws }).lo > 0) keep++;
  }
  return keep / sims;
}
// The smallest lift on a 0.5-point grid up to 100 points with simulated KEEP in at least R3C.power.searchTarget of runs,
// in points, or null. Searched by bisection, which assumes KEEP rises with the lift.
export function mddPoints(posts: number[][], R: number, sims: number, split = false): number | null {
  const at = (k: number) => keepRate(posts, R, (k * R3C.power.stepPoints) / 100, sims, split) >= R3C.power.searchTarget;
  let lo = 1, hi = 100 / R3C.power.stepPoints;
  if (!at(hi)) return null;
  while (lo < hi) { const mid = Math.floor((lo + hi) / 2); if (at(mid)) hi = mid; else lo = mid + 1; }
  return lo * R3C.power.stepPoints;
}

// One grid task-replicate (a none and a told trial with their VM), from the calibration's measured none cost.
export const pairUsd = (m: Model, noneUsd: number) => noneUsd * (1 + R3C.cost[m].toldOverNone) + R3C.vmUsdPerHour * (R3C.cost[m].vmHoursNone + R3C.cost[m].vmHoursTold);

// Admission inputs: the admitted list (more than R3C.maxTasks keeps far tasks first, then mid, then near, each by sha256
// of the id) and each admitted task's posterior under the law refitted on every none trial of this model, in hand and
// calibration, so the law is fitted across every candidate and the keep rule is part of what the posterior conditions
// on; never a task's measured rate. Counts use graded calibration trials only (an infra or ungraded trial is not rerun
// and does not count). The money is the item's cap less settled spend (the item ledger's settled total when admit runs)
// less the reserve headroom. The pair is priced from the mean cost of every calibration none trial on the admitted
// tasks, whatever its outcome (graded, infra or ungraded), since a trial that isn't graded is still paid for and not
// rerun; a null cost on any of them, or no such trial,
// leaves it unpriced, and admission stops, so a missing cost never counts as $0. The mean over every calibration none
// trial, on any task, is report only.
function admissionInputs(h: InHand, m: Model, cal: TrialRow[], o: { capUsd: number; settledUsd: number }) {
  const tr = tiers(h, m), own = new Map(h.none[m]);
  const calRows = cal.filter((t) => t.arm === "none" && /^cal-/.test(t.part)), calNone = calRows.filter(graded);
  for (const t of calNone) { const c = own.get(t.task) ?? { s: 0, n: 0 }; own.set(t.task, { s: c.s + (t.outcome === "pass" ? 1 : 0), n: c.n + 1 }); }
  const cands = new Set([...tr.t1, ...tr.t2, ...tr.t3, ...tr.t4]);
  let list = [...cands].filter((t) => admitted(own.get(t))).sort();
  const cut = list.length > R3C.maxTasks;
  if (cut) list = [...list].sort((a, b) => (h.dist.get(b) ?? 0) - (h.dist.get(a) ?? 0) || (sha(a) < sha(b) ? -1 : 1)).slice(0, R3C.maxTasks).sort();
  const g = npmle([...own.values()]);
  const posts = list.map((t) => posterior(g, own.get(t)));
  const listed = new Set(list), priced = calRows.filter((t) => listed.has(t.task));
  const meanUsd = (rows: TrialRow[]) => (rows.length && rows.every((t) => typeof t.cost_usd === "number" && Number.isFinite(t.cost_usd)) ? rows.reduce((z, t) => z + t.cost_usd!, 0) / rows.length : NaN);
  const noneUsd = meanUsd(priced), poolNoneUsd = meanUsd(calRows);
  const pair = pairUsd(m, noneUsd);
  const money = o.capUsd - o.settledUsd - R3C.reserveUsd;
  const rAfford = list.length && Number.isFinite(pair) && Number.isFinite(money) ? Math.max(0, Math.min(R3C.rMax, Math.floor(money / (list.length * pair)))) : 0;
  return { own, list, cut, g, posts, calNone: calNone.length, priced: priced.length, noneUsd, poolNoneUsd, poolTrials: calRows.length, pair, rAfford };
}
// R: the smallest replicate count from R3C.rMin to the affordable maximum at which simulated KEEP at R3C.mddTarget points
// reaches the level (R3C.power.searchTarget unless a caller names one), found by bisection, which assumes KEEP rises
// with R; when the affordable maximum doesn't reach it, the affordable maximum itself.
export function chooseR(posts: number[][], rAfford: number, sims: number, level = R3C.power.searchTarget): number {
  const reaches = (R: number) => keepRate(posts, R, R3C.mddTarget / 100, sims) >= level;
  let R = rAfford;
  if (reaches(rAfford)) { let lo = R3C.rMin; while (lo < R) { const mid = Math.floor((lo + R) / 2); if (reaches(mid)) R = mid; else lo = mid + 1; } }
  return R;
}

// Admission: the inputs above, R, that R's MDD and null size. GO needs at least R3C.nMin tasks, an affordable maximum of
// at least R3C.rMin, an MDD of at most R3C.mddMax points and a null size of at most R3C.nullMax. The projection is
// settled spend plus that R's grid at the admitted tasks' measured none cost plus the reserve headroom. Report only: the
// MDD when the lift sits on half the tasks, and the MDD and null size with posteriors under the beta-binomial law.
export function admission(h: InHand, m: Model, cal: TrialRow[], o: { capUsd: number; settledUsd: number; sims: number }) {
  const x = admissionInputs(h, m, cal, o);
  const { list, posts, rAfford, pair } = x;
  const far = list.filter((t) => h.dist.get(t) === 2).length;
  const base = { model: m, n: list.length, cut: x.cut, far, listSha256: idsSha(list), list, calNone: x.calNone, priced: x.priced, noneUsd: x.noneUsd, poolNoneUsd: x.poolNoneUsd, poolTrials: x.poolTrials, pairUsd: pair, rAfford, bandMassRefit: bandMass(x.g), meanPosteriorRate: posts.length ? posts.reduce((z, w) => z + w.reduce((y, v, j) => y + v * GRID[j]!, 0), 0) / posts.length : NaN };
  if (list.length < R3C.nMin) return { ...base, go: false, reason: "admitted " + list.length + " tasks, under " + R3C.nMin };
  if (!Number.isFinite(pair)) return { ...base, go: false, reason: "the grid can't be priced: a calibration none trial on an admitted task has no cost, or there is no such trial" };
  if (rAfford < R3C.rMin) return { ...base, go: false, reason: "the money left buys " + rAfford + " replicates, under " + R3C.rMin };
  const R = chooseR(posts, rAfford, o.sims);
  const mdd = mddPoints(posts, R, o.sims), nullSize = keepRate(posts, R, 0, o.sims), mddSplit = mddPoints(posts, R, o.sims, true);
  const bf = betaFit([...x.own.values()]), gb = betaLaw(bf.a, bf.b), postsB = list.map((t) => posterior(gb, x.own.get(t)));
  const beta = { a: bf.a, b: bf.b, bandMass: bandMass(gb), mdd: mddPoints(postsB, R, o.sims), nullSize: keepRate(postsB, R, 0, o.sims) };
  const projection = o.settledUsd + list.length * R * pair + R3C.reserveUsd;
  const more = { R, mdd, mddSplit, nullSize, projection, beta };
  if (mdd === null || mdd > R3C.mddMax) return { ...base, ...more, go: false, reason: "the MDD at " + R + " replicates is " + mdd + " points, over " + R3C.mddMax };
  if (nullSize > R3C.nullMax) return { ...base, ...more, go: false, reason: "null size " + (100 * nullSize).toFixed(1) + "% over " + 100 * R3C.nullMax + "%" };
  return { ...base, ...more, go: true, reason: "ok" };
}

// admit's frozen run: always R3C.power.sims simulated grids per point, whatever R3C_SIMS says.
export const admitOptions = (capUsd: number, settledUsd: number) => ({ capUsd, settledUsd, sims: R3C.power.sims });

// Validate (report only; it decides nothing): K simulated calibrations, each candidate's true chance drawn from its
// posterior under the in-hand NPMLE law (truth "npmle") or the beta-binomial law (truth "beta"), each admitted at the
// given cap; then the power the chosen grid really has at its claimed MDD on the simulated true chances.
const oneHot = (p: number) => GRID.map((q) => (Math.abs(q - p) < 1e-9 ? 1 : 0));
export function validate(h: InHand, m: Model, o: { capUsd: number; k: number; sims: number; first?: number; truth?: "npmle" | "beta" }) {
  const { cands } = candidatePosteriors(h, m, o.truth ?? "npmle");
  const u = rng(R3C.validateSeed);
  const runs: { go: boolean; n: number; R?: number; mdd?: number | null; truePower?: number; spend?: number; reason: string }[] = [];
  const first = o.first ?? 0;
  for (let i = 0; i < first + o.k; i++) {
    const c = simulateCalibration(h, m, cands, u), settled = calSpendUsd(m, c.rows);
    if (i < first) continue;
    const a = admission(h, m, c.rows, { capUsd: o.capUsd, settledUsd: settled, sims: o.sims }) as any;
    if (a.R === undefined || a.mdd === null) { runs.push({ go: false, n: a.n, spend: settled, reason: a.reason }); continue; }
    const truePower = keepRate(a.list.map((t: string) => oneHot(c.truth.get(t)!)), a.R, a.mdd / 100, o.sims);
    runs.push({ go: a.go, n: a.n, R: a.R, mdd: a.mdd, truePower, spend: a.go ? a.projection - R3C.reserveUsd : settled, reason: a.reason });
  }
  return runs;
}

// Calibrate (how R3C.power.searchTarget was set): K simulated calibrations under the NPMLE truth, each with the admitted
// list, R as chooseR picks it at the uncalibrated level R3C.power.target, and simulated KEEP at each lift in
// R3C.calibrateLifts both under the fitted law (what the rule sees) and on the simulated true chances (what the grid
// would get). calread turns the lines into the post's table.
export function calibrate(h: InHand, m: Model, o: { capUsd: number; k: number; sims: number; first?: number }) {
  const { cands } = candidatePosteriors(h, m);
  const u = rng(R3C.validateSeed);
  const runs: { run: number; n: number; R?: number; fitted?: number[]; truth?: number[] }[] = [];
  const first = o.first ?? 0;
  for (let i = 0; i < first + o.k; i++) {
    const c = simulateCalibration(h, m, cands, u), settled = calSpendUsd(m, c.rows);
    if (i < first) continue;
    const x = admissionInputs(h, m, c.rows, { capUsd: o.capUsd, settledUsd: settled });
    if (x.list.length < R3C.nMin || x.rAfford < R3C.rMin) { runs.push({ run: i + 1, n: x.list.length }); continue; }
    const R = chooseR(x.posts, x.rAfford, o.sims, R3C.power.target), tp = x.list.map((t) => oneHot(c.truth.get(t)!));
    runs.push({ run: i + 1, n: x.list.length, R, fitted: R3C.calibrateLifts.map((d) => keepRate(x.posts, R, d / 100, o.sims)), truth: R3C.calibrateLifts.map((d) => keepRate(tp, R, d / 100, o.sims)) });
  }
  return runs;
}

// Reads calibrate's lines: for each search level from 0.80 to 0.95, each run's claimed MDD (the smallest 0.5-point lift
// at which the fitted curve, linearly interpolated between the measured lifts, reaches the level) and its true power
// there (the truth curve interpolated the same way), over the runs whose claimed MDD is at most R3C.mddMax.
export function calibrateTable(runs: { n: number; R?: number; fitted?: number[]; truth?: number[] }[]) {
  const L = R3C.calibrateLifts, rows: { level: number; runs: number; meanMdd: number; meanTrue: number; se: number }[] = [];
  const interp = (ys: number[], d: number) => { const i = Math.min(L.length - 2, Math.max(0, L.findIndex((x) => x >= d) - 1)); const t = (d - L[i]!) / (L[i + 1]! - L[i]!); return ys[i]! + t * (ys[i + 1]! - ys[i]!); };
  for (let lv = 80; lv <= 95; lv++) {
    const level = lv / 100, got: { mdd: number; tp: number }[] = [];
    for (const r of runs) {
      if (!r.fitted || !r.truth) continue;
      let mdd: number | null = null;
      for (let d = L[0]!; d <= L[L.length - 1]! + 1e-9; d += R3C.power.stepPoints) if (interp(r.fitted, d) >= level - 1e-12) { mdd = d; break; }
      if (mdd === null || mdd > R3C.mddMax) continue;
      got.push({ mdd, tp: interp(r.truth, mdd) });
    }
    const m = got.reduce((z, g) => z + g.tp, 0) / got.length, sd = Math.sqrt(got.reduce((z, g) => z + (g.tp - m) ** 2, 0) / Math.max(1, got.length - 1));
    rows.push({ level, runs: got.length, meanMdd: got.reduce((z, g) => z + g.mdd, 0) / got.length, meanTrue: m, se: sd / Math.sqrt(got.length) });
  }
  return rows;
}

// selftest: drives each frozen rule both ways on synthetic inputs, with no file and no key.
function selftest(): number {
  let bad = 0;
  const check = (name: string, ok: boolean) => { console.log((ok ? "PASS " : "FAIL ") + name); if (!ok) bad++; };
  check("band admits 1 of 4 and 3 of 4", admitted({ s: 1, n: 4 }) && admitted({ s: 3, n: 4 }));
  check("band refuses 0 of 4 and 4 of 4", !admitted({ s: 0, n: 4 }) && !admitted({ s: 4, n: 4 }));
  check("band admits 2 of 7 and 5 of 7, refuses 1 of 7 and 6 of 7", admitted({ s: 2, n: 7 }) && admitted({ s: 5, n: 7 }) && !admitted({ s: 1, n: 7 }) && !admitted({ s: 6, n: 7 }));
  check("band admits 1 of 5 (0.2) and 4 of 5 (0.8), both ends included", admitted({ s: 1, n: 5 }) && admitted({ s: 4, n: 5 }));
  check("under 4 trials is never admitted (1 of 2, 1 of 3)", !admitted({ s: 1, n: 2 }) && !admitted({ s: 1, n: 3 }));
  check("changeKey matches a moved owner", changeKey("a__x-1") === changeKey("b__x-1") && changeKey("a__x-1") !== changeKey("a__x-2"));
  const none = (pairs: [string, number, number][]) => new Map(pairs.map(([t, s, n]) => [t, { s, n }]));
  const h: InHand = {
    kept: ["f1", "t1a", "t2a", "t2b", "t4a", "u1", "u2", "l1a", "mv__s-9"],
    none: { opus: none([["f1", 0, 4], ["t1a", 1, 4], ["t2a", 1, 1], ["t2b", 0, 1], ["t4a", 1, 1], ["ow__s-9", 1, 1]]), sol: none([["f1", 0, 4], ["t1a", 0, 4], ["t2a", 0, 1], ["t2b", 1, 1], ["t4a", 1, 1], ["ow__s-9", 1, 1]]) },
    screen: { opus: new Map([["f1", false], ["t1a", false], ["t2a", true], ["t2b", false], ["t4a", true], ["ow__s-9", true]]), sol: new Map([["f1", false], ["t1a", false], ["t2a", false], ["t2b", true], ["t4a", true], ["ow__s-9", true]]) },
    l1: new Set(["l1a"]), dist: new Map([["t1a", 2]]),

function main(argv: string[]): number {
  const [cmd, dir] = argv;
  if (cmd === "selftest") return selftest();
  if (cmd === "calread") {
    if (!dir) { console.error("E_USAGE r3c.ts calread <calibrate output>"); return 2; }
    const runs = readFileSync(dir, "utf8").split("\n").filter((l) => l.trim()).map((l) => JSON.parse(l));
    console.log("r3c calread: " + runs.length + " simulated calibrations, " + runs.filter((r: any) => r.fitted).length + " with at least " + R3C.nMin + " tasks and " + R3C.rMin + " affordable replicates");
    for (const r of calibrateTable(runs)) console.log("search level " + r.level.toFixed(2) + ": " + r.runs + " runs with MDD at most " + R3C.mddMax + ", mean MDD " + r.meanMdd.toFixed(2) + ", true power at it " + (100 * r.meanTrue).toFixed(1) + "% (SE " + (100 * r.se).toFixed(1) + ")");
    return 0;
  }
  const flag = (k: string) => { const i = argv.indexOf(k); return i < 0 ? undefined : argv[i + 1]; };
  const m = flag("--model") as Model;
  if (!cmd || !dir || (m !== "opus" && m !== "sol")) { console.error("E_USAGE r3c.ts plan|forecast|admit|validate|calibrate <in-hand dir> --model opus|sol, or r3c.ts selftest"); return 2; }
  const h = loadInHand(dir);
  if (cmd === "plan") {
    const tr = tiers(h, m), cp = calibrationPlan(tr);
    console.log("r3c plan " + m + ": kept pool " + h.kept.length + "; tiers t1 " + tr.t1.length + ", t2 " + tr.t2.length + ", t3 " + tr.t3.length + ", t4 " + tr.t4.length + ", floor " + tr.floor.length + ", out " + tr.out.length);
    console.log("calibration: cal-1 " + cp.cal1Trials + " none trials (t1 x" + R3C.calTrials.t1 + ", t2 x" + R3C.calTrials.t2 + ", t3 x" + R3C.calTrials.t3 + ", t4 x" + R3C.calTrials.t4 + "), then cal-2: " + R3C.t4Second + " more on each t4 task whose cal-1 trial failed");
    for (const k of ["t1", "t2", "t3", "t4", "floor", "out"] as const) console.log("  " + k + " sha256 " + idsSha(tr[k]) + " (" + tr[k].length + "; far " + tr[k].filter((t) => h.dist.get(t) === 2).length + ")");
    console.log("candidates sha256 (t1 to t4, sorted ids): " + idsSha([...tr.t1, ...tr.t2, ...tr.t3, ...tr.t4]));
    const w = flag("--write");
    if (w) { mkdirSync(w, { recursive: true }); for (const k of ["t1", "t2", "t3", "t4"] as Tier[]) writeFileSync(join(w, "r3c-" + m + "-" + k + ".txt"), tr[k].map((t) => t + "\n").join("")); }
    return 0;
  }
  if (cmd === "forecast") {
    const f = forecast(h, m, Number(process.env.R3C_FORECAST_SIMS ?? 2000));
    const e = f.expectedInBand;
    console.log("r3c forecast " + m + ": the fitted law puts " + (100 * f.lawBandMass).toFixed(1) + "% of tasks in the band; expected in-band candidates t1 " + e.t1.toFixed(1) + ", t2 " + e.t2.toFixed(1) + ", t3 " + e.t3.toFixed(1) + ", t4 " + e.t4.toFixed(1));
    console.log("admitted: mean " + f.admittedMean.toFixed(1) + " (p10 " + f.admittedP10 + ", median " + f.admittedP50 + ", p90 " + f.admittedP90 + "), truly in band " + (100 * f.inBandShare).toFixed(0) + "%; P(at least " + R3C.nMin + ") " + (100 * f.pAtLeastNMin).toFixed(1) + "%; calibration trials mean " + f.calTrialsMean.toFixed(0));
    for (const l of ladderLines(dir, m, f)) console.log(l);
    const e2 = flag("--emit");
    if (e2) writeFileSync(e2, f.lastSim.map((r) => JSON.stringify(r)).join("\n") + "\n");
    return 0;
  }
  if (cmd === "admit") {
    const calPath = flag("--calibration"), cap = Number(flag("--cap-usd")), settled = Number(flag("--settled-usd"));
    if (!calPath || !Number.isFinite(cap) || !Number.isFinite(settled)) { console.error("E_USAGE admit needs --calibration, --cap-usd and --settled-usd"); return 2; }
    const a = admission(h, m, jsonlRows(readFileSync(calPath, "utf8")), admitOptions(cap, settled));
    for (const l of admissionLines(a)) console.log(l);
    return a.go ? 0 : 5;
  }
  if (cmd === "validate" || cmd === "calibrate") {
    const cap = Number(flag("--cap-usd")), k = Number(flag("--runs") ?? 20), first = Number(flag("--first") ?? 0), sims = Number(process.env.R3C_SIMS ?? 2000);
    const truth = (flag("--truth") ?? "npmle") as "npmle" | "beta";
    if (!Number.isFinite(cap) || (truth !== "npmle" && truth !== "beta")) { console.error("E_USAGE " + cmd + " needs --cap-usd, and --truth is npmle or beta"); return 2; }
    if (cmd === "calibrate") { for (const r of calibrate(h, m, { capUsd: cap, k, first, sims })) console.log(JSON.stringify(r)); return 0; }
    const runs = validate(h, m, { capUsd: cap, k, first, sims, truth });
    runs.forEach((r, i) => console.log("run " + (first + i + 1) + ": " + (r.go ? "GO" : "STOP") + " n " + r.n + (r.R !== undefined ? " R " + r.R + " MDD " + r.mdd + " true power " + (100 * r.truePower!).toFixed(1) + "%" : "") + " spend $" + r.spend!.toFixed(2) + (r.go ? "" : " (" + r.reason + ")")));
    const go = runs.filter((r) => r.go), mean = (xs: number[]) => (xs.length ? xs.reduce((a, b) => a + b, 0) / xs.length : NaN);
    const srt = (xs: number[]) => [...xs].sort((a, b) => a - b);
    const mdds = srt(go.map((r) => r.mdd as number)), sp = srt(go.map((r) => r.spend!));
    console.log("r3c validate " + m + " (truth " + truth + ") at cap $" + cap + ": GO in " + go.length + " of " + runs.length + "; MDD mean " + mean(mdds).toFixed(1) + " (min " + mdds[0] + ", max " + mdds[mdds.length - 1] + "); true power at the claimed MDD mean " + (100 * mean(go.map((r) => r.truePower!))).toFixed(1) + "% (target " + 100 * R3C.power.target + "%); spend before reserve mean $" + mean(sp).toFixed(2) + " (max $" + (sp[sp.length - 1] ?? NaN).toFixed(2) + ")");
    return 0;
  }
  console.error("E_USAGE unknown command " + cmd);
  return 2;
}

if (import.meta.main) {
  try { process.exit(main(Bun.argv.slice(2))); } catch (e) { console.error((e as Error).message); process.exit(1); }
}
