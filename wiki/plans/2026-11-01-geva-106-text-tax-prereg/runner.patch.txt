+const bopts = { b: RESAMPLES, seed: SEED };
+export const rate = (map: Map<string, number>) => ({ tasks: map.size, ...bootstrap([...map.keys()].sort(), (s) => mean(s.map((k) => map.get(k)!)), bopts) });

+// Two-sided Monte Carlo sign-flip test of the mean per-task difference: p = (1 + hits) / (1 + draws).
+export function signFlip(diffs: number[], draws = DRAWS, seed = SEED): number {
+  if (!diffs.length) return NaN;
+  const r = rng(seed);
+  const obs = Math.abs(mean(diffs)) - 1e-12;
+  let hits = 0;
+  for (let i = 0; i < draws; i++) {
+    let s = 0;
+    for (const d of diffs) s += r() < 0.5 ? -d : d;
+    if (Math.abs(s / diffs.length) >= obs) hits++;
+  }
+  return (1 + hits) / (1 + draws);
+}
+export function contrast(a: Map<string, number>, b: Map<string, number>, test = true) {
+  const tasks = [...a.keys()].filter((k) => b.has(k)).sort();
+  const ci = bootstrap(tasks, (s) => mean(s.map((k) => a.get(k)! - b.get(k)!)), bopts);
+  return { tasks: tasks.length, ...ci, p: test ? signFlip(tasks.map((k) => a.get(k)! - b.get(k)!)) : NaN };

+export function holm(ps: number[]): number[] {
+  const order = ps.map((p, i) => [p, i] as const).filter(([p]) => Number.isFinite(p)).sort((x, y) => x[0] - y[0]);
+  const out = ps.map(() => NaN);
+  let run = 0;
+  order.forEach(([p, i], k) => { run = Math.max(run, Math.min(1, (order.length - k) * p)); out[i] = run; });
+  return out;

+function cellLine(recs: Rec[], model: string, label: string, told: boolean) {
+  const all = recs.filter((r) => r.model === model && r.label === label);
+  const g = all.filter((r) => r.status === "graded");
+  const passes = g.reduce((s, r) => s + r.pass, 0);
+  const cost = g.reduce((s, r) => s + r.costUsd, 0);
+  const share = (n: number) => (g.length ? n / g.length : NaN);
+  return {
+    graded: g, eligibleShare: share(g.filter((r) => r.eligible).length), bodyLoadShare: share(g.filter((r) => r.bodyLoad).length),
+    text: `${model} ${label}: graded ${g.length}, missing ${all.length - g.length}, refused ${g.filter((r) => r.refused).length}, retried after a response ${g.filter((r) => r.retried).length}; ` +
+      (told ? `body load ${pct(share(g.filter((r) => r.bodyLoad).length))}; ` : `eligible ${pct(share(g.filter((r) => r.eligible).length))}; `) +
+      `pass ${ci(rate(perTask(g, (r) => r.pass)))}; session-cap stops ${g.filter((r) => r.budgetStop).length} (${pct(share(g.filter((r) => r.budgetStop).length))}), agent timeouts ${g.filter((r) => r.agentTimeout).length}; ` +
+      `median requests ${median(g.map((r) => r.requests))}, median first-request tokens ${median(g.flatMap((r) => (r.firstTokens === null ? [] : [r.firstTokens])))}; ` +
+      `cost per trial $${(g.length ? cost / g.length : 0).toFixed(3)}, per success ${passes ? "$" + (cost / passes).toFixed(3) : "n/a"}`,

+// Minimum detectable differences for GEVA-106's preregistration (section 11). No data, no key.
+// Run D, 37 tasks, one trial per task per arm: the per-trial open probability that gives 80% power for
+//   D1: gstack and placebo pooled (two trials per task); a claim needs the bootstrap interval to exclude 0;
+//   D2: rewritten minus pooled, shipped rate 0 or 0.02; Holm across 5 models, so the first step needs p < 0.01,
+//       and the interval must exclude 0.
+// Opens are uniform across tasks, or confined to a random 30% of tasks (11 of 37; the rewritten arm's opens in
+// D2, the shipped opens stay uniform). Each power value is 1,000 simulated datasets with 1,000 bootstrap
+// resamples each (and 1,000 sign-flip draws for D2's p), seed 20261101. The search is a bisection to 0.001.
+// Run T is analytic: MDD = 2.8 x 4.08 x sqrt(3 / r) points, and (2.394 + 0.842) x 4.08 x sqrt(3 / r) for T2's
+// first Holm step across three models.
+//   bun analysis/mdd-sim.ts [datasets] [resamples]
+import { rng } from "./analyze.ts";
+
+const TASKS = 37;
+const CONFINED = Math.round(0.3 * TASKS);
+const [DATASETS, RESAMPLES] = [Number(Bun.argv[2] ?? 1000), Number(Bun.argv[3] ?? 1000)];
+const rand = rng(20261101);
+
+const quantile = (s: Float64Array, q: number) => { const pos = (s.length - 1) * q, lo = Math.floor(pos), hi = Math.ceil(pos); return s[lo]! + (s[hi]! - s[lo]!) * (pos - lo); };
+function interval(v: Float64Array) {
+  const reps = new Float64Array(RESAMPLES);
+  for (let b = 0; b < RESAMPLES; b++) { let s = 0; for (let i = 0; i < TASKS; i++) s += v[Math.floor(rand() * TASKS)]!; reps[b] = s / TASKS; }
+  reps.sort();

+function signFlipP(v: Float64Array) {
+  let obs = 0;
+  for (const x of v) obs += x;
+  obs = Math.abs(obs / TASKS) - 1e-12;
+  let hits = 0;
+  for (let b = 0; b < RESAMPLES; b++) { let s = 0; for (const x of v) s += rand() < 0.5 ? -x : x; if (Math.abs(s / TASKS) >= obs) hits++; }

+}
+// Per-task open probabilities with overall per-trial rate p: uniform, or p / 0.3 on a random 11 tasks.
+function probs(p: number, confined: boolean) {
+  const out = new Float64Array(TASKS).fill(confined ? 0 : p);
+  if (!confined) return out;

+  for (let i = TASKS - 1; i > 0; i--) { const j = Math.floor(rand() * (i + 1)); [idx[i], idx[j]] = [idx[j]!, idx[i]!]; }
+  for (const i of idx.slice(0, CONFINED)) out[i] = Math.min(1, (p * TASKS) / CONFINED);

+  let ok = 0;

+  return ok / DATASETS;
+}
+function powerD2(p: number, shipped: number, confined: boolean) {
+  let ok = 0;
+  for (let d = 0; d < DATASETS; d++) {
+    const rw = trials(probs(p, confined), 1), pool = trials(probs(shipped, false), 2);
+    const v = rw.map((x, i) => x - pool[i]!);

+    if ((ci.lo > 0 || ci.hi < 0) && signFlipP(v) < 0.01) ok++;

+  return ok / DATASETS;

+  return { p: hi, power: power(hi) };

+// regular-file entry's content replaced in place: that entry's size field and header checksum are
+// rewritten, and every other header and byte (name order, mtime 0, owner) is kept, so the output

+// header with a new size and checksum, the new content padded to a block, the bytes after its old

+export function writeReplaced(fd: number, listing: Listing, path: string, content: Uint8Array, outPath: string): { bytes: number; sha256: string } {
+  const e = onlyFile(listing.entries, path);
+  if (content.length > 0o77777777777) throw new Error("E_INPUT content too large for an octal size field");
+  const header = readAt(fd, e.header, BLOCK);

+  const tailFrom = e.data + pad(e.size);
+  const body = e.header + BLOCK + pad(content.length) + (listing.end - tailFrom);
+  const total = Math.ceil((body + 2 * BLOCK) / RECORD) * RECORD;

+    for (let off = 0; off < p.length; ) off += writeSync(out, p, off, p.length - off);
+    bytes += p.length;

+    put(header);

+    put(new Uint8Array(pad(content.length) - content.length));
+    for (const c of chunks(fd, tailFrom, listing.end)) put(c);

+  if (frontmatterFields(out).description !== REWRITTEN_DESCRIPTION) throw new Error("E_INTERNAL rewritten description does not parse back to the frozen text");
+  return out + body;
+}

+export function shortPlacebo(text: string): { output: string; tokens: number } {
+  const { frontmatter } = splitSkill(text);
+  const { name, description } = frontmatterFields(frontmatter);
+  const body = prose(topicTerms(name, description), SHORT_TARGET, prng(SHORT_SEED, "investigate"));
+  const tokens = countTokens(body);
+  const markers = proceduralMarkers(body);
+  if (markers.length) throw new Error("E_INTERNAL placebo-short body has procedural markers: " + markers.join(", "));
+  if (Math.abs(tokens - SHORT_TARGET) > 1) throw new Error("E_INTERNAL placebo-short body has " + tokens + " tokens, not " + SHORT_TARGET + " +- 1");
+  return { output: frontmatter + body, tokens };

+function parseList(tsv: string): Entry[] {
+  return tsv.trimEnd().split("\n").slice(1).map((line) => {
+    const [sha, size, mode, type, ...path] = line.split("\t");
+    return { path: path.join("\t"), type: type as Entry["type"], mode: parseInt(mode!, 8), size: Number(size), sha256: sha! };
+  });
+}

+// A variant's manifest is its base manifest with the one changed file's row replaced and the
+// digests recomputed the way tree.ts computes them.
+export function variantManifest(arm: string, base: any, baseList: string, changed: { sha256: string; size: number }, tar: { file: string; bytes: number; sha256: string }, meta: Record<string, unknown>) {
+  const entries = parseList(baseList);
+  if (treeDigest(entries) !== base.tree.digest) throw new Error("E_INTERNAL " + base.arm + ".files.tsv does not reproduce " + base.arm + ".json's tree digest");
+  const row = entries.find((e) => e.path === SKILL_PATH);
+  if (!row || row.type !== "file") throw new Error("E_INPUT " + base.arm + ".files.tsv has no file row " + SKILL_PATH);
+  row.sha256 = changed.sha256;
+  row.size = changed.size;
+  const prefix = ".claude/skills/";
+  const skillEntries = entries.filter((e) => e.path.startsWith(prefix)).map((e) => ({ ...e, path: e.path.slice(prefix.length) }));
+  const files = entries.filter((e) => e.type === "file");
+  const summary = {
+    arm, claude_code: base.claude_code, gstack_sha: base.gstack_sha, ...meta, tar,
+    tree: { files: files.length, symlinks: base.tree.symlinks, dirs: base.tree.dirs, bytes: files.reduce((n, e) => n + e.size, 0), digest: treeDigest(entries) },
+    skills: {
+      count: base.skills.count, tree_digest: treeDigest(skillEntries),
+      skill_md_digest: treeDigest(skillEntries.filter((e) => /^[^/]+\/SKILL\.md$/.test(e.path))), names: base.skills.names,
+    },
+  };
+  const list = ["sha256\tbytes\tmode\ttype\tpath", ...entries.map((e) => [e.sha256, e.size, e.mode.toString(8), e.type, e.path].join("\t"))];
+  return { summary, list: list.join("\n") + "\n" };
+}

+// The run's settled spend, plus its open reserves, plus 1.4 times the projected cost of the trials not

+//   bun runner/start-rule.ts check --run <discovery-r1|texttax-r1> --ledger <jsonl> --batch <spec> [--batch <spec> ...] [--note]
+//   bun runner/start-rule.ts plan --run <discovery-r1|texttax-r1>
+//
+// A batch spec is a model id for a batch with every trial still to run, or model:arm=n+arm=n for the
+// trials it has left. List the batch about to launch and every batch of the run still running.

+export type Batch = { model: string; remaining: Record<string, number> };

+export function parseBatch(run: RunSpec, spec: string): Batch {
+  const [model, rest] = spec.split(":");
+  const reps = run.replicates[model!];
+  if (!reps) throw new Error(`E_USAGE model ${model} has no batch in ${run.item}`);
+  if (!rest) return { model: model!, remaining: Object.fromEntries(run.arms.map((a) => [a, TASKS * reps])) };
+  const remaining: Record<string, number> = {};
+  for (const part of rest.split("+")) {
+    const [arm, n] = part.split("=");
+    if (!run.arms.includes(arm!) || !/^\d+$/.test(n ?? "") || Number(n) > TASKS * reps) throw new Error(`E_USAGE bad batch part ${part} in ${spec}`);
+    remaining[arm!] = Number(n);
+  }
+  return { model: model!, remaining };

+export const batchMicros = (run: RunSpec, b: Batch) => Object.entries(b.remaining).reduce((s, [arm, n]) => s + n * toMicros(run.price(b.model, arm)), 0);

+export function startRule(run: RunSpec, status: Totals, batches: Batch[]) {
+  const projectedMicros = batches.reduce((s, b) => s + batchMicros(run, b), 0);
+  const boundMicros = status.settledMicros + status.openReservedMicros + Math.ceil(projectedMicros * MARGIN);
+  return {
+    run: run.item, cap_usd: run.cap, settled_usd: fromMicros(status.settledMicros), open_reserved_usd: fromMicros(status.openReservedMicros),
+    batches: batches.map((b) => ({ model: b.model, remaining: b.remaining, projected_usd: fromMicros(batchMicros(run, b)) })),
+    projected_usd: fromMicros(projectedMicros), margin: MARGIN, bound_usd: fromMicros(boundMicros), pass: boundMicros <= toMicros(run.cap),
+  };
+}

+import { describe, expect, test } from "bun:test";
+import { closeSync, mkdirSync, mkdtempSync, openSync, readFileSync, writeFileSync } from "node:fs";
+import { tmpdir } from "node:os";
+import { join } from "node:path";
+import { carriers, entryBytes, fileSha256, onlyFile, REWRITTEN_DESCRIPTION, rewriteDescription, SHIPPED_LINE, shortPlacebo, SHORT_TARGET, tarEntries, writeReplaced } from "../arms/variants.ts";
+import { countTokens, frontmatterFields, proceduralMarkers, splitSkill } from "../arms/placebo/placebo.ts";
+
+const SKILL = ".claude/skills/investigate/SKILL.md";
+const LONG = ".claude/skills/" + "x".repeat(120) + "/SKILL.md";
+const skillMd = (body: string) => "---\nname: investigate\n" + SHIPPED_LINE + "\nallowed-tools:\n  - Bash\n---\n" + body;
+
+// A small GNU tar built the way the arm tars are: sorted names, mtime 0, root owner, a long name.
+function fixture() {
+  const dir = mkdtempSync(join(tmpdir(), "variants-"));
+  const root = join(dir, "root");
+  for (const [path, text] of [[SKILL, skillMd("Shipped body.\n")], [".claude/skills/review/SKILL.md", "---\nname: review\ndescription: Review.\n---\nreview body\n"], [LONG, "long " + SHIPPED_LINE + "\n"]]) {
+    mkdirSync(join(root, path, ".."), { recursive: true });
+    writeFileSync(join(root, path), text);
+  }
+  const run = Bun.spawnSync(["tar", "--format=gnu", "--sort=name", "--mtime=@0", "--owner=0", "--group=0", "--numeric-owner", "-cf", join(dir, "base.tar"), "-C", root, ".claude"]);
+  if (run.exitCode !== 0) throw new Error(run.stderr.toString());
+  return { dir, tar: join(dir, "base.tar") };
+}
+
+const replaced = (tar: string, out: string, content: string) => {
+  const fd = openSync(tar, "r");
+  try {
+    return writeReplaced(fd, tarEntries(fd), SKILL, Buffer.from(content), out);
+  } finally {
+    closeSync(fd);
+  }
+};
+const listing = (tar: string) => Bun.spawnSync(["tar", "-tf", tar]).stdout.toString();
+
+describe("tar entry replacement", () => {
+  test("writing the same content back gives the base tar byte for byte", () => {
+    const { dir, tar } = fixture();
+    const fd = openSync(tar, "r");
+    const same = entryBytes(fd, onlyFile(tarEntries(fd).entries, SKILL));
+    closeSync(fd);
+    const out = replaced(tar, join(dir, "same.tar"), same.toString());
+    expect(out.sha256).toBe(fileSha256(tar));
+    expect(readFileSync(join(dir, "same.tar")).equals(readFileSync(tar))).toBe(true);
+  });
+
+  test("a longer entry keeps every other entry, the order and the long name, and GNU tar reads it", () => {
+    const { dir, tar } = fixture();
+    const body = "y".repeat(3000);
+    const out = replaced(tar, join(dir, "new.tar"), body);
+    const again = replaced(tar, join(dir, "again.tar"), body);
+    expect(again.sha256).toBe(out.sha256);
+    expect(out.bytes % 10240).toBe(0);
+    expect(listing(join(dir, "new.tar"))).toEqual(listing(tar));
+    const x = join(dir, "x");
+    mkdirSync(x);
+    expect(Bun.spawnSync(["tar", "-xf", join(dir, "new.tar"), "-C", x]).exitCode).toBe(0);
+    expect(readFileSync(join(x, SKILL), "utf8")).toBe(body);
+    expect(readFileSync(join(x, LONG), "utf8")).toBe("long " + SHIPPED_LINE + "\n");
+    const fd = openSync(join(dir, "new.tar"), "r");
+    expect(carriers(fd, tarEntries(fd), SHIPPED_LINE.slice(13))).toEqual([LONG]);
+    closeSync(fd);
+  });
+
+  test("a tar whose end is not GNU's record padding is refused", () => {
+    const { dir, tar } = fixture();
+    writeFileSync(join(dir, "cut.tar"), readFileSync(tar).subarray(0, readFileSync(tar).length - 512));
+    const fd = openSync(join(dir, "cut.tar"), "r");
+    expect(() => tarEntries(fd)).toThrow("E_INPUT");
+    closeSync(fd);
+  });
+});
+
+describe("rewritten description", () => {
+  test("only the description line changes, to the frozen 319-character text, YAML double-quoted", () => {
+    expect(REWRITTEN_DESCRIPTION.length).toBe(319);
+    const before = skillMd("Shipped body.\n");
+    const after = rewriteDescription(before);
+    expect(after).toBe(before.replace(SHIPPED_LINE, "description: " + JSON.stringify(REWRITTEN_DESCRIPTION)));
+    expect(frontmatterFields(splitSkill(after).frontmatter).description).toBe(REWRITTEN_DESCRIPTION);
+    expect(() => rewriteDescription(after)).toThrow("E_INPUT");
+  });
+});
+
+describe("short placebo", () => {
+  test("847 tokens within 1, no procedural markers, frontmatter kept, deterministic", () => {
+    const text = skillMd("x".repeat(100));
+    const a = shortPlacebo(text);
+    expect(Math.abs(a.tokens - SHORT_TARGET)).toBeLessThanOrEqual(1);
+    expect(countTokens(splitSkill(a.output).body)).toBe(a.tokens);
+    expect(proceduralMarkers(splitSkill(a.output).body)).toEqual([]);
+    expect(splitSkill(a.output).frontmatter).toBe(splitSkill(text).frontmatter);
+    expect(shortPlacebo(text).output).toBe(a.output);
+  });
+});
