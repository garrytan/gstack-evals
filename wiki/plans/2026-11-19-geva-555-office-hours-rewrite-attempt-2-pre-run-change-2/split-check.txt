// Scratch, read-only: stage 1's split at Sol's measured skill central of $2.04, each item's own need at 1.35 times
// central, the $1 move into ohr-s1-arms, and the lane's own checks on the new split. Run from ~/.capy/work/ohr2/wt.
import { CENTRAL, SCHEDULES, SPLIT, finishesUpTo, scheduleProblems, splitProblems, walkSchedule } from "./lanes/office-hours-rewrite/ledger.ts";
(CENTRAL.session as any).sol = { skill: 2.04, none: 0.3 };
const s1 = SCHEDULES.find((s) => s.name === "stage1")!, rep = SCHEDULES.find((s) => s.name === "stage1-repair")!;
const needs = (split: typeof SPLIT) => {
  const a = walkSchedule(s1, 1.35, split);
  const r = walkSchedule(rep, 1.35, split), r0 = walkSchedule(rep, 0, split);
  const first = Math.round(1.35 * Math.min(r.ran, 3) * CENTRAL.label * 1e6) / 1e6 + 1;
  return { "ohr-s1-arms": a.need.arms, "ohr-s1-sim": a.need.sim, "ohr-s1-labels": a.need.labels, "ohr-s1-repair-arms": r0.need.arms, "ohr-s1-repair-sim": r0.need.sim, "ohr-s1-repair-labels": Math.round(first * 100) / 100 } as Record<string, number>;
};
const show = (name: string, split: typeof SPLIT) => {
  const n = needs(split);
  console.log(name);
  for (const p of split.filter((x) => x.line === "stage1")) console.log("  " + p.item.padEnd(22) + " cap $" + String(p.cap).padStart(4) + "  need at 1.35x $" + n[p.item]!.toFixed(2).padStart(7) + "  room $" + (p.cap - n[p.item]!).toFixed(2).padStart(6));
  console.log("  stage 1 total $" + split.filter((x) => x.line === "stage1").reduce((s, p) => s + p.cap, 0));
  console.log("  " + SCHEDULES.map((s) => s.name + " finishes up to " + finishesUpTo(s, split)).join("; "));
  console.log("  scheduleProblems: " + JSON.stringify(scheduleProblems(split)) + "; splitProblems: " + JSON.stringify(splitProblems(split)));
  return n;
};
const n = show("Sol skill central $2.04, split as built", SPLIT);
const donors = SPLIT.filter((p) => p.line === "stage1" && p.item !== "ohr-s1-arms").map((p) => ({ item: p.item, room: p.cap - n[p.item]! })).sort((a, b) => b.room - a.room);
console.log("donor with the most room: " + donors[0]!.item + " ($" + donors[0]!.room.toFixed(2) + ")");
const moved = SPLIT.map((p) => p.item === "ohr-s1-arms" ? { ...p, cap: p.cap + 1 } : p.item === donors[0]!.item ? { ...p, cap: p.cap - 1 } : p);
show("Sol skill central $2.04, $1 moved from " + donors[0]!.item + " into ohr-s1-arms", moved);
