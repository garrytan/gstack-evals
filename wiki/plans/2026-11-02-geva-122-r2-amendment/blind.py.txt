#!/usr/bin/env python3
# Blinding step for R2 (the R2 amendment's A1, on GEVA-101 prereg section 3.4). The prereg's blind.py with one change:
# ORDINARY also holds every reader name of the R2 build's reader map (/review's Review Army checklists, so
# "data-migration", "api-contract", "state-machines" and "red-team" too) and "batched" and "normalizer". Those are words a
# finding uses for the defect it describes ("breaks the API contract"), so they stay in both arms. R1 ran the prereg's
# version, which is unchanged. Judges must not learn from a finding's text which
# arm or procedure wrote it. In every judged arm alike, before ingest, this step:
#  - rewrites every reader name any judged arm uses (a finding's reader, or a reader:<name> row in cost.tsv), and the
#    procedure name "Review Army", to the neutral token "<reader>" in summary, fix and evidence. A name matches as a whole
#    word, case-insensitive, with each hyphen written as a hyphen, a space or an underscore, so "red-team", "red team"
#    and "Red_Team" all match. Reader names that are ordinary words a finding uses in its own right (ORDINARY) stay;
#  - rewrites each model id an arm lists in summary, fix and evidence as "<model id>";
#  - leaves out a finding whose path names a finding reader with a hyphen or digit, or a model id. The pinned
#    judge-packet refuses such a path as a leak (leakRules in lanes/review-noloss/judge.ts at the rig pin), and a path
#    can't be rewritten. The suite check below fails if any blocker lists such a path, so a left-out finding can never
#    match a blocker. Its id is listed here, and it drops out of the false-blocker and new-blocker counts.
# Paths, lines and every other field stay as they are. Each file is replaced atomically. Logs ids and counts only, never text.
# Usage: python3 blind.py <suite json> <source arm dir>...
import csv, json, os, re, sys

ORDINARY = {"security", "testing", "maintainability", "performance", "design", "simplification", "data-migration", "api-contract", "state-machines", "red-team", "batched", "normalizer", "confirmed", "unconfirmed", "other"}
PROCEDURES = {"Review Army"}
suite, arms = json.load(open(sys.argv[1])), sys.argv[2:]
readers, leaky, models = set(), set(), set()
for a in arms:
    for m in json.load(open(os.path.join(a, "arm.json"))).get("models") or []:
        models.update({m["model"], m["model"].split("/")[-1]})
    if os.path.exists(os.path.join(a, "cost.tsv")):
        for r in csv.DictReader(open(os.path.join(a, "cost.tsv")), delimiter="\t"):
            if (r.get("scope") or "").startswith("reader:"):
                readers.add(r["scope"][7:])
    for f in os.listdir(os.path.join(a, "findings")):
        for line in open(os.path.join(a, "findings", f)):
            if line.strip():
                r = json.loads(line).get("reader")
                if r:
                    readers.add(r)
                    if re.search(r"[-0-9]", r):
                        leaky.add(r)
B, E = r"(?<![A-Za-z0-9_])", r"(?![A-Za-z0-9_])"
names = sorted({r for r in readers if r.lower() not in ORDINARY} | PROCEDURES, key=lambda s: (-len(s), s.lower()))
rp = [(n, re.compile(B + "[-_ ]".join(re.escape(p) for p in re.split(r"[-_ ]", n)) + E, re.I)) for n in names]
mp = [(n, re.compile(re.escape(n), re.I)) for n in sorted(models, key=lambda s: (-len(s), s))]
lp = [re.compile(B + re.escape(n) + E, re.I) for n in sorted(leaky)] + [p for _, p in mp]
names_path = lambda s: any(p.search(s) for p in lp)
print("reader names rewritten to <reader>: " + ", ".join(names) + "; left as ordinary words: " + (", ".join(sorted(r for r in readers if r.lower() in ORDINARY)) or "none") + "; model ids: " + ", ".join(sorted(models)))
for u in suite["units"]:
    for b in u["blockers"]:
        if any(names_path(p) for p in b["paths"]):
            sys.exit("E_BLIND blocker " + u["id"] + " " + b["id"] + " lists a path that names a reader or model id")
for a in arms:
    count = {n: 0 for n in names + sorted(models)}
    rows, dropped = 0, []
    for f in sorted(os.listdir(os.path.join(a, "findings"))):
        p = os.path.join(a, "findings", f)
        out = []
        for line in open(p).read().splitlines():
            if not line.strip():
                continue
            o = json.loads(line)
            if o.get("path") and names_path(o["path"]):
                dropped.append(o.get("id", "?"))
                continue
            changed = False
            for k in ("summary", "fix", "evidence"):
                v = o.get(k)
                if not isinstance(v, str):
                    continue
                for n, pat in rp:
                    v, c = pat.subn("<reader>", v)
                    count[n] += c
                    changed = changed or c > 0
                for n, pat in mp:
                    v, c = pat.subn("<model id>", v)
                    count[n] += c
                    changed = changed or c > 0
                o[k] = v
            rows += changed
            out.append(json.dumps(o, ensure_ascii=False, separators=(",", ":")) if changed else line)
        with open(p + ".tmp", "w") as w:
            w.write("".join(x + "\n" for x in out))
        os.replace(p + ".tmp", p)
    hits = {k: v for k, v in count.items() if v}
    print("%s: %d findings rewritten %s; left out (path names a reader or model id): %s" % (os.path.basename(a), rows, json.dumps(hits, sort_keys=True), ", ".join(dropped) or "none"))
