#!/usr/bin/env bash
# R3c pin post dry run (GEVA-299; GEVA-305 added the H5 grid resume rows), keyless. Drives every rule of R3 amendment 6 (wiki/plans/2026-11-05-geva-284-r3c-calibrated)
# with the Captain's five build rulings through r3c.ts's own subcommands, at the commit given, in the style of the R3b pin
# post's dryrun.sh.txt.
#   Part 0  git archive of the commit outside any git work tree; bun install --frozen-lockfile.
#   Part A  the told arm: rebuilt from the project drive's six parts, every sha256 checked against SHA256SUMS.txt and the
#           committed manifest; arm-check both ways; the none arm rebuilt with arms/build.sh (Docker, keyless) and compared
#           with manifests/r3/none.json.
#   Part B  the in-hand records (the Captain drive's five, by sha256, and the post's r3c-taskfit.tsv); prepare on a temporary
#           ledger approved at $750, against the real SWE-rebench export, checked with tasks/swerebench_r3a4.py. prepare at a
#           $700 ledger and with a changed record are refused.
#   Part C  scenarios. Each copies the prepared work dir, approves a fresh temporary ledger at $750 and runs r3c.ts run (then
#           admit, grid, resume and analyze) for real: launch.sh reserves and settles the VM blocks, starts the item's ledger
#           service and runs the batch. The Ubicloud runner is a stub (the R3b dry run's) and ssh is a stub that plays
#           vm-run.sh: it reserves and settles one $12 session lease per planned trial through the ledger service, as the
#           proxy does, in the part's order, and writes each trial's files in the shape r3c.ts loadTrials reads. No VM starts
#           and no model is called. The start-rule refusal at a lower cap goes through the exported startRule, since
#           itemCheck fixes the cap at $750. The three grid scenarios in section H start from copies of the go run after
#           its admission said GO (work dir, ledger and bound file), since each admission runs the frozen 2,000 simulated grids.
# Every rule line reads "ok" or "FAIL" with its expected and observed outcome; the last line is DRYRUN: ok or DRYRUN: fail.
# Usage: bash dryrun.sh <commit> [<work dir>]   env: G (the gstack-evals git checkout, default /workspace/gstack-evals),
#   ARMPARTS (the told arm's parts), CAPT (the Captain drive's handoffs/), SWEREBENCH (the export; downloaded with the
#   pinned Harbor into the work dir when unset), UBI_OWNER (default GEVA-305), R3C_MDD_SIMS (default 20, report-only).
set -u
COMMIT=${1:?usage: dryrun.sh <commit> [<work dir>]}
G=${G:-/workspace/gstack-evals}
SHORT=$(git -C "$G" rev-parse --short=7 "$COMMIT") || exit 2
FULL=$(git -C "$G" rev-parse "$COMMIT")
W=$(mkdir -p "${2:-$HOME/.capy/work/r3c-pin/run-$SHORT}" && cd "${2:-$HOME/.capy/work/r3c-pin/run-$SHORT}" && pwd)
ARMPARTS=${ARMPARTS:-$HOME/.capy/drive/project-gstack-evals/laneb-arms/r3b-gstack-699a41cb}
CAPT=${CAPT:-$HOME/.capy/drive/captain-gstack-evals/handoffs}
UBI_OWNER=${UBI_OWNER:-GEVA-305}
export R3C_MDD_SIMS=${R3C_MDD_SIMS:-20}
unset ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY UBICLOUD_API_TOKEN
E=$W/export L=$W/export/quick/laneb I=$W/inhand
mkdir -p "$W/log" "$W/bin/vm" "$W/scen" "$W/s" "$W/prep" "$W/arms" "$W/approvals" "$I"
HIDE="s#$W#\$W#g; s#$ARMPARTS#\$ARMPARTS#g; s#$CAPT#\$CAPT#g; s#$HOME#~#g"
fails=0 rules=0
rule() { local label=$1 want=$2 got=$3; shift 3; rules=$((rules + 1)); if "$@" > /dev/null 2>&1; then echo "ok   $label | expected: $want | observed: $got" | sed "$HIDE"; else fails=$((fails + 1)); echo "FAIL $label | expected: $want | observed: $got" | sed "$HIDE"; fi; }
first() { grep -E -m1 -- "$1" "$2" 2>/dev/null | cut -c1-400 || true; }
has() { grep -E -q -- "$1" "$2"; }
for k in ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY UBICLOUD_API_TOKEN; do echo "$k: $([ -n "${!k+x}" ] && echo set || echo unset)"; done
echo "commit $FULL; bun $(bun --version); docker $(docker version --format '{{.Server.Version}}' 2>/dev/null)"

echo; echo "## Part 0: export"
rm -rf "$E" && mkdir -p "$E" && git -C "$G" archive "$FULL" | tar -x -C "$E"
(cd "$E" && bun install --frozen-lockfile > "$W/log/bun-install.txt" 2>&1); echo "bun install exit $?; $(find "$E" -path "$E/node_modules" -prune -o -type f -print | wc -l) files exported"
rule "P0 the export is outside any git work tree" "git rev-parse fails in the export" "$(cd "$E" && git rev-parse --show-toplevel 2>&1 | head -1)" sh -c "cd '$E' && ! git rev-parse --show-toplevel"

echo; echo "## Part A: the told arm and the none arm"
A=$W/arms
(cd "$ARMPARTS" && grep -v whole: SHA256SUMS.txt | sha256sum -c -) > "$W/log/parts.txt" 2>&1; pc=$?
rule "A1 the six drive parts match SHA256SUMS.txt" "6 OK" "$(grep -c ': OK$' "$W/log/parts.txt") OK, exit $pc" test "$pc" = 0 -a "$(grep -c ': OK$' "$W/log/parts.txt")" = 6
cat "$ARMPARTS"/gstack-e93a3aa4e1e4.tar.gz.part-0* > "$A/gstack.tar.gz"
GZ=$(sha256sum "$A/gstack.tar.gz" | cut -c1-64); TAR=$(gzip -dc "$A/gstack.tar.gz" | sha256sum | cut -c1-64)
M=$L/arms/manifests/r3b/gstack.json
CODE_TAR=$(cd "$E" && bun -e 'const { TOLD_TAR_SHA256 } = await import("./quick/laneb/r3/r3b.ts"); console.log(TOLD_TAR_SHA256)')
rule "A2 the told arm's gz and tar sha256 match SHA256SUMS.txt, the manifest and r3b.ts TOLD_TAR_SHA256" "$(jq -r .tar.sha256 "$M")" "gz $GZ, tar $TAR (code $CODE_TAR)" test "$GZ" = "$(jq -r .tar_gz_sha256 "$M")" -a "$TAR" = "$(jq -r .tar.sha256 "$M")" -a "$TAR" = "$CODE_TAR" -a "$GZ" = "$(sed -n 's/ *whole:gstack.tar.gz$//p' "$ARMPARTS/SHA256SUMS.txt")"
(cd "$E" && bun quick/laneb/r3/r3c.ts arm-check) > "$W/log/arm-check.txt" 2>&1; c=$?
rule "A3 r3c.ts arm-check on the committed manifests" "exit 0, pass" "exit $c, $(cut -c1-160 "$W/log/arm-check.txt")" test "$c" = 0
mkdir -p "$W/arm-neg" && cp "$L"/arms/manifests/r3b/*.json "$W/arm-neg/" && jq '.settings.hooks |= map(select(.event != "Stop"))' "$M" > "$W/arm-neg/gstack.json"
(cd "$E" && bun quick/laneb/r3/r3c.ts arm-check "$W/arm-neg") > "$W/log/arm-check-neg.txt" 2>&1; c=$?
rule "A4 arm-check on a manifest copy with the Stop entry removed" "exit 1, FAIL" "exit $c, $(cut -c1-200 "$W/log/arm-check-neg.txt")" sh -c "test $c = 1 && grep -q 'FAIL' '$W/log/arm-check-neg.txt'"
mkdir -p "$W/none-build/manifests"
(cd "$L/arms" && ARMS=none MANIFESTS="$W/none-build/manifests" bash build.sh "$W/none-build/out") > "$W/log/none-build.txt" 2>&1; c=$?
rule "A5 the none arm rebuilt with arms/build.sh matches manifests/r3/none.json and r3b/none.json byte for byte" "identical, tar 8bab4833" "build exit $c, $(jq -r .tar.sha256 "$W/none-build/manifests/none.json" 2>/dev/null)" sh -c "test $c = 0 && cmp '$W/none-build/manifests/none.json' '$L/arms/manifests/r3/none.json' && cmp '$W/none-build/manifests/none.json' '$L/arms/manifests/r3b/none.json'"
cp "$W/none-build/out/none.tar.gz" "$A/none.tar.gz" 2>/dev/null; rm -rf "$W/none-build/out/gstack.tar" "$W/none-build/out/gstack-src" "$W/none-build/out/trees"

echo; echo "## Part B: the in-hand records and prepare against the real SWE-rebench export"
cp "$CAPT/GEVA-248/record/trials.jsonl" "$I/opus-r3a4.jsonl"; cp "$CAPT/GEVA-249/r3a4-sol/trials.jsonl" "$I/sol-r3a4.jsonl"
cp "$CAPT/GEVA-275/trials.tsv" "$I/opus-r3b.tsv"; cp "$CAPT/GEVA-276/trials.tsv" "$I/sol-r3b.tsv"; cp "$CAPT/GEVA-248/r3a4-drop-list.json" "$I/r3a4-drop-list.json"
cp "$E/wiki/plans/2026-11-05-geva-284-r3c-calibrated/r3c-taskfit.tsv" "$I/r3c-taskfit.tsv"
inh=$(cd "$E" && bun -e 'const { INHAND } = await import("./quick/laneb/r3/r3c-rules.ts"); const { createHash } = await import("node:crypto"); const fs = await import("node:fs"); let ok = 0; for (const [f, h] of Object.entries(INHAND)) if (createHash("sha256").update(fs.readFileSync(process.argv[1] + "/" + f)).digest("hex") === h) ok++; console.log(ok + " of " + Object.keys(INHAND).length);' "$I")
rule "B1 the six in-hand records match r3c-rules.ts INHAND by sha256" "6 of 6" "$inh" test "$inh" = "6 of 6"
if [ -z "${SWEREBENCH:-}" ]; then
  if [ ! -d "$W/ref/swe-rebench-leaderboard" ]; then
    UV_TOOL_DIR="$W/uv-tools" UV_TOOL_BIN_DIR="$W/bin" uv tool install --quiet --force "harbor @ git+https://github.com/harbor-framework/harbor@f3a5a923f6d722472c9e797af836ae4425fb215f" > "$W/log/harbor-install.txt" 2>&1
    "$W/bin/harbor" datasets download "swe-rebench/swe-rebench-leaderboard@sha256:ebe7444e313a0d8db94fa541139826eaebe2b0abcd4900c6f73e750494910dca" -o "$W/ref" > "$W/log/download.txt" 2>&1
  fi
  SWEREBENCH=$W/ref/swe-rebench-leaderboard
fi
HIDE="$HIDE; s#$SWEREBENCH#\$SWEREBENCH#g"
(cd "$L/tasks" && python3 swerebench_r3a4.py --export "$SWEREBENCH") > "$W/log/pool-check.txt" 2>&1; c=$?
PARSERS=$(sha256sum "$SWEREBENCH"/*/tests/parser.py | cut -c1-64 | sort | uniq -c | sed 's/^ *//')
rule "B2 the export holds the pool (swerebench_r3a4.py --check) and every task ships the parser r3c.ts pins" "exit 0; 860 bdedd1a6..." "exit $c, $(ls "$SWEREBENCH" | wc -l) tasks; parsers: $PARSERS" sh -c "test $c = 0 && test '$PARSERS' = '860 bdedd1a6f64c511cc1e2c8c4b9e9402d61e68d962dbebc78d9d86ec4ea3cd261'"
cat > "$W/bin/approve.ts" <<'EOF'
const { Ledger } = await import(process.argv[2] + "/runner/ledger.ts");
new Ledger(process.env.LEDGER!, "r3c-opus").approve(Number(process.argv[3]), { words: "dry run, temporary ledger, no spend", by: "dry run", source: "dry run" });
EOF
approve() { LEDGER=$1 bun "$W/bin/approve.ts" "$L" "$2"; }
prep() { (cd "$E" && LEDGER=$1 INHAND=$3 ARMS="$A" SWEREBENCH="$SWEREBENCH" bun quick/laneb/r3/r3c.ts prepare "$2" --item r3c-opus); }
P=$W/prep; rm -rf "$P"/*
approve "$P/at700.jsonl" 700; prep "$P/at700.jsonl" "$P/work700" "$I" > "$W/log/prep-700.txt" 2>&1; c=$?
rule "B3 prepare on a ledger approved at \$700" "refused, nothing prepared" "exit $c, $(first 'cap' "$W/log/prep-700.txt" | sed 's/^E_INTERNAL //')" sh -c "test $c != 0 && test ! -e '$P/work700/plan.json'"
mkdir -p "$W/inhand-bad" && cp "$I"/* "$W/inhand-bad/" && echo >> "$W/inhand-bad/opus-r3b.tsv"
approve "$P/r3c-opus.jsonl" 750; prep "$P/r3c-opus.jsonl" "$P/work-bad" "$W/inhand-bad" > "$W/log/prep-bad.txt" 2>&1; c=$?
rule "B4 prepare with one in-hand record changed by a byte" "exit 2, E_PIN_MISMATCH, nothing prepared" "exit $c, $(cut -c1-160 "$W/log/prep-bad.txt")" sh -c "test $c = 2 && grep -q '^E_PIN_MISMATCH' '$W/log/prep-bad.txt' && test ! -e '$P/work-bad/plan.json'"
prep "$P/r3c-opus.jsonl" "$P/work" "$I" > "$W/log/prep.txt" 2>&1; c=$?
parts=$(cd "$P/work" && for d in cal-1a cal-1b cal-1c; do echo -n "$d:$(jq '.tasks | length' "$d/job.json")/$(jq 'length' "$d/order.json")/$(jq '[.tasks[].path] | unique | length' "$d/oracle.json") "; done)
rule "B5 prepare: cal-1a to cal-1c, each 146 trials in the job (a task once per trial), 146 in order.json, its oracle.json over its tasks" "cal-1a:146/146/n cal-1b:146/146/n cal-1c:146/146/n" "exit $c, $parts" sh -c "test $c = 0 && echo '$parts' | grep -Eq '^cal-1a:146/146/[0-9]+ cal-1b:146/146/[0-9]+ cal-1c:146/146/[0-9]+ $'"
plan=$(jq -c '{item, cap_usd, candidates_sha256: .candidates_sha256[0:8], cal1_seed, grid_seed, parts: [.cal1_parts[].trials], closing: .closing_line_sha256[0:8], parser: .parser_sha256[0:8]}' "$P/work/plan.json")
cl=$(echo -n "$(cd "$E" && bun -e 'const { R3C } = await import("./quick/laneb/r3/r3c-rules.ts"); process.stdout.write(R3C.closingLine)')" | sha256sum | cut -c1-8)
rule "B6 plan.json names the item, the cap, the frozen candidates, both seeds, the closing line and the pinned parser" "r3c-opus \$750 7a2b33c7 20261112 20261113 [146,146,146] closing $cl parser bdedd1a6" "$plan" sh -c "echo '$plan' | jq -e '.item == \"r3c-opus\" and .cap_usd == 750 and .candidates_sha256 == \"7a2b33c7\" and .cal1_seed == 20261112 and .grid_seed == 20261113 and .parts == [146,146,146] and .closing == \"$cl\" and .parser == \"bdedd1a6\"'"
arms=$(jq -c '[.agents[] | {arm: .kwargs.arm, closing: .kwargs.closing_line}]' "$P/work/cal-1a/job.json")
rule "B7 the cal-1 parts run the none arm alone, with the closing line" "[{none, r3c-root-cause-files}]" "$arms" test "$arms" = '[{"arm":"none","closing":"r3c-root-cause-files"}]'

echo; echo "## Part C: run, admit, grid, resume and analyze, scenario by scenario"
cat > "$W/bin/play-vm-run.ts" <<'EOF'
// Plays vm-run.sh for launch.sh's batch ssh, adapted from the R3b pin post's dry run. The "VM" is $VM. It runs the part's
// planned trials in the part's order (order.json for cal-1 and grid parts, Harbor's attempt-major order for cal-2), reserving
// each session at the cells' $12 cap and settling it through the item's ledger service on LEDGER_PORT, as the proxy does,
// writes each trial's files in the shape r3c.ts loadTrials reads (config.json, result.json, the agent's sdk-messages.jsonl
// with parent_tool_use_id, the verifier's report.json, a told trial's hook log), one oracle trial a task (its model.patch
// under artifacts/logs/artifacts/ and its report.json), and the order line in out/harbor.log. The scenario module named by
// SCEN gives, per trial, pass, infra, spend, budget (allowance stop), shown (procedure visible), block (the closing block's
// files, or null for none) and old (a log written before parent_tool_use_id was captured).
import { appendFileSync, existsSync, mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { join } from "node:path";
const scen = (await import(process.env.SCEN!)).default;
const m = /vm-run\.sh work\/run (\S+) (\S+)/.exec(process.argv.at(-1) ?? "");
const prefix = m![1]!, part = prefix.replace(/^r3c-opus-/, "").slice(0, -1);
const run = join(process.env.VM!, "work/run"), port = process.env.LEDGER_PORT!;
appendFileSync(process.env.EVENTS!, "batch started: vm-run.sh work/run " + prefix + " " + m![2] + "\n");
for (const d of ["out", "jobs"]) rmSync(join(run, d), { recursive: true, force: true });
mkdirSync(join(run, "out"), { recursive: true });
const job = JSON.parse(readFileSync(join(run, "job.json"), "utf8"));
const name = (p: string) => String(p).split("/").filter(Boolean).at(-1)!;
const tasks: string[] = job.tasks.map((t: any) => name(t.path));
const arms: string[] = job.agents.map((a: any) => a.kwargs.arm);
const hasOrder = existsSync(join(run, "order.json"));
const plan: [string, string][] = hasOrder ? JSON.parse(readFileSync(join(run, "order.json"), "utf8")) : Array.from({ length: job.n_attempts }, () => tasks.flatMap((t) => arms.map((a) => [t, a] as [string, string]))).flat();
if (hasOrder) writeFileSync(join(run, "out/harbor.log"), "laneb order: applied " + plan.length + " trials from " + join(run, "order.json") + "\n");
const call = async (path: string, body: object) => (await fetch("http://127.0.0.1:" + port + path, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) })).json() as any;
const testsOf = (t: string) => JSON.parse(readFileSync(join(run, "tasks", t, "tests/config.json"), "utf8"));
const report = (t: string, pass: boolean) => { const c = testsOf(t); return { fail_to_pass: Object.fromEntries(c.FAIL_TO_PASS.map((x: string, i: number) => [x, pass || i > 0 ? "PASSED" : "FAILED"])), pass_to_pass: Object.fromEntries(c.PASS_TO_PASS.map((x: string) => [x, "PASSED"])), resolved: pass }; };
const gold = (t: string) => "src/" + t.replace(/[^A-Za-z0-9]/g, "_") + ".py";
for (const t of [...new Set(tasks)]) {
  const d = join(run, "jobs", job.job_name + "-oracle", t + "__oracle");
  mkdirSync(join(d, "artifacts/logs/artifacts"), { recursive: true }); mkdirSync(join(d, "verifier"), { recursive: true });
  writeFileSync(join(d, "config.json"), JSON.stringify({ task: { path: "/vm/run/tasks/" + t }, agent: { name: "oracle" } }));
  writeFileSync(join(d, "result.json"), JSON.stringify({ trial_name: t + "__oracle", verifier_result: { rewards: { reward: 1 } }, exception_info: null }));
  writeFileSync(join(d, "artifacts/logs/artifacts/model.patch"), "diff --git a/" + gold(t) + " b/" + gold(t) + "\n--- a/" + gold(t) + "\n+++ b/" + gold(t) + "\ndiff --git a/tests/test_fix.py b/tests/test_fix.py\n");
  writeFileSync(join(d, "verifier/report.json"), JSON.stringify(report(t, true)));
}
const seen = new Map<string, number>();
let rc = 0;
for (const [k, [task, arm]] of plan.entries()) {
  const j = seen.get(task + "/" + arm) ?? 0; seen.set(task + "/" + arm, j + 1);
  const kw = job.agents.find((a: any) => a.kwargs.arm === arm).kwargs, told = arm === "gstack";
  const t = { pass: false, infra: false, spend: 0.25, budget: false, shown: true, block: [gold(task)] as string[] | null, old: false, ...scen(part, task, j, arm, k) };
  const trial = task.slice(0, 20) + "__" + arm + "__" + k, sid = "s-" + k, d = join(run, "jobs", job.job_name, trial);
  const r = await call("/reserve", { id: prefix + sid, usd_micros: 12_000_000, meta: { trial: trial + "__env" } });
  if (!r.ok) { writeFileSync(join(run, "out/proxy-status-end.json"), JSON.stringify({ cap_stop: "ledger refused a reserve for " + trial })); rc = 130; break; }
  mkdirSync(join(d, "evaluator"), { recursive: true }); mkdirSync(join(d, "verifier"), { recursive: true });
  writeFileSync(join(d, "config.json"), JSON.stringify({ task: { path: "/vm/run/tasks/" + task }, agent: { name: "laneb-claude-code", kwargs: { arm, upstream_model: kw.upstream_model, closing_line: kw.closing_line, ...(told ? { told_skill: kw.told_skill, procedure_line: kw.procedure_line } : {}) } } }));
  writeFileSync(join(d, "result.json"), JSON.stringify({ trial_name: trial, verifier_result: t.infra ? null : { rewards: { reward: t.pass ? 1 : 0 } }, exception_info: null }));
  if (!t.infra) writeFileSync(join(d, "verifier/report.json"), JSON.stringify(report(task, t.pass)));
  const text = (told && t.shown ? "## Phase 1\nDEBUG REPORT\n\n" : "Fixed it.\n\n") + (t.block ? "ROOT CAUSE FILES\n" + t.block.join("\n") : "");
  const recs = [{ type: "AssistantMessage", content: [{ thinking: "ROOT CAUSE FILES\nsrc/thinking.py", signature: "s" }, { text: "Looking." }], parent_tool_use_id: null },
    { type: "AssistantMessage", content: [{ text: "ROOT CAUSE FILES\nsrc/subagent.py" }], parent_tool_use_id: "toolu_sub" }, { type: "AssistantMessage", content: [{ text }], parent_tool_use_id: null }];
  writeFileSync(join(d, "evaluator/sdk-messages.jsonl"), recs.map((x) => { const y: any = { ...x }; if (t.old) delete y.parent_tool_use_id; return JSON.stringify(y); }).join("\n") + "\n");
  if (told) { mkdirSync(join(d, "evaluator/gstack-hooks/hooks"), { recursive: true }); writeFileSync(join(d, "evaluator/gstack-hooks/hooks/skill-contract.log"), "2026-11-05T00:00:00Z session=" + sid + " event=" + (t.shown ? "met" : "blocked") + "\n"); }
  const micros = Math.round(t.spend * 1e6);
  appendFileSync(join(run, "out/proxy-sessions.jsonl"), JSON.stringify({ id: sid, trial: trial + "__env", model: kw.upstream_model, spent_micros: micros, ok_responses: t.infra ? 0 : 3, requests: 4, refused: t.budget ? 1 : 0, refused_budget: t.budget ? 1 : 0, opened_at: "2026-11-05T00:00:00Z", closed_at: "2026-11-05T01:00:00Z", cmd_expanded: told, cmd_names: told ? ["investigate"] : [] }) + "\n");
  await call("/settle", { id: prefix + sid, usd_micros: micros });
}
if (rc === 0) writeFileSync(join(run, "out/proxy-status-end.json"), JSON.stringify({ cap_stop: null }));
writeFileSync(join(run, "out/vm-run.exit"), rc + "\n");
appendFileSync(process.env.EVENTS!, "batch ended " + rc + "\n");
process.exit(rc);
EOF
printf '#!/usr/bin/env bash\nexec bun %s "$@"\n' "$W/bin/play-vm-run.ts" > "$W/bin/vm/ssh" && chmod +x "$W/bin/vm/ssh"
cat > "$W/bin/launch-ubi" <<'EOF'
#!/usr/bin/env bash
# Stub ubi-runner, from the R3b pin post's dry run: the "VM" is the directory $VM, made fresh at each up; sync uses hard
# links so the arm tars aren't copied; every call is logged to $EVENTS. No Ubicloud call.
set -u
cmd=$1; shift
echo "$cmd $*" >> "$EVENTS"
case "$cmd" in
  up) rm -rf "$VM"; mkdir -p "$VM" "$STATE/vm-1"; echo "IP=127.0.0.1" > "$STATE/vm-1/env"; touch "$STATE/vm-1/key"; echo vm-1 ;;
  sync) mkdir -p "$VM/$3"; cp -rl "$2/." "$VM/$3/" ;;
  ssh) case "$2" in
         *STOP*) touch "$VM/work/run/STOP" ;;
         *vm-run.exit*) test -f "$VM/work/run/out/vm-run.exit" ;;
         *"find out jobs"*) cd "$VM/work/run" && find out jobs -type f | wc -l ;;
         *) timeout 5 cat > /dev/null || true ;;
       esac ;;
  pull) cp -r "$VM/$2" "$3/" ;;
  down) ;;
esac
EOF
chmod +x "$W/bin/launch-ubi"
# Scenario modules: scen(part, task, j, arm, k) gives a trial's fields; j counts the task and arm's earlier trials in the part.
# h is a stable 0/1 from the task id, so about half of the candidates lean each way.
scen() { printf 'const h = (t: string) => [...t].reduce((a, c) => a + c.charCodeAt(0), 0) %% 2;\nexport default (part: string, task: string, j: number, arm: string, k: number) => (%s);\n' "$2" > "$W/scen/$1.ts"; }
CAL='part.startsWith("cal") ? { pass: (j + h(task)) % 2 === 0 } : '
KEEP='{ pass: arm === "gstack" ? j % 4 !== 3 : j % 2 === 0, spend: arm === "gstack" ? 0.4 : 0.25 }'
scen go "$CAL$KEEP"
scen stop "part.startsWith(\"cal\") ? { pass: (j + h(task)) % 2 === 0, spend: 0.7 } : $KEEP"
scen proj-cal "part === \"cal-1a\" ? { pass: (j + h(task)) % 2 === 0, spend: 1.2 } : $CAL$KEEP"
scen infra-cal "part === \"cal-1b\" ? { pass: true, infra: k < 15 } : $CAL$KEEP"
scen unreadable "$CAL k === 5 && part === \"grid-1\" ? { pass: true, old: true } : $KEEP"
scen proj-grid "$CAL part === \"grid-1\" ? { pass: arm === \"gstack\", spend: arm === \"gstack\" ? 1.5 : 0.25 } : $KEEP"
scen cap-grid "$CAL part === \"grid-1\" ? { pass: arm === \"gstack\", spend: arm === \"gstack\" ? 6 : 0.25 } : $KEEP"
scen cut "$CAL{ pass: j % 2 === 0, spend: 0.3 }"

# setup <name>: a copy of the prepared work dir and a fresh ledger approved at $750.
setup() { local x=$W/s/$1; rm -rf "$x"; mkdir -p "$x/state"; cp -a "$P/work" "$x/work"; approve "$x/r3c-opus.jsonl" 750; }
# r3c <name> <cmd> <scenario>: r3c.ts with the stub runner and the stub ssh; output in log/<name>-<cmd>.txt.
r3c() {
  local x=$W/s/$1 cmd=$2 sc=$3; port=$((port + 1))
  (cd "$E" && env PATH="$W/bin/vm:$PATH" SCEN="$W/scen/$sc.ts" UBICLOUD_API_KEY=dry-run-placeholder-not-a-key LEDGER="$x/r3c-opus.jsonl" INHAND="$I" ARMS="$A" SWEREBENCH="$SWEREBENCH" \
    UBI_OWNER="$UBI_OWNER" UBI_RUNNER="$W/bin/launch-ubi" UBI_RUNNER_STATE="$x/state" STATE="$x/state" VM="$x/vm" EVENTS="$x/events.log" \
    PULL_RETRY_SEC=0 UBI_TTL_HOURS="${TTL:-12}" VM_USD_PER_HOUR=0.6 VM_RESERVE_HOURS=4 LEDGER_PORT="$port" \
    bun quick/laneb/r3/r3c.ts "$cmd" "$x/work" --item r3c-opus) < /dev/null > "$W/log/$1-$cmd.txt" 2>&1
}
an() { local c; r3c "$1" analyze x; c=$?; cp "$W/log/$1-analyze.txt" "$W/log/$1-analyze-$c.txt"; echo $c; }
outcome() { first '^R3c outcome' "$W/log/$1-analyze.txt"; }
launched() { ls "$W/s/$1/work" | grep -E 'launch-exit$|projection-stop|raised$|grid.ready|admit.stop' | tr '\n' ' '; }
rows() { jq -s -c '[(map(select(.kind == "reserve")) | length), (map(select(.kind == "settle")) | length)]' "$W/s/$1/r3c-opus.jsonl"; }
