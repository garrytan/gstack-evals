#!/usr/bin/env python3
# gen-synth.py <staging>: writes a synthetic 44-unit panel in the authors' layout (units/<ID>/ with plan.md, context/,
# key.json) and passing audit files, with the assignment table's kinds, item kinds and modes. Test material only: the
# text is filler, never a real unit. GEVA-115.
import json, os, sys
st = os.path.abspath(sys.argv[1])
T = {"K": [("hard","H-X+H-F","greenfield"),("hard","H-X+H-T","added"),("hard","H-F+H-T","fix"),("hard","H-X+H-F","greenfield"),("premise","P-premise","added"),("premise","P-size","greenfield"),("ambition","U","added"),("ambition","U","fix"),("decoy","D-H","fix"),("decoy","D-P","greenfield"),("decoy","D-U","added")],
     "L": [("hard","H-X+H-T","greenfield"),("hard","H-F+H-T","added"),("hard","H-X+H-F","fix"),("hard","H-X+H-T","added"),("premise","P-alt","fix"),("premise","P-premise","greenfield"),("ambition","U","fix"),("ambition","U","added"),("decoy","D-H","added"),("decoy","D-P","fix"),("decoy","D-U","fix")],
     "M": [("hard","H-F+H-T","greenfield"),("hard","H-X+H-F","added"),("hard","H-X+H-T","fix"),("hard","H-F+H-T","fix"),("premise","P-size","added"),("premise","P-alt","greenfield"),("ambition","U","added"),("ambition","U","fix"),("decoy","D-H","greenfield"),("decoy","D-P","added"),("decoy","D-U","added")],
     "N": [("hard","H-X+H-F","added"),("hard","H-X+H-T","greenfield"),("hard","H-F+H-T","fix"),("hard","H-X+H-F","greenfield"),("premise","P-premise","fix"),("premise","P-size","added"),("ambition","U","fix"),("ambition","U","added"),("decoy","D-H","fix"),("decoy","D-P","greenfield"),("decoy","D-U","fix")]}
os.makedirs(os.path.join(st, "audit"))
verd = {"A": [], "B": []}
for a, rows in T.items():
    for n, (kind, ik, mode) in enumerate(rows, 1):
        uid = a + "%02d" % n; d = os.path.join(st, "units", uid); os.makedirs(os.path.join(d, "context"))
        heads = ["Goal", "Background", "Approach", "Milestones", "Team", "Rollout", "Measures"]
        open(os.path.join(d, "plan.md"), "w").write("# Plan " + uid + "\n\n" + "".join("## " + h + "\n\n" + ("Filler text for " + h + ". ") * 12 + "\n\n" for h in heads))
        for i in range(4): open(os.path.join(d, "context", "doc%d.md" % i), "w").write(("Context filler %d. " % i) * 90)
        ev = [{"path": "context/doc0.md", "what": "filler"}]
        if kind == "decoy":
            key = {"unit": uid, "kind": kind, "mode": mode, "bait": {"resembles": ik, "plan_section": "Approach", "choice": "A choice.", "why_not_a_flaw": "The context backs it. It is right.", "justification": ev}}
        else:
            items = []
            for j, k in enumerate(ik.split("+"), 1):
                it = {"id": uid + "-" + str(j), "item_kind": k, "plan_section": heads[j + 1], "cause": "Synthetic cause " + k + ".", "evidence": ev, "deserved_severity": "High", "fix": "Fix it."}
                if k == "H-X": it["other_places"] = ["Rollout"]
                if kind in ("premise", "ambition"): it.update(decision="The plan builds X.", should_become="Build Y instead.")
                items.append(it)
            key = {"unit": uid, "kind": kind, "mode": mode, "items": items}
        json.dump(key, open(os.path.join(d, "key.json"), "w"))
        verd["A" if a in "KL" else "B"].append({"unit": uid, "verdict": "pass"})
for k, v in verd.items(): json.dump(v, open(os.path.join(st, "audit", "audit-" + k + ".json"), "w"))
print("synthetic units", sum(len(r) for r in T.values()))
