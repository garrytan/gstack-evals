#!/usr/bin/env python3
# exposure.py: the CEO panel v2 admission pilot's procedure gate and exposure read (prereg section 6). Reads each
# session's host/transcript.ndjson and host/init.json under <out>/runs/<model>/<arm>/<task>/p<pass>/ and the run's
# modes.tsv. Prints per-arm counts only, never transcript text. Writes per-session rows to --tsv when given.
# Usage: python3 exposure.py --out <review-units out dir> --model claude-opus-5-5 --arms none,gstack,gstack-forced
#        --sessions-per-arm 16 [--floor 21300] [--tsv <file>] [--json]
import argparse, json, os, re, statistics, sys

MODES = ["SCOPE EXPANSION", "SELECTIVE EXPANSION", "HOLD SCOPE", "SCOPE REDUCTION"]
SECTION_TITLES = {
    1: [r"architecture"], 2: [r"error\s*(?:&|and)\s*rescue", r"\berror\b"], 3: [r"security", r"threat model"],
    4: [r"data flow", r"edge case"], 5: [r"code quality"], 6: [r"\btests?\b"], 7: [r"performance"],
    8: [r"observability", r"debuggability"], 9: [r"deployment", r"rollout"], 10: [r"trajectory", r"long[- ]term"],
}
SECTIONS_FILE = "plan-ceo-review/sections/review-sections.md"
PRINT_CMD = re.compile(r"(^|[\s;&|(])(cat|sed|head|tail|awk)(\s|$)")
HEADING = re.compile(r"^\s{0,3}(#{1,6}\s+.+|\*\*[^*].*\*\*\s*:?\s*|section\s+\d{1,2}\b.*)$", re.I)
SECTION_NUM = re.compile(r"\bsection\s+(\d{1,2})\b", re.I)
MIN_READ_CHARS, MIN_BODY_CHARS, MIN_SECTIONS = 1000, 150, 9

def result_text(c):
    if isinstance(c, str): return c
    if isinstance(c, list): return "".join(x.get("text", "") if isinstance(x, dict) else str(x) for x in c)
    return ""

def is_sections_read(name, inp):
    if name == "Read":
        return str(inp.get("file_path", "")).endswith(SECTIONS_FILE)
    if name == "Bash":
        cmd = str(inp.get("command", ""))
        paths = re.findall(r"[^\s;&|()'\"<>]*review-sections\.md", cmd)
        own = all(p.endswith(SECTIONS_FILE) or re.fullmatch(r"(\./)?sections/review-sections\.md", p) for p in paths)
        return bool(paths) and own and "plan-ceo-review" in cmd and bool(PRINT_CMD.search(cmd))
    return False

def section_of(line):
    m = SECTION_NUM.search(line)
    if m and 1 <= int(m.group(1)) <= 10: return int(m.group(1))
    low = line.lower()
    for n, pats in SECTION_TITLES.items():
        if any(re.search(p, low) for p in pats): return n
    return None

def sections_ran(text):
    lines = text.split("\n")
    heads = []
    for i, l in enumerate(lines):
        if HEADING.match(l):
            n = section_of(l)
            if n: heads.append((i, n))
    # longest strictly increasing run of section numbers, in order of appearance (O(k^2), k is small)
    best = [[h] for h in heads]
    for j in range(len(heads)):
        for i in range(j):
            if heads[i][1] < heads[j][1] and len(best[i]) + 1 > len(best[j]): best[j] = best[i] + [heads[j]]
    chain = max(best, key=len) if best else []
    counted = 0
    for k, (i, n) in enumerate(chain):
        end = chain[k + 1][0] if k + 1 < len(chain) else len(lines)
        body = "\n".join(lines[i + 1:end]).strip()
        if len(body) >= MIN_BODY_CHARS: counted += 1
    return counted, len({n for _, n in chain})

def read_session(d):
    tp = os.path.join(d, "host", "transcript.ndjson")
    if not os.path.exists(tp): return None
    msgs = []
    for l in open(tp, encoding="utf8", errors="replace"):
        l = l.strip()
        if not l: continue
        try: msgs.append(json.loads(l))
        except Exception: pass
    uses, results, order = {}, {}, []
    first_tokens, helpers, agents, subtype, ids = None, 0, 0, None, set()
    for m in msgs:
        t = m.get("type")
        if t == "assistant":
            msg = m.get("message", {})
            if msg.get("id"): ids.add(msg["id"])
            u = msg.get("usage")
            if first_tokens is None and isinstance(u, dict):
                first_tokens = sum(int(u.get(k) or 0) for k in ("input_tokens", "cache_creation_input_tokens", "cache_read_input_tokens"))
            for b in msg.get("content", []) or []:
                if b.get("type") == "text": order.append(("text", b.get("text", "")))
                elif b.get("type") == "tool_use":
                    uses[b.get("id")] = (b.get("name"), b.get("input") or {})
                    order.append(("use", b.get("id")))
                    if b.get("name") == "Bash" and re.search(r"gstack-[a-z]", str((b.get("input") or {}).get("command", ""))): helpers += 1
                    if b.get("name") in ("Task", "Agent"): agents += 1
        elif t == "user":
            c = m.get("message", {}).get("content", [])
            for b in c if isinstance(c, list) else []:
                if isinstance(b, dict) and b.get("type") == "tool_result":
                    results[b.get("tool_use_id")] = (bool(b.get("is_error")), result_text(b.get("content")))
                    order.append(("result", b.get("tool_use_id")))
        elif t == "result":
            subtype = m.get("subtype")
    read_at = None
    for k, (kind, v) in enumerate(order):
        if kind == "result" and v in uses and is_sections_read(*uses[v]):
            err, txt = results.get(v, (True, ""))
            if not err and len(txt) >= MIN_READ_CHARS:
                read_at = k
                break
    texts = [v for kind, v in order if kind == "text"]
    mode_text = any(sum(1 for mo in MODES if mo in t) == 1 for t in texts)
    after = "\n".join(v for kind, v in order[read_at + 1:] if kind == "text") if read_at is not None else ""
    counted, distinct = sections_ran(after) if read_at is not None else (0, 0)
    sp = None
    ip = os.path.join(d, "host", "init.json")
    if os.path.exists(ip):
        try: sp = json.load(open(ip)).get("system_prompt")
        except Exception: sp = None
    return dict(read=read_at is not None, mode_text=mode_text, sections=counted, distinct=distinct,
                first_tokens=first_tokens, system_prompt=sp, helpers=helpers, agents=agents, turns=len(ids), subtype=subtype)

def modes_tsv(out):
    # review-units writes arm_id, arm, model, unit, pass, mode; the unit id (panel/plans#<pr>@<head>) and the session's task
    # dir (plans-<pr>-<head8>) share the PR number, so a row is keyed by (arm, pr, pass).
    p = os.path.join(out, "modes.tsv")
    got = {}
    if not os.path.exists(p): return got
    rows = [l.rstrip("\n").split("\t") for l in open(p) if l.strip()]
    if not rows: return got
    head = rows[0]
    if not all(n in head for n in ("arm", "unit", "pass", "mode")): raise SystemExit("E_INPUT exposure.py: modes.tsv lacks the arm, unit, pass and mode columns")
    col = {n: head.index(n) for n in ("arm", "unit", "pass", "mode")}
    for r in rows[1:]:
        m = re.search(r"#(\d+)(?:@[0-9a-f]+)?$", r[col["unit"]].strip())
        mode = r[col["mode"]].strip()
        if m and mode in MODES: got[(r[col["arm"]].strip(), m.group(1), r[col["pass"]].strip())] = mode
    return got

def task_pr(task):
    m = re.search(r"-(\d+)-[0-9a-f]{8}$", task)
    return m.group(1) if m else task

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", required=True); ap.add_argument("--model", required=True); ap.add_argument("--arms", required=True)
    ap.add_argument("--sessions-per-arm", type=int, required=True); ap.add_argument("--floor", type=int, default=21300)
    ap.add_argument("--forced", default="gstack-forced"); ap.add_argument("--control", default="none")
    ap.add_argument("--tsv"); ap.add_argument("--json", action="store_true")
    a = ap.parse_args()
    root = os.path.join(a.out, "runs", a.model.replace("/", "_"))
    mt = modes_tsv(a.out)
    summary, rows = {}, []
    for arm in a.arms.split(","):
        ad = os.path.join(root, arm)
        sess = []
        if os.path.isdir(ad):
            for task in sorted(os.listdir(ad)):
                for p in sorted(os.listdir(os.path.join(ad, task))):
                    if not re.fullmatch(r"p\d+", p): continue
                    s = read_session(os.path.join(ad, task, p))
                    if s is None: continue
                    s["mode_tsv"] = (arm, task_pr(task), p[1:]) in mt
                    s["mode"] = s["mode_text"] or s["mode_tsv"]
                    s["exposed"] = s["read"] and s["mode"]
                    s["ran"] = s["exposed"] and s["sections"] >= MIN_SECTIONS
                    sess.append(s)
                    rows.append([arm, task, p] + [str(s[k]) for k in ("read", "mode_text", "mode_tsv", "exposed", "sections", "ran", "first_tokens", "system_prompt", "helpers", "agents", "turns", "subtype")])
        ft = [s["first_tokens"] for s in sess if s["first_tokens"] is not None]
        summary[arm] = dict(scheduled=a.sessions_per_arm, transcripts=len(sess), read=sum(s["read"] for s in sess), mode=sum(s["mode"] for s in sess),
                            exposed=sum(s["exposed"] for s in sess), ran=sum(s["ran"] for s in sess),
                            preset=sum(s["system_prompt"] == "claude-code-preset" for s in sess),
                            first_tokens_median=statistics.median(ft) if ft else None, first_tokens_min=min(ft) if ft else None,
                            above_floor=sum(1 for x in ft if x >= a.floor), helpers=sum(s["helpers"] for s in sess), agents=sum(s["agents"] for s in sess),
                            max_turn_endings=sum(1 for s in sess if s["subtype"] == "error_max_turns"))
    total = sum(v["transcripts"] for v in summary.values())
    sp_ok = all(v["preset"] == v["transcripts"] for v in summary.values())
    ctl = summary.get(a.control, {})
    floor_ok = ctl.get("transcripts") == a.sessions_per_arm and ctl.get("above_floor") == a.sessions_per_arm
    f = summary.get(a.forced)
    if f is None: proc = "NOT RUN"
    elif f["transcripts"] < a.sessions_per_arm: proc = "NO READ"
    else: proc = "READY" if f["ran"] >= a.sessions_per_arm - 1 else "FIX-METHOD"
    verdict = dict(arms=summary, system_prompt_check="PASS" if (sp_ok and floor_ok) else "FAIL", procedure=proc, sessions=total, floor=a.floor)
    if a.tsv:
        with open(a.tsv, "w") as fh:
            fh.write("\t".join(["arm", "task", "pass", "read", "mode_text", "mode_tsv", "exposed", "sections", "ran", "first_tokens", "system_prompt", "helpers", "agents", "turns", "subtype"]) + "\n")
            for r in rows: fh.write("\t".join(r) + "\n")
    if a.json: print(json.dumps(verdict, indent=2))
    else:
        for arm, v in summary.items():
            print(f"{arm}: transcripts {v['transcripts']}/{v['scheduled']}, read {v['read']}, mode {v['mode']}, exposed {v['exposed']}, ran {v['ran']}, preset {v['preset']}, first-request tokens median {v['first_tokens_median']} min {v['first_tokens_min']} (>= {a.floor}: {v['above_floor']}), helpers {v['helpers']}, agent calls {v['agents']}, turn-cap endings {v['max_turn_endings']}")
        print(f"system prompt check: {verdict['system_prompt_check']}; procedure: {proc}")

if __name__ == "__main__":
    main()
