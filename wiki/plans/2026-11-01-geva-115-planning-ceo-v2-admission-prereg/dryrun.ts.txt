async function cli(label: string, argv: string[], o: { show?: number } = {}) {
  const out: string[] = [], err: string[] = [];
  const code = await nolossCli(argv, { out: (t: string) => out.push(t), err: (t: string) => err.push(t) });
  say("$ noloss " + argv.map((a) => (/\s/.test(a) ? JSON.stringify(a) : a)).join(" "));
  say("  exit " + code + "  (" + label + ")");
  if (o.show !== 0) show(out.join("\n") + "\n" + err.join("\n").split("\n").filter((l) => /^E_[A-Z_]+ /.test(l)).join("\n"), o.show ?? 6);
  return { code, out: out.join("\n"), err: err.join("\n") };
}
function units(argv: string[], label: string, n = 14) {
  const r = run([join(scorer, "bin/gstack-evals"), "review-units", ...argv]);
  say("$ review-units " + argv.map((a) => (/\s/.test(a) ? JSON.stringify(a) : a)).join(" ").replaceAll(GSTACK, "<gstack>"));
  say("  exit " + r.status + "  (" + label + ")");
  show(((r.stdout ?? "") + "\n" + (r.stderr ?? "").split("\n").filter((l) => /^E_[A-Z_]+ /.test(l)).join("\n")).replaceAll(GSTACK, "<gstack>"), n);
  return { code: r.status, out: r.stdout ?? "", err: r.stderr ?? "" };
}

function session(dir: string, events: Ev[], o: { sp?: string | null; first?: number } = {}) {

  let k = 0;
  const usage = () => (k === 1 ? { usage: { input_tokens: 3, cache_creation_input_tokens: (o.first ?? 22270) - 3, cache_read_input_tokens: 0 } } : {});
  const msgs: object[] = events.map((e) => {

    return { type: "assistant", message: { id: "msg_" + ++k, content: [{ type: "text", text: e[1] }], ...usage() } };
  });
  msgs.push({ type: "result", subtype: "success" });
  writeFileSync(join(dir, "host", "transcript.ndjson"), msgs.map((m) => JSON.stringify(m)).join("\n") + "\n");
  if (o.sp !== null) writeFileSync(join(dir, "host", "init.json"), JSON.stringify({ model: OPUS, claude_code_version: "2.1.314", system_prompt: o.sp ?? "claude-code-preset" }));
