#!/usr/bin/env bash
# GEVA-101 prereg dry run: every frozen rule of R1, R2 and R3, keyless, through the code at the pins, with one ledger
# lease per frozen run on temporary ledgers. No step calls a model: judging runs the pinned judge-run in process against
# a fake provider (judge-fake.ts), and R2's and R3's sessions and trials are synthetic fixtures.
# Usage: R=<gstack-evals at 715f837> TAG=<gstack-evals checkout holding tag laneb-stacks-1-runner> ARMC=<org drive
#   handoffs/GSTA-167/arm-C> REV=<GEVA-89 archive judge dir> GSRC=<gstack clone> GBSRC=<gbrain clone> SCRUB=<dir with scrub.ts, scrub2.ts and r1-scrub3.ts> SWEBV=<SWE-bench/experiments
#   checkout with evaluation/verified/2026021*/per_instance_details.json> bash dry-run.sh <empty work dir>
set -u
RAW=$REV/raw/stacks-r1
W=$(mkdir -p "$1" && cd "$1" && pwd)
D=$(cd "$(dirname "$0")" && pwd)
P=$R/wiki/plans/2026-10-31-geva-76-stacks-round-1-prereg
G="$R/bin/gstack-evals noloss"
L="$R/bin/gstack-evals ledger"
export R
unset ANTHROPIC_API_KEY OPENAI_API_KEY OPENROUTER_API_KEY UBICLOUD_API_KEY
mkdir -p "$W/log" "$W/bin"
for f in "$D"/*.txt; do b=$(basename "$f" .txt); case "$b" in *.ts|*.py|*.sh) cp "$f" "$W/bin/$b";; esac; done
cp "$D"/*.json "$W/bin/"
for f in scrub.ts scrub2.ts r1-scrub3.ts; do cp "$SCRUB/$f" "$W/bin/$f"; done
n=0
# step <grep pattern for the lines to show> <command...>: runs it, keeps its full output in log/, shows matching lines.
step() {
  local pat=$1; shift
  n=$((n + 1))
  echo
  echo "\$ $*" | sed "s#$W#\$W#g; s#$R#\$R#g; s#$D#\$D#g"
  "$@" > "$W/log/$n.txt" 2>&1
  local c=$?
  grep -E -- "$pat" "$W/log/$n.txt" | head -"${SHOW:-24}" | sed "s#$W#\$W#g; s#$R#\$R#g"
  echo "exit $c (log/$n.txt)"
}
JUDGES="--primary anthropic/claude-opus-5-5 --second openai/gpt-6.1-sol --third anthropic/claude-sonnet-5-5"
arms() { local root=$1; shift; for a in "$@"; do printf -- ' --arm %s' "$root/$a"; done; }

echo "## Pins"
step . git -C "$R" log -1 --format='rig %H %s'
step . sh -c "git -C $R status --porcelain | wc -l"
step . bash -c "cd $R && sha256sum lanes/review-noloss/{cli,contract,derive,ingest,judge,judge-run,score,secondary,stats,suite,verdict}.ts lanes/review-noloss/suites/gsta167-14pr.json quick/laneb/analysis/analyze.ts && cd $P && sha256sum stacks-r1.json judge-brief.md"
step . sh -c "cd $D && sha256sum *.txt *.json"

step 'OK|FAILED' sh -c "cd $W/bin && printf '%s  %s\\n' c3829af672462a22e22991cb7d80bedffdad3fb9c181c5ce549f8d881a2c4f8e scrub.ts cad13e6af4444df657d3b48c1978b08184e2c4c056f8b8d0f73e9629cdd1b237 scrub2.ts 3a462a557026665d6a367b40dde48f2f9c863f567915c9bff3c4cb4461ae9886 r1-scrub3.ts | sha256sum -c"
sh -c "cd $W/bin && printf '%s  %s\\n' c3829af672462a22e22991cb7d80bedffdad3fb9c181c5ce549f8d881a2c4f8e scrub.ts cad13e6af4444df657d3b48c1978b08184e2c4c056f8b8d0f73e9629cdd1b237 scrub2.ts 3a462a557026665d6a367b40dde48f2f9c863f567915c9bff3c4cb4461ae9886 r1-scrub3.ts | sha256sum -c --status" || { echo "E_R1_SCRUB a scrub script differs from its pinned digest; stopping before any scrub runs"; exit 1; }

echo; echo "## Ledger items at their frozen caps (section 5)"
step '^(review|lane)' bun "$W/bin/items.ts" "$W/record" "$W/laneb"

echo; echo "## R1: source arms, scrub and blinding (section 2)"
step '^(arm|suite|E_)' python3 "$W/bin/r1-arms.py" "$ARMC" "$RAW" "$R" "$W/r1src"
S1=$R/lanes/review-noloss/suites/gsta167-14pr.json
step '^(scan|left)' bash "$W/bin/r1-scrub.sh" "$W/r1src/C" "$W/bin/scrub.ts" "$W/bin/scrub2.ts" "$W/bin/r1-scrub3.ts"
step '^(reader|C:|none|gstack|E_)' python3 "$W/bin/blind.py" "$S1" "$W/r1src/C" "$W/r1src/none-opus" "$W/r1src/gstack-opus"
R1ARMS="C none-opus gstack-opus"
for a in $R1ARMS; do step 'ingested|published|E_' $G ingest "$W/r1src/$a" --suite "$S1" --root "$W/r1/raw"; done
step . sh -c "cd $W/r1src/C && python3 -c \"import csv,sys; r=[x for x in csv.DictReader(open('cost.tsv'),delimiter='\\t')]; print('arm C cost rows', len(r))\""
step '^(exposure|gate|E_)' python3 "$W/bin/r1-exposure.py" "$W/r1src/C"


# judge_round <round> <scenario> <FAKE_RATES>: both judging batches of a round on that round's own items, then the
# scores, the secondaries and the decision. Reads SUITE, RAWR (raw arms root), ARMS, COMP, IDS, VAR, REF and DECIDE.
judge_round() {
  local X=$W/$1/$2; local SID; SID=$(jq -r .id "$SUITE")
  mkdir -p "$X"; cp -r "$W/record" "$X/record"
  step 'packet|E_' $G judge-packet --suite "$SUITE" $(arms $RAWR $ARMS) --seed 20261031 --out "$X/match" --key "$X/match-key.json"
  local J="--suite $SUITE --packets $X/match --key $X/match-key.json $(arms $RAWR $ARMS) --brief $P/judge-brief.md $JUDGES --ledger $X/ledger.sqlite --record $X/record"
  step '^(worst case|item )|E_' $G judge-run $J --item $1-judge-match --out "$X/judge-match" --dry-run
  FAKE_RATES=$3 step '^(judge-run|E_)|unjudged' bun "$W/bin/judge-fake.ts" $J --item $1-judge-match --out "$X/judge-match" --concurrency 4
  step 'lease|state|E_' sh -c "jq -c '{state, unjudged, lease: (.lease // .ledger // null)}' $X/judge-match/run.json"
  step 'judge-join|E_' $G judge-join --suite "$SUITE" --key "$X/match-key.json" --reply "$X/judge-match/primary.json" --second "$X/judge-match/second.json" --third "$X/judge-match/third.json" $(arms $RAWR $ARMS) --out "$X/judge-match/joined"
  for a in $ARMS; do step 'published|E_' $G ingest "$X/judge-match/joined/$a" --suite "$SUITE" --root "$X/judged"; done
  step 'packet|E_' $G judge-packet --suite "$SUITE" $(arms $X/judged/$SID $ARMS) --seed 20261031 --out "$X/adj" --key "$X/adj-key.json" --adjudicate
  local A="--suite $SUITE --packets $X/adj --key $X/adj-key.json $(arms $X/judged/$SID $ARMS) --brief $P/judge-brief.md $JUDGES --ledger $X/ledger.sqlite --record $X/record --source garrytan/gstack=$GSRC --source garrytan/gbrain=$GBSRC"
  step '^(worst case|item )|E_' $G judge-run $A --item $1-judge-adjudicate --out "$X/judge-adj" --dry-run
  echo; echo "# A bound item refuses a second run: the matching run bound $1-judge-match."
  FAKE_RATES=$3 step 'E_' bun "$W/bin/judge-fake.ts" $A --item $1-judge-match --out "$X/judge-adj-on-match-item"
  FAKE_RATES=$3 step '^(judge-run|unverifiable|E_)' bun "$W/bin/judge-fake.ts" $A --item $1-judge-adjudicate --out "$X/judge-adj" --concurrency 4
  step 'judge-join|E_' $G judge-join --suite "$SUITE" --key "$X/adj-key.json" --reply "$X/judge-adj/primary.json" --second "$X/judge-adj/second.json" --third "$X/judge-adj/third.json" $(arms $X/judged/$SID $ARMS) --out "$X/judge-adj/joined"
  for a in $ARMS; do step 'published|E_' $G ingest "$X/judge-adj/joined/$a" --suite "$SUITE" --root "$X/final"; done
  step 'committed|settled|open|E_' $L status --ledger "$X/ledger.sqlite" --ledger-record "$X/record"
  for id in $IDS; do
    local v r; v=$(jq -r --arg i $id '.comparisons[] | select(.id == $i) | .variant' "$COMP"); r=$(jq -r --arg i $id '.comparisons[] | select(.id == $i) | .reference' "$COMP")
    step '^([A-Za-z0-9-]+ vs [a-z0-9-]+ at a |verdict|false blockers|E_)' $G score --suite "$SUITE" --root "$X/final" --prereg "$COMP" --variant $v --reference $r --out "$X/score-$id.json"
  done
  step '^(recall|new|idea|E_)' $G secondary --suite "$SUITE" --root "$X/final" --variant $VAR --reference $REF --seed 20261031
  $DECIDE "$X" "$SID"
}
# incomplete <score json> <out json>: the same record with a judgment marked coverage incomplete, as the scorer writes it
# (no_verdict_reason set, verdict null). Synthetic: it drives the decision guard, not the scorer.
incomplete() { jq '.result.no_verdict_reason = "a judgment has coverage incomplete" | .result.verdict = null' "$1" > "$2"; }
r1_decide() { step '^(R1|exposure|gate|delta|decision|CUT|R2|volume|E_)' python3 "$W/bin/r1-decide.py" "${3:-$1/score-C-vs-none.json}" "$W/r1src/C" "$1/final/$2"; }
SUITE=$S1; RAWR=$W/r1/raw/gsta167-14pr; ARMS=$R1ARMS; COMP=$W/bin/comparisons.r1.json; IDS="C-vs-none C-vs-gstack gstack-vs-none"; VAR=C; REF=none-opus; DECIDE=r1_decide
echo; echo "## R1 judging, scenario 1: fake answers that give arm C a lead (section 2)"
judge_round r1 lead "C=40,none-opus=30,gstack-opus=30"
echo; echo "## R1 judging, scenario 2: fake answers with no lead for arm C"
judge_round r1 nolead "C=2,none-opus=30,gstack-opus=30"
echo; echo "# R1 with a judgment marked coverage incomplete on scenario 1's scores: no decision."
incomplete "$W/r1/lead/score-C-vs-none.json" "$W/r1/lead/score-incomplete.json"
r1_decide "$W/r1/lead" gsta167-14pr "$W/r1/lead/score-incomplete.json"

echo; echo "## R2: the 19-unit suite, fixture arms and the procedure gate (section 3)"
mkdir -p "$W/r2"
(echo unit; jq -r '.units[].id' "$S1") > "$W/r2/cohort14.tsv"
step . sha256sum "$W/r2/cohort14.tsv"
step '^(derive|suite|E_)|units|blocking' $G derive --suite "$P/stacks-r1.json" --comparisons "$REV/comparisons.stacks-r1.nopstack.json" --id stacks-r1-out19 --out "$W/r2/suite" --root "$REV/final/arms" --drop-file "$W/r2/cohort14.tsv"
S2=$W/r2/suite/stacks-r1-out19.json
step . sha256sum "$S2"
step '^(arm|E_)' python3 "$W/bin/r2-arms.py" "$RAW" "$S2" "$W/r2src" pass
step '^(arm|E_)' python3 "$W/bin/r2-arms.py" "$RAW" "$S2" "$W/r2src-fail" fail
step '^(reader|gstack|none|E_)' python3 "$W/bin/blind.py" "$S2" "$W/r2src/gstack-full" "$W/r2src/none-r2"
for a in gstack-full none-r2; do step 'ingested|published|E_' $G ingest "$W/r2src/$a" --suite "$S2" --root "$W/r2/raw"; done
step '^(exposure|gate)' python3 "$W/bin/r1-exposure.py" "$W/r2src/gstack-full"
step '^(exposure|gate)' python3 "$W/bin/r1-exposure.py" "$W/r2src-fail/gstack-full"
echo; echo "# R2's sessions: one lease per frozen run, none-r2 on r2a-grid and gstack-full on r2b-grid; a bound item refuses a second run."
mkdir -p "$W/r2/grid"; cp -r "$W/record" "$W/r2/grid/record"
LG="--ledger $W/r2/grid/ledger.sqlite --ledger-record $W/r2/grid/record"
for pair in r2a-grid:r2-none:25 r2b-grid:r2-full:325; do IFS=: read -r it run amt <<< "$pair"; step 'lease_id|state|E_' sh -c "$L lease --item $it --run $run --amount $amt $LG --json | tee $W/r2/grid/$run.json"; done
step 'E_' $L lease --item r2b-grid --run r2-full-again --amount 1 $LG
for run in r2-none r2-full; do step 'settle|E_' $L settle --lease "$(jq -r .lease_id "$W/r2/grid/$run.json")" $LG; done
step 'committed|settled|open|E_' $L status $LG
r2_decide() { step '^(R2|exposure|gate|E_)' python3 "$W/bin/r2-decide.py" "${3:-$1/score-gstack-full-vs-none.json}" "$W/r2src/gstack-full"; }
SUITE=$S2; RAWR=$W/r2/raw/stacks-r1-out19; ARMS="gstack-full none-r2"; COMP=$W/bin/comparisons.r2.json; IDS=gstack-full-vs-none; VAR=gstack-full; REF=none-r2; DECIDE=r2_decide
echo; echo "## R2 judging, scenario 1: fake answers that favour gstack-full"
judge_round r2 lift "gstack-full=90,none-r2=10"
echo; echo "## R2 judging, scenario 2: fake answers with no lift (the kill rule)"
judge_round r2 nolift "gstack-full=10,none-r2=50"
echo; echo "# R2 with the procedure gate failing on scenario 1's scores: no outcome."
step '^(R2|exposure|gate|E_)' python3 "$W/bin/r2-decide.py" "$W/r2/lift/score-gstack-full-vs-none.json" "$W/r2src-fail/gstack-full"
echo; echo "# R2 with a judgment marked coverage incomplete on scenario 1's scores: no outcome."
incomplete "$W/r2/lift/score-gstack-full-vs-none.json" "$W/r2/lift/score-incomplete.json"
r2_decide "$W/r2/lift" stacks-r1-out19 "$W/r2/lift/score-incomplete.json"

echo; echo "## R3: the in-band set and every frozen rule on synthetic trials (section 4)"
step . git -C "$TAG" log -1 --format='lane b runner %H %s'
step . sh -c "cd $TAG && sha256sum quick/laneb/runner/ledger.ts quick/laneb/analysis/analyze.ts quick/laneb/tasks/prescreen_swebv.py"
mkdir -p "$W/r3"
step '^band' python3 "$W/bin/r3-band.py" "$SWEBV" "$W/r3/band.txt"
for s in lift small nolift gatefail saturated pilotfail capstop; do
  echo; echo "# R3 scenario $s"
  SHOW=30 step '^(r3|admission|R3|cell|contrast|procedure|contamination|gap)' env TAG="$TAG" bun "$W/bin/r3-dry.ts" "$W/r3/band.txt" "$W/laneb" "$W/r3/$s" $s
done
echo; echo "## Done"
