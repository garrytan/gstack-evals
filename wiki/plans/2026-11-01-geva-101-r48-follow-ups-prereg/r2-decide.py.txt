#!/usr/bin/env python3
# R2 decision (GEVA-101 prereg, section 3). Reads the pinned scorer's JSON record for gstack-full-vs-none and the
# procedure gate on gstack-full's source arm (r1-exposure.py, the same gate as R1), and prints the frozen outcome:
#  - none, when the gate failed, an arm stopped, or the scorer gave no verdict, no delta or no interval
#    (no_verdict_reason set): R2 is incomplete and goes to the Captain as FIX-METHOD (the procedure under test didn't
#    run, so nothing about it was measured);
#  - CUT-CANDIDATE, when gstack-full found no more blocking blockers than none-r2 (the kill rule);
#  - KEEP, when the 95% PR-clustered interval of the recall difference is above zero;
#  - otherwise FIX-METHOD: gstack-full found more, but 13 blockers can't show it (power is the Why-null reason).
# Exit 0 KEEP, 1 CUT-CANDIDATE, 2 FIX-METHOD, 4 none, 5 an error (bad input or a gate error), which is never an outcome.
# Usage: python3 r2-decide.py <score-gstack-full-vs-none.json> <gstack-full source dir> <suite json>
import json, os, subprocess, sys


def main():
    s = json.load(open(sys.argv[1]))
    here = os.path.dirname(os.path.abspath(__file__))
    gate = subprocess.run([sys.executable, os.path.join(here, "r1-exposure.py"), sys.argv[3], sys.argv[2]], capture_output=True, text=True)
    print(gate.stdout.strip())
    if gate.returncode not in (0, 3):
        raise RuntimeError("the procedure gate exited %d: %s" % (gate.returncode, gate.stderr.strip()[-300:]))
    res = s["result"]
    ci = res.get("ci95") or [None, None]
    stopped = s["variant"]["status"] != "complete" or s["reference"]["status"] != "complete"
    if gate.returncode or stopped or res.get("no_verdict_reason") or res.get("delta") is None or None in ci:
        why = "procedure gate failed" if gate.returncode else "an arm stopped" if stopped else res.get("no_verdict_reason") or "the scorer gave no delta or no interval"
        print("R2 outcome: none (" + why + "). FIX-METHOD to the Captain; no CUT-CANDIDATE and no KEEP.")
        sys.exit(4)
    b = res["blockers"]
    f, n = round(res["recall"]["variant"] * b), round(res["recall"]["reference"] * b)
    lo, hi = ci[0] * 100, ci[1] * 100
    print("R2 count: gstack-full found %d of %d blocking blockers, none-r2 %d; difference %+.1f points [95%% CI %+.1f, %+.1f]" % (f, b, n, res["delta"] * 100, lo, hi))
    if f <= n:
        print("R2 outcome: CUT-CANDIDATE. The full /review procedure found no more blocking blockers than plain Opus 5.5; the Review Army and adversarial passes go to Garry with this record.")
        sys.exit(1)
    if lo > 0:
        print("R2 outcome: KEEP. The full procedure found more blocking blockers than plain Opus 5.5, and the interval is above zero.")
        sys.exit(0)
    print("R2 outcome: FIX-METHOD. The full procedure found more, but the interval includes zero; 13 blockers give an MDD of about 43 points (section 6). A larger suite needs its own prereg.")
    sys.exit(2)


try:
    main()
except Exception as e:
    print("E_R2_DECIDE %s: %s. This is an error, not an outcome." % (type(e).__name__, e))
    sys.exit(5)
