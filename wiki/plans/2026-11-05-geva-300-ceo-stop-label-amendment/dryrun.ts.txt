#!/usr/bin/env bun
// dryrun.ts: keyless dry run of the CEO outcome eval's preregistration (GEVA-285), as amended by
// 2026-11-05-geva-289-ceo-rule-guards-amendment, 2026-11-05-geva-295-ceo-ladder-amendment and
// 2026-11-05-geva-300-ceo-stop-label-amendment (two stop labels in section f3, each with its own check). It drives every frozen rule
// in ceo-rule.ts on synthetic readings, drives each input guard on its repro and on its accept twin (sections f2 and f3),
// and opens and settles one ledger lease per frozen item on a temporary ledger.
// Every unit, review, label and cost here is synthetic. No model is called and no key is in the environment.
// Run from a directory holding the preregistration's bundle files with the guard amendment's files, the ladder
// amendment's files and then the stop label amendment's files copied over them, and ceo-rule.ts.txt and dryrun.ts.txt
// renamed to drop .txt:
//   env -u ANTHROPIC_API_KEY -u OPENAI_API_KEY -u OPENROUTER_API_KEY bun dryrun.ts <scorer checkout> <scratch dir>
// It writes dryrun.txt beside itself.

// fill() adds the ladder amendment's row fields to any fixture row or call that lacks them, so the guard amendment's cases
// read as before: a scored row's issues follow its words (each cut to 40 words), a row that isn't scored carries no issues
// and no words, and a call on a keyed unit carries rem following its own D call (stock calls: no), null on any other unit.
// Each row's fixes follow its issues the same way, with fix_words equal to its words.
// A case that tests a missing field fills first, deletes the field and runs raw.
const wordsOf = (n: unknown) => (typeof n === "number" && Number.isFinite(n) ? Array(Math.min(40, n)).fill("word").join(" ") : "");
const remOf = (c: any, stock: boolean) => Object.fromEntries(["sol", "opus", "sonnet"].map((j) => [j, c[j] === null || c[j] === undefined ? null : !stock && c[j] === "D"]));
function fill(d: any): any {
  if (d?.stage1) { fill(d.stage1); return d; }
  if (!Array.isArray(d?.units)) return d;
  const keyed = new Map(d.units.map((u: any) => [u.id, u.remedy === true]));
  const calls = (ls: any, stock: boolean, unit: string) => { if (Array.isArray(ls)) for (const c of ls) if (c && typeof c === "object" && !("rem" in c)) c.rem = keyed.get(unit) ? remOf(c, stock) : null; };
  for (const r of Array.isArray(d.reviews) ? d.reviews : []) {
    if (!("issues" in r)) { if (r.status === "scored") r.issues = Array.isArray(r.words) ? r.words.map(wordsOf) : []; else { r.issues = []; if (Array.isArray(r.words)) r.words = []; } }
    if (!("fixes" in r)) { r.fix_words = Array.isArray(r.words) ? [...r.words] : []; r.fixes = r.fix_words.map(wordsOf); }
    calls(r.labels, false, r.unit);
  }
  for (const s of Array.isArray(d.stock) ? d.stock : []) calls(s.labels, true, s.unit);
  return d;
}
// compare reads the projection output beside the readings. Unless a case gives its own, projText writes the PASS and
// Primary lines the readings' own family, passes, held passes and lens imply, and runRule passes its sha256.
function projText(d: any, primary = "hit@3 (stage 1 HEADROOM); G_R form rem") {
  const fam = d?.family ?? {}, passes = d?.passes ?? {}, lens = d?.lens === true;
  const cells = ["opus", "sol", "particle"].flatMap((m) => ["none:" + m, ...(Array.isArray(fam[m]) ? fam[m] : []).filter((a: string) => a !== "lens").map((a: string) => a + ":" + m), ...(m === "opus" && lens ? ["lens:opus"] : [])]);
  const one = ["sol", "particle"].filter((m) => passes[m] === 1);
  const label = one.length ? "Sol and particle-1 cells at 1 pass" : !lens ? "lens dropped" : d?.held_passes === 1 ? "held units at 1 pass" : "as planned";
  return "Stage 2 projection check (synthetic)\nPrimary: " + primary + "\nProjection: PASS at '" + label + "'; record these caps; cells: " + cells.join(", ") + "; held passes " + (d?.held_passes === 1 ? 1 : 2) + (one.length ? "; 1 pass on " + one.join(", ") : "") + "\n";
}
type Opt = { raw?: boolean; proj?: string | null; sha?: string; primary?: string };
const runRule = (cmd: string, data: unknown, tag: string, o: Opt = {}) => {
  const f = join(OUT, tag + ".json");
  const d = o.raw ? data : fill(structuredClone(data));
  writeFileSync(f, JSON.stringify(d).replaceAll('"__1e999__"', "1e999"));
  const args = [join(HERE, "ceo-rule.ts"), cmd, f];
  if (cmd === "compare" && o.proj !== null) {
    const text = o.proj ?? projText(d, o.primary);
    writeFileSync(join(OUT, tag + ".projection.txt"), text);
    args.push(join(OUT, tag + ".projection.txt"), o.sha ?? createHash("sha256").update(text).digest("hex"));
  }
  return spawnSync("bun", args, { encoding: "utf8", env: { ...KEYLESS, GSTACK_EVALS_SCORER: scorer }, maxBuffer: 1 << 26 });
};
const refusal = (cmd: string, data: unknown, tag: string) => {
  const r = runRule(cmd, data, tag);
  return { status: r.status, msg: (r.stderr.split("\n").find((l) => l.startsWith("error: ")) ?? "").slice(7) };
};
const rule = (cmd: string, data: unknown, tag: string) => {
  const r = runRule(cmd, data, tag);
  if (r.status !== 0) throw new Error("ceo-rule.ts " + cmd + " exited " + r.status + ": " + r.stderr.trim().split("\n").slice(-2).join(" "));
  return r.stdout;
};
const PLAN = "E_USAGE readings must carry one row per planned session";
const verdictOf = (o: string) => (o.split("\n").find((l) => l.startsWith("Verdict: ")) ?? "").slice(9);
const show = (o: string, keep: RegExp) => { for (const l of o.split("\n").filter((l) => keep.test(l))) say("    " + l); };
const tagOf = (s: string) => s.slice(0, 48).replace(/[^a-z0-9]+/gi, "-");

type Label = "D" | "R1" | "R2" | "R3" | "none";
const agree = (l: Label) => ({ sol: l, opus: l, sonnet: l });
const missLabels = () => [agree("none"), agree("R1"), agree("none")];
const hitLabels = (rank = 1) => [0, 1, 2].map((i) => agree(i === rank - 1 ? "D" : "none"));
const PLANNED: Record<string, Record<string, number>> = { opus: { none: 0.25, "autoplan-ceo": 3, standalone: 0.7, lens: 0.3 }, sol: { none: 0.15, "autoplan-ceo": 1.8, standalone: 0.42 }, particle: { none: 0.06, "autoplan-ceo": 0.33, standalone: 0.15 } };
const POSTED = ["2026-07-20", "2026-09-12", "2026-10-03", "2026-08-02"];
const CATS = ["scope", "premise or demand", "user impact or migration", "direction or strategy fit"];
let rid = 0;
const review = (unit: string, arm: string, model: string, passNo: number, isHit: boolean, extra: Record<string, unknown> = {}) => ({ id: "r" + ++rid, unit, arm, model, pass: passNo, status: "scored", exposed: arm === "none" || arm === "lens" ? null : true, labels: isHit ? hitLabels(1 + (rid % 3)) : missLabels(), words: [28, 35, 41], leak: { sol: false, opus: false }, forecast: isHit ? 30 : 70, usd: PLANNED[model]![arm]!, ...extra });
const KINDS = ["scope", "alternative", "approach", "not_build"];
// Changed unit ci is keyed unless i % 4 is 3 (a fixture may choose), with a remedy kind, decoy R1 on even units and the
// second-order flag on every fifth; held units carry false, null, null and false.
const unitsOf = (nc: number, nh: number, keyed: (i: number) => boolean = (i) => i % 4 !== 3): any[] => [...Array(nc)].map((_, i) => ({ id: "c" + i, kind: "changed", strict: POSTED[i % 4], category: CATS[i % 4], role: i % 3 === 0 ? "maintainer" : "outside", remedy: keyed(i), remedy_kind: keyed(i) ? KINDS[Math.floor(i / 4) % 4] : null, decoy: i % 2 === 0 ? "R1" : null, second_order: i % 5 === 0 })).concat([...Array(nh)].map((_, i) => ({ id: "h" + i, kind: "held", strict: "2026-07-10", category: "held", role: "outside", remedy: false, remedy_kind: null, decoy: null, second_order: false })));
const remAll = (b: boolean | null) => ({ sol: b, opus: b, sonnet: b });
const stockOf = (units: any[], hitA: (i: number) => boolean, hitB: (i: number) => boolean): any[] => units.filter((u) => u.kind === "changed").flatMap((u, i) => [{ unit: u.id, row: "A", labels: hitA(i) ? hitLabels(2) : missLabels() }, { unit: u.id, row: "B", labels: hitB(i) ? hitLabels(3) : missLabels() }]);

// Stage 1: 16 changed + 4 held calibration units; Opus none 3 passes, Sol and particle-1 none 2 passes; smoke: 2 sessions
// per gstack arm per model on c0 and c1.
// Opus none's hits are at rank 1 unless ranks is mixed or rank1 sets how many hits, in reading order, are at rank 1 (the
// rest at rank 2). remYes gives rem yes to the first keyed hits and no to the rest; remNull makes the D call's rem
// unjudged on the readings it names and remYesAt makes it yes there; solOnlyRem gives Sol alone a rem yes on that many cells of misses on keyed units;
// stockRem and stockRemUnjudged set the first stock issue of row A's rem; splitRem turns the first miss on a keyed unit into
// a split D at rank 1 with a split rem.
type S1 = { ranks?: "mixed"; rank1?: number; keyed?: (i: number) => boolean; remYes?: number; remNull?: number[]; remYesAt?: number[]; solOnlyRem?: number; stockRem?: string[]; stockRemUnjudged?: string[]; splitRem?: boolean; m: number; incomplete?: number; stockA?: number; stockB?: number; stockSame?: boolean; leakUnits?: number; gaps?: number; smokeFail?: string[]; smokeSlow?: string[]; smokeNoBlock?: string[]; splitDecides?: boolean; judgeFail?: boolean; lowAgree?: boolean; smokeUsd?: Record<string, [number, number]>; noneUsd?: Record<string, number> };
function stage1(o: S1) {
  rid = 0;
  const units = unitsOf(16, 4, o.keyed);
  const isKeyed = (u: string) => units.find((x) => x.id === u)!.remedy;
  const reviews: any[] = [], opusNone: any[] = [];
  const usdNone = (m: string) => ({ usd: o.noneUsd?.[m] ?? PLANNED[m]!.none! });
  let k = 0;
  for (let p = 1; p <= 3; p++) for (let i = 0; i < 16; i++) { const r: any = review("c" + i, "none", "opus", p, k < o.m, usdNone("opus")); if (k < o.m && o.ranks !== "mixed") r.labels = hitLabels(o.rank1 === undefined || k < o.rank1 ? 1 : 2); opusNone.push(r); if (k >= 48 - (o.gaps ?? 0)) { r.status = "gap"; r.labels = []; } if (k >= 48 - (o.incomplete ?? 0)) { r.status = "incomplete"; r.labels = []; r.forecast = null; r.usd = 0; } reviews.push(r); k++; }
  const misses = () => reviews.filter((r) => r.status === "scored" && r.labels[0]?.sol === "none" && r.labels[1]?.sol === "R1");
  if (o.splitDecides) {
    const [r0, r1, r2] = misses();
    r0.labels = [{ sol: "D", opus: "none", sonnet: "D" }, agree("none"), agree("none")];
    r1.labels = [{ sol: "D", opus: "none", sonnet: "none" }, agree("none"), agree("none")];
    r2.labels = [{ sol: "R1", opus: "none", sonnet: "D" }, agree("none"), agree("none")];
  }
  if (o.judgeFail) {
    const [r0, r1] = misses();
    r0.labels = [{ sol: null, opus: "D", sonnet: "D" }, agree("none"), agree("none")];
    r1.labels = [{ sol: null, opus: "D", sonnet: "none" }, agree("none"), agree("none")];
  }
  for (let i = 0; i < 4; i++) for (let p = 1; p <= 3; p++) reviews.push(review("h" + i, "none", "opus", p, false, usdNone("opus")));
  for (const model of ["sol", "particle"]) for (let p = 1; p <= 2; p++) for (const u of units) reviews.push(review(u.id, "none", model, p, u.kind === "changed" && Number(u.id.slice(1)) % 2 === 0, usdNone(model)));
  if (o.lowAgree) for (const r of reviews) if (r.status === "scored" && r.labels[0]?.sol === "none" && r.labels[1]?.sol === "R1") r.labels[0] = { sol: "D", opus: "none", sonnet: "none" };
  if (o.remYes !== undefined) { let n = 0; for (const r of opusNone) if (r.status === "scored" && isKeyed(r.unit) && r.labels.some((c: any) => c.sol === "D")) for (const c of r.labels) c.rem = remAll(c.sol === "D" ? n++ < o.remYes : false); }
  for (const k of o.remNull ?? []) opusNone[k].labels.find((c: any) => c.sol === "D").rem = { sol: true, opus: false, sonnet: null };
  for (const k of o.remYesAt ?? []) opusNone[k].labels.find((c: any) => c.sol === "D").rem = remAll(true);
  if (o.splitRem) { const r = opusNone.find((r) => r.status === "scored" && isKeyed(r.unit) && !r.labels.some((c: any) => c.sol === "D")); r.labels = [{ sol: "D", opus: "none", sonnet: "D", rem: { sol: true, opus: false, sonnet: true } }, agree("none"), agree("none")]; }
  if (o.solOnlyRem) { let n = 0; for (const r of reviews) if (r.arm === "none" && r.status === "scored" && isKeyed(r.unit) && !r.labels.some((c: any) => c.sol === "D")) for (const c of r.labels) if (n < o.solOnlyRem) { c.rem = { sol: true, opus: false, sonnet: false }; n++; } }
  for (let l = 0; l < (o.leakUnits ?? 0); l++) for (const r of reviews.filter((r) => r.unit === "c" + (15 - l)).slice(0, 2)) r.leak = { sol: true, opus: true };
  for (const model of ["opus", "sol", "particle"]) for (const arm of ["autoplan-ceo", "standalone"]) ["c0", "c1"].forEach((u, i) => {
    const key = arm + ":" + model;
    const usd = (o.smokeUsd?.[key] ?? [0.85 * PLANNED[model]![arm]!, PLANNED[model]![arm]!])[i];
    const nb = (o.smokeNoBlock ?? []).includes(key) && i === 1;
    reviews.push(review(u, arm, model, 1, false, { smoke: true, exposed: !(o.smokeFail ?? []).includes(key), status: nb ? "no-block" : "scored", labels: [], wall_s: (o.smokeSlow ?? []).includes(key) && i === 0 ? 2400 : 900, usd }));
  });
  const stock = stockOf(units, (i) => i < (o.stockA ?? 0) || (!!o.stockSame && i === 0), (i) => (i >= 8 && i < 8 + (o.stockB ?? 0)) || (!!o.stockSame && i === 0));
  for (const u of o.stockRem ?? []) stock.find((s) => s.unit === u && s.row === "A")!.labels[0]!.rem = remAll(true);
  for (const u of o.stockRemUnjudged ?? []) stock.find((s) => s.unit === u && s.row === "A")!.labels[0]!.rem = { sol: true, opus: false, sonnet: null };
  return { stage: 1, units, reviews, stock };
}

// Stage 2: changed + 16 held units, 2 passes (held units at min(passes, heldPasses)). hits[model][arm] is the number of
// changed units hit in both passes. lens is true when the fixture runs lens on Opus, unless the case sets it.
// remNull makes the D call's rem unjudged on the first pass of an arm's first hit units that are keyed; stockRem gives the
// first stock issue of row A a rem yes on the first keyed changed units.
type S2 = { remNull?: Record<string, number>; stockRem?: number; hits: Record<string, Record<string, number>>; passes?: Record<string, number>; heldPasses?: number; lens?: boolean; exposure?: Record<string, number>; gapUnits?: Record<string, number>; incomplete?: Record<string, number>; noBlock?: Record<string, number>; unjudged?: Record<string, number>; nullForecast?: Record<string, number>; changed?: number; skip?: string[]; family?: Record<string, string[]>; stockB?: number };
function stage2(o: S2) {
  rid = 0;
  const nc = o.changed ?? 48;
  const units = unitsOf(nc, 16);
  const reviews: any[] = [];
  for (const [model, arms] of Object.entries(o.hits)) for (const [arm, h] of Object.entries(arms)) {
    const key = arm + ":" + model;
    if ((o.skip ?? []).includes(key)) continue;
    const exp = o.exposure?.[key] ?? 1;
    let n = 0, nulls = o.remNull?.[key] ?? 0;
    for (const u of units) for (let p = 1; p <= (u.kind === "changed" ? (o.passes?.[model] ?? 2) : Math.min(o.passes?.[model] ?? 2, o.heldPasses ?? 2)); p++) {
      const idx = Number(u.id.slice(1)), ch = u.kind === "changed";
      const r: any = review(u.id, arm, model, p, ch && idx < h, { forecast: ch ? 25 : 80 });
      if (arm !== "none" && arm !== "lens") r.exposed = ch ? n++ < Math.round(exp * nc * 2) : true;
      if (ch && idx >= nc - (o.gapUnits?.[key] ?? 0)) { r.status = "gap"; r.labels = []; r.forecast = null; }
      if (ch && idx >= nc - (o.incomplete?.[key] ?? 0)) { r.status = "incomplete"; r.labels = []; r.forecast = null; r.usd = 0; r.exposed = null; }
      if (ch && idx < (o.noBlock?.[key] ?? 0)) { r.status = "no-block"; r.labels = []; r.forecast = null; }
      if (ch && p === 1 && idx >= h && idx < h + (o.unjudged?.[key] ?? 0)) r.labels[0] = { sol: null, opus: "D", sonnet: "none" };
      if (ch && idx < (o.nullForecast?.[key] ?? 0)) r.forecast = null;
      if (ch && p === 1 && idx < h && u.remedy && nulls > 0 && r.status === "scored") { nulls--; r.labels.find((c: any) => c.sol === "D").rem = { sol: true, opus: false, sonnet: null }; }
      reviews.push(r);
    }
  }
  const stock = stockOf(units, () => false, (i) => i < (o.stockB ?? 0));
  for (const u of units.filter((u) => u.kind === "changed" && u.remedy).slice(0, o.stockRem ?? 0)) stock.find((s) => s.unit === u.id && s.row === "A")!.labels[0]!.rem = remAll(true);
  return { stage: 2, units, reviews, stock, ...(o.family ? { family: o.family } : {}), passes: Object.fromEntries(["opus", "sol", "particle"].map((m) => [m, o.passes?.[m] ?? 2])), held_passes: o.heldPasses ?? 2, lens: o.lens ?? ("lens" in (o.hits.opus ?? {}) && !(o.skip ?? []).includes("lens:opus")) };
}

  const cases: [string, S1, RegExp][] = [

await section("f2. Input guards added by the guard amendment: each refusal on its repro, then its accept twin", () => {
  let gi = 0;
  const read = (o: string, shows: RegExp) => o.split("\n").filter((l) => shows.test(l)).map((l) => l.trim()).join(" | ");
  const refuse = (cmd: string, label: string, data: unknown, want: string, shows: RegExp) => {
    const r = runRule(cmd, data, "guard-" + ++gi + "-" + tagOf(label));
    const err = r.stderr.split("\n");
    const msg = (err.find((l) => l.startsWith("error: ")) ?? "").slice(7) || (err.find((l) => /^[A-Za-z]*Error: /.test(l)) ?? "");
    check("refuse " + label + " -> exit " + r.status + ": " + (msg || "no refusal; read " + read(r.stdout, shows)), r.status === 1 && msg === want);
  };
  const accept = (cmd: string, label: string, data: unknown, wants: RegExp[]) => {
    const r = runRule(cmd, data, "guard-" + ++gi + "-" + tagOf(label));
    const found = wants.map((w) => r.stdout.split("\n").find((l) => w.test(l)));
    check("  accept twin: " + label + " -> exit " + r.status + (r.status === 0 ? ": " + found.map((l) => (l ?? "(missing)").trim()).join(" | ") : ""), r.status === 0 && found.every((l) => l !== undefined));
  };
  const s1 = (o: Partial<S1> = {}) => stage1({ m: 20, ...o }) as any;
