const n=70, reps=3, conc=1.78;
function sim(m, f, d, sims){
  const a=m*conc, b=(1-m)*conc+1; // posterior after one failed screen trial
  let prim0=0, aFire=0, bFire=0, infoSum=0, subSum=0;
  for(let s=0;s<sims;s++){
    let sumAll=0, sumSub=0, nSub=0, info=0;
    for(let t=0;t<n;t++){
      const unfit = R()<f;
      const p = unfit?0:beta(a,b);
      const q = unfit?0:Math.min(1,p+d);
      let kn=0,kt=0; for(let r=0;r<reps;r++){ if(R()<p)kn++; if(R()<q)kt++; }
      const diff=(kt-kn)/reps; sumAll+=diff;
      if(kn+kt>0) info++;
      const other = R()<p; // other model's one screen trial, shared latent p
      if(other){ sumSub+=diff; nSub++; }
    }
    infoSum+=info; subSum+=nSub;
    if(sumAll<=0){ prim0++; if(nSub>0 && sumSub>0) aFire++; if(nSub===0) aFire++; if(info<30) bFire++; }
  }
  const pc=x=>(100*x/sims).toFixed(1);
  return {m,f,d, primaryAtOrBelow0:pc(prim0), A_fires:pc(aFire), B_fires:pc(bFire), cutAfterA:pc(prim0-aFire), cutAfterB:pc(prim0-bFire), meanInformative:(infoSum/sims).toFixed(1), meanOtherSolved:(subSum/sims).toFixed(1)};
}
